113bool HASHCRC16::Do(XBYTE* input, XQWORD size)
114{
115 if(!size) return false;
116
117 XWORD value = 0;
118
119 resulttmp->Extract(value);
120
121 for(XDWORD c=0; c<size; c++)
122 {
123 value = Update(value, (*input));
124 input++;
125 }
126
127 resulttmp->Add((XWORD)value);
128
129 return true;
130}
