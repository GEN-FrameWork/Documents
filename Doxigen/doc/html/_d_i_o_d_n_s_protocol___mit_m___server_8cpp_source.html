300bool DIODNSPROTOCOL_MITM_SERVER::Update()
301{
302 XBUFFER receivedbuffer[2];
303 DIODNSPROTOCOL_HEADER* header = NULL;
304 DIODNSPROTOCOL_QUESTION* question = NULL;
305 XWORD askcount = 0;
306 XBYTE* reader = NULL;
307 int stop = 0;
308 bool status = false;
309
310 if(!diostreamudp)
311 {
312 return false;
313 }
314
315 if(diostreamudp->GetInXBuffer()->GetSize() < sizeof(DIODNSPROTOCOL_HEADER))
316 {
317 return false;
318 }
319
320 receivedbuffer[0].Empty();
321 receivedbuffer[0].Resize(65535);
322
323 XDWORD sizeread = diostreamudp->Read(receivedbuffer[0]);
324 if(!sizeread)
325 {
326 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[DSN MitM Server] Error to Read header"));
327 return false;
328 }
329
330 header = (DIODNSPROTOCOL_HEADER*)receivedbuffer[0].Get();
331 askcount = SwapWORD(header->q_count);
332
333 reader = &receivedbuffer[0].Get()[sizeof(DIODNSPROTOCOL_HEADER)];
334
335 XSTRING nameURLstr;
336
337 for(XWORD c=0; c<askcount; c++)
338 {
339 XBYTE* nameURL = NULL;
340
341 nameURL = DIODNSPROTOCOLCLIENT::GetBufferName(reader, receivedbuffer[0].Get(), &stop);
342 nameURLstr.Set((char*)nameURL);
343
344 DIODNSPROTOCOL_MITM_SERVER_XEVENT xevent(this, DIODNSPROTOCOL_MITM_SERVER_XEVENT_TYPE_ASKDNS);
345 xevent.GetAskedURL()->Set(nameURLstr);
346 xevent.GetAskedBuffer()->CopyFrom(receivedbuffer[0]);
347
348 PostEvent(&xevent);
349
350 reader += stop;
351 question = (DIODNSPROTOCOL_QUESTION*)reader;
352 }
353
354 for(XDWORD c=0; c<dnsservers.GetSize(); c++)
355 {
356 DIODNSRESOLVER_DNSSERVER* DNSserver = dnsservers.Get(c);
357 if(DNSserver)
358 {
359 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[DSN MitM Server] %d server..."), c);
360
361 status = Detour(DNSserver, receivedbuffer[0], receivedbuffer[1]);
362 if(status)
363 {
364 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[DSN MitM Server] Buffer receiver"));
365 XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_BLUE, receivedbuffer[1]);
366
367 break;
368 }
369 else
370 {
371 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[DSN MitM Server] Error to Detour"));
372 }
373 }
374 }
375
376 status = diostreamudp->Write(receivedbuffer[1]);
377 if(status)
378 {
379 //status = diostreamudp->WaitToFlushOutXBuffer(3);
380 if(status)
381 {
382 DIODNSPROTOCOL_MITM_SERVER_XEVENT xevent(this, DIODNSPROTOCOL_MITM_SERVER_XEVENT_TYPE_ANSWERDNS);
383 xevent.GetAskedURL()->Set(nameURLstr);
384 xevent.GetAnsweredBuffer()->CopyFrom(receivedbuffer[1]);
385
386 PostEvent(&xevent);
387
388 status = true;
389 }
390 else
391 {
392 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[DSN MitM Server] Error to Write answer buffer"));
393 }
394 }

397 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[DSN MitM Server] Error to Write answer"));

606bool DIODNSPROTOCOL_MITM_SERVER::Detour(DIODNSRESOLVER_DNSSERVER* DNSserver, XBUFFER& sendbuffer, XBUFFER& receivedbuffer)
607{
608 if(!DNSserver)
609 {
610 return false;
611 }
612
613 DIOSTREAMUDPCONFIG dioudpcfg;
614 DIOSTREAMUDP* dioudp = NULL;
615 XSTRING remoteIP;
616 int serverport = 0;
617 bool status = false;
618
619 DNSserver->GetIP()->GetXString(remoteIP);
620
621 dioudpcfg.SetMode(DIOSTREAMMODE_CLIENT);
622 dioudpcfg.SetIsUsedDatagrams(false);
623 dioudpcfg.GetRemoteURL()->Set(remoteIP);
624
625 serverport = DNSserver->GetPort();
626
627 if(!serverport)
628 {
629 dioudpcfg.SetRemotePort(DIODNSPROTOCOL_DEFAULTPORT);
630 }
631 else
632 {
633 dioudpcfg.SetRemotePort(serverport);
634 }
635
636
637 dioudp = (DIOSTREAMUDP*)GEN_DIOFACTORY.CreateStreamIO(&dioudpcfg);
638 if(dioudp)
639 {
640 if(dioudp->Open())
641 {
642 status = dioudp->Write(sendbuffer);
643 if(status)
644 {
645 status = diostreamudp->WaitToFlushOutXBuffer(3);
646 }
647
648 if(status)
649 {
650 receivedbuffer.Empty();
651 receivedbuffer.Resize(65535);
652
653 status = diostreamudp->WaitToFilledReadingBuffer(sizeof(DIODNSPROTOCOL_HEADER), 2);
654 if(status)
655 {
656 status = diostreamudp->Read(receivedbuffer);
657 }
658 }
659
660 dioudp->Close();
661 }
662
663 GEN_DIOFACTORY.DeleteStreamIO(dioudp);
664 }
665
666 return status;
667}
