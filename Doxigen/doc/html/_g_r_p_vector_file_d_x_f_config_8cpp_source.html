121bool GRPVECTORFILEDXFCONFIG::SetHeader(GRPVECTORFILEDXFTEXTSECTIONHEADER* header)
122{
123 this->header = header;
124
125 fileCFG->DeleteAllValues();
126
127 for(XDWORD c=0; c<header->GetVariables()->GetSize(); c++)
128 {
129 XSTRING* namekey = header->GetVariables()->GetKey(c);
130 XVARIANT* variant = header->GetVariables()->GetElement(c);
131
132 if(namekey && variant)
133 {
134 switch(variant->GetType())
135 {
136 case XVARIANT_TYPE_BOOLEAN : fileCFG->AddValue(XFILECFG_VALUETYPE_BOOLEAN , VECTORFILEDXFCONFIG_SECTION_DFX_CFG, namekey->Get(), (bool*)(variant->GetData())); break;
137 case XVARIANT_TYPE_INTEGER : fileCFG->AddValue(XFILECFG_VALUETYPE_INT , VECTORFILEDXFCONFIG_SECTION_DFX_CFG, namekey->Get(), (int*)(variant->GetData())); break;
138 case XVARIANT_TYPE_FLOAT : fileCFG->AddValue(XFILECFG_VALUETYPE_FLOAT , VECTORFILEDXFCONFIG_SECTION_DFX_CFG, namekey->Get(), (float*)(variant->GetData())); break;
139 case XVARIANT_TYPE_STRING : fileCFG->AddValue(XFILECFG_VALUETYPE_STRING , VECTORFILEDXFCONFIG_SECTION_DFX_CFG, namekey->Get(), (XSTRING*)(variant->GetData())); break;
140
141 case XVARIANT_TYPE_NULL :
142 case XVARIANT_TYPE_DOUBLEINTEGER :
143 case XVARIANT_TYPE_DOUBLE :
144 case XVARIANT_TYPE_CHAR :
145 case XVARIANT_TYPE_XCHAR :
146 case XVARIANT_TYPE_DATE :
147 case XVARIANT_TYPE_TIME :
148 case XVARIANT_TYPE_DATETIME :
149 default : return false;
150 break;
151 }
152
153 XCHAR* remark = header->GetVariableRemark((*namekey));
154 if(remark)
155 {
156 fileCFG->AddRemark(VECTORFILEDXFCONFIG_SECTION_DFX_CFG, namekey->Get(), remark, 64, 0);
157 }
158
159 }
160 }
161
162 return true;
163}

203bool GRPVECTORFILEDXFCONFIG::Load_File(XCHAR* pathfile)
204{
205 if(!fileCFG)
206 {
207 return false;
208 }
209
210 if(!header)
211 {
212 return false;
213 }
214
215 XPATH path;
216 bool status = false;
217
218 path = pathfile;
219
220 header->DeleteAllVariables();
221
222 fileCFG->DeleteAllValues();
223
224 if(fileCFG->IniFile(path))
225 {
226 fileCFG->AjustRemarks();
227
228 XFILEINISECTION* sectionINI = fileCFG->GetFileINI()->GetSection(VECTORFILEDXFCONFIG_SECTION_DFX_CFG);
229 if(sectionINI)
230 {
231 XSTRING key;
232 XSTRING value;
233
234 while(sectionINI->GetNextKey(key, value))
235 {
236 XVARIANT variant;
237 XSTRING trash;
238 int type = 0;
239
240 trash.AdjustSize(_MAXSTR);
241
242 key.UnFormat(__L("$%s_%d"), trash.Get(), &type, NULL);
243
244 GRPVECTORFILEDXFTEXTSECTION::GetVariableFromLine(trash.Get(), type, &value, variant);
245
246 if(variant.GetType() != XVARIANT_TYPE_NULL)
247 {
248 header->AddVariable(key.Get(), &variant);
249 }
250 }
251 }
252
253 //status = fileCFG->EndFile();
254
255 fileCFG->GetFileINI()->XFILETXT::Close();
256 fileCFG->DeleteAllRemarks();
257 }
258
259 #ifdef XTRACE_ACTIVE
260 header->ShowTraceAllVariables();
261 #endif
262
263 return status;
264}
