333XDWORD DIOSTREAMICMP::Read(XBYTE* buffer, XDWORD size)
334{
335 XDWORD esize = 0;
336
337 int index = GetFirstDatagram(false);
338 if(index != DIOSTREAMICMP_NOTFOUND)
339 {
340 DIOSTREAMICMPDATAGRAM* datagram = GetDatagram(index);
341 if(datagram)
342 {
343 esize = datagram->GetData()->GetSize();
344 if(esize >= size)
345 {
346 esize = size;
347
348 memcpy(buffer,(XBYTE*)datagram->GetData()->Get(), size);
349
350 nbytesread+= size;
351 if(xtimernotactivity) xtimernotactivity->Reset();
352
353 DeleteDatagram(index);
354
355 inbuffer->Extract(buffer, 0, size);
356 }
357 }
358 }
359
360 return esize;
361}

688int DIOSTREAMICMP::GetFirstDatagram(bool tosend)
689{
690 int index = DIOSTREAMICMP_NOTFOUND;
691
692 if(datagramsmutex) datagramsmutex->Lock();
693
694 for(int c=0;c<(int)datagrams.GetSize();c++)
695 {
696 DIOSTREAMICMPDATAGRAM* datagram = (DIOSTREAMICMPDATAGRAM*)datagrams.Get(c);
697 if(datagram)
698 {
699 if(datagram->IsToSend() == tosend)
700 {
701 index = c;
702 break;
703 }
704 }
705 }
706
707 if(datagramsmutex) datagramsmutex->UnLock();
708
