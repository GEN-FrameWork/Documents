726bool XSYSTEM::HostFile(bool add, XCHAR* domain, XCHAR* IP)
727{
728 XPATH xpath;
729 XFILETXT xfileTXT;
730 bool status = false;
731
732 #ifdef WINDOWS
733 xpath = __L("C:\\Windows\\System32\\drivers\\etc\\hosts");
734 #endif
735
736 #if defined(LINUX) || defined(ANDROID)
737 xpath = __L("/etc/hosts");
738 #endif
739
740 if(xfileTXT.Open(xpath, false))
741 {
742 xfileTXT.ReadAllFile();
743
744 if(add)
745 {
746 int nline = IsLineInHostFile(xfileTXT, domain, IP);
747 if(nline == XSTRING_NOTFOUND)
748 {
749 XSTRING line;
750
751 line.Format(__L("%s %s"), IP, domain);
752
753 xfileTXT.AddLine(line);
754 xfileTXT.WriteAllFile();
755 }
756 }
757 else
758 {
759 int nline = IsLineInHostFile(xfileTXT, domain, IP);
760 if(nline != XSTRING_NOTFOUND)
761 {
762 xfileTXT.DeleteLine(nline);
763 status = xfileTXT.WriteAllFile();
764 }
765 }
766
767 xfileTXT.Close();
768 }
769
770 return status;
771}

806int XSYSTEM::IsLineInHostFile(XFILETXT& xfileTXT, XCHAR* domain, XCHAR* IP)
807{
808 bool found = false;
809 int c = 0;
810
811 if(!xfileTXT.GetNLines()) return XSTRING_NOTFOUND;
812
813 do{ XSTRING* line = xfileTXT.GetLine(c);
814 if(line)
815 {
816 int index = line->Find(domain, true);
817 if(index != XSTRING_NOTFOUND)
818 {
819 found = true;
820
821 for(XDWORD d=0; d<line->GetSize(); d++)
822 {
823 XCHAR character = line->Get()[d];
824
825 if((character != __C('\t') && (character != __C(' '))))
826 {
827 if(character == __C('#'))
828 {
829 found = false;
830 }
831
832 break;
833 }
834 }
835
836 if(found) break;
837 }
838 }
839
840 c++;
841
842 } while(c<xfileTXT.GetNLines());
843
844 if(!found) return XSTRING_NOTFOUND;
845
846 return c;
847}
