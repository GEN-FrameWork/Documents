617bool CIPHERCERTIFICATEX509::IsValidDates()
618{
619 XDATETIME* xdatetime_notbefore = GetDateNotBefore();
620 XDATETIME* xdatetime_notafter = GetDateNotAfter();
621
622 if(!xdatetime_notbefore || !xdatetime_notafter)
623 {
624 return false;
625 }
626
627 XQWORD seconds_notbefore = xdatetime_notbefore->GetSeconsFromDate();
628 XQWORD seconds_notafter = xdatetime_notafter->GetSeconsFromDate();
629 XQWORD seconds_actual = 0;
630
631 XDATETIME* xdatetime = GEN_XFACTORY.CreateDateTime();
632 if(xdatetime)
633 {
634 xdatetime->Read();
635 seconds_actual = xdatetime->GetSeconsFromDate();
636
637 GEN_XFACTORY.DeleteDateTime(xdatetime);
638 }
639
640 if((seconds_actual > seconds_notbefore) && (seconds_actual < seconds_notafter))
641 {
642 return true;
643 }
644
645 return false;
646}

915bool CIPHERCERTIFICATEX509::ConvertDateTime(XCHAR* datestr, XDATETIME* datetime)
916{
917 if(!datestr)
918 {
919 return false;
920 }
921
922 if(!datetime)
923 {
924 return false;
925 }
926
927 XSTRING year;
928 XSTRING month;

930
931 XSTRING hour;

938 {
939 year = __L("19");

941 else
942 {
943 year = __L("20");
944 }
945
946 year += datestr[0];
947 year += datestr[1];
948
949 month += datestr[2];
950 month += datestr[3];
951

953 day += datestr[5];
954
955 hour += datestr[6];
956 hour += datestr[7];
957

960
961 seconds += datestr[10];
