130bool DIOWEBSERVER_PLUGIN_CGI::PageExtension(XPATH& pathfile, DIOWEBSERVER_REQUEST* request, DIOWEBSERVER_QUERYSTRINGS* querystrings, DIOWEBHEADER_RESULT& headerresult, XSTRING& result)

132 if(!request) return false;

194 GEN_XSYSTEM.DelEnviromentVariable(__L("SCRIPT_FILENAME"));

198 GEN_XSYSTEM.DelEnviromentVariable(__L("QUERY_STRING"));
199
