107bool INPWINDOWSCAPTURE::Activate()
108{
109 if(keyhook) return false;
110
111 RegisterEvent(INPCAPTURE_XEVENT_TYPE_PRESSKEY);
112 RegisterEvent(INPCAPTURE_XEVENT_TYPE_UNPRESSKEY);
113
114 // Set up the hook
115
116 HINSTANCE hinst = (HINSTANCE)applicationhandle;
117 if(!hinst) return false;
118
119 keyhook = SetWindowsHookEx(WH_KEYBOARD_LL, LowLevelKeyboardProc, hinst, 0);
120 if(keyhook == NULL)
121 {
122 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[Windows Hook Keyboard] Error: Could not hook keyboard. Another program might be interfering."));
123 return false;
124 }
125
126 return true;
127}

139bool INPWINDOWSCAPTURE::Deactivate()
140{
141 if(!keyhook) return false;
142
143 DeRegisterEvent(INPCAPTURE_XEVENT_TYPE_PRESSKEY);
144 DeRegisterEvent(INPCAPTURE_XEVENT_TYPE_UNPRESSKEY);
145
146 //Remove keyboard hook
147 if(UnhookWindowsHookEx(keyhook) == 0)
148 {
149 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[Windows Hook Keyboard] Error: Could not unhook keyboard."));
150 return false;
151 }
152
153 //Success
154 keyhook = NULL;
155
156 return true;
157}

173LRESULT CALLBACK INPWINDOWSCAPTURE::LowLevelKeyboardProc(int ncode, WPARAM wparam, LPARAM lparam)
174{
175 bool keydown = false;
176
177 if(ncode == HC_ACTION)
178 {
179 if(wparam == WM_KEYDOWN || wparam == WM_SYSKEYDOWN) keydown = true;
180 if(wparam == WM_KEYUP || wparam == WM_SYSKEYUP) keydown = false;
181 }
182
183 KBDLLHOOKSTRUCT* keydata = (KBDLLHOOKSTRUCT*)lparam;
184
185 //XTRACE_PRINTCOLOR(XTRACE_COLOR_PURPLE, __L("[Windows Hook keyboard] vkCode %04X, scanCode %08X, flags %08X [%c]"), keydata->vkCode, keydata->scanCode, keydata->flags, keydata->vkCode);
186
187 if(capture)
188 {
189 INPCAPTURE_XEVENT xevent(capture, (keydown?INPCAPTURE_XEVENT_TYPE_PRESSKEY:INPCAPTURE_XEVENT_TYPE_UNPRESSKEY));
190
191 if(!keydown)
192 {
193 XCHAR character = (XCHAR)keydata->vkCode;
194
195 capture->SetNKeys(capture->GetNKeys());
196
197 if(capture->GetLimit())
198 {
199 if(capture->GetLimit() >= capture->GetNKeys())
200 {
201 capture->GetString()->Empty();
202 capture->GetBuffer()->Empty();
203 }
204 }
205
206 if(((character >= 0x20) && (character < 0x80)) || (character == 0x13) || (character == 0x10) || (character == 0x09))
207 {
208 capture->GetString()->Add(character);
209 }
210
211 capture->GetBuffer()->Add((XDWORD)keydata->vkCode);
212 }
213
214 xevent.SetVKCode((XDWORD)keydata->vkCode);
215 xevent.SetScanCode((XWORD)keydata->scanCode);
216 xevent.SetFlags((XDWORD)keydata->flags);
217
218 xevent.SetNKeys(capture->GetNKeys());
219 xevent.SetLimit(capture->GetLimit());
220 xevent.GetString()->Set(capture->GetString()->Get());
221 xevent.GetBuffer()->Add(capture->GetBuffer()->Get(), capture->GetBuffer()->GetSize());
222
223 capture->PostEvent(&xevent);
224 if(xevent.IsKeyLocked())
225 {
226 return 1;
227 }
228 else
229 {
230 return 0;
231 }
232 }
233
234 return CallNextHookEx(NULL, ncode, wparam, lparam);
235}
