284bool DIOMACMANUFACTURED::File_Convert(XBUFFER& xbuffer, XPATH& xpath)
285{
286 if(xpath.IsEmpty()) return false;
287
288 XFILETXT* xfiletxtoui = GEN_NEW XFILETXT();
289 if(!xfiletxtoui) return false;
290
291 XMAP<XDWORD, XSTRING*> idmap;
292 XMAP<XDWORD, int> idmapindex;
293 XDWORD c;
294
295 idmap.DeleteAll();
296 idmapindex.DeleteAll();
297
298 idmap.SetAddInLimit(10000);
299 idmapindex.SetAddInLimit(10000);
300
301 xfiletxtoui->AddBufferLines(XFILETXTFORMATCHAR_ASCII, xbuffer);
302
303 int nlines = xfiletxtoui->GetNLines();
304 for(int c=0;c<nlines;c++)
305 {
306 XSTRING* line = xfiletxtoui->GetLine(c);
307 if(line)
308 {
309 if(!line->IsEmpty())
310 {
311 int start = line->Find(__L("(base 16)"),true);
312 if(start!=XSTRING_NOTFOUND)
313 {
314 XDWORD ID = 0;
315
316 start+=11;
317 line->UnFormat(__L("%X (base 16)\t\t"),&ID);
318
319 if(idmap.Find(ID)==-1)
320 {
321 XSTRING* name = GEN_NEW XSTRING();
322 if(name)
323 {
324 line->Copy(start,(*name));
325 idmap.Add(ID,name);
326 }
327 }
328 }
329 }
330 }
331 }
332
333 GEN_DELETE xfiletxtoui;
334
335 XDWORD nrecords = idmap.GetSize();
336 HASHMD5 hashmd5;
337 XDWORD initablepos = 0;
338 XDWORD filepos = 0;
339
340 XFILEXDB* xfilexdb = GEN_NEW XFILEXDB( &hashmd5, xpath);
341 if(xfilexdb)
342 {
343 xfilexdb->Set(XFILE_XDB_ID, XFILEXDB_TYPE, DIOMACMANUFACTURED_FILE_VERSION, DIOMACMANUFACTURED_FILE_IDSTRING);
344
345 if(xfilexdb->Create(xpath))
346 {
347 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&nrecords, sizeof(XDWORD));
348 xfilexdb->GetPrimaryFile()->GetPosition(initablepos);
349
350 XDWORD ID = 0;
351
352 for(c=0;c<nrecords;c++)
353 {
354 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&ID, sizeof(XDWORD));
355 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&ID, sizeof(XDWORD));
356 }
357
358 for(c=0;c<nrecords;c++)
359 {
360 size_t _name = (size_t)idmap.GetElement(c);
361 XSTRING* name = (XSTRING*)_name;
362 if(name)
363 {
364 ID = (XDWORD)idmap.GetKey(c);
365
366 xfilexdb->GetPrimaryFile()->GetPosition(filepos);
367
368 XDWORD size = (name->GetSize()+1)*sizeof(XWORD);
369
370 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&size , sizeof(XDWORD));
371
372 XBUFFER xbufferexchange;
373
374 (*name)->ConvertToBufferExchange(xbufferexchange, true);
375 xfilexdb->GetPrimaryFile()->Write(xbufferexchange->Get(), size);
376
377 idmapindex.Add(ID,filepos);
378
379 GEN_DELETE name;
380 }
381 }
382
383 xfilexdb->GetPrimaryFile()->SetPosition(initablepos-sizeof(XDWORD));
384 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&nrecords,sizeof(XDWORD));
385
386 for(c=0;c<nrecords;c++)
387 {
388 ID = (XDWORD)idmapindex.GetKey(c);
389 filepos = (XDWORD)idmapindex.GetElement(c);
390
391 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&ID , sizeof(XDWORD));
392 xfilexdb->GetPrimaryFile()->Write((XBYTE*)&filepos, sizeof(XDWORD));
393 }
394
395 xfilexdb->CloseFile();
396 }
397 }
398
399 idmap.DeleteAll();
400 idmapindex.DeleteAll();
401
402 return true;
403}
