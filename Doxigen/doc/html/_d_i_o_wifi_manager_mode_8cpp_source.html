188bool DIOWIFIMANAGERMODE::CheckNetInterface(DIOIP* IP)
189{
190 bool status = false;
191
192 DIOSTREAMENUMDEVICES* enumdevicesIPlocal = NULL;
193
194 enumdevicesIPlocal = GEN_DIOFACTORY.CreateStreamEnumDevices(DIOSTREAMENUMTYPE_IP_LOCAL);
195 if(enumdevicesIPlocal)
196 {
197 enumdevicesIPlocal->Search();
198 while(enumdevicesIPlocal->IsSearching())
199 {
200 XSLEEP::GetInstance().MilliSeconds(10);
201 }
202
203 for(int c=0; c<(int)enumdevicesIPlocal->GetDevices()->GetSize(); c++)
204 {
205 DIOSTREAMDEVICEIP* deviceIP = (DIOSTREAMDEVICEIP*)enumdevicesIPlocal->GetDevices()->Get(c);
206 if(deviceIP)
207 {
208 bool isdeviceselect = false;
209
210 if(!netinterface.IsEmpty())
211 {
212 if(!deviceIP->GetName()->Compare(netinterface, true)) isdeviceselect = true;
213 }
214 else
215 {
216 if(deviceIP->GetIPType() == DIOSTREAMIPDEVICE_TYPE_WIFI) isdeviceselect = true;
217 }
218
219
220 if(isdeviceselect)
221 {
222 status = true;
223
224 if(deviceIP->GetIP()->IsEmpty()) status = false;
225 if(!deviceIP->GetIP()->IsLocal()) status = false;
226
227 if(IP) IP->Set(deviceIP->GetIP()->Get());
228
229 break;
230 }
231 }
232 }
233
234 GEN_DIOFACTORY.DeleteStreamEnumDevices(enumdevicesIPlocal);
235 }
236
237 return status;
238}
