650

652 t1 = count & (XMPINTEGER_BITSINLIMB - 1);
653
654 i = GetMSB() + count;

656 if(nlimbs * XMPINTEGER_BITSINLIMB < i)
657 {

1249 if(!T1.Grow(2)) return false;
1250 if(!T2.Grow(3)) return false;

1258 if(!Y.LeftShift(k)) return false;

1263 t = Y.GetNLimbs() - 1;

1300 T1.GetLimbs()[1] = Y.GetLimbs()[t];

1307 T2.GetLimbs()[1] = (i < 1) ? 0 : X.GetLimbs()[i - 1];
1308 T2.GetLimbs()[2] = X.GetLimbs()[i];
1309
1310 } while(T1.CompareSignedValues(T2) > 0 );

1319 {

1321 if(!T1.LeftShift(XMPINTEGER_BITSINLIMB * (i - t - 1))) return false;
1322

1341
1342 if(xmpintegerR->CompareSignedValues(0)==0) xmpintegerR->SetSign(1);

1346 Y.End();

1349 T2.End();

1429bool XMPINTEGER::Module(XLIMB* r, XMPINTEGER* xmpintegerA, int integerb)
1430{

1432 XDWORD i;
1433
1434 if(integerb == 0) return false;

1436
1437 // handle trivial cases
1438
1439 if(integerb == 1 )
1440 {
1441 (*r) = 0;

1443 }
1444

1463 }

1466

1468
1469 *r = y;
1470
1471 return true;

1589bool XMPINTEGER::SlidingWindowExponentiation(XMPINTEGER* A, XMPINTEGER* E, XMPINTEGER* N, XMPINTEGER* _RR)
1590{
1591 XDWORD wbits, wsize, one = 1;
1592 XDWORD i, j, nblimbs;
1593 XDWORD bufsize, nbits;
1594 XLIMB ei, mm, state;
1595 XMPINTEGER RR;
1596 XMPINTEGER T;
1597 XMPINTEGER W[2 << XMPINTEGER_WINDOWSIZE];
1598 XMPINTEGER Apos;
1599 int neg;
1600
1601 if(N->CompareSignedValues(0) < 0 || ( N->GetLimbs()[0] & 1 ) == 0 ) return false;
1602 if(E->CompareSignedValues(0) < 0) return false;
1603
1604 //Init temps and window size
1605
1606 MontgomeryIni( &mm, N );
1607 RR.Ini();
1608 T.Ini();
1609 Apos.Ini();
1610
1611 memset((XBYTE*)W, 0, sizeof(W));
1612
1613 i = E->GetMSB();
1614
1615 wsize = ( i > 671 ) ? 6 : ( i > 239 ) ? 5 :
1616 ( i > 79 ) ? 4 : ( i > 23 ) ? 3 : 1;
1617
1618 if(wsize > XMPINTEGER_WINDOWSIZE) wsize = XMPINTEGER_WINDOWSIZE;
1619
1620 j = N->GetNLimbs() + 1;
1621
1622 if(!Grow(j)) return false;
1623 if(!W[1].Grow(j)) return false;
1624 if(!T.Grow(j*2)) return false;
1625
1626 // Compensate for negative A (and correct at the end)
1627
1628 neg = ( A->GetSign() == -1 );
1629 if(neg)
1630 {
1631 if(!Apos.CopyFrom(A)) return false;
1632 Apos.SetSign(1);
1633 A = &Apos;
1634 }
1635
1636 // If 1st call, pre-compute R^2 mod N
1637
1638 if( _RR == NULL || _RR->GetLimbs() == NULL )
1639 {
1640 if(!RR.LeftSet(1)) return false;
1641 if(!RR.LeftShift(N->GetNLimbs() * 2 * XMPINTEGER_BITSINLIMB)) return false;
1642 if(!RR.Module(&RR, &RR, N)) return false;
1643
1644 _RR->CopyFrom(&RR);
1645
1646 if(_RR != NULL) _RR->CopyFrom(&RR);
1647
1648 } else RR.CopyFrom(_RR);
1649
1650 // W[1] = A * R^2 * R^-1 mod N = A * R mod N
1651
1652 if(A->CompareSignedValues((*N)) >= 0)
1653 {
1654 if(!Module( &W[1], A, N )) return false;
1655 }
1656 else
1657 {
1658 if(!W[1].CopyFrom(A)) return false;
1659 }
1660
1661 MontgomeryMultiplication(&W[1], &RR, N, mm, &T);
1662
1663 // X = R^2 * R^-1 mod N = R mod N
1664
1665 if(!CopyFrom(&RR)) return false;
1666 MontgomeryReduction(this, N, mm, &T);
1667
1668 if(wsize > 1)
1669 {
1670 // W[1 << (wsize - 1)] = W[1] ^ (wsize - 1)
1671
1672 j = one << (wsize - 1);
1673
1674 if(!W[j].Grow(N->GetNLimbs() + 1)) return false;
1675
1676 if(!W[j].CopyFrom(&W[1])) return false;
1677
1678 for(i = 0; i < wsize - 1; i++)
1679 {
1680 MontgomeryMultiplication( &W[j], &W[j], N, mm, &T);
1681 }
1682
1683 // W[i] = W[i - 1] * W[1]
1684
1685 for(i = j + 1; i < (one << wsize); i++)
1686 {
1687 if(!W[i].Grow(N->GetNLimbs() + 1)) return false;
1688 if(!W[i].CopyFrom(&W[i-1])) return false;
1689
1690 MontgomeryMultiplication( &W[i], &W[1], N, mm, &T);
1691 }
1692 }
1693
1694 nblimbs = E->GetNLimbs();
1695 bufsize = 0;
1696 nbits = 0;
1697 wbits = 0;
1698 state = 0;
1699
1700 while(1)
1701 {
1702 if(bufsize == 0)
1703 {
1704 if(nblimbs == 0) break;
1705 nblimbs--;
1706 bufsize = sizeof(XLIMB ) << 3;
1707 }
1708
1709 bufsize--;
1710
1711 ei = (E->GetLimbs()[nblimbs] >> bufsize) & 1;
1712
1713 // skip leading 0s
1714 if(ei == 0 && state == 0) continue;
1715 if(ei == 0 && state == 1)
1716 {
1717 // out of window, square X
1718 MontgomeryMultiplication(this, this, N, mm, &T );
1719 continue;
1720 }
1721
1722 // add ei to current window
1723 state = 2;
1724
1725 nbits++;
1726 wbits |= (ei << (wsize - nbits));
1727
1728 if(nbits == wsize)
1729 {
1730 // X = X^wsize R^-1 mod N
1731 for( i = 0; i < wsize; i++ )
1732 {
1733 MontgomeryMultiplication(this, this, N, mm, &T);
1734 }
1735
1736 // X = X * W[wbits] R^-1 mod N
1737 MontgomeryMultiplication(this, &W[wbits], N, mm, &T);
1738
1739 state--;
1740 nbits = 0;
1741 wbits = 0;
1742 }
1743 }
1744
1745 // process the remaining bits
1746 for(i=0; i<nbits; i++)
1747 {
1748 MontgomeryMultiplication(this, this, N, mm, &T);
1749 wbits <<= 1;
1750 if((wbits & (one << wsize)) != 0) MontgomeryMultiplication(this, &W[1], N, mm, &T);
1751 }
1752
1753 // X = A^E * R * R^-1 mod N = A^E mod N
1754 MontgomeryReduction(this, N, mm, &T);
1755
1756 if(neg)
1757 {
1758 SetSign(-1);
1759 if(!AdditionSigned(N, this)) return false;
1760 }
1761
1762 for(i = (one << (wsize - 1)); i < (one << wsize); i++)
1763 {
1764 W[i].End();
1765 }
1766
1767 W[1].End();
1768 T.End();
1769 Apos.End();
1770
1771 if(_RR == NULL) RR.End();
1772
1773 return true;
1774}

2046 T.Ini();
2047 A.Ini();
2048 RR.Ini();

2050 // W = |X| - 1
2051 // R = W >> lsb( W )
2052
2053 if(!W.SubtractionSigned(this, 1)) return false;
2054

2058 if(!R.RightShift(s)) return false;

2062 // HAC, table 4.4
2063

2065 ( i >= 650 ) ? 4 : ( i >= 350 ) ? 8 :
2066 ( i >= 250 ) ? 12 : ( i >= 150 ) ? 18 : 27 );

2069 {
