139bool DIOSTREAMTCPIPSERVER::WaitToConnected(int timeout)
140{
141 if(!xtimerconnection)
142 {
143 return false;
144 }
145
146 DIOSTREAMTCPIPCONFIG* config = (DIOSTREAMTCPIPCONFIG*)GetConfig();
147 bool status = false;
148
149 if(!config)
150 {
151 return false;
152 }
153
154 xtimerconnection->Reset();
155
156 while(1)
157 {
158 if(config->GetHandleMultiServer() > 0)

160 status = true;
161 break;
162 }
