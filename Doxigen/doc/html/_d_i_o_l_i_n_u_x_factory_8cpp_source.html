237DIOSTREAM* DIOLINUXFACTORY::CreateStreamIO(DIOSTREAMCONFIG* config)
238{
239 if(!config)
240 {
241 return NULL;
242 }
243
244 DIOSTREAM* _class=NULL;
245
246
247 if(!config->IsTLS())
248 {
249 switch(config->GetType())
250 {
251 case DIOSTREAMTYPE_UNKNOWN : return NULL;
252
253 #ifdef DIO_STREAMUART_ACTIVE
254 case DIOSTREAMTYPE_UART : _class = GEN_NEW DIOLINUXSTREAMUART();
255 break;
256 #endif
257
258 #ifdef DIO_STREAMUSB_ACTIVE
259 case DIOSTREAMTYPE_USB : _class = GEN_NEW DIOLINUXSTREAMUSB();
260 break;
261 #endif
262
263 #ifdef DIO_STREAMICMP_ACTIVE
264 case DIOSTREAMTYPE_ICMP : _class = GEN_NEW DIOLINUXSTREAMICMP();
265 break;
266 #endif
267
268 #ifdef DIO_STREAMUDP_ACTIVE
269 case DIOSTREAMTYPE_UDP : _class = GEN_NEW DIOLINUXSTREAMUDP();
270 break;
271 #endif
272
273 #ifdef DIO_STREAMTCPIP_ACTIVE
274 case DIOSTREAMTYPE_TCPIP : if(config->GetMode() == DIOSTREAMMODE_SERVERMULTISOCKET)
275 {
276 _class = GEN_NEW DIOLINUXSTREAMTCPIPSERVER();
277 }
278 else
279 {
280 _class = GEN_NEW DIOLINUXSTREAMTCPIP();
281 }
282 break;
283 #endif
284
285 #if (defined(DIO_STREAMBLUETOOTH_ACTIVE) || defined(DIO_STREAMBLUETOOTHLE_ACTIVE))
286 case DIOSTREAMTYPE_BLUETOOTH : _class = GEN_NEW DIOLINUXSTREAMBLUETOOTH(); break;
287 #endif
288
289 #ifdef DIO_STREAMSPI_ACTIVE
290 case DIOSTREAMTYPE_SPI : _class = GEN_NEW DIOLINUXSTREAMSPI(); break;
291 #endif
292
293 #ifdef DIO_STREAMI2C_ACTIVE
294 case DIOSTREAMTYPE_I2C : _class = GEN_NEW DIOLINUXSTREAMI2C(); break;
295 #endif
296 default : break;
297 }
298 }
299 else
300 {
301 switch(config->GetType())
302 {
303 case DIOSTREAMTYPE_UNKNOWN : return NULL;
304
305 #if defined(DIO_STREAMTCPIP_ACTIVE) && defined(DIO_STREAMTLS_ACTIVE)
306 case DIOSTREAMTYPE_TCPIP : if(config->GetMode() == DIOSTREAMMODE_SERVERMULTISOCKET)
307 {
308 //_class = GEN_NEW DIOWINDOWSSTREAMTCPIPSERVER();
309 }
310 else
311 {
312 _class = GEN_NEW DIOSTREAMTLS<DIOLINUXSTREAMTCPIP>();
313 }
314 break;
315 #endif
316
317 default : return NULL;
318 }
319
320 }
321
322
323 if(_class)
324 {
325 _class->SetType (config->GetType());
326 _class->SetConfig (config);
327 }
328
329 return _class;
330}
