615void XWINDOWSCONSOLE::SendEnterKey()
616{
617 #ifndef APPFLOW_GRAPHICS_NOTCONSOLE_ACTIVE
618
619 INPUT ip;
620 // Set up a generic keyboard event.
621 ip.type = INPUT_KEYBOARD;
622 ip.ki.wScan = 0; // hardware scan code for key
623 ip.ki.time = 0;
624 ip.ki.dwExtraInfo = 0;
625
626 // Send the "Enter" key
627 ip.ki.wVk = 0x0D; // virtual-key code for the "Enter" key
628 ip.ki.dwFlags = 0; // 0 for key press
629 SendInput(1, &ip, sizeof(INPUT));
630
631 // Release the "Enter" key
632 ip.ki.dwFlags = KEYEVENTF_KEYUP; // KEYEVENTF_KEYUP for key release
633 SendInput(1, &ip, sizeof(INPUT));
634
635 #endif
636}
