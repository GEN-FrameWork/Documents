210bool XWINDOWSPROCESSMANAGER::Application_Execute(XCHAR* applicationpath, XCHAR* params, XBUFFER* in, XBUFFER* out, int* returncode)
211{
212 if(!applicationpath)
213 {
214 return false;
215 }
216
217 if(!applicationpath[0])
218 {
219 return false;
220 }
221
222 if(returncode)
223 {
224 (*returncode) = 0;
225 }
226
227 //#define OUTBUF_SIZE (10*1024*1024)
228
229 #define OUTBUF_SIZE (4*1024)
230
231 PROCESS_INFORMATION processinfo;
232 STARTUPINFOW startupinfo;
233 SECURITY_ATTRIBUTES saattr;
234 XBYTE* outbuftmp = NULL;
235 DWORD bytes_read;
236 DWORD exitcode;
237 HANDLE stdhandle_out_read = NULL;
238 HANDLE stdhandle_out_write = NULL;
239 HANDLE stdhandle_in_read = NULL;
240 HANDLE stdhandle_in_write = NULL;
241 XSTRING command;
242 bool status = false;
243
244 outbuftmp = GEN_NEW XBYTE[OUTBUF_SIZE];
245
246 if(outbuftmp)
247 {
248 command.Format(__L("\"%s\""), applicationpath);
249
250 if(params)
251 {
252 command.AddFormat(__L(" %s"), params);
253 }
254
255 memset(&saattr, 0, sizeof(saattr));
256 saattr.nLength = sizeof(SECURITY_ATTRIBUTES);
257 saattr.bInheritHandle = TRUE;
258 saattr.lpSecurityDescriptor = NULL;
259
260 // Create a pipe for the child process's STDOUT.
261 if(!CreatePipe(&stdhandle_out_read, &stdhandle_out_write, &saattr, 0)) return false;
262 // Ensure the read handle to the pipe for STDOUT is not inherited.
263 if(!SetHandleInformation(stdhandle_out_read, HANDLE_FLAG_INHERIT, 0)) return false;
264
265 // Create a pipe for the child process's STDIN.
266 if(!CreatePipe(&stdhandle_in_read, &stdhandle_in_write, &saattr, 0)) return false;
267 // Ensure the read handle to the pipe for STDIN is not inherited.
268 if(!SetHandleInformation(stdhandle_in_write, HANDLE_FLAG_INHERIT, 0)) return false;
269
270 ZeroMemory(&processinfo, sizeof(PROCESS_INFORMATION));
271
272 ZeroMemory(&startupinfo, sizeof(STARTUPINFO));
273 startupinfo.cb = sizeof(startupinfo);
274 startupinfo.hStdError = stdhandle_out_write;
275 startupinfo.hStdOutput = stdhandle_out_write;
276 startupinfo.hStdInput = stdhandle_in_read; // GetStdHandle(STD_INPUT_HANDLE);
277 startupinfo.dwFlags |= STARTF_USESTDHANDLES;
278
279 if(CreateProcessW(NULL, (LPWSTR)command.Get(), NULL, NULL, TRUE, CREATE_NO_WINDOW, NULL, NULL, &startupinfo, &processinfo))
280 {
281 CloseHandle(stdhandle_out_write);
282
283 if(in)
284 {
285 if(in->GetSize())
286 {
287 DWORD bytes_write;
288
289 WriteFile(stdhandle_in_write, in->Get(), in->GetSize(), &bytes_write, NULL);
290
291 FlushFileBuffers(stdhandle_in_write);
292 }
293 }
294
295 CloseHandle(stdhandle_in_write);
296
297 if(out)
298 {
299 memset(outbuftmp, 0, OUTBUF_SIZE);
300
301 for(;;)
302 {
303 if(!ReadFile(stdhandle_out_read, outbuftmp, OUTBUF_SIZE, &bytes_read, NULL))
304 {
305 break;
306 }
307
308 if(bytes_read > 0)
309 {
310 out->Add(outbuftmp, bytes_read);
311 }
312 }
313 }
314
315
316 if(WaitForSingleObject(processinfo.hProcess, (out?INFINITE:500)) == WAIT_OBJECT_0)
317 {
318 if(out)
319 {
320 if(GetExitCodeProcess(processinfo.hProcess, &exitcode))
321 {
322 if(returncode) (*returncode) = exitcode;
323 }
324
325 CloseHandle(processinfo.hProcess);
326 CloseHandle(processinfo.hThread);
327
328 }
329
330 status = true;
331 }
332 }
333 }
334
335 if(outbuftmp)
336 {
337 GEN_DELETE_ARRAY outbuftmp;
338 }
339
340 return status;
341}

730BOOL XWINDOWSPROCESSMANAGER::EnablePrivilege(LPCWSTR privname)
731{
732 HANDLE hToken = NULL;

734 {
735 return FALSE;
736 }
737
738 LUID luid;
739 if(!LookupPrivilegeValueW(NULL, privname, &luid))
740 {
741 CloseHandle(hToken);
742 return FALSE;
743 }
744
745 TOKEN_PRIVILEGES tp;
746 tp.PrivilegeCount = 1;
747 tp.Privileges[0].Luid = luid;
748 tp.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
749
750 if(!AdjustTokenPrivileges(hToken, FALSE, &tp, sizeof(tp), NULL, NULL))
751 {
752 CloseHandle(hToken);
753 return FALSE;
754 }
755
756 DWORD err = GetLastError();
757 CloseHandle(hToken);
758
759 return (err == ERROR_SUCCESS);

920
921 SIZE_T pos = 2 + exeLen;
922 if(argsLen)
923 {
924 buf[pos - 0] = L' ';
925 CopyMemory(buf + pos, args, argsLen * sizeof(WCHAR));
926 pos += argsLen + 1; // +1 por el espacio ya puesto
927 }
928
