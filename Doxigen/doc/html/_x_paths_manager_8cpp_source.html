177bool XPATHSMANAGER::GetPathOfSection(XPATHSMANAGERSECTIONTYPE sectiontype, XPATH& xpath, bool addroot)
178{
179 xpath.Empty();
180
181 if(addroot && (sectiontype!=XPATHSMANAGERSECTIONTYPE_ROOT))
182 {
183 for(XDWORD c=0; c<pathsections.GetSize(); c++)
184 {
185 XPATHSMANAGERSECTION* pathsection = pathsections.Get(c);
186 if(pathsection)
187 {
188 if(pathsection->type == XPATHSMANAGERSECTIONTYPE_ROOT)

306bool XPATHSMANAGER::AddPathSection(XPATHSMANAGERSECTIONTYPE sectiontype, XSTRING& xpath)
307{
308 XPATHSMANAGERSECTION* xpathsection = GetPathSection(sectiontype);
309 if(!xpathsection)
310 {
311 xpathsection = GEN_NEW XPATHSMANAGERSECTION();
312 if(!xpathsection) return false;
313
314 xpathsection->type = sectiontype;

436bool XPATHSMANAGER::AdjustRootPathDefault(XCHAR* assetsdirname)
437{
438 XDIR* xdir;
439 XPATH xpathroot;
440 XPATH xpathtest;
441 bool founddir = false;
442 int indexfound = 0;
443 bool status = false;
444
445 GEN_XFACTORY_CREATE(xdir, Create_Dir())
446 if(!xdir) return false;
447
448 GetPathOfSection(XPATHSMANAGERSECTIONTYPE_ROOT, xpathroot);
449 xpathroot.Slash_Add();
450
451 do{ indexfound = xpathroot.Find(__L("/"), false, indexfound);
452 if(indexfound == XSTRING_NOTFOUND) indexfound = xpathroot.Find(__L("\\"), false, indexfound);
453
454 if(indexfound != XSTRING_NOTFOUND)
455 {
456 xpathroot.Copy(0, indexfound, xpathtest);
457
458 xpathtest.Slash_Add();
459 xpathtest.Add(assetsdirname);
460
461 founddir = xdir->Exist(xpathtest);
462 if(founddir)
463 {
464 AddPathSection(XPATHSMANAGERSECTIONTYPE_ROOT, xpathtest);
465 status = true;
466 break;
467 }
468
469 indexfound++;
470 }
471
472 } while(indexfound != XSTRING_NOTFOUND);
473
474 if(!founddir)
475 {
476 xpathtest = xpathroot;
477 xpathtest.Slash_Add();
478 xpathtest.Add(assetsdirname);
479
480 if(xdir->Make(xpathtest, false))
481 {
482 AddPathSection(XPATHSMANAGERSECTIONTYPE_ROOT, xpathtest);
483 status = true;
484 }
485 }
486
487 GEN_XFACTORY.Delete_Dir(xdir);
488
489 return status;
490}
