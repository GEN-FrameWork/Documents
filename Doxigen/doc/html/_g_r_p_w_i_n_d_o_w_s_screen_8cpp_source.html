585GRPBITMAP* GRPWINDOWSSCREEN::CaptureContent(GRPRECTINT* rect, void* handle_window)
586{
587 XBUFFER xbuffer;
588 HDC hdcsource = GetDC(handle_window?(HWND)handle_window:hwnd);
589 HDC hdcmemory = CreateCompatibleDC(hdcsource);
590 int x = 0;
591 int y = 0;
592 int cx = GetWidth();
593 int cy = GetHeight();
594 bool status = false;
595
596 if(!hdcsource)
597 {
598 return NULL;
599 }
600
601 if(!hdcmemory)
602 {
603 DeleteDC(hdcsource);
604 return NULL;
605 }
606
607 if(rect)
608 {
609 x = rect->x1;
610 y = rect->y1;
611 cx = (rect->x2 - rect->x1);
612 cy = (rect->y2 - rect->y1);
613 }
614
615 HBITMAP hbitmap = CreateCompatibleBitmap(hdcsource, cx, cy);
616 HBITMAP hbitmapold = (HBITMAP)SelectObject(hdcmemory, hbitmap);
617 BITMAP bitmap;
618
619 BitBlt(hdcmemory, 0, 0, cx, cy, hdcsource, x, y, SRCCOPY);
620 hbitmap = (HBITMAP)SelectObject(hdcmemory, hbitmapold);
621
622 DeleteDC(hdcsource);
623 DeleteDC(hdcmemory);
624
625 GetObject(hbitmap, sizeof(bitmap), (LPSTR)&bitmap);
626
627 HDC dcbitmap = CreateCompatibleDC(NULL);
628 SelectObject(dcbitmap, hbitmap);
629
630 BITMAPINFO bmpinfo;
631
632 bmpinfo.bmiHeader.biSize = sizeof(BITMAPINFOHEADER);
633 bmpinfo.bmiHeader.biWidth = bitmap.bmWidth;
634 bmpinfo.bmiHeader.biHeight = bitmap.bmHeight;
635 bmpinfo.bmiHeader.biPlanes = bitmap.bmPlanes;
636 bmpinfo.bmiHeader.biBitCount = bitmap.bmBitsPixel;
637 bmpinfo.bmiHeader.biCompression = BI_RGB;
638 bmpinfo.bmiHeader.biSizeImage = 0;
639 bmpinfo.bmiHeader.biClrImportant = 0;
640 bmpinfo.bmiHeader.biClrUsed = 0;
641
642 GRPBITMAP* grpbitmap = GRPFACTORY::GetInstance().CreateBitmap(bitmap.bmWidth, bitmap.bmHeight, GRPPROPERTYMODE_32_BGRA_8888);
643 if(grpbitmap)
644 {
645 status = GetDIBits(dcbitmap, hbitmap, 0, bitmap.bmHeight, grpbitmap->GetBuffer(), &bmpinfo, DIB_RGB_COLORS)?true:false;
646
647 XDWORD* buffer = (XDWORD*)grpbitmap->GetBuffer();
648 for(XDWORD c=0; c<(grpbitmap->GetBufferSize()/sizeof(XDWORD)); c++)
649 {
650 (*buffer) |= 0xFF000000;
651 buffer++;
652 }
653 }
654
655 if(!status)
656 {
657 GEN_DELETE grpbitmap;
658 grpbitmap = NULL;
659 }
660
661 return grpbitmap;
