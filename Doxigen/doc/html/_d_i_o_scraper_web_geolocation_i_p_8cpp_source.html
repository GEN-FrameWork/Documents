485bool DIOSCRAPERWEBGEOLOCATIONIP::Get(DIOIP& IP, DIOGEOLOCATIONIP_RESULT& geolocationIPresult, int timeoutforurl, XSTRING* localIP, bool usecache)
486{
487 bool status = false;
488
489 if(xmutexdo) xmutexdo->Lock();
490
491 this->IP.Set(IP.Get());
492
493 if(usecache)
494 {
495 XSTRING IPstring;
496
497 IP.GetXString(IPstring);
498
499 DIOGEOLOCATIONIP_RESULT* geoIPresult = (DIOGEOLOCATIONIP_RESULT*)cache->Get(IPstring);
500 if(geoIPresult)
501 {
502 geolocationIPresult.CopyFrom(geoIPresult);
503
504 if(xmutexdo) xmutexdo->UnLock();
505 return true;
506 }
507 }
508
509 if(Load(DIOSCRAPERWEBGEOLOCATIONIP_NAMEFILE))
510 {
511 if(Do(DIOSCRAPERWEBGEOLOCATIONIP_NAMESERVICE, timeoutforurl, localIP))
512 {
513 XSTRING country;
514 XSTRING state;
515 XSTRING city;
516 XSTRING ISP;
517 XSTRING organization;
518 double latitude = 0;
519 double longitude = 0;
520 XSTRING string;
521
522 country = GetValue(__L("COUNTRY"));
523 state = GetValue(__L("STATE"));
524 city = GetValue(__L("CITY"));
525 ISP = GetValue(__L("ISP"));
526 organization = GetValue(__L("ORGANIZATION"));
527
528 string = GetValue(__L("LATITUDE"));
529 if(!string.IsEmpty()) string.UnFormat(__L("%f"), &latitude);
530
531 string = GetValue(__L("LONGITUDE"));
532 if(!string.IsEmpty()) string.UnFormat(__L("%f"), &longitude);
533
534 geolocationIPresult.Set(country, state, city, ISP, organization);
535 geolocationIPresult.Set((float)latitude, (float)longitude);
536
537 if(!geolocationIPresult.IsEmpty())
538 {
539 if(usecache)
540 {
541 DIOGEOLOCATIONIP_RESULT* geoIPresult = GEN_NEW DIOGEOLOCATIONIP_RESULT();
542 if(geoIPresult)
543 {
544 XSTRING IPstring;
545
546 IP.GetXString(IPstring);
547
548 geoIPresult->CopyFrom(&geolocationIPresult);
549 cache->Add(IPstring, geoIPresult);
550 }
551 }
552
553 status = true;
554 }
555 }
556 }
557
558 if(xmutexdo) xmutexdo->UnLock();
559
560 return status;
561}
