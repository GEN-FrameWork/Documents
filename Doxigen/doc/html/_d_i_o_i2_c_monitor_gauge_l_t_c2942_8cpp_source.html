105float DIOI2CMONITORGAUGELTC2942::GetVoltage (void)
106{
107 if(!diostream) return false;
108
109 float voltage_return = 0;
110 XBYTE buffer[2] = { 0, 0 };
111
112 if(!ReadRegister(DIOI2CMONITORGAUGELTC2942_REG_VOL_H, buffer, 2)) return 0;
113
114 XDWORD read_value = buffer[1] | (buffer[0] << 8);
115
116 if(read_value == DIOI2CMONITORGAUGELTC2942_VOLTAGE_READ_ERROR)
117 {
118 return 0;
119 }
120
121 voltage_return = (float)read_value;
122
123 return ((voltage_return* 6000) / 65535)/1000;
124}

138bool DIOI2CMONITORGAUGELTC2942::SetAC(XWORD ac)
139{
140 XBYTE ctrl;
141
142 // Backup control register
143 if(!ReadRegister(DIOI2CMONITORGAUGELTC2942_REG_CONTROL,ctrl)) return false;
144
145 // Shutdown analog section before programming GEN_NEW AC value
146 if(!WriteRegister(DIOI2CMONITORGAUGELTC2942_REG_CONTROL, ctrl | DIOI2CMONITORGAUGELTC2942_CTL_SHUTDOWN)) return false;
147
148 // Program GEN_NEW AC value
149 if(!WriteRegister(DIOI2CMONITORGAUGELTC2942_REG_AC_H, ac >> 8)!=0) return false;
150
151 if(!WriteRegister(DIOI2CMONITORGAUGELTC2942_REG_AC_L, ac & 0xFF)!=0) return false;
152
153 // Restore control register
154 if(!WriteRegister(DIOI2CMONITORGAUGELTC2942_REG_CONTROL, ctrl)!=0) return false;
155
156 return true;
157}

250bool DIOI2CMONITORGAUGELTC2942::Initialize()
251{
252 XBYTE ctrl;
253
254 if(!ReadRegister(DIOI2CMONITORGAUGELTC2942_REG_CONTROL, ctrl)) return false;
255
256 // Enable auto measurement of battery voltage and temperature
257 ctrl &= DIOI2CMONITORGAUGELTC2942_CTL_ADC_MSK; // clear ADC mode bits
258 ctrl |= DIOI2CMONITORGAUGELTC2942_ADC_AUTO; // configure mode
259
260 // Enable analog section of the chip
261 ctrl &= ~DIOI2CMONITORGAUGELTC2942_CTL_SHUTDOWN;
262
263 // Set prescaler M value
264 //
265 // 0.085mAh 50momh M
266 // qLSB = ----------------------- =
267 // RSENSE 128
268
269 // 0.085mAh 50momh 16
270 // qLSB = ----------------------- = 0,0133mAh -> MaxValue: 870,38mAh (not valid)
271 // 40momh 128
272
273 // 0.085mAh 50momh 32
274 // qLSB = ----------------------- = 0,0265625mAh -> MaxValue: 1.740,7mAh valid
275 // 40momh 128
276
277 // M=32 for 1050mA battery AC max value 39.530
278
279 ctrl &= DIOI2CMONITORGAUGELTC2942_CTL_PSCM_MSK; // clear prescaler M bits
280 ctrl |= DIOI2CMONITORGAUGELTC2942_PSCM_32; // configure GEN_NEW value
281
282 // Disable AL/CC pin
283 ctrl &= DIOI2CMONITORGAUGELTC2942_CTL_ALCC_MSK; // clear AL/CC bits
284 ctrl |= DIOI2CMONITORGAUGELTC2942_ALCC_DISABLED; // configure GEN_NEW value
285
286 // Write control register
287 if(!WriteRegister(DIOI2CMONITORGAUGELTC2942_REG_CONTROL, ctrl))
288 {
289 return false;
290 }
291
292 // Checks if there was power off and sets de acumulator in middle value
293 XWORD currentAC = GetAC();
294 if(currentAC >= 0x7FF0) SetAC(DIOI2CMONITORGAUGELTC2942_AC_MAXVALUE/2);
295
296 return true;
297}

331bool DIOI2CMONITORGAUGELTC2942::ReadRegister(XBYTE offset, XBYTE* data, XDWORD sizedata)
332{
333 if(!diostream) return false;
334
335 bool status;
336
337 status = diostream->Write(&offset, 1)?true:false;
338 if(status) status = diostream->WaitToFlushOutXBuffer(timeout);
339
340 if(!status) return false;
341
342 status = diostream->WaitToFilledReadingBuffer(sizedata, timeout);
343 if(status) status = (diostream->Read(data, sizedata) == sizedata)?true:false;
344
345 return status;
346}
