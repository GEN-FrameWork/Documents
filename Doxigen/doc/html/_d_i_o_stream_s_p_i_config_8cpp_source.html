140bool DIOSTREAMSPICONFIG::SetLocalDeviceNameByPort()
141{
142 int revision = 0;
143
144 switch(GEN_XSYSTEM.GetTypeHardware(&revision))
145 {
146 case XSYSTEM_HARDWARETYPE_PC :
147 #ifdef LINUX
148 localdevicename.Format(__L("/dev/spi%d"), port-1);
149 #endif
150
151 #ifdef WINDOWS
152 localdevicename.Format(__L("spi%d"), port-1);
153 #endif
154 break;
155
156 case XSYSTEM_HARDWARETYPE_RASPBERRYPI : localdevicename.Format(__L("/dev/spidev%d.%d"), port-1, chipselect);
157 break;
158
159 default : break;
160 }
161
162 return true;
163}
