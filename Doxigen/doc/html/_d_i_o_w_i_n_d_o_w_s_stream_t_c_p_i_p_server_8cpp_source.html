252SOCKET DIOWINDOWSSTREAMTCPIPSERVER::Accept(SOCKET handlesocket, void* addr, void* addrlen, XDWORD usec)
253{
254 fd_set fds;
255 struct timeval time_out;
256 SOCKET status;
257
258 if(handlesocket==INVALID_SOCKET)
259 {
260 return INVALID_SOCKET;
261 }
262
263 FD_ZERO(&fds);
264 FD_SET((unsigned int)handlesocket,&fds);
265
266 time_out.tv_sec = 0;
267 time_out.tv_usec = usec;
268
269 status = (SOCKET)select((int)(handlesocket)+1, &fds, NULL, NULL, &time_out);
270 if(status>0)
271 {
272 status = accept(handlesocket,(SOCKADDR*)addr, (int*)addrlen);
273 }
274 else
275 {
276 status = INVALID_SOCKET;
277 }
278
279 return status;
280}

294int DIOWINDOWSSTREAMTCPIPSERVER::IsReadyConnect(SOCKET handlesocket)
295{
296 if(handlesocket==INVALID_SOCKET)
297 {
298 return -1;
299 }
300
301 struct timeval tv;
302 int rc;
303 fd_set fdr;
304 fd_set fdw;
305 fd_set fds;
306
307 FD_ZERO(&fdr);
308 FD_ZERO(&fdw);
309 FD_ZERO(&fds);
310
311 FD_SET((unsigned int)handlesocket, &fdr);
312 FD_SET((unsigned int)handlesocket, &fdw);
313 FD_SET((unsigned int)handlesocket, &fds);
314
315 tv.tv_sec = 0;
316 tv.tv_usec = 100;
317
318 rc = select((int)(handlesocket)+1, &fdr, &fdw, &fds, &tv);
319 if(rc == SOCKET_ERROR)
320 {
321 return -1;
322 }
323
324 int status1 = FD_ISSET(handlesocket,&fdr) ? 1 : 0;
325 int status2 = FD_ISSET(handlesocket,&fdw) ? 1 : 0;
326 int status3 = FD_ISSET(handlesocket,&fds) ? 1 : 0;
327
328 if(status1 || status2 )
329 {
330 return 1;
331 }
332
333 if(status3)
334 {
335 return -1;
336 }
337
338 return 0;
339}

465bool DIOWINDOWSSTREAMTCPIPSERVER::GetHandleServer(DIOSTREAMTCPIP* diostream)
466{
467 SOCKET handleserver = (SOCKET)config->GetHandleMultiServer();
468 SOCKADDR_IN loc_addr;
469 SOCKADDR_IN rem_addr;
470 XSTRING IPstring;
471
472 // ------------------------------------------------------------------------------------------
473
474 memset(&loc_addr,0,sizeof(SOCKADDR_IN));
475 memset(&rem_addr,0,sizeof(SOCKADDR_IN));
476
477 if(!diostream)
478 {
479 return false;
480 }
481
482 // ------------------------------------------------------------------------------------------
483
484 if(handleserver == INVALID_SOCKET)
485 {
486 config->GetXMutexMultiServer()->Lock();
487
488 handleserver = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
489 if(handleserver == INVALID_SOCKET)
490 {
491 return false;
492 }
493
494 int opt = 1;
495 if(setsockopt(handleserver, SOL_SOCKET, SO_REUSEADDR, (char*)&opt, sizeof(opt)) == SOCKET_ERROR)
496 {
497 SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
498 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
499
500 config->GetXMutexMultiServer()->UnLock();
501 return false;
502 }
503
504 config->GetLocalIP()->GetXString(IPstring);
505
506 loc_addr.sin_family = AF_INET;
507 if(!config->GetLocalIP()->IsEmpty())
508 {
509 XBUFFER charstr;
510
511 IPstring.ConvertToASCII(charstr);
512
513 #ifndef BUILDER
514 inet_pton(loc_addr.sin_family, charstr.GetPtrChar(), &loc_addr.sin_addr.s_addr);
515 #else
516 loc_addr.sin_addr.s_addr = inet_addr(charstr.GetPtrChar());
517 #endif
518
519 } else loc_addr.sin_addr.s_addr = htonl(INADDR_ANY);
520
521 loc_addr.sin_port = htons(config->GetRemotePort());
522
523 if(bind(handleserver, (LPSOCKADDR)&loc_addr, sizeof(SOCKADDR_IN)) == SOCKET_ERROR)
524 {
525 int errortype = WSAGetLastError();
526 if(errortype == WSAEADDRINUSE)
527 {
528 SetLastDIOError(DIOSTREAMERROR_ADDRINUSE);
529 }
530 else SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
531
532 SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
533
534 config->GetXMutexMultiServer()->UnLock();
535 return false;
536 }
537
538 if(listen(handleserver, DIOSTREAMTCPIP_MAXLISTEN) == SOCKET_ERROR)
539 {
540 SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
541 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
542
543 config->GetXMutexMultiServer()->UnLock();
544 return false;
545 }
546
547 DWORD nonblock = 1;
548 ioctlsocket(handleserver, FIONBIO, &nonblock);
549
550 config->SetHandleMultiServer((int)handleserver);
551
552 config->GetXMutexMultiServer()->UnLock();
553 }
554
555 // ------------------------------------------------------------------------------------------
556
557 if(enumservers)
558 {
559 if(enumservers->GetXTimer())
560 {
561 if(enumservers->GetXTimer()->GetMeasureSeconds() >= DIOSTREAMTCPIP_TIMEOUTENUMSERVERSIGNAL)
562 {
563 enumservers->SendSignal();
564 enumservers->GetXTimer()->Reset();
565 }
566 }
567 }
568
569 // ------------------------------------------------------------------------------------------
570
571
572 XDWORD sizeaddr = sizeof(SOCKADDR_IN);
573 SOCKET handlesocket = Accept(handleserver, (SOCKADDR*)&rem_addr, &sizeaddr, 50*1000);
574 if(handlesocket != INVALID_SOCKET)
575 {
576 int opt = 1;
577 if(setsockopt(handlesocket, SOL_SOCKET, SO_KEEPALIVE, (char*)&opt, sizeof(opt)) == SOCKET_ERROR)
578 {
579 SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
580 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
581 return false;
582 }
583
584 DWORD nonblock = 1;
585 ioctlsocket(handlesocket, FIONBIO, &nonblock);
586
587 #ifndef BUILDER
588 char IPcharstring[_MAXSTR];
589 inet_ntop(AF_INET, &(rem_addr.sin_addr), IPcharstring, _MAXSTR);
590 clientIP.Set(IPcharstring);
591 #else
592 clientIP.Set(inet_ntoa(rem_addr.sin_addr));
593 #endif
594 }
595 else
596 {
597 int errortype = WSAGetLastError();
598 // consider what's better, check for individual error codes or just anything now WSAEWOULDBLOCK
599 if(errortype == WSAEADDRINUSE)
600 {
601 SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
602 SetLastDIOError(DIOSTREAMERROR_ADDRINUSE);
603 return false;
604 }
605 }
606
607 DIOWINDOWSSTREAMTCPIP* diowindowsstreamtcpip = (DIOWINDOWSSTREAMTCPIP*)diostream;
608 if(diowindowsstreamtcpip)
609 {
610 diowindowsstreamtcpip->SetHandleSocket(handlesocket);
611 }
612
613 // ------------------------------------------------------------------------------------------
614
615 return true;
616}

628void DIOWINDOWSSTREAMTCPIPSERVER::ThreadConnection(void* data)
629{
630 DIOWINDOWSSTREAMTCPIPSERVER* diostreamserver = (DIOWINDOWSSTREAMTCPIPSERVER*)data;
631 if(!diostreamserver)
632 {
633 return;
634 }
635
636 diostreamserver->DeleteAllStreamDisconnected();
637
638 for(XDWORD c=0; c<diostreamserver->GetMultiSocketStreams()->GetSize(); c++)
639 {
640 DIOWINDOWSSTREAMTCPIP* diostream = (DIOWINDOWSSTREAMTCPIP*)diostreamserver->GetMultiSocketStreams()->Get(c);
641 if(diostream)
642 {
643 if(diostream->GetEvent()==DIOWINDOWSTCPIPSERVERFSMEVENT_NONE) // No hay nuevos Eventos
644 {
645 switch(diostream->GetCurrentState())
646 {
647 case DIOWINDOWSTCPIPSERVERFSMSTATE_NONE : break;
648
649 case DIOWINDOWSTCPIPSERVERFSMSTATE_GETTINGCONNECTION : if(diostream->GetHandleSocket() == INVALID_SOCKET)
650 {
651 diostreamserver->GetHandleServer(diostream);
652 }
653 else
654 {
655 switch(diostreamserver->IsReadyConnect(diostream->GetHandleSocket()))
656 {
657 case -1: diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
658 break;
659
660 case 1: diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_CONNECTED);
661 diostream->SetStatus(DIOSTREAMSTATUS_CONNECTED);
662 break;
663
664 default: diostream->SetStatus(DIOSTREAMSTATUS_GETTINGCONNECTION);
665 break;
666 }
667 }
668 break;
669
670 case DIOWINDOWSTCPIPSERVERFSMSTATE_CONNECTED : break;
671
672 case DIOWINDOWSTCPIPSERVERFSMSTATE_WAITINGTOREAD : { struct timeval waitd;
673 fd_set read_flags;
674 fd_set write_flags;
675
676 if(diostream->GetHandleSocket() == INVALID_SOCKET)
677 {
678 diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
679 break;
680 }
681
682 waitd.tv_sec = 0;
683 waitd.tv_usec = 100;
684
685 FD_ZERO(&read_flags);
686 FD_ZERO(&write_flags);
687
688 if(!diostream->IsBlockRead()) FD_SET(diostream->GetHandleSocket(), &read_flags);
689 if(!diostream->IsBlockWrite())
690 {
691 if(diostream->outbuffer->GetSize()) FD_SET(diostream->GetHandleSocket(), &write_flags);
692 }
693
694 int error = select((int)(diostream->GetHandleSocket())+1, &read_flags, &write_flags, (fd_set*)NULL, &waitd);
695 if(error == SOCKET_ERROR) //Socket ready for reading
696 {
697 diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
698 break;
699 }
700
701 if(FD_ISSET(diostream->GetHandleSocket(), &read_flags)) //Socket ready for reading
702 {
703 XBYTE buffer[DIOSTREAM_MAXBUFFER];
704
705 FD_CLR(diostream->GetHandleSocket(), &read_flags);
706
707 memset(buffer, 0, DIOSTREAM_MAXBUFFER);
708
709 XDWORD size = recv(diostream->GetHandleSocket(), (char*)buffer, DIOSTREAM_MAXBUFFER, 0);
710 if(size==SOCKET_ERROR)
711 {
712 diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
713 break;
714 }
715
716 if(size)
717 {
718 diostream->inbuffer->Add(buffer,size);
719
720 diostream->AddNBytesRead(size);
721 diostreamserver->nbytesread += size;
722 }
723 else
724 {
725 diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
726 break;
727 }
728 }
729
730 if(FD_ISSET(diostream->GetHandleSocket(), &write_flags)) //Socket ready for writing
731 {
732 FD_CLR(diostream->GetHandleSocket(), &write_flags);
733
734 int esize = diostream->outbuffer->GetSize();
735 if(esize>DIOSTREAM_MAXBUFFER) esize = DIOSTREAM_MAXBUFFER;
736
737 if(esize)
738 {
739 diostream->outbuffer->SetBlocked(true);
740 XDWORD size = send(diostream->GetHandleSocket(),(char*)diostream->outbuffer->Get(), esize,0);
741 diostream->outbuffer->SetBlocked(false);
742 if(size==SOCKET_ERROR)
743 {
744 diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_DISCONNECTING);
745 break;
746 }
747
748 if(size)
749 {
750 diostream->outbuffer->Extract(NULL, 0 ,size);
751
752 diostream->AddNBytesWrite(size);
753 diostreamserver->nbyteswrite += size;
754 }
755 }
756 }
757 }
758 break;
759
760 case DIOWINDOWSTCPIPSERVERFSMSTATE_SENDINGDATA : break;
761
762 case DIOWINDOWSTCPIPSERVERFSMSTATE_DISCONNECTING : break;
763
764 }
765 }
766 else
767 {
768 if(diostream->GetEvent()<DIOWINDOWSTCPIPSERVER_LASTEVENT)
769 {
770 diostream->CheckTransition();
771
772 switch(diostream->GetCurrentState())
773 {
774 case DIOWINDOWSTCPIPSERVERFSMSTATE_NONE : break;
775
776 case DIOWINDOWSTCPIPSERVERFSMSTATE_GETTINGCONNECTION : if(diostreamserver->enumservers)
777 {
778 if(diostreamserver->enumservers->GetXTimer())
779 {
780 diostreamserver->enumservers->GetXTimer()->AddSeconds(DIOSTREAMTCPIP_TIMEOUTENUMSERVERSIGNAL);
781 }
782 }
783 break;
784
785 case DIOWINDOWSTCPIPSERVERFSMSTATE_CONNECTED : { DIOSTREAM_XEVENT xevent(diostream, DIOSTREAM_XEVENT_TYPE_CONNECTED);
786 xevent.SetDIOStream(diostream);
787 diostreamserver->PostEvent(&xevent);
788
789 diostream->SetEvent(DIOWINDOWSTCPIPSERVERFSMEVENT_WAITINGTOREAD);
790
791 diostreamserver->CreateStream();
792 }
793 break;
794
795 case DIOWINDOWSTCPIPSERVERFSMSTATE_WAITINGTOREAD : break;
796
797 case DIOWINDOWSTCPIPSERVERFSMSTATE_SENDINGDATA : break;
798
799 case DIOWINDOWSTCPIPSERVERFSMSTATE_DISCONNECTING : { DIOSTREAM_XEVENT xevent(diostream, DIOSTREAM_XEVENT_TYPE_DISCONNECTED);
800 xevent.SetDIOStream(diostream);
801 diostreamserver->PostEvent(&xevent);
802
803 diostream->SetStatus(DIOSTREAMSTATUS_DISCONNECTED);
804 }
805 break;
806
807 }
808 }
809 }
810 }
811 }
