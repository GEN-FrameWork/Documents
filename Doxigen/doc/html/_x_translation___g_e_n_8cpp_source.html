236bool XTRANSLATION_GEN::Sentence_Add(XDWORD ID, XDWORD codelanguage, XCHAR* sentence, XDWORD fixed)
237{
238 if(!ID) return false;
239 if(!codelanguage) return false;
240 if(!sentence) return false;
241
242 if(Sentence_Get(ID, codelanguage)) return false;
243
244 XTRANSLATION_GEN_SENTENCE* sentenceGEN = GEN_NEW XTRANSLATION_GEN_SENTENCE();
245 if(!sentenceGEN) return false;
246
247 int sizesentence = XSTRING::GetSize(sentence);
248
249 sentenceGEN->ID = ID;
250 sentenceGEN->codelanguage = codelanguage;
251 sentenceGEN->sentence = GEN_NEW XCHAR[sizesentence+1];
252 sentenceGEN->fixed = fixed;
253 if(!sentenceGEN->sentence)
254 {
255 GEN_DELETE sentenceGEN;
256 return false;
257 }
258
259 memset(sentenceGEN->sentence, 0, (sizesentence+1) * sizeof(XCHAR));
260 memcpy(sentenceGEN->sentence, sentence, sizesentence * sizeof(XCHAR));
261
262 sentences.Add(sentenceGEN);
263
264 return true;
265}

280XTRANSLATION_GEN_SENTENCE* XTRANSLATION_GEN::Sentence_Get(XDWORD ID, XDWORD codelanguage)
281{

289 }
290
291 return NULL;
