598bool XINTERPOLATOR::Update(XQWORD delta)
599{
600 switch(type)
601 {
602 case XINTERPOLATOR_TYPE_LIMIT : Interpolate(delta);
603
604 if(mode == XINTERPOLATOR_LIMITMODE_INCREMENTAL)
605 {
606 if(target<=current) current=target;
607 }
608 else
609 {
610 if(mode == XINTERPOLATOR_LIMITMODE_DECREMENTAL)
611 if(target>=current) current = target;
612 }
613
614 break;
615
616 case XINTERPOLATOR_TYPE_RELATIVE :
617 case XINTERPOLATOR_TYPE_ABSOLUTE : if(current > target) current = target;
618 if(current < 0) current = 0;
619 break;
620
621 case XINTERPOLATOR_TYPE_NONE : break;
622 }
623
624 return true;
625}

729{
730 float fdelta = (float) (delta/100.0f);
731 int itarget = (int) (target);
732 float result = (float) (current);
733 int margin = 0;
734
735 if(result == target) return false;

737 if(result < itarget)

739 if(delta!=0)
740 {
741 if(carry<0.0000001f) carry=0.0f;
742
743 float dif = (float)(itarget - result);

745
