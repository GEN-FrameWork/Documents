91bool DIOAIOBIMAINBOARD::Ini()
92{
93 for(int c=0; c<3; c++)
94 {
95 pcf8574[c] = GEN_NEW DIOI2CGPIOPCF8574();
96 if(!pcf8574[c]) return false;
97 }
98
99 if(!pcf8574[DIOAIOBIMAINBOARD_I2CDIGIGPIO_IN]->Ini(DIOAIOBIMAINBOARD_I2CDEVICENAME , DIOAIOBIMAINBOARD_I2CDIGIGPIO_ID_IN , DIOAIOBIMAINBOARD_DEFAULTTIMEOUT))
100 {
101 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! I2C GPIO IN not init!"));
102 return false;
103 }
104
105 if(!pcf8574[DIOAIOBIMAINBOARD_I2CDIGIGPIO_OUT]->Ini(DIOAIOBIMAINBOARD_I2CDEVICENAME , DIOAIOBIMAINBOARD_I2CDIGIGPIO_ID_OUT , DIOAIOBIMAINBOARD_DEFAULTTIMEOUT))
106 {
107 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! I2C GPIO OUT not init!"));
108 return false;
109 }
110
111 if(!pcf8574[DIOAIOBIMAINBOARD_I2CDIGIGPIO_RELAYS]->Ini(DIOAIOBIMAINBOARD_I2CDEVICENAME , DIOAIOBIMAINBOARD_I2CDIGIGPIO_ID_RELAYS , DIOAIOBIMAINBOARD_DEFAULTTIMEOUT))
112 {
113 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! I2C GPIO Relays not init!"));
114 return false;
115 }
116
117 for(int c=1; c<9; c++)
118 {
119 SetDigitalOutput(c, false);
120 }
121
122 eeprom = GEN_NEW DIOI2CEEPROM24XXX();
123 if(!eeprom) return false;
124
125 eeprom->SetType(DIOI2CEEPROM24XXXTYPE_16);
126
127 if(!eeprom->Ini(DIOAIOBIMAINBOARD_I2CDEVICENAME, DIOAIOBIMAINBOARD_I2CEEPROM, DIOAIOBIMAINBOARD_DEFAULTTIMEOUT))
128 {
129 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! I2C Serial EEPROM not init!"));
130 return false;
131 }
132
133
134 pca9685 = GEN_NEW DIOI2CPWMCONTROLERPCA9685();
135 if(!pca9685) return false;
136
137 if(!pca9685->Ini(DIOAIOBIMAINBOARD_I2CDEVICENAME, DIOAIOBIMAINBOARD_I2CPWM, DIOAIOBIMAINBOARD_DEFAULTTIMEOUT))
138 {
139 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! I2C PCA9685 (PWM) not init!"));
140 return false;
141 }
142
143
144 gpio = GEN_DIOFACTORY.CreateGPIO();
145 if(!gpio)
146 {
147 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! GPIO not created!"));
148 return false;
149 }
150
151 if(!gpio->Ini())
152 {
153 GEN_XLOG.AddEntry(XLOGLEVEL_ERROR, DIOAIOBIMAINBOARD_LOGSECTIONID, false, __L("AIOBI Main Board: ERROR! GPIO not init!"));
154 return false;
155 }
156
157 isini = true;
158
159 return isini;
160}

642bool DIOAIOBIMAINBOARD::ResetMicroControler()
643{
644 if(!SetModeHeaderPin(DIOAIOBIMAINBOARD_HEADER_SLOT, 9, false)) return false; // Reset Micro
645 if(!SetStatusHeaderPin(DIOAIOBIMAINBOARD_HEADER_SLOT, 9, true)) return false;
646
647 bool pinstatus = false;
648
649 if(SetStatusHeaderPin(DIOAIOBIMAINBOARD_HEADER_SLOT, 9, pinstatus))
650 {
651 GEN_XSLEEP.MilliSeconds(10);
652
653 pinstatus = !pinstatus;
654
655 return SetStatusHeaderPin(DIOAIOBIMAINBOARD_HEADER_SLOT, 9, pinstatus);
656 }
657
658 return false;
659}

671bool DIOAIOBIMAINBOARD::End()
672{
673 for(int c=0; c<3; c++)
674 {
675 if(pcf8574[c])
676 {
677 pcf8574[c]->End();
678
679 GEN_DELETE pcf8574[c];
680 pcf8574[c] = NULL;
681 }
682 }
683
684 if(eeprom)
685 {
686 eeprom->End();
687 GEN_DELETE eeprom;
688 eeprom = NULL;
689 }
690
691
692 if(pca9685)
693 {
694 pca9685->GetDIOStream()->WaitToFlushXBuffers(3);
695 pca9685->End();
696 GEN_DELETE pca9685;
697 pca9685 = NULL;
698 }
699
700 if(gpio)
701 {
702 SetLedA(false);
703 SetLedB(false);
704
705 gpio->End();
706 GEN_DIOFACTORY.DeleteGPIO(gpio);
707 gpio = NULL;
708 }
709
710 return true;
711}
