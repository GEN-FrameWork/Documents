135bool INPLINUXDEVICETOUCHSCREENDIRECT::Update()
136{
137 if((!created)||(!enabled)) return false;
138
139 struct input_event event;
140
141 int eventsize = sizeof(struct input_event);
142 int bytesread = 0;
143
144 //--------------- clean up old states
145
146 INPBUTTON* button = GetButton(INPBUTTON_ID_TOUCHSCREEN);
147 if(button)
148 {
149 button->SetPressed(false);
150
151 touchscr_x = -1;
152 touchscr_y = -1;
153
154 UpdateTouchScreenData();
155 }
156
157 //--------------- process
158
159 if(touchscr.GetSize() == 0)
160 {
161 OpenAllDevicesID();
162 }
163
164 for(XDWORD e=0; e<touchscr.GetSize(); e++)
165 {
166 while(true)
167 {
168 bytesread=0;
169
170 memset(&event, 0, sizeof(event));
171
172 errno = 0;
173 bytesread = read(touchscr.Get(e)->GetFileDescriptor(), &event, sizeof(event)/*eventsize*/);
174
175 // if device is diconnected we receive a ENODEV message (happens for example every time a touchscreen is poweredoff)
176
177 if(bytesread == -1 && errno == ENODEV)
178 {
179 // we lost the handle to the event file, need to reclaim it
180 touchscr.DeleteContents();
181 OpenAllDevicesID();
182 }
183
184 if(bytesread==-1 && errno!=EAGAIN) break;
185
186 if((bytesread < sizeof(struct input_event)) || (bytesread == -1)) break;
187
188 if(bytesread == sizeof(event)/*eventsize*/)
189 {
190 switch(event.type)
191 {
192 case EV_ABS : switch(event.code)
193 {
194 case ABS_X : { touchscr_x = event.value;
195 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[Touch] x: %d"), touchscr_x);
196
197 }
198 break;
199
200 case ABS_Y : { touchscr_y = grpscreen->GetHeight() - event.value;
201 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[Touch] y: %d"), touchscr_y);
202
203 INPBUTTON* button = GetButton(INPBUTTON_ID_TOUCHSCREEN);
204 if(button) button->SetPressed(true);
205 }
206 break;
207 }
208 break;
209
210 default : continue;
211 }
212
213 } else break;
214 }
215 }
216
217 UpdateTouchScreenData();
218
219 return true;
220}

273bool INPLINUXDEVICETOUCHSCREENDIRECT::OpenAllDevicesID()
274{
275 INPLINUXFACTORY* inpdevices = (INPLINUXFACTORY*)(&INPFACTORY::GetInstance());
276 XVECTOR<INPLINUXDEVICEID*> devicesID;
277
278 if(inpdevices) inpdevices->GetDeviceHandlers(INPDEVICE_TYPE_TOUCHSCREEN, devicesID);
279
280 for(XDWORD c=0; c<devicesID.GetSize(); c++)
281 {
282 INPLINUXDEVICEID* deviceID = devicesID.Get(c);
283 if(deviceID)
284 {
285 XSTRING event;
286 int filedescriptor = INPLINUXDEVICEID_INVALID;
287
288 event.Format(__L("/dev/input/event%d"), deviceID->GetEventIndex());
289
290 XBUFFER charstr;
291
292 event.ConvertToASCII(charstr);
293 filedescriptor = open(charstr.GetPtrChar(), O_RDONLY | O_NONBLOCK);
294
295 if(filedescriptor != -1)
296 {
297 char mouseOEMname[_MAXSTR];
298 XSTRING namehardware;
299
300 // we get the keyboard name
301 memset(mouseOEMname, 0, _MAXSTR);
302
303 ioctl(filedescriptor, EVIOCGNAME(_MAXSTR), mouseOEMname);
304 namehardware.Set(mouseOEMname);
305
306 deviceID->SetFileDescriptor(filedescriptor);
307 //deviceID->GetName()->Set(namehardware);
308
309 touchscr.Add(deviceID);
310 }
311 }
312 }
313
314 devicesID.DeleteAll();
315
316 return touchscr.GetSize()?true:false;
317}
