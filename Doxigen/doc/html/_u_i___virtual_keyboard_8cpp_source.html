334bool UI_VIRTUALKEYBOARD::SelectInput(UI_ELEMENT* key_select)
335{
336 XCHAR* literals[] = { __L("RETURN") , // 0
337 __L("CAPS") , // 1
338 __L("SYMBOLS") , // 2
339 __L("BACK") , // 3
340 __L("ARROWLEFT") , // 4
341 __L("ARROWRIGHT") , // 5
342 __L("SPACE") , // 6
343 };
344
345 if(!key_select) return false;
346
347 XSTRING elementID;
348 XSTRING keystr;
349
350 UI_ELEMENT_EDITTEXT* element_edittext = (UI_ELEMENT_EDITTEXT*)element_editable;
351 if(!element_edittext) return false;
352
353 elementID = UI_VIRTUALKEYBOARD_ELEMENTID;
354
355 int index = key_select->GetName()->Find(elementID, true);
356 if(index == XSTRING_NOTFOUND) return false;
357
358 index += elementID.GetSize();
359
360 key_select->GetName()->Copy(index, keystr);
361
362 int index_literal = -1;
363 for(int c=0; c<(sizeof(literals) / sizeof(XCHAR*)); c++)
364 {
365 if(!keystr.Compare(literals[c], true))
366 {
367 index_literal = c;
368 break;
369 }
370 }
371
372 switch(index_literal)
373 {
374 case 0 : //-------------------------------------------------------
375 // RETURN
376
377 {
378 if(contentchanged)
379 {
380 GEN_USERINTERFACE.SendEvent(UI_XEVENT_TYPE_OUTPUT_CHANGECONTENTS, element_editable);
381 contentchanged = false;
382 }
383
384 Show(false, NULL);
385 }
386 break;
387
388
389 case 1 : //-------------------------------------------------------
390 // CAPS
391
392 { bool _contentchanged = contentchanged;
393 XDWORD cursor_position = element_edittext->Cursor_GetPosition();
394
395 if(actualset == UI_VIRTUALKEYBOARD_SET_UPPERCASE)
396 actualset = UI_VIRTUALKEYBOARD_SET_LOWERCASE;
397 else actualset = UI_VIRTUALKEYBOARD_SET_UPPERCASE;
398
399 Show(true, element_editable);
400
401 contentchanged = _contentchanged;
402 element_edittext->Cursor_SetPosition(cursor_position);
403 }
404 break;
405
406 case 2 : //-------------------------------------------------------
407 // SYMBOLS
408
409 { bool _contentchanged = contentchanged;
410 XDWORD cursor_position = element_edittext->Cursor_GetPosition();
411
412 if(actualset == UI_VIRTUALKEYBOARD_SET_SYMBOLS)
413 actualset = UI_VIRTUALKEYBOARD_SET_UPPERCASE;
414 else actualset = UI_VIRTUALKEYBOARD_SET_SYMBOLS;
415
416 Show(true, element_editable);
417
418 contentchanged = _contentchanged;
419 element_edittext->Cursor_SetPosition(cursor_position);
420 }
421 break;
422
423 case 3 : //-------------------------------------------------------
424 // BACK
425
426 { if(element_editable->GetType() == UI_ELEMENT_TYPE_EDITTEXT)
427 {
428 XDWORD position = element_edittext->Cursor_GetPosition();
429 if(position)
430 {
431 element_edittext->GetText()->DeleteCharacters(position-1, 1);
432 element_edittext->Cursor_SetPosition(position-1);
433
434 contentchanged = true;
435 }
436 }
437 }
438 break;
439
440 case 4 : //-------------------------------------------------------
441 // ARROW RIGHT
442
443 { XDWORD position = element_edittext->Cursor_GetPosition();
444
445 if(position) position--;
446
447 element_edittext->Cursor_SetPosition(position);
448 }
449 break;
450
451 case 5 : //-------------------------------------------------------
452 // ARROW LEFT
453
454 {
455 XDWORD position = element_edittext->Cursor_GetPosition();
456
457 position++;
458 if(position > element_edittext->GetText()->GetSize()) position = element_edittext->GetText()->GetSize();
459
460 element_edittext->Cursor_SetPosition(position);
461 }
462 break;
463
464 case 6 : //-------------------------------------------------------
465 // SPACE
466
467 {
468 keystr = __L(" ");
469 }
470
471 default : //-------------------------------------------------------
472 // Other key
473
474 { if(element_editable->GetType() == UI_ELEMENT_TYPE_EDITTEXT)
475 {
476 XDWORD position = element_edittext->Cursor_GetPosition();
477 if(element_edittext->GetText()->GetSize() + 1 <= element_edittext->GetMaxSizeText())
478 {
479 element_edittext->GetText()->Insert(keystr, position);
480 element_edittext->Cursor_SetPosition(position+1);
481
482 contentchanged = true;
483 }
484 }
485 }
486 break;
487
488
489 }
490
491 return true;
492}

522UI_ANIMATION* UI_VIRTUALKEYBOARD::AddImageCache(XCHAR* name, XCHAR* resource)
523{
524 UI_ANIMATION* animation = NULL;
525
526 if(!skin) return NULL;
527
528 animation = GEN_UI_ANIMATIONS.Get(name);
529 if(animation) return animation;
530
531 animation = GEN_UI_ANIMATIONS.Get(resource);
532 if(animation) return animation;
533
534 animation = GEN_NEW UI_ANIMATION();
535 if(!animation) return NULL;
536
537 XSTRING resourcename;
538 bool status = false;
539
540 resourcename = resource;
541
542 switch(skin->GetDrawMode())
543 {
544 case UI_SKIN_DRAWMODE_UNKNOWN : break;
545
546 case UI_SKIN_DRAWMODE_CANVAS : { UI_SKINCANVAS* skincanvas = (UI_SKINCANVAS*)skin;
547 if(skincanvas) status = animation->LoadFromFile(resourcename, skincanvas->GetCanvas()->GetMode());
548 }
549 break;
550
551 case UI_SKIN_DRAWMODE_CONTEXT : break;
552 }
553
554 if(status)
555 {
556 animation->GetName()->Set(name);
557 animation->GetResource()->Set(resource);
558
559 GEN_UI_ANIMATIONS.Add(animation);
560
561 return animation;
562 }
563
564 GEN_DELETE animation;
565
566 return NULL;
567}
