714XTRACE_STATUS_MSG* XTRACE_STATUS_MSGS::StatusMsg_Add(XSTRING* line)

718 XSTRING ID;
719 XSTRING name;
720 XCHAR typechar;
721 XSTRING valuestr;
722 bool isnew = false;
723
724
725 int indexstatus = line->Find(XTRACE_IDMSGSTATUSCLEAR, false);

727 {
728 statusmsgs.DeleteContents();
729 statusmsgs.DeleteAll();

731 return NULL;
732 }

752 ID.AdjustSize();

754 valuestr.AdjustSize();

757 XTRACE_STATUS_MSG* status_msg = StatusMsg_Get(name.Get());
758 if(!status_msg)

760 status_msg = GEN_NEW XTRACE_STATUS_MSG();

769 if(!valuestr.Compare(__L("true"), true)) status_msg->Value_SetBoolean(true); else status_msg->Value_SetBoolean(false);
770 }

774 status_msg->Value_SetInteger(valuestr.ConvertToInt());
775 }

777

779 status_msg->Value_GetString()->Set(valuestr);
780 }
781 break;
