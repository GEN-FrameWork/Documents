254bool APPFLOWWEBSERVER::SendRequest(DIOWEBSERVER_CONNECTION* connection, DIOWEBHEADER_RESULT headerresult, XBUFFER* data, int timeout, XSTRING* addhead)
255{
256 DIOWEBSERVER_HEADER webserverheader;
257 XSTRING entitytag;
258 HASHMD5 hashmd5;
259 bool iscached = false;
260 bool status = false;
261
262 if(!connection) return false;
263 if(!connection->GetDIOStream()) return false;
264 if(!connection->GetDIOStream()->IsConnected()) return false;
265
266 if(data)
267 {
268 hashmd5.Do((*data));
269 hashmd5.GetResultString(entitytag);
270 iscached = connection->GetRequest()->GetIfNoneMatch()->Compare(entitytag)?false:true;
271 }
272
273 if(iscached)
274 {
275 webserverheader.Create(connection->GetRequest()->GetResource(), 1, 1, DIOWEBHEADER_RESULT_NOTMODIFIED, connection->GetRequest()->GetSize());
276 status = webserverheader.Write(connection->GetDIOStream(), 5);
277 }
278 else
279 {
280 XPATH resourceext;
281 int size = (data ?(int)data->GetSize():-1);
282
283 resourceext = connection->GetRequest()->GetResource()->Get();
284 resourceext.SetOnlyExt();
285
286 webserverheader.Create(connection->GetRequest()->GetResource(), 1, 1, headerresult, size, &entitytag, connection->GetServer()->IsPlayablePage(resourceext));
287 if(addhead) webserverheader.AddLine(addhead->Get());
288 status = webserverheader.Write(connection->GetDIOStream(), 5);
289 if(status)
290 {
291 if(size > 0) status = connection->Send((*data), timeout);
292 }
293 }
294
295 return status;
296}

391bool APPFLOWWEBSERVER::ResolveRequest(DIOWEBSERVER* server, DIOWEBSERVER_CONNECTION* connection, DIOWEBSERVER_REQUEST* request)
392{
393 if(!server) return false;
394 if(!connection) return false;
395 if(!request) return false;
396
397 XSTRING* resource = request->GetResource();
398 if(!resource) return false;
399
400 if(!cfg) return false;
401
402 DIOWEBSERVER_HEADER webserverheader;
403 DIOURL resourceconv;
404 int nparams = 0;
405 DIOWEBSERVER_QUERYSTRINGS querystring;
406 XSTRING extension;
407 bool ispagepluging = false;
408 bool status = false;
409
410 resourceconv.Set(resource->Get());
411 resourceconv.DecodeUnsafeChars();
412
413 if(isauthenticatedaccess)
414 {
415 if(!cfg->WebServer_GetPassword()->IsEmpty())
416 {
417 if(!request->GetLoginPassword(page_login, page_password))
418 {
419 SendRequest(connection, DIOWEBHEADER_RESULT_UNAUTHORIZED, NULL, 5);
420
421 return true;
422 }
423 else
424 {
425 bool ispasswordvalid;
426
427 ispasswordvalid = !page_password.Compare(cfg->WebServer_GetPassword()->Get(), true);
428 if(!ispasswordvalid)
429 {
430 XSTRING leyend;
431
432 leyend = __L("Invalid user or password!");
433 GenerateResponse_Error(connection, DIOWEBHEADER_RESULT_NOTFOUND, leyend);
434 }
435
436 XSTRING IPstring;
437 connection->GetDIOStream()->GetClientIP()->GetXString(IPstring);
438
439 APPFLOW_LOG_ENTRY(XLOGLEVEL_ERROR, DIOWEBSERVER_LOGSECTIONID, false, __L("Request from [%s] to the web server password [%s]: %s"), IPstring.Get(), cfg->WebServer_GetPassword()->Get(), (ispasswordvalid)? __L("Ok."): __L("INVALID!") );
440
441 if(!ispasswordvalid) return true;
442 }
443 }
444 else
445 {
446 bool invalidpassword = false;
447
448 if(page_login.IsEmpty() || page_password.IsEmpty())
449 {
450 DIOWEBSERVER_XEVENT xevent(this, DIOWEBSERVER_XEVENT_TYPE_DOAUTHENTICATE);
451
452 xevent.SetServer(server);
453 xevent.SetConnection(connection);
454 xevent.SetRequest(request);
455
456 PostEvent(&xevent);
457
458 if(xevent.IsRequestAuthenticate())
459 {
460 page_login = __L("usr");
461 page_password = __L("pwd");
462
463 } else invalidpassword = true;
464 }
465 else
466 {
467 DIOWEBSERVER_XEVENT xevent(this, DIOWEBSERVER_XEVENT_TYPE_CHECKAUTHENTICATE);
468
469 xevent.SetServer(server);
470 xevent.SetConnection(connection);
471 xevent.SetRequest(request);
472
473 PostEvent(&xevent);
474
475 if(!xevent.IsRequestAuthenticate()) invalidpassword = true;
476 }
477
478 if(invalidpassword)
479 {
480 XSTRING leyend;
481
482 leyend = __L("Invalid user or password!");
483 GenerateResponse_Error(connection, DIOWEBHEADER_RESULT_NOTFOUND, leyend);
484
485 APPFLOW_LOG_ENTRY(XLOGLEVEL_ERROR, DIOWEBSERVER_LOGSECTIONID, false, leyend.Get());
486
487 return true;
488 }
489 }
490 }
491
492 //-------------------------------------------------------------------------------------------------------------------------------
493 /*
494 if(useragentID)
495 {
496 XSTRING browser;
497 XSTRING SO;
498
499 if(useragentID->Get(request->GetUserAgent()->Get(), browser, SO, 5))
500 {
501 APPFLOW_LOG_ENTRY(XLOGLEVEL_INFO, DIOWEBSERVER_LOGSECTIONID_VERBOSE, false, __L("Web browser : %s, with: %s."), browser.Get(), SO.Get());
502 }
503 }
504 */
505
506 //-------------------------------------------------------------------------------------------------------------------------------
507
508 status = false;
509 nparams = querystring.GetParamsFromURL(resourceconv.Get())?true:false;
510
511 querystring.DeleteParamsFromURL(resourceconv);
512
513 if(!resourceconv.Compare(__L("/"),true)) resourceconv.Set(__L("/index.html"));
514
515
516 //-------------------------------------------------------------------------------------------------------------------------------
517
518
519 XPATH xpathfile;
520
521 if(cfg->WebServer_PathResources()->IsEmpty())
522 GEN_XPATHSMANAGER.GetPathOfSection(XPATHSMANAGERSECTIONTYPE_WEB, xpathfile);
523 else xpathfile = cfg->WebServer_PathResources()->Get();
524
525 xpathfile.Slash_Delete();
526 xpathfile += resourceconv.Get();
527
528 //-------------------------------------------------------------------------------------------------------------------------------
529
530 if(resourceconv.GetExtension(extension))
531 {
532 int nplugins = webserver->GetPluginsManager()->Plugins_GetNPlugins();
533 for(int c=0; c<nplugins; c++)
534 {
535 DIOWEBSERVER_PLUGIN* plugin = webserver->GetPluginsManager()->Plugins_Get(c);
536 if(plugin)
537 {
538 if(plugin->PageExtension_IsContent(extension))
539 {
540 DIOWEBHEADER_RESULT headerresult = DIOWEBHEADER_RESULT_OK;
541 XSTRING result;
542
543 status = plugin->PageExtension(xpathfile, request, &querystring, headerresult, result);
544 if(status)
545 {
546 XSTRING head;
547 XSTRING data;
548
549 plugin->SeparedResult(result, head, data);
550
551 status = SendRequest(connection, headerresult, data, 5, &head);
552 }
553 else
554 {
555 XSTRING leyend;
556
557 leyend.Format(__L("Error to execute the %s."), plugin->GetName()->Get());
558
559 GenerateResponse_Error(connection, DIOWEBHEADER_RESULT_NOTFOUND, leyend);
560 APPFLOW_LOG_ENTRY(XLOGLEVEL_ERROR, DIOWEBSERVER_LOGSECTIONID, false, leyend.Get());
561 }
562
563 XSTRING IPstring;
564 connection->GetDIOStream()->GetClientIP()->GetXString(IPstring);
565
566 if(!status)
567 {
568 APPFLOW_LOG_ENTRY((status?XLOGLEVEL_INFO:XLOGLEVEL_ERROR), DIOWEBSERVER_LOGSECTIONID, false, __L("Request from [%s] to the web server \"%s\" %s."), IPstring.Get(), resourceconv.Get(), status?__L("sent"):__L("not send"));
569 }
570
571 ispagepluging = true;
572 }
573 }
574 }
575 }
576
577 if(!ispagepluging)
578 {
579 if(!nparams && !isapirestonly)
580 {
581 status = SendRequest(connection, DIOWEBHEADER_RESULT_OK, xpathfile, 5);
582 if(!status)
583 {
584 // Brower close the conextion in thread
585 if(!connection->GetDIOStream()->IsConnected()) status = true;
586 }
587 }
588
589 if(!status)
590 {
591 if(!nparams && !isapirestonly)
592 {
593 DIOWEBSERVER_XEVENT xevent(this, DIOWEBSERVER_XEVENT_TYPE_REQUEST);
594
595 xevent.SetServer(server);
596 xevent.SetConnection(connection);
597 xevent.SetRequest(request);
598
599 PostEvent(&xevent);
600
601 status = xevent.GetStatus();
602 }
603 }
604
605 //-------------------------------------------------------------------------------------------------------------------------------
606
607 if(!status)
608 {
609 DIOWEBPAGEHTMLCREATOR* webHTMLpage = GEN_NEW DIOWEBPAGEHTMLCREATOR();
610 if(webHTMLpage)
611 {
612 DIOWEBSERVER_XEVENT xevent(this, DIOWEBSERVER_XEVENT_TYPE_REQUEST_ENDPOINT);
613
614 xevent.SetServer(server);
615 xevent.SetConnection(connection);
616 xevent.SetRequest(request);
617 xevent.SetQueryString(&querystring);
618
619 PostEvent(&xevent);
620
621 //DIOWEBHEADER_RESULT result = xevent.GetResult();
622
623 status = xevent.GetStatus();
624 if(status)
625 {
626 if(!xevent.GetOutputString()->IsEmpty())
627 {
628 webHTMLpage->Set(xevent.GetOutputString()->Get());
629 status = SendRequest(connection, DIOWEBHEADER_RESULT_OK, (*webHTMLpage), 5);
630 }
631 else
632 {
633 status = SendRequest(connection, DIOWEBHEADER_RESULT_OK, xevent.GetOutputBuffer(), 5);
634 }
635 }
636 else
637 {
638 XSTRING leyend;
639
640 leyend = __L("Error: command or erroneous parameters.");
641 GenerateMessagePage(leyend, (*webHTMLpage));
642
643 GenerateResponse_Error(connection, DIOWEBHEADER_RESULT_NOTFOUND, leyend);
644
645 APPFLOW_LOG_ENTRY(XLOGLEVEL_ERROR, DIOWEBSERVER_LOGSECTIONID, false, leyend.Get());
646 }
647
648 XSTRING IPstring;
649 connection->GetDIOStream()->GetClientIP()->GetXString(IPstring);
650
651 if(!status)
652 {
653 APPFLOW_LOG_ENTRY((status?XLOGLEVEL_INFO:XLOGLEVEL_ERROR), DIOWEBSERVER_LOGSECTIONID, false, __L("Request from [%s] to the web server\"%s\" %s."), IPstring.Get(), resourceconv.Get(), status?__L("sent"):__L("not send"));
654 }
655
656 GEN_DELETE webHTMLpage;
657 }
658 }
659 else
660 {
661 XSTRING IPstring;
662 connection->GetDIOStream()->GetClientIP()->GetXString(IPstring);
663
664 // APPFLOW_LOG_ENTRY(XLOGLEVEL_INFO, DIOWEBSERVER_LOGSECTIONID, false, __L("Request from [%s] to the web server of file \"%s\" sent."), IPstring.Get(), xpathfile.Get());
665 }
666 }
667
668 return true;
669}

832void APPFLOWWEBSERVER::HandleEvent_WebServer(DIOWEBSERVER_XEVENT* event)
833{
834 switch(event->GetEventType())
835 {
836 case DIOWEBSERVER_XEVENT_TYPE_REQUEST : ResolveRequest(event->GetServer(), event->GetConnection(), event->GetRequest());
837 break;
838
839 case DIOWEBSERVER_XEVENT_TYPE_WEBSOCKET_CONNECTED : if(!PostEvent(event))
840 {
841 // Accept connection with any version, protocol o resources.
842 event->SetStatus(true);
843 break;
844 }
845 break;
846
847 case DIOWEBSERVER_XEVENT_TYPE_WEBSOCKET_READDATA : PostEvent(event);
848 break;
849
850 case DIOWEBSERVER_XEVENT_TYPE_WEBSOCKET_DISCONNECTED : PostEvent(event);
851 break;
852 }
853}
