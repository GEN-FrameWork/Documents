274XFILETXTFORMATCHAR XFILETXT::GetFormatCharFromFile(XDWORD* sizeBOM)
275{
276 XFILETXTFORMATCHAR formatchar;
277
278 XBYTE BOM[3];
279 XQWORD position = 0;
280
281 file->GetPosition(position);
282 file->SetPosition(0);
283
284 if(sizeBOM) (*sizeBOM) = 0;
285
286 if(file->Read((XBYTE*)&BOM,3))
287 {
288 if((BOM[0]==0xEF)&&(BOM[1]==0xBB)&&(BOM[2]==0xBF))
289 {
290 formatchar = XFILETXTFORMATCHAR_UTF8;
291 if(sizeBOM) (*sizeBOM) = 3;
292 }
293 else
294 {
295 if((BOM[0]==0xFE)&&(BOM[1]==0xFF))
296 {
297 formatchar = XFILETXTFORMATCHAR_UTF16_BE;
298 if(sizeBOM) (*sizeBOM) = 2;
299 }
300 else
301 {
302 if((BOM[0]==0xFF)&&(BOM[1]==0xFE))
303 {
304 formatchar = XFILETXTFORMATCHAR_UTF16_LE;
305 if(sizeBOM) (*sizeBOM) = 2;
306 }
307 else formatchar = XFILETXTFORMATCHAR_ASCII;
308 }
309 }
310
311 } else formatchar = XFILETXTFORMATCHAR_ASCII;
312
313 file->SetPosition(position);
314
315 return formatchar;
316}

637bool XFILETXT::IsBinaryFile()
638{
639 if(!IsOpen()) return false;
640 if(!file) return false;
641
642 bool istextfile = false;
643
644 #define MAX_BUFFER_TEST 1024*5
645
646 XBUFFER buffer;
647 XDWORD size = MAX_BUFFER_TEST;
648 XDWORD controlchars = 0;
649
650 buffer.Resize(MAX_BUFFER_TEST);
651 file->Read(buffer.Get(), &size);
652
653 if(size)
654 {
655 for(XDWORD c=0; c<size; c++)
656 {
657 if((((int)buffer.Get()[c] >= 0x00) && ((int)buffer.Get()[c] < 0x09)) ||
658 (((int)buffer.Get()[c] > 0x0D) && ((int)buffer.Get()[c] < 0x20)))
659 {
660 controlchars++;
661 }
662 }
663
664 int percent = (controlchars*100/size);
665 if(percent < 10) istextfile = true;
666 }
667
668 return !istextfile;
669}

840{
841 if(!file) return false;
842 if(!file->IsOpen()) return false;
843
844 if(lines.IsEmpty()) return false;
845
846 XBYTE BOM[3] = { 0,0,0 };

867 if(buffer)

871 for(int c=0;c<(int)lines.GetSize();c++)

873 XSTRING* string = (XSTRING*)lines.Get(c);
874 bw = 0;
875
876 memset(buffer, 0, XFILETXT_MAXBUFFER);
877
878 if(string)
879 {
880 switch(formatchar)
881 {
882 case XFILETXTFORMATCHAR_UNKNOWN : break;
883
884 case XFILETXTFORMATCHAR_ASCII : { bw = string->GetSize();
885
886 XBUFFER charstr;
887
888 (*string).ConvertToASCII(charstr);
889 memcpy(buffer,(XBYTE*)charstr.Get(), bw);
890
891 }
892 break;

895 break;

899 bw = (string->GetSize()*2);

910
911 bw = (string->GetSize()*2);

915 bufferw[d] = (XWORD)string->Get()[d];
916 }
917 }
918 break;

921 }

925 }

930 GEN_DELETE_ARRAY buffer;

932
933 if(status)

956bool XFILETXT::ReadNLines(int nlines)
957{
958 if(!file) return false;
959 if(!file->IsOpen()) return false;
960
961 DeleteAllLines();
962
963 XDWORD sizeBOM = 0;
964 XFILETXTFORMATCHAR formatchar = GetFormatCharFromFile(&sizeBOM);
965 XDWORD sizebytescharacter = SizeOfCharacter(formatchar);
966
967 if(this->formatchar==XFILETXTFORMATCHAR_UNKNOWN) this->formatchar = formatchar;
968
969 file->SetPosition(sizeBOM);

971 bool endfile;
972 XDWORD br;

975 XBYTE* readbuffer = GEN_NEW XBYTE[XFILETXT_MAXBUFFER];

978 memset(readbuffer, 0, XFILETXT_MAXBUFFER);

983 endfile = !file->Read(readbuffer, &br);

987 XDWORD sizeLF = 0;

995 dataline.Add(&readbuffer[bufferpos], sizebytescharacter*sizeline);
996 AddLine(formatchar, dataline.Get(), (int)(dataline.GetSize()/sizebytescharacter));
997 dataline.Delete();
998
999 sizeline *= sizebytescharacter;
1000 sizeline += (sizeLF*sizebytescharacter);
1001 bufferpos += sizeline;
1002 }
1003 else
1004 {
1005 int lack = (br-bufferpos);
1006 dataline.Add(&readbuffer[bufferpos], lack);
1007 bufferpos += lack;
1008
1009 if(endfile && (bufferpos == br))
1010 {
1011 AddLine(formatchar, dataline.Get(), (int)(lack/sizebytescharacter));
1012 dataline.Delete();
1013 }
1014 }

1016 if(nlines != XFILETXT_ALLLINES)
1017 {
1018 if(lines.GetSize() > (XDWORD)nlines)
1019 {
1020 break;
1021 }
1022 }
1023
1024 } while(bufferpos < br);
1025
1026 } while(!endfile);
1027
1028 GEN_DELETE_ARRAY readbuffer;
1029
1030 return true;
1031}
