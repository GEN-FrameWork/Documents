150bool XFILECOBOL_PIC::ResolveMask()
151{
152 if(mask.IsEmpty())
153 {
154 XTRACE_PRINTCOLOR(4, __L("Cobol Reader: Error PIC Missing"));
155 return false;
156 }
157
158 hassign = false;
159 ndecimals = 0;
160 sizedata = 0;
161
162 XDWORD index = 0;
163
164 while(index < mask.GetSize())
165 {
166 switch(mask.Get()[index])
167 {
168 case __C('X') : type = XFILECOBOL_PIC_TYPE_BINARY;
169 maskdecondensed += __C('X');
170 sizedata++;
171 index++;
172 break;
173
174 case __C('A') : type = XFILECOBOL_PIC_TYPE_ALPHA;
175 maskdecondensed += __C('A');
176 sizedata++;
177 index++;
178 break;
179
180 case __C('9') : if(type != XFILECOBOL_PIC_TYPE_DECIMAL) type = XFILECOBOL_PIC_TYPE_NUMERIC;
181 if(type == XFILECOBOL_PIC_TYPE_DECIMAL) ndecimals++;
182 maskdecondensed += __C('9');
183 sizedata++;
184 index++;
185 break;
186
187 case __C(',') : type = XFILECOBOL_PIC_TYPE_DECIMAL;
188 maskdecondensed += __C(',');
189 sizedata++;
190 index++;
191 break;
192
193 case __C('S') : hassign = true;
194 maskdecondensed += __C('S');
195 index++;
196 break;
197
198 case __C('(') : { XSTRING modificator;
199 int sizemodificator;
200
201 int end = mask.Find(__L(")"), true, index+1);
202 if(end == XSTRING_NOTFOUND)
203 {
204 XTRACE_PRINTCOLOR(4, __L("Cobol Reader: Error PIC invalid %s. Missing ')'"), mask.Get());
205 return false;
206 }
207
208 mask.Copy(index, end+1, modificator);
209 modificator.UnFormat(__L("(%d)"), &sizemodificator);
210
211 if(!sizemodificator)
212 {
213 XTRACE_PRINTCOLOR(4, __L("Cobol Reader: Error PIC invalid %s. Modificador is zero '(0)'"), mask.Get());
214 return false;
215 }
216
217 for(int c=0; c<sizemodificator-1; c++)
218 {
219 maskdecondensed += mask.Get()[index-1];
220 sizedata++;
221 if(type == XFILECOBOL_PIC_TYPE_DECIMAL) ndecimals++;
222 }
223
224 index = end;
225 }
226 break;
227
228 case __C(')') : index++;
229 break;
230
231 case __C('V') : type = XFILECOBOL_PIC_TYPE_DECIMAL;
232 maskdecondensed += __C('V');
233 index++;
234 break;
235
236 default : XTRACE_PRINTCOLOR(4, __L("Cobol Reader: Error PIC invalid %s. Unknown Specifier '%c'"), mask.Get(), mask.Get()[index]);
237 return false;
238 break;
239 }
240 }
241
242 switch(computationaltype)
243 {
244 case 1 : // Float point 4 Bytes
245 break;
246
247 case 2 : // Float Point 8 Bytes
248 break;
249
250 case 3 : // BCD + Sing
251 sizedata = ((sizedata + 1) / 2) + (((sizedata + 1) % 2)?1:0);
252 break;
253
254 case 4 : break;
255
256 case 5 : // Conversion Packet number standard binary data
257 if((sizedata >= 1) && (sizedata <= 2))
258 {
259 sizedata = 1;
260 }
261 else
262 {
263 if((sizedata >=3) && (sizedata <= 4))
264 {
265 sizedata = 2;
266 }
267 else
268 {
269 if((sizedata >= 5) && (sizedata <= 9))
270 {
271 sizedata = 4;
272 }
273 else
274 {
275 if((sizedata >= 10) && (sizedata <= 18))
276 {
277 sizedata = 8;
278 }
279 else
280 {
281 XTRACE_PRINTCOLOR(4, __L("Cobol Reader: Error PIC invalid %s. Size of modificator to long %s > 18 "), mask.Get(), sizedata);
282 return false;
283 }
284 }
285 }
286 }
287 break;
288 }
289
290 return true;
291}

1118 {

1120 {
