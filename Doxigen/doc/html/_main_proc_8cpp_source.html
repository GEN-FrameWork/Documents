258bool MAINPROC::CreateParams(XCHAR* commandline)
259{
260 XSTRING _commandline = commandline;
261
262 if(_commandline.IsEmpty()) return false;
263
264 int start = 0;
265 XDWORD c = 0;
266
267 do{ if((_commandline.Get()[c] == __C(' ')) || (_commandline.GetSize() == c))
268 {
269 XSTRING* param = GEN_NEW XSTRING();
270 if(param)
271 {
272 _commandline.Copy(start, c, (*param));
273
274 GetExecParams()->Add(param);
275 start = c+1;
276 }
277 }
278
279 c++;
280
281 } while(c <= _commandline.GetSize());
282
283 return true;
284}
