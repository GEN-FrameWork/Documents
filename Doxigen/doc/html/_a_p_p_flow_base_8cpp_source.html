113bool APPFLOWBASE::Ini(XVECTOR<XSTRING*>* execparams)
114{
115 this->execparams = execparams;
116
117 GEN_XFACTORY_CREATE(xtimerglobal, CreateTimer())
118 if(!xtimerglobal) return false;
119
120 xtimerglobal->Reset();
121
122 firstupdate = false;
123
124 XDWORD code = GEN_XSYSTEM.GetLanguageSO();
125
126 if(code == XLANGUAGE_ISO_639_3_CODE_INVALID)
127 {
128 code = XLANGUAGE_ISO_639_3_CODE_ENG;
129 }
130
131 if(code != XLANGUAGE_ISO_639_3_CODE_INVALID)
132 {
133 XTRANSLATION_GEN::GetInstance().Sentences_AddToTranslation(code);
134 }
135
136 if(!AppProc_Ini()) return false;
137
138 return true;
139}

456void APPFLOWBASE::SetSystemChangeStatus(XSYSTEM_CHANGESTATUSTYPE systemchangestatustype)
457{
458 if(this->systemchangestatustype == systemchangestatustype)
459 {
460 return;
461 }
462
463 this->systemchangestatustype = systemchangestatustype;
464
465
466 XSTRING statusstr;
467
468 switch(systemchangestatustype)
469 {
470 case XSYSTEM_CHANGESTATUSTYPE_NONE :
471 default : statusstr = __L("normal"); break;
472 case XSYSTEM_CHANGESTATUSTYPE_REBOOT : statusstr = __L("reboot"); break;
473 case XSYSTEM_CHANGESTATUSTYPE_CONSOLE_CONNECT : statusstr = __L("console connect"); break;
474 case XSYSTEM_CHANGESTATUSTYPE_CONSOLE_DISCONNECT : statusstr = __L("console disconnect"); break;
475 case XSYSTEM_CHANGESTATUSTYPE_SESSION_LOGON : statusstr = __L("session logon"); break;
476 case XSYSTEM_CHANGESTATUSTYPE_SESSION_LOGOFF : statusstr = __L("session logooff"); break;
477 case XSYSTEM_CHANGESTATUSTYPE_SESSION_LOCK : statusstr = __L("session lock"); break;
478 case XSYSTEM_CHANGESTATUSTYPE_SESSION_UNLOCK : statusstr = __L("session unlock"); break;
479 case XSYSTEM_CHANGESTATUSTYPE_POWEROFF : statusstr = __L("power off"); break;
480 case XSYSTEM_CHANGESTATUSTYPE_SUSPEND : statusstr = __L("suspend"); break;
481 }
482
483 //XTRACE_PRINTCOLOR(XTRACE_COLOR_GREEN, __L("[APP Base System] Mode change to [%s]"), statusstr.Get());
484
485 APPFLOW_XEVENT xevent(this, APPFLOW_XEVENT_TYPE_CHANGESTATUSTYPE);
486 xevent.SetChangeStatusType(systemchangestatustype);
487 PostEvent(&xevent);
488}
