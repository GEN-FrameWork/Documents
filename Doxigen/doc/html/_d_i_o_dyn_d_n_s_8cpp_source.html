135bool DIODYNDNS::GetPublicIP(DIOIP& publicIP, int timeout, XSTRING* IPlocal)
136{
137 XBUFFER xbuffer;
138 bool status;
139
140 status = webclient->Get(__L("checkip.dyndns.com"), xbuffer, DIOWEBCLIENT_DEFAULTUSERAGENT, timeout, IPlocal);
141 if(status)
142 {
143 XSTRING startstring;
144 XSTRING result;
145
146 result.Set(xbuffer.Get(), xbuffer.GetSize());
147
148 startstring = __L("Current IP Address: ");
149

154

197 bool status = false;
198 DIOURL* _url = NULL;
199
200 ischanged = false;
201
202 _url = GEN_NEW DIOURL();
203 if(!_url)

205 return false;
206 }
207

209
210 if(_url->ResolveURL(publicIP[0]))
211 {

222 }
223 }

245DIODYNDNS_STATUSCODE DIODYNDNS::Update(DIOURL& url, DIOIP& newpublicip, int timeout, XSTRING* IPlocal)
246{
247 if(!webclient) return DIODYNDNS_STATUSCODE_UNKNOWN;
248
249 DIOURL askurl;
250 XSTRING IPstring;
251 XBUFFER xbuffer;
252 XSTRING result;
253 bool status;
254
255 newpublicip.GetXString(IPstring);
256
257 askurl.Format(__L("%s:%s@members.dyndns.org/nic/update?hostname=%s&myip=%s&wildcard=NOCHG&mx=NOCHG&backmx=NOCHG"), login.Get(), password.Get(), url.Get(), IPstring.Get());
258
259 status = webclient->Get(askurl.Get(), xbuffer, DIOWEBCLIENT_DEFAULTUSERAGENT, timeout, IPlocal);

262 result.Set(xbuffer.Get(), xbuffer.GetSize());
263
264 return GetError(result);

279DIODYNDNS_STATUSCODE DIODYNDNS::GetError(XSTRING& result)
280{
281 // Other Unknown result from dyndns service.
282 // "good" The update was successful, and the hostname is now updated.
283 // "nochg" The update changed no settings, and is considered abusive. (Additional nochg updates will cause the hostname to become blocked).
284 // "badsys" The system parameter given is not valid. Valid system parameters are dyndns, statdns and custom.
285 // "badagent" The user agent that was sent has been blocked for not following these specifications or no user agent was specified.
286 // "badauth" The username or password specified are incorrect.
287 // "!donator" An option available only to credited users (suchas offline URL) was specified, but the user is not a credited user.
288 // "notfqdn" The hostname specified is not a fully-qualified domain name (not in the form hostname.dyndns.org or domain.com).
289 // "nohost" The hostname specified does not exist (or is not in the service specified in the system parameter).
290 // "!yours" The hostname specified exists, but not under the username specified.
291 // "abuse" The hostname specified is blocked for update abuse.
292 // "numhost" Too many or too few hosts found.
293 // "dnserr" DNS error encountered.
294 // "911" There is a serious problem on our side, such as a database or DNS server failure. The client should stop updating until notified via the status page that the service is back up.
295
296
297 DIODYNDNS_STATUSCODE code = DIODYNDNS_STATUSCODE_UNKNOWN;
298 XCHAR* stringcodes[] = { __L("good") ,
299 __L("nochg") ,
300 __L("badsys") ,
301 __L("badagent") ,
302 __L("badauth") ,
303 __L("!donator") ,
304 __L("notfqdn") ,
305 __L("nohost") ,
306 __L("!yours") ,
307 __L("abuse") ,
308 __L("numhost") ,
309 __L("dnserr") ,
310 __L("911")
311 };
312
313
314 for(int c=0; c<(sizeof(stringcodes)/sizeof(XCHAR*)); c++)
315 {
316 if(result.Find(stringcodes[c],true) != XSTRING_NOTFOUND)
317 {
318 code = (DIODYNDNS_STATUSCODE)(c);
319 break;
320 }
321 }
322
323 return code;
324}
