96bool XWINDOWSDRIVEIMAGEMANAGER::GetDrives(XVECTOR<XDRIVEIMAGEMANAGER_DRIVE*>& drives)
97{
98 drives.DeleteContents();
99 drives.DeleteAll();
100
101 XPATH path;
102 XDWORD drivemask = GetLogicalDrives();
103 XQWORD ID = 0;
104 int c = 0;
105
106 // GetLogicalDrives returns 0 on failure, or a bitmask representing the drives available on the system (bit 0 = A:, bit 1 = B:, etc)
107
108 while(drivemask != 0)
109 {
110 if(drivemask & 1)
111 {
112 // the "A" in path will get incremented by the # of bits
113 // we've shifted
114 path.Format( __L("\\\\.\\%c:\\"), __C('A') + c);
115 if(CheckDriveType(path, ID))
116 {
117 XDRIVEIMAGEMANAGER_DRIVE* drive = GEN_NEW XDRIVEIMAGEMANAGER_DRIVE();
118 if(drive)
119 {
120 XSTRING label;
121
122 drive->GetName()->Format(__L("%c:"), __C('A') + c);
123 drive->GetPath()->Set(path);
124 drive->SetVolume(c);
125 drive->GetLabel()->Set(label);
126 drive->SetID(ID);
127
128 if(GetDriveLabel(drive->GetName()->Get(), label)) drive->GetLabel()->Set(label);
129
130 drives.Add(drive);
131 }
132 }
133 }
134
135 drivemask >>= 1;
136
137 ++c;
138 }
139
140 return true;
141}

155bool XWINDOWSDRIVEIMAGEMANAGER::Open(XDRIVEIMAGEMANAGER_DRIVE& drive)
156{
157 HANDLE handlevolume = GetHandleFromVolume(drive.GetVolume(), GENERIC_READ | GENERIC_WRITE);
158 if(handlevolume == INVALID_HANDLE_VALUE) return false;
159
160 drive.SetHandle((XDWORD)handlevolume);
161
162 XDWORD deviceID = 0;
163 if(!GetDeviceID(handlevolume, deviceID)) return false;
164 drive.SetDeviceID(deviceID);
165
166 if(!LockVolume(handlevolume, true)) return false;
167
168 if(!UnMountVolume(handlevolume)) return false;
169
170 HANDLE rawhandle = GetHandleFromDevice(deviceID, GENERIC_READ | GENERIC_WRITE);
171 if(rawhandle == INVALID_HANDLE_VALUE) return false;
172
173 drive.SetRawHandle((XDWORD)rawhandle);
174
175 XQWORD sectorsize = 0;
176 XQWORD numbersectors = 0;
177
178 if(!GetNumberOfSectors(rawhandle, sectorsize, numbersectors)) return false;
179
180 drive.SetSectorSize(sectorsize);
181 drive.SetNumberSectors(numbersectors);
182
183 return true;
184}

201bool XWINDOWSDRIVEIMAGEMANAGER::Read(XDRIVEIMAGEMANAGER_DRIVE& drive, XQWORD startsector, XQWORD numbersectors, XBUFFER& buffer)
202{
203 XBYTE* _buffer;
204 XDWORD maxbuffer = ((XDWORD)numbersectors * (XDWORD)drive.GetSectorSize());
205 XQWORD maxnsectorsblock = (maxbuffer / drive.GetSectorSize());
206 XQWORD readnsectorsblock = 0;
207 XQWORD sizeread = 0;
208 bool status = false;
209
210 buffer.Delete();
211
212 _buffer = GEN_NEW XBYTE[maxbuffer];
213 if(!_buffer) return false;
214
215 if(maxnsectorsblock > numbersectors) maxnsectorsblock = numbersectors;
216
217 while(readnsectorsblock < numbersectors)
218 {
219 XDRIVEIMAGEMANAGER_XEVENT xevent(this, XDRIVEIMAGEMANAGER_XEVENT_TYPE_READSECTORS);
220
221 xevent.SetDrive(&drive);
222 xevent.SetIndexSector(startsector);
223 xevent.SetNumberSectorsInOperation(maxnsectorsblock);
224
225 sizeread = (maxnsectorsblock * drive.GetSectorSize());
226
227 memset(_buffer, 0, maxbuffer);
228
229 status = ReadSectorDataFromHandle((HANDLE)drive.GetRawHandle(), startsector, drive.GetSectorSize(), _buffer, sizeread);
230 if(status)
231 {
232 xevent.SetError(XDRIVEIMAGEMANAGER_ERROR_NONE);
233
234 buffer.Add(_buffer, (XDWORD)sizeread);
235
236 readnsectorsblock += maxnsectorsblock;
237
238 if((readnsectorsblock + maxnsectorsblock) > numbersectors) maxnsectorsblock = (numbersectors - readnsectorsblock);
239 if((readnsectorsblock + maxnsectorsblock) > drive.GetNumberSectors()) maxnsectorsblock = (drive.GetNumberSectors() - readnsectorsblock);
240 }
241 else
242 {
243 xevent.SetError(XDRIVEIMAGEMANAGER_ERROR_READSECTORS);
244 }
245
246 PostEvent(&xevent);
247
248 if(!status) break;
249
250 }
251
252 GEN_DELETE_ARRAY _buffer;
253
254 return status;
255}

596 DWORD nbytesreturned;
597
598
599 path_noslash = path.Get();
600
601 path_noslash.Slash_Delete();
602
603 // some calls require no tailing slash, some require a trailing slash...
604
605 drivetype = GetDriveType(path.Get());
606 switch(drivetype)
607 {
608 case DRIVE_REMOVABLE : // The media can be removed from the drive.
609
610 case DRIVE_FIXED : { // The media cannot be removed from the drive. Some USB drives report as this.
611 hdevice = CreateFile(path_noslash.Get(), FILE_READ_ATTRIBUTES, FILE_SHARE_READ | FILE_SHARE_WRITE, NULL, OPEN_EXISTING, 0, NULL);
612 if(hdevice == INVALID_HANDLE_VALUE) break;
613
614 int sizearray = sizeof(STORAGE_DEVICE_DESCRIPTOR) + 512 - 1;
615 pdevdesc = (PSTORAGE_DEVICE_DESCRIPTOR)GEN_NEW BYTE[sizearray];

617 pdevdesc->Size = sizearray;
618
619 // get the device number if the drive is removable or (fixed AND on the usb bus, SD, or MMC (undefined in XP/mingw))

621 if(GetMediaType(hdevice))
