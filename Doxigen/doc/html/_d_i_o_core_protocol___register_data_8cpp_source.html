282bool DIOCOREPROTOCOL_REGISTERDATA::CreateIDMachine(XUUID& ID)
283{
284 XSTRING origin;
285 XBUFFER originbuffer;
286 HASHSHA2* sha2 = NULL;
287 XBUFFER* result = NULL;
288
289 ID.Empty();
290
291 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetBIOSSerialNumber()->Get());
292 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetCPUSerialNumber()->Get());
293
294 origin.ConvertToUTF8(originbuffer);
295
296 sha2 = GEN_NEW HASHSHA2(HASHSHA2TYPE_256);
297 if(!sha2)
298 {
299 return false;
300 }
301
302 sha2->Do(originbuffer);
303 result = sha2->GetResult();
304
305 if(!result)
306 {
307 return false;
308 }
309
310 if(!result->GetSize())
311 {
312 return false;
313 }
314
315 XDWORD data1 = 0;
316 XWORD data2 = 0;
317 XWORD data3 = 0;
318 XBYTE data4 = 0;
319 XBYTE data5 = 0;
320 XBYTE* data6 = &result->Get()[12];
321
322 result->Get((XDWORD&)data1);
323 result->Get((XWORD&)data2);
324 result->Get((XWORD&)data3);
325 result->Get((XBYTE&)data4);
326 result->Get((XBYTE&)data5);
327
328 ID.Set(data1, data2, data3, data4, data5, data6);
329
330 ID.GetToString(origin);
331
332 GEN_DELETE sha2;
333
334 return true;
335}
