102bool UI_ANIMATION::LoadFromFile(XSTRING& resourcename, GRPPROPERTYMODE mode)
103{
104 XPATH xpathimg;
105 GRPBITMAPFILE bitmapfile;
106 GRPBITMAP* bitmap = NULL;
107 XSTRING mask;
108 int indexmask = 0;
109 int indeximg = 0;
110 GRPPROPERTYMODE modealpha = mode;
111 GRPBITMAPFILE_TYPE typefile = bitmapfile.GetTypeFromExtenxion(resourcename.Get());
112
113 modealpha = ChangeModeWithAlpha(typefile, mode);
114
115 while(1)
116 {
117 XSTRING namefile;
118
119 indexmask = resourcename.Copy(UI_ANIMATION_FRAME_INI, UI_ANIMATION_FRAME_END, true, 0, mask);
120
121 namefile = resourcename;
122
123 if(!mask.IsEmpty())
124 {
125 XSTRING result;
126 int masksize = mask.GetSize()+3;
127
128 namefile.DeleteCharacters(indexmask-masksize, masksize);
129
130 mask.Insert(__L("%"), 0);
131 mask.Add(__L("d"));
132
133 result.Format(mask.Get(), indeximg);
134 indeximg++;
135
136 namefile.Insert(result.Get(), indexmask-masksize);
137 }
138
139 GEN_XPATHSMANAGER.GetPathOfSection(XPATHSMANAGERSECTIONTYPE_GRAPHICS, xpathimg);
140 xpathimg.Slash_Add();
141 xpathimg.Add(namefile);
142
143 if(GEN_USERINTERFACE.IsZippedFile())
144 {
145 XFILEUNZIP* unzipfile = GEN_USERINTERFACE.GetUnzipFile();
146 if(unzipfile)
147 {
148 XPATH pathnamefilecmp(APPFLOW_DEFAULT_DIRECTORY_GRAPHICS);

151 pathnamefilecmp.Slash_Add();
152 pathnamefilecmp += namefile;
153
154 pathnamefilecmp.GetNamefileExt(namefileonly);
155

157 if(status)

159 XPATH unzippathfile_tmp;
160
161 unzippathfile_tmp = GEN_USERINTERFACE.GetUnzipPathFile()->Get();
162 unzippathfile_tmp += namefileonly;
163
164 bitmap = bitmapfile.Load(unzippathfile_tmp, modealpha);
165
166 GEN_USERINTERFACE.DeleteTemporalUnZipFile(unzippathfile_tmp);
167
168 if(bitmap)
169 {
170 bitmaps.Add(bitmap);
171 if(mask.IsEmpty())

173 break;
174 }
175 }

178 break;
179 }
180 }

182 {
183 break;
184 }
185 }
