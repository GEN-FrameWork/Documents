483bool DIONODEITEM::Deserialize()

486

490 Primitive_Extract<bool>(issimulated , __L("simulated"));
491 Primitive_Extract<XQWORD>(timetoupdate , __L("timetoupdate"));
492
493 DIONODEITEMHANDLER* itemhandler = ItemHandler_Get();
494 if(itemhandler)
495 {
496 XDWORD type;
497
498 Primitive_Extract<XDWORD>(type, __L("handler_type"));
499 itemhandler->SetType(type);
500
501
502 Primitive_Extract<XSTRING>((*itemhandler->GetName()), __L("handler_name"));
503 }
504
505 XVectorClass_Extract<DIONODEITEMVALUE>(&values, __L("values"), __L("value"));
506
507 UUID.SetFromString(ID);
