271
272 if(!GetHW_ID(hardwareID)) return false;
273 if(hardwareID == DIOI2CTOUCHSENSORAT42QT1060_HW_ID) return false;
274
275 // Key Mask Enable CHG interrupt for KEYs 0-3
276 if(!WriteRegister(DIOI2CTOUCHSENSORAT42QT1060_KEY_MASK_REG, 0x0F)) return false;
