349bool XFILEHEX::DecodeLine(XSTRING* line, XFILEHEX_ENTRY& entry)
350{
351 if(!line) return false;
352 if(line->IsEmpty()) return false;
353
354 if(line->Get()[0] !=__C(':')) return false;
355
356 int sizedata;
357 int address;
358 int entrytype;
359 XSTRING string;
360 int CRC;
361 XBUFFER dataCRC;
362
363 line->ToUpperCase();
364
365 line->Copy( 1, 3, string);
366 string.UnFormat(__L("%02X"), &sizedata);
367 entry.SetSizeData((XBYTE)sizedata);
368
369 line->Copy( 3, 7, string);
370 string.UnFormat(__L("%04X"), &address);
371 entry.SetAddress((XWORD)address);
372
373 line->Copy( 7, 9, string);
374 string.UnFormat(__L("%02X"), &entrytype);
375 entry.SetEntryType((XFILEHEX_ENTRYTYPE)entrytype);
376
377 if(sizedata)
378 {
379 line->Copy(9, line->GetSize()-2, string);
380 string.ConvertHexStringToBuffer((*entry.GetData()));
381 }
382
383 line->Copy(line->GetSize()-2, string);
384 string.UnFormat(__L("%02X"), &CRC);
385
386 line->Copy(1, line->GetSize()-2, string);
387 string.ConvertHexStringToBuffer(dataCRC);
388
389 XBYTE calculeCRC = entry.CalculeCRC(dataCRC);
390
391 if((XBYTE)CRC != calculeCRC) return false;
392
393 return true;
394}

409bool XFILEHEX::EncodeLine(XFILEHEX_ENTRY* entry, XSTRING& line)
410{
411 if(!entry) return false;
412
413 XSTRING string;
414
415 string.Format(__L("%02X"), entry->GetSizeData()); line.Add(string);
416 string.Format(__L("%04X"), entry->GetAddress()); line.Add(string);
417 string.Format(__L("%02X"), (XWORD)entry->GetEntryType()); line.Add(string);
418
419 if(entry->GetSizeData())
420 {
421 string.ConvertHexStringFromBuffer((*entry->GetData()), true);
422 line.Add(string);
423 }
424
425 XBUFFER dataCRC;
426
427 line.ConvertHexStringToBuffer(dataCRC);
428 string.Format(__L("%02X"), entry->CalculeCRC(dataCRC)); line.Add(string);
429
430 line.Insert(__L(":"), 0);
431 line.ToUpperCase();
432
433 return true;
434}
