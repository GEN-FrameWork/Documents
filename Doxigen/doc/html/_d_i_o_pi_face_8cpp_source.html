98bool DIOPIFACE::Configure()
99{
100 Write_Register(DIOSPIGPIOMCP23S17_WRITE_CMD, DIOSPIGPIOMCP23S17_IOCON , 8); // Enable hardware addressing
101 Write_Register(DIOSPIGPIOMCP23S17_WRITE_CMD, DIOSPIGPIOMCP23S17_GPIOA , 0x00); // Turn on port A
102 Write_Register(DIOSPIGPIOMCP23S17_WRITE_CMD, DIOSPIGPIOMCP23S17_IODIRA , 0); // Set port A as an output
103 Write_Register(DIOSPIGPIOMCP23S17_WRITE_CMD, DIOSPIGPIOMCP23S17_IODIRB , 0xFF); // Set port B as an input
104 Write_Register(DIOSPIGPIOMCP23S17_WRITE_CMD, DIOSPIGPIOMCP23S17_GPPUB , 0xFF); // Turn on port B pullups
105
106 return true;
107}

121bool DIOPIFACE::ReadInputs(XBYTE& value)
122{
123 XBYTE _value[3] = { 0xFF, 0xFF, 0xFF };
124
125 if(!diostream) return false;
126
127 diostream->GetInXBuffer()->Delete();
128
129 if(!Write_Register(DIOSPIGPIOMCP23S17_READ_CMD, DIOPIFACE_INPUTPORT, 0xFF)) return false;
130
131 if(!diostream->WaitToFilledReadingBuffer(3, timeout)) return false;
132
133 bool status = diostream->Read(_value, 3)?true:false;
134
135 value = _value[2] ^ 0xFF;
136
137 return status;
138}
