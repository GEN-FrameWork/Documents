191bool DIONODE::GetSubGroups(XVECTOR<XSTRING*>& listsubgroups)
192{
193 XPATH pathgroup;
194
195 pathgroup = group.Get();
196
197 pathgroup.Slash_Delete();
198
199 do{ XSTRING* _subgroup = GEN_NEW XSTRING();
200 if(_subgroup)
201 {
202 pathgroup.GetNamefileExt(*_subgroup);
203 pathgroup.GetPath(pathgroup);
204
205 if(!_subgroup->Compare(pathgroup, true))
206 {
207 listsubgroups.Insert(0, _subgroup);
208 break;
209 }
210
211 pathgroup.Slash_Delete();
212
213 if(_subgroup->IsEmpty())
214 {
215 GEN_DELETE _subgroup;
216 break;
217 }
218 else
219 {
220 listsubgroups.Insert(0, _subgroup);
221 }
222 }
223 else
224 {
225 break;
226 }
227
228 } while(1);
229
230
231 return listsubgroups.IsEmpty()?false:true;
232}
