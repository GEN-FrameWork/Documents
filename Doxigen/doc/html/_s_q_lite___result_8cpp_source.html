199bool SQLITE_RESULT::ProcessRow()
200{
201 row.Clear();
202 header.DeleteContents();
203 header.DeleteAll();
204
205 int ncol = (int)this->GetNumColumns();
206
207 for(int e=0; e<ncol; e++)
208 {
209 DB_SQL_VARIANT* variant=GEN_NEW DB_SQL_VARIANT();
210 if(!variant) return false;
211
212 DB_SQL_STRING columnvalue;
213 char* text=(char*)sqlite3_column_text(query->ppstmt, (int)e);
214 int len = sqlite3_column_bytes(query->ppstmt, (int)e);
215 columnvalue.Set(text);
216
217 DB_SQL_STRING* columnname=GEN_NEW DB_SQL_STRING();
218 if(!columnname)
219 {
220 delete(variant);
221 return false;
222 }
223
224 columnname->Set(sqlite3_column_name(query->ppstmt, (int)e));
225
226 int sqtype=sqlite3_column_type(query->ppstmt, (int)e);
227 switch(sqtype)
228 {
229 case SQLITE_INTEGER : { int i = columnvalue.ConvertToInt();
230 (*variant) = i;
231 }
232 break;
233
234 case SQLITE_FLOAT : { float i = columnvalue.ConvertToFloat();
235 (*variant) = i;
236 }
237 break;
238
239 case SQLITE_BLOB : { // unsupported
240 // XTRACE_PRINTCOLOR(0,__L("Unsupported BLOB type"));
241 }
242 break;
243
244 case SQLITE_NULL : break;
245
246 case SQLITE_TEXT : { XSTRING received;
247
248 received.ConvertFromUTF8((XBYTE*)text,len);
249 (*variant) = received;
250 }
251 break;
252 }
253
254 header.Add(columnname);
255
256 row.Add(variant);
257 }
258
259 return true;
260}
