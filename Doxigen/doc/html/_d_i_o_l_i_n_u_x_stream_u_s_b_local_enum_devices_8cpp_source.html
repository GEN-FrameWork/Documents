302bool DIOLINUXSTREAMUSBLOCALENUMDEVICES::SearchAndDeleteEqualDevices()
303{
304 if(devices.IsEmpty()) return false;
305
306 bool found;
307
308 do{ found = false;
309
310 for(int c=0; c<devices.GetSize(); c++)
311 {
312 DIOSTREAMDEVICEUSB* device1 = (DIOSTREAMDEVICEUSB*)devices.Get(c);
313 if(device1)
314 {
315 for(int d=0; d<devices.GetSize(); d++)
316 {
317 DIOSTREAMDEVICEUSB* device2 = (DIOSTREAMDEVICEUSB*)devices.Get(d);
318 if(device2 && (c!=d))
319 {
320 if(!device1->GetName()->Compare(device2->GetName()->Get()))
321 {
322 GEN_DELETE device1;
323 devices.Delete(device1);
324
325 found = true;
326 break;
327 }
328 }
329 }
330
331 if(found) break;
332 }
333 }
334
335 if(!found) break;
336
337 } while(1);
338
339 return true;
