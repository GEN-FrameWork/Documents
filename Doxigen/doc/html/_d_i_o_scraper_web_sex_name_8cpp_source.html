217{
218 bool status = false;
219
220 if(xmutexdo) xmutexdo->Lock();
221
222 this->name = name;
223
224 DIOSEXNAME_RESULT* sexnameresult;
225 if(usecache)
226 {
227 sexnameresult = (DIOSEXNAME_RESULT*)cache->Get(this->name);
228 if(sexnameresult)
229 {
230 sexnametype = sexnameresult->Get();
231 if(nameclear) nameclear->Set(sexnameresult->GetNameClear()->Get());
232
233 if(xmutexdo) xmutexdo->UnLock();

239 sexnametype = DIOSCRAPERWEBSEXNAMETYPE_NOTOBTAINED;
240 if(this->name.IsEmpty())
241 {
242 if(Load(DIOSCRAPERWEBSEXNAME_NAMEFILE))

250 sexnamestr = GetValue(__L("SEX"));
251 if(!sexnamestr.Compare(__L("male"))) sexnametype = DIOSCRAPERWEBSEXNAMETYPE_MALE;
252 if(!sexnamestr.Compare(__L("female"))) sexnametype = DIOSCRAPERWEBSEXNAMETYPE_FEMALE;
253
254 if(nameclear)
255 {
256 if((sexnametype == DIOSCRAPERWEBSEXNAMETYPE_MALE) || (sexnametype == DIOSCRAPERWEBSEXNAMETYPE_FEMALE))
257 (*nameclear) = GetValue(__L("NAME"));
258 else (*nameclear) = this->name.Get();

261 if(usecache)
262 {
263 sexnameresult = GEN_NEW DIOSEXNAME_RESULT();
264 if(sexnameresult)
265 {
266 if(nameclear) sexnameresult->GetNameClear()->Set(nameclear->Get());
267 sexnameresult->Set(sexnametype);
268
269 cache->Add(this->name, sexnameresult);
270 }
271 }
272
273 status = true;
274 }
275 }
276 }
277
278 if(xmutexdo) xmutexdo->UnLock();
279
