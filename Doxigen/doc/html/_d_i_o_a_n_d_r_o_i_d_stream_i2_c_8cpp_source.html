125bool DIOANDROIDSTREAMI2C::Open()
126{
127 if(!config) return false;
128
129 if(!threadconnection) return false;
130
131 XBUFFER charstr;
132
133 (*config->GetLocalDeviceName()).ConvertToASCII(charstr);
134 handle = open(charstr.GetPtrChar(), O_RDWR);
135 if(handle<0) return false;
136
137 if(ioctl(handle, DIOANDROIDSTREAMI2C_SLAVE, config->GetRemoteAddress()) < 0) return false;
138
139 SetEvent(DIOSTREAMI2C_FSMEVENT_CONNECTED);
140
141 status = DIOSTREAMSTATUS_CONNECTED;
142
143 ResetXBuffers();
144 ResetConnectionStatistics();
145
146 return threadconnection->Ini();

206void DIOANDROIDSTREAMI2C::ThreadConnection(void* data)
207{
208 DIOANDROIDSTREAMI2C* diostream = (DIOANDROIDSTREAMI2C*)data;
209 if(!diostream) return;
210
211 if(diostream->GetEvent()==DIOSTREAMI2C_FSMEVENT_NONE) // No hay nuevos Eventos
212 {
213 switch(diostream->GetCurrentState())
214 {
215 case DIOSTREAMI2C_FSMSTATE_NONE : break;
216
217 case DIOSTREAMI2C_FSMSTATE_CONNECTED : break;
218
219 case DIOSTREAMI2C_FSMSTATE_WAITINGTOREAD : { if(diostream->outbuffer->IsBlocked()) break;
220
221 if(!diostream->IsBlockWrite())
222 {
223 XDWORD size = diostream->outbuffer->GetSize();
224 if(size)
225 {
226 diostream->outbuffer->SetBlocked(true);
227 XDWORD bw = diostream->I2C_Write(diostream->config->GetRemoteAddress(), diostream->outbuffer->Get(), size);
228 diostream->outbuffer->SetBlocked(false);
229
230 if(bw)
231 {
232 diostream->outbuffer->Extract(NULL, 0 , size);
233 }
234 }
235 }
236
237
238 if(!diostream->IsBlockRead())
239 {
240 if(diostream->sizeread)
241 {
242 XBYTE* _data = GEN_NEW XBYTE[diostream->sizeread];
243
244 if(_data)
245 {
246 if(diostream->I2C_Read(diostream->config->GetRemoteAddress(), _data, diostream->sizeread))
247 {
248 diostream->inbuffer->Add(_data, diostream->sizeread);
249 //XTRACE_PRINTCOLOR(3, __L("(Addr %04X, size:%d)"), diostream->config->GetRemoteAddress(), diostream->inbuffer->GetSize());
250 //XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_BLUE, (*diostream->inbuffer));
251
252 diostream->sizeread = 0;
253 }
254
255 GEN_DELETE_ARRAY _data;
256 }
257 }
258 }
259 }
260 break;
261
262
263 case DIOSTREAMI2C_FSMSTATE_SENDINGDATA : break;
264
265 case DIOSTREAMI2C_FSMSTATE_DISCONNECTING : break;
266
267 }
268 }
269 else
270 {
271 if(diostream->GetEvent()<DIOSTREAMI2C_LASTEVENT)
272 {
273 diostream->CheckTransition();
274
275 switch(diostream->GetCurrentState())
276 {
277 case DIOSTREAMI2C_FSMSTATE_NONE : break;
278
279 case DIOSTREAMI2C_FSMSTATE_CONNECTED : { DIOSTREAM_XEVENT xevent(diostream,DIOSTREAM_XEVENT_TYPE_CONNECTED);
280 xevent.SetDIOStream(diostream);
281
282 diostream->PostEvent(&xevent);
283
284 diostream->SetEvent(DIOSTREAMI2C_FSMEVENT_WAITINGTOREAD);
285 }
286 break;
287
288 case DIOSTREAMI2C_FSMSTATE_WAITINGTOREAD : break;
289
290 case DIOSTREAMI2C_FSMSTATE_SENDINGDATA : break;
291
292 case DIOSTREAMI2C_FSMSTATE_DISCONNECTING : { DIOSTREAM_XEVENT xevent(diostream,DIOSTREAM_XEVENT_TYPE_DISCONNECTED);
293 xevent.SetDIOStream(diostream);
294 diostream->PostEvent(&xevent);
295
296 diostream->threadconnection->Run(false);
297 diostream->status = DIOSTREAMSTATUS_DISCONNECTED;
298 }
299 break;
300 }
301 }
302 }
303}
