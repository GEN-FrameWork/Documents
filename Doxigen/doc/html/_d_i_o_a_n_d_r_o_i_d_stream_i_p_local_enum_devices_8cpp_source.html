116
117 ifconf.ifc_buf = (char *) ifreq;
118 ifconf.ifc_len = sizeof ifreq;

120 if(ioctl(sock, SIOCGIFCONF, &ifconf) == -1)

122 close(sock);

145 device->GetDescription()->Set(__L(""));
146 device->GetMAC()->Set((XBYTE*)ifreq[c].ifr_hwaddr.sa_data);
