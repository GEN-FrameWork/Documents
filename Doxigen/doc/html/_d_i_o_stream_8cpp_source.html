481bool DIOSTREAM::ReadStr(XSTRING& string)
482{
483 bool foundreturn = false;
484 int stringsize = 0;
485
486 inbuffer->SetBlocked(true);
487
488 for(int c=0;c<(int)inbuffer->GetSize();c++)
489 {
490 if((inbuffer->GetByte(c) =='\n') || (inbuffer->GetByte(c) == '\r'))
491 {
492 stringsize++;
493
494 if(c+1<(int)inbuffer->GetSize())
495 {
496 if((inbuffer->GetByte(c+1) == '\n') || (inbuffer->GetByte(c+1) =='\r')) stringsize++;
497 }
498
499 foundreturn = true;
500
501 break;
502
503 } else stringsize++;
504 }
505
506 inbuffer->SetBlocked(false);
507
508 if(!foundreturn) return false;
509
510 XBYTE* buffer = GEN_NEW XBYTE[stringsize+1];
511 if(!buffer) return false;
512
513 memset(buffer, 0, stringsize+1);
514
515 int br = Read((XBYTE*)buffer, stringsize);
516
517 if(br != stringsize) return false;
518
519 int c=(int)strlen((const char*)buffer);
520 while(c>=0)
521 {
522 if((buffer[c]=='\n')||(buffer[c]=='\r')) buffer[c]=0;
523 c--;
524 }
525
526 string.Empty();
527 string = (char*)buffer;
528
529 GEN_DELETE_ARRAY buffer;
530
531 return true;
532}
