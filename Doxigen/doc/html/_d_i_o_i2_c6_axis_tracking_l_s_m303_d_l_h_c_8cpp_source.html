388bool DIOI2C3DACCELEROMETERLSM303DLHC::Configure()
389{
390 if(!IsInitialized()) return false;
391
392 XBYTE value;
393
394 value = 0x00;
395 value |= 0x01 << 5; // Normal mode
396 value |= 0x07; // X/Y/Z axis enable.
397 if(!WriteRegister(DIOI2CLSM303DLHC_CTRL_REG1_A , value)) return false;
398 value = 0x00;
399 ReadRegister(DIOI2CLSM303DLHC_CTRL_REG1_A, value);
400
401 value = 0x00;
402 value |= 0x01 << 6; // 1: data MSB @ lower address
403 value |= 0x01 << 4; // +/- 4g
404 if(!WriteRegister(DIOI2CLSM303DLHC_CTRL_REG4_A, value)) return false;
405
406 return true;

707bool DIOI2C6AXISTRACKINGLSM303DLHC::Read(DIOI2CLSM303DLHCVECTOR& a, DIOI2CLSM303DLHCVECTOR& c)
708{
709 if(!accelerometer) return false;
710 if(!compass) return false;
711
712 XWORD ax = 0;
713 XWORD ay = 0;
714 XWORD az = 0;
715 XWORD cx = 0;
716 XWORD cy = 0;
717 XWORD cz = 0;
718
719 if(!Read(ax, ay, az, cx, cy, cz)) return false;
720
721 // Perform simple lowpass filtering Intended to stabilize heading despite device vibration such as on a UGV
722 filter_ax += ax - (filter_ax >> DIOI2CLSM303DLHC_FILTERSHIFT);
723 filter_ay += ay - (filter_ay >> DIOI2CLSM303DLHC_FILTERSHIFT);
724 filter_az += az - (filter_az >> DIOI2CLSM303DLHC_FILTERSHIFT);
725
726 a.x = (float)(filter_ax >> DIOI2CLSM303DLHC_FILTERSHIFT);
727 a.y = (float)(filter_ay >> DIOI2CLSM303DLHC_FILTERSHIFT);
728 a.z = (float)(filter_az >> DIOI2CLSM303DLHC_FILTERSHIFT);
729
730 // offset and scale
731 c.x = (cx + offset_x) * scale_x;
732 c.y = (cy + offset_y) * scale_y;
733 c.z = (cz + offset_z) * scale_z;
734
735 return true;
736}

771float DIOI2C6AXISTRACKINGLSM303DLHC::Heading(DIOI2CLSM303DLHCVECTOR& from)
772{
773 DIOI2CLSM303DLHCVECTOR a;
774 DIOI2CLSM303DLHCVECTOR c;
775
776 if(!Read(a, c)) return 0.0f;
777
778 DIOI2CLSM303DLHCVECTOR temp_a;
779
780 temp_a.x = a.x;
781 temp_a.y = a.y;
782 temp_a.z = a.z;
783
784 // normalize
785 DIOI2CLSM303DLHCVECTOR::Normalize(temp_a);
786
787 // compute E and N
788 DIOI2CLSM303DLHCVECTOR e;
789 DIOI2CLSM303DLHCVECTOR n;
790
791 DIOI2CLSM303DLHCVECTOR::Cross(c ,temp_a, e);
792
793 DIOI2CLSM303DLHCVECTOR::Normalize(e);
794
795 DIOI2CLSM303DLHCVECTOR::Cross(temp_a, e , n);
796
797 // compute heading
798 float heading = (float)atan2( DIOI2CLSM303DLHCVECTOR::Dot(e , from), DIOI2CLSM303DLHCVECTOR::Dot(n , from)) * (180/PI);
799 if (heading < 0) heading += 360;
800
801 return float(heading);
802}
