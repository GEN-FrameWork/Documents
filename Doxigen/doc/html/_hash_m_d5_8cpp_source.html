182void HASHMD5::Update(XBYTE* input, XQWORD size)
183{
184 // compute number of bytes mod 64
185 XDWORD index = count[0] / 8 % HASHMD5_BLOCKSIZE;
186
187 // Update number of bits
188 if ((count[0] += (XDWORD)(size << 3)) < (XDWORD)(size << 3)) count[1]++;
189
190 count[1] += (XDWORD)(size >> 29);
191
192 // number of bytes we need to fill in buffer
193 XDWORD firstpart = 64 - index;
194
195 XDWORD i;
196
197 // transform as many times as possible.
198 if(size >= firstpart)
199 {
200 // fill buffer first, transform
201 memcpy(&buffer[index], input, firstpart);
202
203 Transform(buffer);
204
205 // transform chunks of HASHMD5_BLOCKSIZE (64 bytes)
206 for(i = firstpart; i + HASHMD5_BLOCKSIZE <= (XDWORD)size; i += HASHMD5_BLOCKSIZE)
207 {
208 Transform(&input[i]);
209 }
210
211 index = 0;
212
213 } else i = 0;
214
215 // buffer remaining input
216 memcpy(&buffer[index], &input[i], (XDWORD)(size-i));
217}

377 };
378
379 if(finalized) return;
380
381 // Save number of bits
382 XBYTE bits[8];
383
384 Encode(bits, count, 8);
385
386 // pad out to 56 mod 64.
387 XDWORD index = count[0] / 8 % 64;
388 XDWORD padLen = (index < 56) ? (56 - index) : (120 - index);
389
390 Update(padding, padLen);
391
392 // Append size (before padding)
393 Update(bits, 8);
394
395 // Store state in digest
396 Encode(digest, state, 16);
397
398 // Zeroize sensitive information.
399 memset(buffer, 0, sizeof buffer);
400 memset(count, 0, sizeof count);
401
402 finalized=true;
403}
