442bool DIODNSRESOLVER::DNSServer_AddDNSServer(XCHAR* server)
443{
444 if(!server) return false;
445
446 DIODNSRESOLVER_DNSSERVER* DNSserver = GEN_NEW DIODNSRESOLVER_DNSSERVER();
447 if(!DNSserver) return false;
448
449 XSTRING string;
450 XSTRING IPstring;
451 int port = 0;
452
453 string = server;
454 IPstring.AdjustSize(_MAXSTR);
455 string.UnFormat(__L("%s:%d"), IPstring.Get(), &port);
456 IPstring.AdjustSize();
457
458 if(IPstring.IsEmpty())
459 {
460 GEN_DELETE DNSserver;
461
462 return false;
463 }
464
465 DNSserver->GetIP()->Set(IPstring);
466 DNSserver->SetPort(port);
467
468 if(DNSServer_IsIPonList((*DNSserver->GetIP())))
469 {
470 GEN_DELETE DNSserver;

472 return false;
473 }
474
475 return listDNSservers.Add(DNSserver);
