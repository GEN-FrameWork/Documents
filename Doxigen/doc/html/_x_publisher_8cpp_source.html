448bool XPUBLISHER::PostEvent(XEVENT* event, XSUBJECT* subject)
449{
450 if(!on) return false;
451
452 inevent = true;
453
454 if(xmutexevents) xmutexevents->Lock();
455
456 XVECTOR<XOBSERVER*> observers;
457
458 for(XDWORD c=0; c<eventsvector.GetSize(); c++)
459 {
460 XPUBLISHERENTRY* publisherentry = eventsvector.Get(c);
461 if(publisherentry)
462 {
463 if((publisherentry->type == event->GetEventType()) && (publisherentry->subject == subject))
464 {
465 observers.Add(publisherentry->observer);
466 }
467 }
468 }
469
470 if(xmutexevents) xmutexevents->UnLock();
471
472 for(XDWORD c=0; c<observers.GetSize(); c++)
473 {
474 XOBSERVER* observer = observers.Get(c);
475 if(observer) observer->HandleEvent(event);
476 }
477
478 observers.DeleteAll();
479
480 inevent = false;
481
482 return true;
483}
