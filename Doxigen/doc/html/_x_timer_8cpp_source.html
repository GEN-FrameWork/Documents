415bool XTIMER::GetMeasureString(XSTRING& measure,bool large)
416{
417 XQWORD allseconds = GetMeasureSeconds();
418
419 XDWORD years = XDATETIME_SECONDSYEARS(allseconds);
420 XDWORD months = XDATETIME_SECONDSMONTHS(allseconds);
421 XDWORD days = XDATETIME_SECONDSDAYS(allseconds);
422
423 XDWORD hours = XDATETIME_SECONDSHOURS(allseconds);

431 for(int c=0;c<6;c++)

435 switch(c)
436 {
437 case 0: if(!years) break;
438 if(years>1)
439 string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_XX_YEARS), years);
440 else string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_ONE_YEAR));
441 break;
442
443 case 1: if(!months) break;
444 if(months>1)
445 string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_XX_MONTHS), months);
446 else string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_ONE_MONTH));
447 break;
448
449 case 2: if(!days) break;
450 if(days>1)
451 string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_XX_DAYS), days);
452 else string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_ONE_DAY));
453 break;
454
455 case 3: if(large)
456 {
457 if(!hours) break;
458 if(hours>1)
459 string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_XX_HOURS), hours);
460 else string.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_ONE_HOUR));
461 }

464 string.Format(__L("%02d"), hours);

467
468 case 4: if(large)

493

500 if(!measure.IsEmpty()) measure += ", ";

509 measure += string;
510 }
511 }
512 }

516 measure.Format(XT_L(XTRANSLATION_GEN_ID_XTIMER_ZERO_SECONDS));

518 }

520 return true;
