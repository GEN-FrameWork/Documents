106bool DIOSTM32GPIO::SetMode(DIOGPIO_ENTRY* entry, XWORD mode)
107{
108 if(!entry) return false;
109
110 entry->SetMode(mode);
111
112 GPIO_InitTypeDef GPIO_initstruct = {0};
113 GPIO_TypeDef* GPIOgroup = NULL;
114
115 GPIO_initstruct.Pin = entry->GetPin();
116
117 switch(entry->GetGroup())
118 {
119 case DIOGPIO_PINSGROUP_NONE : break;
120 case DIOGPIO_PINSGROUP_A : GPIOgroup = GPIOA; break;
121 case DIOGPIO_PINSGROUP_B : GPIOgroup = GPIOB; break;
122 case DIOGPIO_PINSGROUP_C : GPIOgroup = GPIOC; break;
123 case DIOGPIO_PINSGROUP_D : GPIOgroup = GPIOD; break;
124 case DIOGPIO_PINSGROUP_E : GPIOgroup = GPIOE; break;
125 #if defined(HW_STM32F4XX)
126 case DIOGPIO_PINSGROUP_F : GPIOgroup = GPIOF; break;
127 #else
128 case DIOGPIO_PINSGROUP_F : break;
129 #endif
130 }
131
132 if((mode & DIOGPIO_MODE_INTERRUPT) == DIOGPIO_MODE_INTERRUPT)
133 {
134 GPIO_initstruct.Mode = GPIO_MODE_IT_RISING_FALLING;
135 GPIO_initstruct.Pull = GPIO_NOPULL;
136 }
137 else
138 {
139 GPIO_initstruct.Mode = GPIO_MODE_INPUT;
140 if((mode & DIOGPIO_MODE_INPUT) == DIOGPIO_MODE_INPUT) GPIO_initstruct.Mode = GPIO_MODE_INPUT;
141 if((mode & DIOGPIO_MODE_OUTPUT) == DIOGPIO_MODE_OUTPUT) GPIO_initstruct.Mode = GPIO_MODE_OUTPUT_PP;
142
143 switch(entry->GetPullMode())
144 {
145 case DIOGPIO_PULLMODE_OFF : GPIO_initstruct.Pull = GPIO_NOPULL; break;
146 case DIOGPIO_PULLMODE_UP : GPIO_initstruct.Pull = GPIO_PULLUP; break;
147 case DIOGPIO_PULLMODE_DOWN : GPIO_initstruct.Pull = GPIO_PULLDOWN; break;
148 }
149
150 GPIO_initstruct.Speed = GPIO_SPEED_FREQ_LOW;
151 }
152
153 HAL_GPIO_Init(GPIOgroup, &GPIO_initstruct);
154
155 if((mode & DIOGPIO_MODE_INTERRUPT) == DIOGPIO_MODE_INTERRUPT)
156 {
157 // EXTI interrupt init
158 HAL_NVIC_SetPriority((IRQn_Type)entry->GetIntIRQ(), entry->GetIntPriority(), 0);
159 HAL_NVIC_EnableIRQ((IRQn_Type)entry->GetIntIRQ());
160 }
161
162 return true;
163}
