561{
562 if(!applicationfilepath) return false;
563 if(!desktopname) return false;
564

566 if(!XSTRING::GetSize(desktopname)) return false;
567
568 XPATH directoryapp;

570

572
573 directoryapp.SetOnlyDriveAndPath();
574

577
578 sinfo.cb = sizeof(sinfo);
579 sinfo.lpDesktop = desktopname;

583 , NULL
584 , NULL
585 , TRUE
586 , NORMAL_PRIORITY_CLASS
587 , NULL
588 , directoryapp.Get()

590 , &pinfo);
