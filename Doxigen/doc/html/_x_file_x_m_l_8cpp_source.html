1206{
1207 DeleteAllElements();
1208
1209 XSTRING string;
1210 XSTRING* stringml;
1211 bool quote = false;
1212 bool cdata = false;
1213 int c = 0;
1214
1215 string.Empty();
1216
1217 while(c<GetNLines())
1218 {
1219 stringml = GetLine(c);
1220
1221 for(int d=0;d<(int)stringml->GetSize();d++)
1222 {
1223 XCHAR letter[2] = { 0, 0 };
1224 letter[0] = (*stringml)[d];
1225
1226 if((*stringml)[d]==__C('\"')) quote=!quote;
1227
1228 if((*stringml)[d]==__C('<')&&(!quote))
1229 {
1230 if(string.Find(__L("[CDATA["),false)!=XSTRING_NOTFOUND) cdata=true;
1231
1232 if(!string.IsEmpty() && string.HaveCharacters() && (!cdata))
1233 {
1234 DecodeLine(string,false);
1235 string.Empty();
1236 }
1237
1238 quote = false;
1239 string += letter;
1240 }
1241 else
1242 {
1243 if(((*stringml)[d]==__C('>'))&&(!quote))
1244 {
1245 if(string.Find(__L("]]"),false)!=XSTRING_NOTFOUND) cdata=false;
1246
1247 string += letter;
1248 if(!string.IsEmpty() && string.HaveCharacters() && (!cdata))
1249 {
1250 DecodeLine(string,true);
1251 string.Empty();
1252 }
1253 }
1254 else string += letter;
1255 }
1256 }

1258 c++;

1597int XFILEXML::DecodeLine(XSTRING& string,bool iselement)

1600 XFILEXMLELEMENTTYPELINE typeline = XFILEXMLELEMENTTYPELINE_UNKNOWN;

1605
1606 str.DeleteNoCharacters(XSTRINGCONTEXT_ALLSTRING);
1607
1608 int iini = str.FindCharacter(__C('<'));
1609 int iend = str.FindCharacter(__C('>'),0,true);
1610
1611 if(iini!=XSTRING_NOTFOUND) iini++;
1612 if(iend!=XSTRING_NOTFOUND) iend--;
1613

1624 elementtype = XFILEXMLELEMENTTYPE_COMMENT;

1627 elementtype = XFILEXMLELEMENTTYPE_CDATA;
