551{
552 error = DIOAPPLICATIONUPDATE_ERROR_INVALIDOPERATION;
553
554 if(!webclient) return false;
555
556 bool status = false;
557
558 DIOURL urlfile;
559 XPATH xpath;
560
561 urlfile = url;
562 urlfile.Slash_Add();

569 xpath += DIOAPPLICATIONUPDATE_INIFILE;
570
571 webclient->Set_Port(port);
572

574 if(!status)
575 {

577 {
578 error = DIOAPPLICATIONUPDATE_ERROR_BREAKFORUSER;

581 {
582 error = DIOAPPLICATIONUPDATE_ERROR_DOWNLOADFILE;

587 error = DIOAPPLICATIONUPDATE_ERROR_INVALIDOPERATION;
588
589 XFILEINI* ini = GEN_NEW XFILEINI();

591 {
592 if(ini->Open(xpath))
593 {
594 XSTRING key;
595 XSTRING value;
596 int index;

606
607 index = 0;
608
609 do{ key.Format(__L("%s%d"), DIOAPPLICATIONUPDATE_FILESSECTION_FILES, index);
610
611 ini->ReadValue(DIOAPPLICATIONUPDATE_FILESSECTION, key, value);
612 if(!value.IsEmpty())
613 {
614 XPATH xpathfile(_MAXSTR);
615 int sizefile = 0;
616 XDWORD CRC32 = 0;
617
618 int comma=value.FindCharacter(__C(','));
619 XSTRING name;
620 value.Copy(0,comma,name);
621 value.DeleteCharacters(0,comma+1);
622 value.UnFormat(__L("%d,%08X"), &sizefile, &CRC32);
623 xpathfile.Set(name.Get());
624
625 if(xpathfile.GetSize() && sizefile && CRC32)
626 {
627 DIOAPPLICATIONUPDATE_FILE* updatefile = GEN_NEW DIOAPPLICATIONUPDATE_FILE(xpathfile, CRC32, sizefile);
628 if(updatefile) filestoupdate.Add(updatefile);
629 }
630 }
631
632 index++;
633
634 } while(index<DIOAPPLICATIONUPDATE_MAXINDEXFILES);
635
636 //----------------------------------------------------------------------------------------
637
638 ini->Close();
639 error = DIOAPPLICATIONUPDATE_NOT_ERROR;
640 }
641
642 GEN_DELETE ini;
643 }
644 }
645

647 DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_GETVERSIONAVAILABLE);
648
649 xevent.SetVersionData(&updateversiondata);
650 xevent.SetFilesToUpdate(&filestoupdate);
651 xevent.SetError(error);
652
653 PostEvent(&xevent);
654 //----------------------------------------------------------------------------------------
655
656 XFILE* GEN_XFACTORY_CREATE(file, Create_File())
657 if(file)
658 {
659 file->Erase(xpath);
660 GEN_XFACTORY.Delete_File(file);
661 }
662
663 return status;
664}

683bool DIOAPPLICATIONUPDATE::Download(DIOURL& url, int port, XCHAR* addtargetpath, DIOAPPLICATIONUPDATE_VERSIONDATA& updateversiondata, int timeout, DIOAPPLICATIONUPDATE_ERROR& error)

692 int c;

709
710 urlfile = url;
711 urlfile.Slash_Add();
712 urlfile += updatefile->GetXPathFile()->Get();
713 urlfile.Slash_Normalize();
714
715 xpathfile.Empty();
716
717 if(addtargetpath)
718 {
719 if(addtargetpath[0])

737 if(xdir) xdir->Make(xpath);

743

749 if(updatefile->GetCRC32() != hashCRC32.GetResultCRC32())

765

773

792 if(unzip)
793 {

795 {
796 XPATH xpath;
797 int sourceindex = 0;
798
799 xpath = xpathfile;
800 xpath.SetOnlyDriveAndPath();
801
802 status = true;
803
804 do{ XFILECMPZIP* xfilecmp = unzip->GotoFile(sourceindex);
805 if(!xfilecmp) break;
806
807 XPATH xpathcomplete;
808
809 xpathcomplete = xpath;
810 xpathcomplete += xfilecmp->GetName();
811
812 if(xfilecmp->IsDirectory())

814 xdir->Make(xpathcomplete);
815 status = true;
816 }
817 else
818 {
819 hashCRC32.ResetResult();
820 hashCRC32.Do(xpathcomplete);
821
822 if(xfilecmp->GetCRC32() != hashCRC32.GetResultCRC32())
823 {
824 status = unzip->DecompressFile(sourceindex, xpath);
825
826 } else status = true;
827
828 DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_UNZIPFILE);
829
830 xevent.SetVersionData(&updateversiondata);
831 xevent.SetFilesToUpdate(&filestoupdate);
832 xevent.SetIndexOfOperation(c);
833 xevent.GetActualUnzipFileName()->Set(xfilecmp->GetName());
834 xevent.SetError(status?DIOAPPLICATIONUPDATE_NOT_ERROR:DIOAPPLICATIONUPDATE_ERROR_INVALIDOPERATION);
835
836 PostEvent(&xevent);
837 }
838
839 sourceindex++;
840
841 GEN_DELETE xfilecmp;
842
843 } while(status);
844
845 unzip->Close();
846
847 XFILE* GEN_XFACTORY_CREATE(file, Create_File())
848 if(file)
849 {
850 file->Erase(xpathfile);
851 GEN_XFACTORY.Delete_File(file);
852 }
853 }
854
855 GEN_DELETE unzip;
856 }
857 }
858
859 //-------------------------------------------------------------
860 }
861
862 if(!status) error = DIOAPPLICATIONUPDATE_ERROR_INVALIDOPERATION;
863
864 return status;
865}
