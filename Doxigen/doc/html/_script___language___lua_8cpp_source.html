110int SCRIPT_LNG_LUA::Run(int* returnval)
111{
112 XBUFFER charnamescript;
113 XBUFFER charscript;
114 XSTRING currenttoken;
115
116 namescript.ConvertToASCII(charnamescript);
117 script.ConvertToASCII(charscript);
118
119 int status = LUA_OK;
120
121 status = luaL_loadbuffer(state, charscript.GetPtrChar(), strlen(charscript.GetPtrChar()), charnamescript.GetPtrChar());
122 if(status == LUA_OK )
123 {
124 errorcode = SCRIPT_ERRORCODE_NONE;
125 iscancelexec = false;
126
127 if(!HaveMainFunction())
128 {
129 // Exec to ajust the Stack in Lua with Main Function. lua_getglobal dont work well without this.
130 lua_pcall(state, 0, LUA_MULTRET, 0);
131 }
132 else
133 {
134 lua_getglobal(state, SCRIPT_LNG_LUA_MAINFUNCTIONNAME);
135
136 status = lua_isfunction(state, lua_gettop(state));
137 if(status)
138 {
139 status = lua_pcall(state, 0, 1, 0);
140 }
141 else
142 {
143 lua_setglobal(state, SCRIPT_LNG_LUA_MAINFUNCTIONNAME);
144 status = lua_pcall(state, 0, 1, 0);
145 currenttoken = lua_tostring(state, -1);
146 }
147 }
148
149 int returnvalue = (int)lua_tonumber(state, -1);
150 lua_pop(state, -1);
151
152 if(returnval)
153 {
154 (*returnval) = (int)returnvalue;
155
156 }
157 }
158
159 if(status != LUA_OK)
160 {
161 switch(status)
162 {
163 case LUA_YIELD : errorcode = SCRIPT_LNG_LUA_ERRORCODE_YIELD; break;
164 case LUA_ERRRUN : errorcode = SCRIPT_LNG_LUA_ERRORCODE_RUN; break;
165 case LUA_ERRSYNTAX : errorcode = SCRIPT_LNG_LUA_ERRORCODE_SYNTAX; break;
166 case LUA_ERRMEM : errorcode = SCRIPT_LNG_LUA_ERRORCODE_MEM; break;
167 case LUA_ERRGCMM : errorcode = SCRIPT_LNG_LUA_ERRORCODE_GCMM; break;
168 case LUA_ERRERR : errorcode = SCRIPT_LNG_LUA_ERRORCODE_ERRERR; break;
169 }
170
171 HaveError(currenttoken, errorcode);
172
173 return errorcode;
174 }
175
176 return errorcode;
177}

220bool SCRIPT_LNG_LUA::HaveError(XSTRING& currenttoken, int errorcode)
221{
222 if(errorcode != SCRIPT_ERRORCODE_NONE)
223 {
224 static XCHAR* errorstr[]= { __L("None") ,
225 __L("Internal error") ,
226 __L("Syntax error") ,
227 __L("Insufficient parameters") ,
228 __L("Thread not yield") ,
229 __L("runtime error") ,
230 __L("memory allocation error") ,
231 __L("error while running a __gc metamethod") ,
232 __L("error while running the error handler function") ,
233 };
234 XPATH namefile;
235 XSTRING errorstring;
236 int nline = 0;
237
238 namefile.AdjustSize(_MAXSTR);
239 errorstring.AdjustSize(_MAXSTR);
240 currenttoken.UnFormat(__L("[string \"%s\"]:%d:%s"), namefile.Get(), &nline, errorstring.Get());
241 namefile.AdjustSize();
242 errorstring.AdjustSize();
243
244 //namescript.Set(namefile);
245 //namefile.SetOnlyNamefileExt();
246
247 SCRIPT_XEVENT xevent(this, SCRIPT_XEVENT_TYPE_ERROR);
248
249 xevent.SetError(errorcode);
250 xevent.GetErrorText()->Set(errorstr[errorcode]);
251 xevent.GetCurrentToken()->Set(currenttoken);
252 xevent.SetNLine(nline);
253
254 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("Script [%s] ERROR %d: %s, line %d \"%s\""), namescript.Get(), errorcode, errorstr[errorcode], nline, errorstring.Get());
255
256 PostEvent(&xevent);
257
258 this->errorcode = errorcode;
259 iscancelexec = true;
260 }
261
262 return (errorcode == SCRIPT_ERRORCODE_NONE)?false:true;
263}

275bool SCRIPT_LNG_LUA::HaveMainFunction()
276{
277 XSTRING mainfunctionname;
278 XSTRING script;
279 int start = 0;
280 int end = 0;
281 XSTRING line;
282
283 script = GetScript()->Get();
284 if(script.IsEmpty()) return false;
285
286 mainfunctionname = SCRIPT_LNG_LUA_MAINFUNCTIONNAME;
287
288 int index = GetScript()->Find(mainfunctionname, false);
289 if(index == XSTRING_NOTFOUND) return false;
290
291 for(start=index; start>0; start--)
292 {
293 if((script.Get()[start] == __C('\r')) || (script.Get()[start] == __C('\n')))
294 {
295 break;
296 }
297 }
298
299 for(end=index; end<(int)script.GetSize() ; end++)
300 {
301 if(script.Get()[end] == __C(')'))
302 {
303 break;
304 }
305 }
306
307 script.Copy(start, end+1, line);
308
309 line.DeleteCharacter(__C('\n'));
310 line.DeleteCharacter(__C('\r'));
311 line.DeleteCharacter(__C(' '));
312
313 XSTRING searchmask;
314
315 searchmask.Format(__L("function%s()"), mainfunctionname.Get());
316
317 if(!line.Compare(searchmask)) return true;
318
319 return false;
320}

348int LUA_LibraryCallBack(lua_State* state)
349{
350 lua_Debug ar;
351 XSTRING namefunction;
352 SCRIPT_LNG_LUA* script = *static_cast<SCRIPT_LNG_LUA**>(lua_getextraspace(state));
353
354 if(!script)
355 {
356 return 0;
357 }
358
359 lua_getstack (state, 0, &ar);
360 if(lua_getinfo(state, "Snlu", &ar)) namefunction = ar.name;
361
362 if(namefunction.IsEmpty())
363 {
364 return 0;
365 }
366
367 SCRIPT_LIB_FUNCTION* libfunction = script->GetLibraryFunction(namefunction.Get());
368 if(!libfunction)
369 {
370 return 0;
371 }
372
373 if(!libfunction->GetFunctionLibrary())
374 {
375 return 0;
376 }
377
378 XVECTOR<XVARIANT*> params;
379 XVARIANT returnvalue;
380
381 int nargs = lua_gettop(state);
382
383 for(int c=1; c <= nargs; ++c)
384 {
385 XVARIANT* variant = GEN_NEW XVARIANT();
386 if(variant)
387 {
388 switch(lua_type(state, c))
389 {
390 case LUA_TNONE : break;
391 case LUA_TNIL : /*(*variant) = NULL;*/ break;
392 case LUA_TBOOLEAN : (*variant) = (lua_toboolean(state, c)?true:false); break;
393 case LUA_TLIGHTUSERDATA : break;
394 case LUA_TNUMBER : /*
395 {
396 float number = (float)lua_tonumber(state, c);
397 double intpart = 0.0f;
398 double fracpart = 0.0f;
399
400 fracpart = modf(number, &intpart);
401
402 if(!fracpart)
403 (*variant) = (int)(number);
404 else (*variant) = (float)(number);
405 }
406 */
407
408 (*variant) = (double)lua_tonumber(state, c);
409 break;
410
411
412 case LUA_TSTRING : (*variant) = (char*)(lua_tostring(state, c)); break;
413 case LUA_TTABLE : break;
414 case LUA_TFUNCTION : // XVARIANT dont suppor pointers yet
415 //(*variant) = (void*)(lua_tocfunction(state,c));
416 break;
417 case LUA_TUSERDATA : break;
418 case LUA_TTHREAD : break;
419 case LUA_NUMTAGS : break;
420 }
421
422 params.Add(variant);
423
424 }
425 }
426
427 libfunction->GetFunctionLibrary()(libfunction->GetLibrary(), script, &params, &returnvalue);
428
429 params.DeleteContents();
430 params.DeleteAll();
431
432 // Push the return value on top of the stack. NOTE: We haven't popped the
433 // input arguments to our function. To be honest, I haven't checked if we
434 // must, but at least in stack machines like the JVM, the stack will be
435 // cleaned between each function call.
436
437 int nreturnvalues = 0;
438
439 switch(returnvalue.GetType())
440 {
441 case XVARIANT_TYPE_NULL : lua_pushnil (state); nreturnvalues++; break;
442 case XVARIANT_TYPE_BOOLEAN : lua_pushboolean (state, (int)(returnvalue)); nreturnvalues++; break;
443 case XVARIANT_TYPE_INTEGER : lua_pushnumber(state, (int)(returnvalue)); nreturnvalues++; break;
444 case XVARIANT_TYPE_CHAR : lua_pushnumber(state, (int)(returnvalue)); nreturnvalues++; break;
445 case XVARIANT_TYPE_XCHAR : break;
446 case XVARIANT_TYPE_FLOAT : lua_pushnumber(state, (float)(returnvalue)); nreturnvalues++; break;
447
448 case XVARIANT_TYPE_STRING : { XSTRING stringreturnvalue;
449
450 stringreturnvalue = (const XSTRING&)returnvalue;
451
452 XBUFFER charstr;
453
454 stringreturnvalue.ConvertToASCII(charstr);
455 lua_pushstring(state, charstr.GetPtrChar());
456
457 nreturnvalues++;
458 }
459 break;
460
461 case XVARIANT_TYPE_DATE : break;
462 case XVARIANT_TYPE_TIME : break;
463 case XVARIANT_TYPE_DATETIME : break;
464
465 default : break;
466 }
467
468
469 // Let Lua know how many return values we've passed
470 return nreturnvalues;
471}
