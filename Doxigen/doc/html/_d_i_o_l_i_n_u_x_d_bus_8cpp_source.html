576 if(in) return false;

596 result = value_str_ptr;
597 status = true;

1023void DIOLINUXDBUS::Thread_Signals(void* param)
1024{
1025 DIOLINUXDBUS* dbus = (DIOLINUXDBUS*)param;
1026 if(!dbus) return;
1027
1028 if(!dbus->connection) return;
1029 if(dbus->signals.IsEmpty()) return;
1030
1031 DBusMessage* handler_message = NULL;
1032
1033 dbus_connection_read_write(dbus->connection, 0);
1034 handler_message = dbus_connection_pop_message(dbus->connection);
1035
1036 if(!handler_message)
1037 {
1038 GEN_XSLEEP.Seconds(1);
1039 return;
1040 }
1041
1042 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[DBUS] Message ..."));
1043
1044 DIOLINUXDBUS_SIGNAL* signal = NULL;
1045 bool found_signal = false;
1046
1047 for(XDWORD c=0; c<dbus->signals.GetSize(); c++)
1048 {
1049 signal = dbus->signals.Get(c);
1050 if(signal)
1051 {
1052 // ---------------------------------------------------------------------------------------
1053 // check if the message is a signal from the correct interface and with the correct name
1054
1055 XBUFFER typechar;
1056 XBUFFER namechar;
1057
1058 (*signal->GetType()).ConvertToASCII(typechar);
1059 (*signal->GetName()).ConvertToASCII(namechar);
1060
1061 if(dbus_message_is_signal(handler_message, typechar.GetPtrChar(), namechar.GetPtrChar()))
1062 {
1063 found_signal = true;
1064 }
1065
1066 if(found_signal)
1067 {
1068 break;
1069 }
1070 else
1071 {
1072 signal = NULL;
1073 }
1074 }
1075 }
1076
1077
1078 if(found_signal && signal)
1079 {
1080 DIOLINUXDBUS_MESSAGE message;
1081
1082 message.SetHandler(handler_message);
1083
1084 DIOLINUXDBUS_XEVENT xevent(dbus, DIOLINUXDBUS_XEVENT_TYPE_SIGNAL);
1085
1086 xevent.SetMessage(&message);
1087 xevent.SetSignal(signal);
1088
1089 dbus->PostEvent(&xevent);
1090
1091 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[DBUS] Signal produced [%s] [%s] ..."), signal->GetType()->Get(), signal->GetName()->Get());
1092 }
1093 else
1094 {
1095 dbus_message_unref(handler_message);
1096 }
1097}
