104bool DIOALERTSUDPSERVER::Ini(XCHAR* URL, int port)
105{
106 bool status = false;
107
108 UDPdiostreamcfg = GEN_NEW DIOSTREAMUDPCONFIG();
109 if(!UDPdiostreamcfg) return false;
110
111 UDPdiostreamcfg->SetMode(DIOSTREAMMODE_SERVER);
112 UDPdiostreamcfg->GetRemoteURL()->Set(URL);
113 UDPdiostreamcfg->SetRemotePort(port);
114 UDPdiostreamcfg->SetIsUsedDatagrams(true);
115
116 UDPdiostream = GEN_NEW DIOSTREAMUDPACKNOWLEDGE(UDPdiostreamcfg);
117 if(UDPdiostream)
118 {
119 status = UDPdiostream->Open();
120 if(status)
121 {
122 GEN_XFACTORY_CREATE(xmutexalert, Create_Mutex())
123 if(xmutexalert)
124 {
125 threadread = CREATEXTHREAD(XTHREADGROUPID_DIOALERTS, __L("DIOALERTSUDPSERVER::Ini"), ThreadReadFunction, (void*)this);
126 if(threadread) status = threadread->Ini();
127 }
128 }
129 }
130
131 return status;
132}

357bool DIOALERTSUDPSERVER::ReceivedEvents()
358{
359 if(!UDPdiostream) return false;
360
361 XBUFFER data;
362 XSTRING address;
363 XWORD port;
364 bool status = false;
365
366 if(UDPdiostream->ReadDatagram(address, port, data))
367 {
368 HASHCRC32 hashcrc32;
369 XDWORD CRC32;
370 DIOALERT* alert = NULL;
371
372 if(data.GetSize())
373 {
374 alert = GEN_NEW DIOALERT();
375 if(alert)
376 {
377 XSTRING string;
378 XDWORD dataxdword;
379 XDWORD sizestring;
380 XDWORD applicationversion;
381 XDWORD applicationsubversion;
382 XDWORD applicationsubversionerr;
383
384 data.Extract(CRC32, (data.GetSize()-sizeof(XDWORD)));
385
386 hashcrc32.ResetResult();
387 hashcrc32.Do(data);
388
389 if(CRC32 == hashcrc32.GetResultCRC32())
390 {
391 // Date time
392 data.Extract(dataxdword); alert->GetDateTime()->SetDay(dataxdword); // Day
393 data.Extract(dataxdword); alert->GetDateTime()->SetMonth(dataxdword); // Month
394 data.Extract(dataxdword); alert->GetDateTime()->SetYear(dataxdword); // Year
395 data.Extract(dataxdword); alert->GetDateTime()->SetHours(dataxdword); // Hours
396 data.Extract(dataxdword); alert->GetDateTime()->SetMinutes(dataxdword); // Minutes
397 data.Extract(dataxdword); alert->GetDateTime()->SetSeconds(dataxdword); // Seconds
398
399 data.Extract(sizestring); data.Extract(string, 0, sizestring); alert->Application_GetID()->Set(string); // Aplication String
400 data.Extract(applicationversion);
401 data.Extract(applicationsubversion);
402 data.Extract(applicationsubversionerr);
403
404 alert->Application_SetVersion(applicationversion, applicationsubversion, applicationsubversionerr);
405
406 data.Extract(dataxdword); alert->SetID(dataxdword); // ID
407 data.Extract(dataxdword); alert->SetType(dataxdword); // Type
408 data.Extract(dataxdword); alert->SetLevel((DIOALERTLEVEL)dataxdword); // Level
409 data.Extract(sizestring); data.Extract(string, 0, sizestring); alert->GetOrigin()->Set(string); // Origin
410 data.Extract(sizestring); data.Extract(string, 0, sizestring); alert->GetTitle()->Set(string); // Title
411 data.Extract(sizestring); data.Extract(string, 0, sizestring); alert->Get_Message()->Set(string); // Message
412
413 status = alerts.Add(alert);
414 }
415
416 if(!status) GEN_DELETE alert;
417 }
418 }
419 }
420
421 return status;
422}
