199bool DIOTEMHUMSENSORAM2301::ReadDirect(float& temperature, float& humidity)
200{
201 if(!xtimer) return false;
202 if(!IsInitialized()) return false;
203
204 int timeelapsedlow;
205 int timeelapsedhigh;
206 XBYTE value[5] = { 0, 0, 0, 0, 0 };
207
208 // Leave it high for a while
209 GEN_DIOGPIO.SetMode(pindata, DIOGPIO_MODE_OUTPUT);
210 GEN_DIOGPIO.SetValue(pindata, DIOTEMHUMSENSORAM2301_HIGH);
211 GEN_XSLEEP.MilliSeconds(1);
212
213 // Set it low to give the start signal
214 GEN_DIOGPIO.SetValue(pindata, DIOTEMHUMSENSORAM2301_LOW);
215 GEN_XSLEEP.MilliSeconds(1);
216 GEN_DIOGPIO.SetValue(pindata, DIOTEMHUMSENSORAM2301_HIGH);
217
218 // Now set the pin high to let the sensor start communicating
219 GEN_DIOGPIO.SetMode(pindata, DIOGPIO_MODE_INPUT);
220
221 GEN_XSLEEP.NanoSeconds(30);
222
223 // Read ACK
224 WaitTo(false, 50);
225
226 WaitTo(true , 100);
227 WaitTo(false, 100);
228
229 // Read values
230 for(int d=0; d<5; d++)
231 {
232 value[d] = 0x00;
233
234 for(int c=0; c<8; c++)
235 {
236 WaitTo(false, 50);
237 WaitTo(true , 100, &timeelapsedlow);
238 WaitTo(false, 100, &timeelapsedhigh);
239
240 value[d]|=(timeelapsedlow>timeelapsedhigh?0:1);
241
242 if(c<7) value[d] <<= 1;
243 }
244 }
245
246
247 // Final State of SDA
248 GEN_DIOGPIO.SetMode(pindata, DIOGPIO_MODE_OUTPUT);
249 GEN_DIOGPIO.SetValue(pindata, DIOTEMHUMSENSORAM2301_HIGH);
250
251 GEN_XSLEEP.MilliSeconds(1);
252
253 // Verify checksum
254 XBYTE x = value[0] + value[1] + value[2] + value[3];
255 if(x != value[4]) return false;
256
257 humidity = (float)(((XWORD)value[0] << 8) | (XWORD)value[1]);
258 humidity /= 10.0f;
259
260 temperature = (float)(((XWORD)value[2] << 8) | (XWORD)value[3]);
261 temperature /= 10.0f;
262
263 if((humidity > 100.0) || (humidity <= 0.0) || (temperature > 80.0) || (temperature < -40.0)) return false;
264
265 return true;
266}

346void DIOTEMHUMSENSORAM2301::ThreadRunFunction(void* param)
347{
348 DIOTEMHUMSENSORAM2301* sensor = (DIOTEMHUMSENSORAM2301*)param;
349 if(!sensor) return;
350
351 if(!sensor->IsInitialized()) return;
352
353 float temperature;
354 float humidity;
355
356 bool status = sensor->ReadDirect(temperature, humidity);
357 if(status)
358 {
359 if(sensor->xmutexread) sensor->xmutexread->Lock();
360
361 if(sensor->nreads>1)
362 {
363 if((abs((int)sensor->temperaturecache - (int)temperature)>10) || (abs((int)sensor->humiditycache - (int)humidity)>10)) status = false;
364 }
365
366 if(status)
367 {
368 sensor->temperaturecache = temperature;
369 sensor->humiditycache = humidity;
370 sensor->nreads++;
371 }
372
373 if(sensor->xmutexread) sensor->xmutexread->UnLock();
374 }
375
376 if(sensor->xtimer) GEN_XSLEEP.MilliSeconds(500);
377
378}
