120bool DIOI2CPWMCONTROLERPCA9685::SetPWMFrequency(float frequency)
121{
122 if(!IsInitialized()) return false;
123
124 float prescaleval;
125
126 prescaleval = 25000000; // 25MHz
127 prescaleval /= 4096; // 12 bits
128 prescaleval /= frequency;
129 prescaleval -= 1;
130
131 XBYTE prescale = (XBYTE)floor(prescaleval + 0.5f);
132 XBYTE oldmode;
133 bool status;
134
135 status = ReadCommand(DIOI2CPWMCONTROLERPCA9685_MODE1, oldmode);
136 if(!status) return false;
137
138 XBYTE newmode = (oldmode&0x7F) | 0x10; // sleep
139
140 status = SendCommand(DIOI2CPWMCONTROLERPCA9685_MODE1 , newmode); // go to sleep
141 if(!status) return false;
142
143 status = SendCommand(DIOI2CPWMCONTROLERPCA9685_PRESCALE , prescale); // set the prescaler
144 if(!status) return false;
145
146 status = SendCommand(DIOI2CPWMCONTROLERPCA9685_MODE1 , oldmode);
147 if(!status) return false;
148
149 xtimerout->SetMilliSeconds(50);
150
151 return SendCommand(DIOI2CPWMCONTROLERPCA9685_MODE1, (oldmode | 0x80));
152}

169bool DIOI2CPWMCONTROLERPCA9685::SetPWM(XBYTE channel, XWORD on, XWORD off, bool wait)
170{
171 if(!IsInitialized()) return false;
172
173 if(!SendCommand(DIOI2CPWMCONTROLERPCA9685_LED0_ON_L + (4*channel), (on & 0xFF), wait)) return false;
174 if(!SendCommand(DIOI2CPWMCONTROLERPCA9685_LED0_ON_H + (4*channel), (on >> 0x08), wait)) return false;
175 if(!SendCommand(DIOI2CPWMCONTROLERPCA9685_LED0_OFF_L + (4*channel), (off & 0xFF), wait)) return false;
176 if(!SendCommand(DIOI2CPWMCONTROLERPCA9685_LED0_OFF_H + (4*channel), (off >> 0x08), wait)) return false;
177
178 return true;
179}
