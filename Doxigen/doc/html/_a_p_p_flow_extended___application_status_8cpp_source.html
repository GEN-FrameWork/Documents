312bool APPFLOWEXTENDED_APPLICATIONSTATUS::Update()
313{
314 if(checkresourceshardware && cfg)
315 {
316 static bool iniappcheck = false;
317
318 if(!iniappcheck)
319 {
320 checkresourceshardware->Ini(cfg);
321 iniappcheck = true;
322 }
323
324 bool ischecktotalCPUusage = cfg->CheckResourcesHardware_GetTotalCPUUsageCheckCadence()?true:false;
325 bool ischeckappCPUusage = cfg->CheckResourcesHardware_GetAppCPUUsageCheckCadence()?true:false;
326
327 if(ischecktotalCPUusage || ischeckappCPUusage)
328 {
329 averange.Empty();
330
331 if(ischecktotalCPUusage)
332 {
333 averange.AddFormat(__L("avg. %d%% (max. %d%%)") , checkresourceshardware->GetCPUTotalCPUUsageAverange()
334 , checkresourceshardware->GetCPUTotalCPUUsageMax());
335 }
336
337 if(ischeckappCPUusage)
338 {
339 if(ischecktotalCPUusage)
340 {
341 averange.AddFormat(__L(", "));
342 }
343
344 if(ischeckappCPUusage)
345 {
346 averange.AddFormat(__L("app %s avg. %d%% (max. %d%%)") , cfg->CheckResourcesHardware_GetAppCPUUsageProcessName()->Get()
347 , checkresourceshardware->GetCPUAppCPUUsageAverange()
348 , checkresourceshardware->GetCPUAppCPUUsageMax());
349 }
350 }
351 }
352 }
353
354 GEN_XSYSTEM.GetMemoryInfo(memory_total, memory_free);
355 memory_freepercent = GEN_XSYSTEM.GetFreeMemoryPercent();
356
357 XDATETIME* datetime = GEN_XFACTORY.CreateDateTime();
358 if(datetime)
359 {
360 datetime->Read();
361 datetime->GetDateTimeToString(XDATETIME_FORMAT_STANDARD | XDATETIME_FORMAT_TEXTMONTH | XDATETIME_FORMAT_ADDDAYOFWEEK, currentdate);
362
363 GEN_XFACTORY.DeleteDateTime(datetime);
364 }
365
366 if(xtimeroperatingtime)
367 {
368 xtimeroperatingtime->GetMeasureString(operatingtime, true);
369 }
370
371 return true;
372}
