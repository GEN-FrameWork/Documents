331bool GRPVIEWPORT::SetCanvasPosition(float x, float y)
332{
333 bool status = true;
334
335 canvas_x = x;
336 canvas_y = y;
337
338 if(canvas_x < 0)
339 {
340 canvas_x = 0;
341 status = false;
342 }
343
344 if(canvas_y < 0)
345 {
346 canvas_y = 0;
347 status = false;
348 }
349
350 if(canvas)
351 {
352 if((canvas_x + width) > canvas->GetWidth())

355 status = false;
356 }
357
358 if((canvas_y + height) > canvas->GetHeight())
359 {
360 canvas_y = canvas->GetHeight() - height;
361 status = false;
362 }
363 }
364 else
365 {
366 status = false;
367 }
