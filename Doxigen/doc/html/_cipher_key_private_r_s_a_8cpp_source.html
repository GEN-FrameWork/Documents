185bool CIPHERKEYPRIVATERSA::Check(CIPHERKEYPUBLICRSA& publickey)
186{
187 XMPINTEGER PQ;
188 XMPINTEGER DE;
189 XMPINTEGER P1;
190 XMPINTEGER Q1;
191 XMPINTEGER H;
192 XMPINTEGER I;
193 XMPINTEGER G;
194 XMPINTEGER G2;
195 XMPINTEGER L1;
196 XMPINTEGER L2;
197 XMPINTEGER publicmodulus;
198 XMPINTEGER publicexponent;
199
200 if(!publickey.Check()) return false;
201
202 if((!prime1factor.GetLimbs()) || (!prime2factor.GetLimbs()) || (!exponent.GetLimbs())) return false;
203
204 publickey.Get(publicmodulus, publicexponent);
205
206 PQ.Ini();
207 DE.Ini();
208 P1.Ini();
209 Q1.Ini();
210 H.Ini();
211 I.Ini();
212 G.Ini();
213 G2.Ini();
214 L1.Ini();
215 L2.Ini();
216
217 if(!PQ.Multiplication(&prime1factor, &prime2factor)) return false;
218 if(!DE.Multiplication(&exponent, &publicexponent)) return false;
219 if(!P1.SubtractionSigned(&prime1factor, 1)) return false;
220 if(!Q1.SubtractionSigned(&prime2factor, 1)) return false;
221 if(!H.Multiplication(&P1, &Q1)) return false;
222 if(!G.GreatestCommonDivisor(&G, &publicexponent, &H)) return false;
223 if(!G2.GreatestCommonDivisor(&G2, &P1, &Q1)) return false;
224 if(!L1.Division(&L1, &L2, &H, &G2 )) return false;
225 if(!I.Module(&I, &DE, &L1)) return false;
226
227 // Check for a valid PKCS1v2 private key
228 if((PQ.CompareSignedValues(publicmodulus) != 0) ||
229 (L2.CompareSignedValues(0) != 0) ||
230 (I.CompareSignedValues(1) != 0) ||
231 (G.CompareSignedValues(1) != 0)) return false;
232
233 PQ.End();
234 DE.End();
235 P1.End();
236 Q1.End();
237 H.End();
238 I.End();
239 G.End();
240 G2.End();
241 L1.End();
242 L2.End();
243
244 return true;
245}
