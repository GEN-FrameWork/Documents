178bool SQLITE_DATABASE::Rollback(XCHAR* savepoint)
179{
180 if(IsTransactionStarted())
181 {
182 if(savepoint == NULL)
183 {
184 int rc = sqlite3_exec(sqlite3database, "ROLLBACK", 0, 0, 0);
185 transactionstarted = false;
186
187 return (rc==SQLITE_OK);

189 else
190 {
191 DB_SQL_STRING text;

195
196 text.ConvertToASCII(rollbacktext);

198 transactionstarted = false;

200 return (rc==SQLITE_OK);
201 }
202 }

304bool SQLITE_DATABASE::GetTables()
305{
306 DB_SQL_QUERY* query = CreateQuery();
307 if(!query) return false;
308
309 bool status = true;
310
311 tables.DeleteContents();
312 tables.DeleteAll();
313
314 query->Set(__L("SELECT name FROM sqlite_master WHERE type='table'"));
315 if(query->Execute())
316 {
317 DB_SQL_RESULT* result=query->GetResult();
318 if(result!=NULL) //hay un result
319 {
320 result->GetNumRows();
321 while(result->HasNext())
322 {
323 result->ProcessRow();
324
325 DB_SQL_ROW* row = result->GetRow();
326 for(XDWORD e=0; e<result->GetNumColumns(); e++)
327 {
328 if(row->Get(e).GetType()==(XVARIANT_TYPE)DB_SQL_VARIANT_TYPE_STRING)
329 {
330 DB_SQL_STRING* tablename = GEN_NEW DB_SQL_STRING();
331 if(tablename)
332 {
333 tablename->Set((XCHAR*)row->Get(0));
334 tables.Add(tablename);
335 }
336 }
337 }
338
339 result->Next();
340 }
341 }
342 }
343 else
344 {
345 Error(__L("Unable to obtain Tables"));
346 status = false;
347 }
348
349 delete(query);
350
351 return status;
352}
