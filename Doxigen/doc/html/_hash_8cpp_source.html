202bool HASH::Do(XPATH& xpath, XQWORD size, XQWORD pos)
203{
204 XFILE* xfile;
205 bool status = false;
206
207 GEN_XFACTORY_CREATE(xfile, Create_File())
208 if(!xfile) return false;
209
210 ResetResult();
211
212 if(xfile->Open(xpath, true))
213 {
214 XQWORD _size;
215
216 if(size == HASHALLFILESIZE) _size = xfile->GetSize(); else _size = size;
217
218 status = Do(xfile, _size, pos);
219
220 xfile->Close();
221 }
222
223 if(xfile) GEN_XFACTORY.Delete_File(xfile);
224
225 return status;
226}

242bool HASH::Do(XFILE* xfile, XQWORD size, XQWORD pos)
243{
244 if(!xfile) return false;
245 if(!xfile->IsOpen()) return false;
246
247 XBUFFER xbuffer;
248 XQWORD block = (((size==HASHALLFILESIZE)?xfile->GetSize():size) / HASHMAXFILESIZEBUFFER);
249 XDWORD mod = (((size==HASHALLFILESIZE)?xfile->GetSize():size) % HASHMAXFILESIZEBUFFER);
250 bool status = true;
251
252 if(!xfile->SetPosition(pos)) return false;
253
254 xbuffer.Resize((XDWORD)HASHMAXFILESIZEBUFFER);
255
256 for(XDWORD c=0;c<block;c++)
257 {
258 if(xfile->Read(xbuffer.Get(), HASHMAXFILESIZEBUFFER))
259 {
260 if(!Do(xbuffer))
261 {
262 status = false;
263 break;
264 }
265 }
266 else
267 {
268 status = false;
269 break;
270 }
271 }
272
273 if(status)
274 {
275 xbuffer.Resize(mod);
276 if(xfile->Read(xbuffer.Get(),mod))
277 {
278 if(!Do(xbuffer))
279 {
280 status = false;
281 }
282 }
283 else
284 {
285 status = false;
286 }
287 }
288
289 return status;
290}
