920bool XPROCESSMANAGER::Application_Terminate(XCHAR* processname, XDWORD exitcode)
921{
922 XVECTOR<XPROCESS*> apprunning;
923 bool status = false;
924
925 if(!GEN_XPROCESSMANAGER.Application_GetRunningList(apprunning)) return false;
926
927 for(XDWORD c=0; c<apprunning.GetSize(); c++)
928 {
929 XPROCESS* xprocess = apprunning.Get(c);
930 if(xprocess)
931 {
932 if(!xprocess->GetName()->Compare(processname, true))
933 {
934 status = Application_Terminate((*xprocess), exitcode);
935 break;
936 }
937 }
938 }
939
940 apprunning.DeleteContents();
941 apprunning.DeleteAll();
942
943 return status;
944}
