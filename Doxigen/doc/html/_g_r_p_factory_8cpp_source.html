243GRPCANVAS* GRPFACTORY::CreateCanvas(GRPPROPERTIES* properties)
244{
245 GRPCANVAS* canvas = NULL;
246 GRPPROPERTYMODE mode = GRPPROPERTYMODE_XX_UNKNOWN;
247
248 if(!properties)
249 {
250 return NULL;
251 }
252
253 mode = properties->GetMode();
254 if(mode == GRPPROPERTYMODE_XX_UNKNOWN)
255 {
256 return NULL;
257 }
258
259 if(!properties->GetWidth() || !properties->GetHeight())
260 {
261 return NULL;
262 }
263
264 switch(mode)
265 {
266 case GRPPROPERTYMODE_08_INDEX : break;
267 case GRPPROPERTYMODE_16_RGBA_4444 : break;
268 case GRPPROPERTYMODE_16_RGBA_5551 : canvas = (GRPCANVAS*)GEN_NEW GRPCANVASAGG<agg::pixfmt_rgb555, agg::rgba8>(); break;
269 case GRPPROPERTYMODE_16_RGB_565 : canvas = (GRPCANVAS*)GEN_NEW GRPCANVASAGG<agg::pixfmt_rgb565, agg::rgba8>(); break;
270 case GRPPROPERTYMODE_16_SRGB_565 : break;
271 case GRPPROPERTYMODE_24_RGB_888 : canvas = (GRPCANVAS*)GEN_NEW GRPCANVASAGG<agg::pixfmt_rgb24 , agg::rgba8>(); break;
272 case GRPPROPERTYMODE_24_BGR_888 : canvas = (GRPCANVAS*)GEN_NEW GRPCANVASAGG<agg::pixfmt_bgr24 , agg::rgba8>(); break;
273 case GRPPROPERTYMODE_32_RGBA_8888 : canvas = (GRPCANVAS*)GEN_NEW GRPCANVASAGG<agg::pixfmt_rgba32, agg::rgba8>(); break;
274 case GRPPROPERTYMODE_32_BGRA_8888 : canvas = (GRPCANVAS*)GEN_NEW GRPCANVASAGG<agg::pixfmt_bgra32, agg::rgba8>(); break;
275 default : break;
276 }
277
278 if(canvas)
279 {
280 canvas->CopyPropertysFrom(properties);
281 }
282
283 return canvas;
284}
