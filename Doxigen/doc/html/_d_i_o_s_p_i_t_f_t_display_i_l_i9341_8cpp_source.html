373bool DIOSPITFTDISPLAYILI9341::TFT_RotateRGB(int rotate, XBYTE bgr)
374{
375 #define MEM_Y (7) /* MY row address order */
376 #define MEM_X (6) /* MX column address order */
377 #define MEM_V (5) /* MV row / column exchange */
378 #define MEM_L (4) /* ML vertical refresh order */
379 #define MEM_H (2) /* MH horizontal refresh order */
380 #define MEM_BGR (3) /* RGB-BGR Order */
381
382 bool status = false;
383
384 switch(rotate)
385 {
386 case 0 : status = WriteRegister(0x36, 1, (1 << MEM_X) | (bgr << MEM_BGR)); break;
387 case 90 : status = WriteRegister(0x36, 1, (1 << MEM_Y) | (1 << MEM_X) |(1 << MEM_V) | (bgr << MEM_BGR)); break;
388 case 180 : status = WriteRegister(0x36, 1, (1 << MEM_Y) | (bgr << MEM_BGR)); break;
389 case 270 : status = WriteRegister(0x36, 1, (1 << MEM_V) | (1 << MEM_L) | (bgr << MEM_BGR)); break;
390 default : break;
391 }
392
393 return status;
394}
