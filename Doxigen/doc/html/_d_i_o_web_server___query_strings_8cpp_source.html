457int DIOWEBSERVER_QUERYSTRINGS::GetParamsFromURL(XCHAR* url)
458{
459 DIOURL _url;
460 int start;
461 int separator;
462 int end;
463 XSTRING param;
464
465 DelAllParam();
466
467 _url = url;
468
469 start = _url.Find(__L("?"),true);
470 if(start != XSTRING_NOTFOUND)
471 {
472 XSTRING ID;
473 XSTRING value;
474
475 start++;
476
477 do{
478 value.Empty();
479 end = _url.Find(__L("&"),true,start);
480 if(end == XSTRING_NOTFOUND)
481 _url.Copy(start, param);
482 else _url.Copy(start, end, param);
483
484 separator = param.Find(__L("="),true);
485 if(separator == XSTRING_NOTFOUND)
486 {
487 ID = param;
488 }
489 else
490 {
491 param.Copy(0 , separator,ID);
492 param.Copy(separator+1, value);
493 }
494
495 AddParam(ID, value);
496
497 start = end+1;
498
499 } while(end != XSTRING_NOTFOUND);
500 }
501
502
503 return GetNParams();
504}
