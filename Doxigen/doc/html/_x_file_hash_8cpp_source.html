108bool XFILEHASH::Open(XPATH& pathname, bool readonly, bool checkhash, bool checkversion)
109{
110 if(!hash) return false;
111 if(!file) return false;
112
113 if(!hash->GetDefaultSize()) return false;
114
115 if(file->Exist(pathname)!=true) return false;
116
117 bool status = file->Open(pathname,readonly);
118
119 XSTRING IDini(16);
120 XWORD _ID = 0;
121 XWORD _type = 0;
122 XWORD _version = 0;
123
124 if(!file->Read((XBYTE*)IDini.Get(), XSTRING::GetSize(XFILE_ID)*sizeof(XWORD)+1)) return false;
125
126 IDini.AdjustSize();
127
128 if(!file->Read((XBYTE*)&_ID, sizeof(XWORD))) return false;
129 if(_ID!=ID) return false;
130
131 if(!file->Read((XBYTE*)&_type, sizeof(XWORD))) return false;
132 if(_type!=type) return false;
133
134 if(!file->Read((XBYTE*)&_version,sizeof(XWORD))) return false;
135 if(checkversion)
136 {
137 if(version!=_version) return false;
138 }
139
140 version = _version;
141
142 XDWORD sizeIDstring = 0;
143
144 if(!file->Read((XBYTE*)&sizeIDstring, sizeof(XDWORD))) return false;
145 if(sizeIDstring)
146 {
147 XWORD* IDstr = GEN_NEW XWORD[sizeIDstring+1];
148 if(IDstr)
149 {
150 memset((XBYTE*)IDstr, 0, sizeof(XWORD)*(sizeIDstring+1));
151
152 if(!file->Read((XBYTE*)IDstr, sizeof(XWORD)*sizeIDstring))
153 {
154 GEN_DELETE_ARRAY IDstr;
155 return false;
156 }
157
158 IDstring.Set((XWORD*)IDstr);
159
160 GEN_DELETE_ARRAY IDstr;
161 }
162 }
163
164 XBUFFER hashbuffer((XDWORD)hash->GetDefaultSize(), true);
165 if(!file->Read((XBYTE*)hashbuffer.Get(), hash->GetDefaultSize())) return false;
166
167 file->GetPosition(dataposition);
168
169 XQWORD size = file->GetSize() - GetDataPosition();
170
171 hash->ResetResult();
172
173 if(!hash->Do(file, size, GetDataPosition())) return false;
174
175 if(checkhash)
176 {
177 if(!hashbuffer.Compare(hash->GetResult()))
178 {
179 return false;
180 }
181 }
182
183 if(readonly) hashisupdate = false;
184
185 file->SetPosition(dataposition);
186
187 return status;
188}

476bool XFILEHASH::UpdateHash()
477{

481 file->Flush();
482
483 hash->ResetResult();
484

488
