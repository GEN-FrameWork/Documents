740bool XFILECSV::CreateRecordOfLine(XSTRING& line, XFILECSV_RECORD* record)
741{
742 if(line.IsEmpty()) return false;
743
744 int start = 0;
745 int end = 0;
746
747 do{ end = line.Find(__L(";"),true,start);
748 if(end == XSTRING_NOTFOUND)
749 {
750 end = line.Find(__L(","),true,start);
751 if(end != XSTRING_NOTFOUND) SetSeparator(__C(','));
752
753 } else SetSeparator(__C(';'));
754
755 XSTRING element;
756
757 if(end != XSTRING_NOTFOUND)
758 {
759 line.Copy(start,end,element);
760
761 record->AddElement(element);
762
763 start = end+1;
764 }
765 else
766 {
767 line.Copy(start,element);
768 record->AddElement(element);
769
770 break;
771 }
772
773 } while(1);
774
775 return true;
776}
