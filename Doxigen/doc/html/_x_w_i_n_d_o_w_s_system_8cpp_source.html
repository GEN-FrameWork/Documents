135double XWINDOWSSYSTEM_CPUUSAGE::GetTotalCpuUsage()
136{
137 #ifndef BUILDER
138 FILETIME idletime;
139 FILETIME kerneltime;
140 FILETIME usertime;
141
142 if(GetSystemTimes(&idletime, &kerneltime, &usertime) == 0)
143 {
144 return -1;
145 }
146
147 ULONGLONG sysidlediff = SubtractTimes(idletime , lastsysidletime);
148 ULONGLONG syskerneldiff = SubtractTimes(kerneltime , lastsyskerneltime);
149 ULONGLONG sysuserdiff = SubtractTimes(usertime , lastsysusertime);
150
151 ULONGLONG systotal = syskerneldiff + sysuserdiff;
152 ULONGLONG totalsystime = systotal + sysidlediff;
153
154 lastsysidletime = idletime;
155 lastsyskerneltime = kerneltime;
156 lastsysusertime = usertime;
157
158 if(totalsystime == 0)
159 {
160 return -1;
161 }
162
163 return (systotal * 100.0) / totalsystime;
164
165 #else
166 return 0.0f;
167 #endif
168}

182double XWINDOWSSYSTEM_CPUUSAGE::GetProcessCpuUsage(DWORD processID)
183{
184 #ifndef BUILDER
185 HANDLE hprocess = OpenProcess(PROCESS_QUERY_INFORMATION | PROCESS_VM_READ, FALSE, processID);
186 if(hprocess == NULL)
187 {
188 return -1;
189 }
190
191 FILETIME proccreationtime;
192 FILETIME procexittime;
193 FILETIME prockerneltime;
194 FILETIME procusertime;
195
196 if(GetProcessTimes(hprocess, &proccreationtime, &procexittime, &prockerneltime, &procusertime) == 0)
197 {
198 CloseHandle(hprocess);
199 return -1;
200 }
201
202 ULONGLONG prockerneldiff = SubtractTimes(prockerneltime, lastprockerneltime);
203 ULONGLONG procuserdiff = SubtractTimes(procusertime, lastprocusertime);
204
205 lastprockerneltime = prockerneltime;
206 lastprocusertime = procusertime;
207
208 CloseHandle(hprocess);
209
210 if(!lastsystemtime)
211 {
212 lastsystemtime = GetTickCount64();
213 lastprocesstime = prockerneldiff + procuserdiff;
214
215 return -1;
216 }
217
218 ULONGLONG currenttime = GetTickCount64();
219 ULONGLONG timediff = currenttime - lastsystemtime;
220 ULONGLONG processtime = prockerneldiff + procuserdiff;
221 ULONGLONG processtimediff = processtime - lastprocesstime;
222
223 lastsystemtime = currenttime;
224 lastprocesstime = processtime;
225
226 return (processtimediff * 100.0) / (timediff * GetNumberOfProcessors());
227 #else
228 return 0.0f;
229 #endif
230}

633 #ifndef BUILDER

639 LARGE_INTEGER newperftime100nsec;
640 XSTRING processstr;
641 XWINDOWSSYSTEM_CPUUSAGESTATUS* cus = NULL;
642
643 newperftime100nsec.HighPart = 0;
644 newperftime100nsec.LowPart = 0;
645
646 objectindex = XWINDOWSSYSTEM_PROCESSOR_OBJECT_INDEX;
647 cpuusageindex = XWINDOWSSYSTEM_PROCESSOR_TIME_COUNTER_INDEX;
648
649 processstr = __L("_Total");
650
651 cus = AddCPUUsageStatus(processstr.Get());
652 if(!cus) return 0;
653
654 newvalue = perfcounters.GetCounterValue(&perfdata, objectindex, cpuusageindex, (LPCTSTR)processstr.Get());
655
656 newperftime100nsec = perfdata->PerfTime100nSec;
657
658 if(cus->firsttime)
659 {
660 cus->firsttime = false;
661 cus->oldvalue = newvalue;
662 cus->oldperftime100nsec = newperftime100nsec;
663
664 return 0;
665 }
666
667 LONGLONG lnValueDelta = newvalue - cus->oldvalue;
668 double DeltaPerfTime100nSec = (double)newperftime100nsec.QuadPart - (double)cus->oldperftime100nsec.QuadPart;
669
670 cus->oldvalue = newvalue;
671 cus->oldperftime100nsec = newperftime100nsec;
672
673 double a = (double)lnValueDelta / DeltaPerfTime100nSec;
674
675 double f = (1.0 - a) * 100.0;
676 cpuusage = (int)(f + 0.5); // rounding the result
677
678 if(cpuusage < 0) return 0;
679 #endif
680
681 return cpuusage;

696int XWINDOWSSYSTEM::GetCPUUsageForProcessName(XCHAR* processname)
697{
698 int cpuusage = 0;
699
700 #ifndef BUILDER
701 HANDLE hprocesssnap;
702 PROCESSENTRY32 pe32;
703
704 // Take a snapshot of all processes in the system.
705 hprocesssnap = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
706 if(hprocesssnap == INVALID_HANDLE_VALUE ) return false;
707
708 // Set the size of the structure before using it.
709 memset(&pe32, 0, sizeof(PROCESSENTRY32));
710 pe32.dwSize = sizeof(PROCESSENTRY32);
711
712 // Retrieve information about the first process, and exit if unsuccessful
713 if(!Process32First(hprocesssnap, &pe32))
714 {
715 CloseHandle(hprocesssnap); // clean the snapshot object
716 return false;
717 }
718
719 // Now walk the snapshot of processes
720 do{ XSTRING nametask(pe32.szExeFile);
721
722 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("Process %s : [%s]"), processname, nametask.Get());
723
724 if(!nametask.Compare(processname, true))
725 {
726 cpuusage = GetCPUUsageForProcessID((XDWORD)pe32.th32ProcessID);
727 }
728
729 } while(Process32Next(hprocesssnap, &pe32));
730
731 CloseHandle(hprocesssnap);
732
733 #endif
734
735 return cpuusage;

997bool XWINDOWSSYSTEM::ShutDown(XSYSTEM_CHANGESTATUSTYPE type)
998{
999 if(type == XSYSTEM_CHANGESTATUSTYPE_NONE) return false;
1000
1001 HANDLE hToken;
1002 TOKEN_PRIVILEGES tkp;
1003
1004 // Get a token for this process.
1005 if(!OpenProcessToken(GetCurrentProcess(), TOKEN_ADJUST_PRIVILEGES | TOKEN_QUERY, &hToken)) return false;
1006
1007 // Get the LUID for the shutdown privilege.
1008 LookupPrivilegeValue(NULL, SE_SHUTDOWN_NAME, &tkp.Privileges[0].Luid);
1009
1010 tkp.PrivilegeCount = 1; // one privilege to set
1011 tkp.Privileges[0].Attributes = SE_PRIVILEGE_ENABLED;
1012
1013 // Get the shutdown privilege for this process.
1014 AdjustTokenPrivileges(hToken, FALSE, &tkp, 0, (PTOKEN_PRIVILEGES)NULL, 0);
1015 if(GetLastError() != ERROR_SUCCESS) return false;
1016
1017 switch(type)
1018 {
1019 case XSYSTEM_CHANGESTATUSTYPE_REBOOT : if(!ExitWindowsEx(EWX_REBOOT | EWX_FORCE, SHTDN_REASON_MAJOR_OPERATINGSYSTEM | SHTDN_REASON_MINOR_UPGRADE | SHTDN_REASON_FLAG_PLANNED)) return false;
1020 break;
1021
1022 case XSYSTEM_CHANGESTATUSTYPE_SESSION_LOGOFF : if(!ExitWindowsEx(EWX_LOGOFF | EWX_FORCE, SHTDN_REASON_MAJOR_OPERATINGSYSTEM | SHTDN_REASON_MINOR_UPGRADE | SHTDN_REASON_FLAG_PLANNED)) return false;
1023 break;
1024
1025 case XSYSTEM_CHANGESTATUSTYPE_POWEROFF : if(!ExitWindowsEx(EWX_POWEROFF | EWX_FORCE, SHTDN_REASON_MAJOR_OPERATINGSYSTEM | SHTDN_REASON_MINOR_UPGRADE | SHTDN_REASON_FLAG_PLANNED)) return false;
1026 break;
1027
1028 case XSYSTEM_CHANGESTATUSTYPE_SUSPEND :
1029 #ifndef BUILDER
1030 if(!SetSuspendState(TRUE, FALSE, FALSE)) return false;
1031 #endif
1032 break;
1033
1034 default : return false;
1035
1036
1037 }
1038
1039 return true;
1040}
