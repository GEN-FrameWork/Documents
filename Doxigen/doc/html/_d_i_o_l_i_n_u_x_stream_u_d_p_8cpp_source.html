280int DIOLINUXSTREAMUDP::IsReadyConnect(int socket)
281{
282 struct timeval tv;
283 int rc;
284 fd_set fdr;
285 fd_set fdw;
286 fd_set fds;
287
288 if(socket==-1) return -1;
289
290 FD_ZERO(&fdr);
291 FD_ZERO(&fdw);
292 FD_ZERO(&fds);
293
294 FD_SET(socket,&fdr);
295 FD_SET(socket,&fdw);
296 FD_SET(socket,&fds);
297
298 tv.tv_sec = 0;
299 tv.tv_usec = 100;
300
301 rc = select(socket+1, &fdr, &fdw, &fds, &tv);
302 if(rc==-1) return -1;
303
304 int status1 = FD_ISSET(socket,&fdr) ? 1 : 0;
305 int status2 = FD_ISSET(socket,&fdw) ? 1 : 0;
306 int status3 = FD_ISSET(socket,&fds) ? 1 : 0;
307
308 if(config->IsServer())
309 {
310 if(status1 || status2 ) return 1;
311 if(status3) return -1;
312 }
313 else
314 {
315 if((!status1) && status2) return 1;
316 if(status3) return -1;
317 }
318
319 return 0;
320}
