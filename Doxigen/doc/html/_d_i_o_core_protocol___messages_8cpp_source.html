287bool DIOCOREPROTOCOL_MESSAGES::Delete(XUUID* IDmessage)
288{
289 if(!IDmessage)
290 {
291 return false;
292 }
293
294 if(GetAll()->IsEmpty())
295 {
296 return false;
297 }
298
299 if(xmutexmessages)
300 {
301 xmutexmessages->Lock();
302 }
303
304
305 bool status = false;
306 XDWORD c = 0;
307
308 while(c < GetAll()->GetSize())
309 {
310 DIOCOREPROTOCOL_MESSAGE* request = NULL;
311 DIOCOREPROTOCOL_MESSAGE* response = NULL;
312 bool statusmessageID[2] = { false, false };
313
314 request = (DIOCOREPROTOCOL_MESSAGE*)GetAll()->GetKey(c);
315 if(request)
316 {
317 if(request->GetHeader())
318 {
319 if(request->GetHeader()->GetIDMessage()->Compare((*IDmessage)))
320 {
321 statusmessageID[0] = true;
322 }
323 }
324 }
325
326 response = (DIOCOREPROTOCOL_MESSAGE*)GetAll()->GetElement(c);
327 if(response)
328 {
329 if(response->GetHeader())
330 {
331 if(response->GetHeader()->GetIDMessage()->Compare((*IDmessage)))
332 {
333 statusmessageID[1] = true;
334 }
335 }
336 }
337
338 if(statusmessageID[0] && statusmessageID[1])
339 {
340 status = GetAll()->Delete(request);
341 if(status)
342 {
343 GEN_DELETE request;
344 GEN_DELETE response;
345
346 status = true;
347 break;
348 }
349 }
350
351 c++;
352 }
353
354 // ----------------------------------------------------------------------------------------------------------------
355
356 if(!status)
357 {
358 XSTRING UUID;
359 IDmessage->GetToString(UUID);
360
361 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[DIO Core Protocol Messages] Delete message %s"), UUID.Get());
362 }
363
364 // ----------------------------------------------------------------------------------------------------------------
365
366 if(xmutexmessages)
367 {
368 xmutexmessages->UnLock();
369 }
370
371 return status;
372}

710 case DIOCOREPROTOCOL_MESSAGE_TYPE_ACQUISITION_READ : operation = __L("Read"); break;
