590 for(int c=0; c<(int)services.GetSize(); c++)
591 {
592 DIOSTREAMDEVICEBLUETOOTHSDPSERVICE* service = (DIOSTREAMDEVICEBLUETOOTHSDPSERVICE*)services.Get(c);
593 if(service)
594 {
595 if(servicetype == service->GetID())
596 {
597 for(int d=0; d<(int)service->GetNProtocols(); d++)
598 {
599 DIOSTREAMDEVICEBLUETOOTHSDPSERVICEPROTOCOL* protocol = (DIOSTREAMDEVICEBLUETOOTHSDPSERVICEPROTOCOL*)service->GetProtocol(d);

601 {
602 int channel = protocol->GetChannel();
603 if(channel) return channel;
604 }
