734bool MAINPROCWINDOWS::IsRunningAsService()
735{
736 // return IsRunningAsService(APPMODE_SERVICE_NAME);
737
738 SERVICE_STATUS_PROCESS ssstatus;
739 DWORD bytesneeded;
740 SC_HANDLE schscmanager;
741
742 schscmanager = OpenSCManager( NULL, // local computer
743 NULL , // servicesActive database
744 SC_MANAGER_ALL_ACCESS ); // full access rights
745 if(!schscmanager) return false;
746
747 // Get a handle to the service.
748 SC_HANDLE schservice = OpenService( schscmanager , // SCM database
749 APPMODE_SERVICE_NAME , // name of service
750 SERVICE_ALL_ACCESS); // full access
751
752 if(!schservice)
753 {
754 CloseServiceHandle(schscmanager);
755 return false;
756 }
757
758 // Check the status in case the service is not stopped.
759
760 if(!QueryServiceStatusEx( schservice, // handle to service
761 SC_STATUS_PROCESS_INFO, // information level
762 (LPBYTE) &ssstatus, // address of structure
763 sizeof(SERVICE_STATUS_PROCESS), // size of structure
764 &bytesneeded ) ) // size needed if buffer is too small
765 {
766 CloseServiceHandle(schservice);
767 CloseServiceHandle(schscmanager);
768 return false;
769 }
770
771 DWORD processid = GetCurrentProcessId();
772 if(processid == ssstatus.dwProcessId)
773 {
774 CloseServiceHandle(schservice);
775 CloseServiceHandle(schscmanager);
776
777 return true;
778 }
779
780 CloseServiceHandle(schservice);
781 CloseServiceHandle(schscmanager);
782
783 if((!ssstatus.dwProcessId) && (ssstatus.dwCurrentState == SERVICE_START_PENDING)) return true;
784
785 return false;
786}

1063{
1064 WINDOWSSERVICE* service = (WINDOWSSERVICE*)lpparam;
1065 bool userstop = false;
1066 int status = 0;
1067
1068 if(!service) return false;
1069
1070 #ifdef APPFLOW_ACTIVE
1071 if(!mainprocwindows.Ini(&GEN_appmain, APPFLOWBASE_MODE_TYPE_SERVICE))
1072 #else
1073 if(!mainprocwindows.Ini())
1074 #endif
1075 {
1076 status = 1;
1077 }
1078
1079 if(!status)
1080 {
1081 while(1)
1082 {
1083 if(!mainprocwindows.Update())
1084 {
1085 userstop = true;
1086 break;
1087 }
1088 }
1089 }
1090
1091
1092 APPFLOWBASE_EXITTYPE exittype = APPFLOWBASE_EXITTYPE_UNKNOWN;
1093
1094 #ifdef APPFLOW_ACTIVE
1095 exittype = mainprocwindows.GetAppMain()->GetApplication()->GetExitType();
1096 #endif
1097
1098 mainprocwindows.GetXPathExec()->Empty();
1099 mainprocwindows.End();
1100
1101
1102 if(exittype == APPFLOWBASE_EXITTYPE_BY_SERVICERELOAD)
1103 {
1104 service->SetMustRestartService(true);
1105 }
1106
1107 if(service->GetHandleStoppedEvent())
1108 {
1109 SetEvent(service->GetHandleStoppedEvent());
1110 }
1111
1112 if(userstop)
1113 {
1114 service->Stop();
1115 }
1116
1117 return ERROR_SUCCESS;
