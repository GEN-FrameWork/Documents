150void Call_System_GetType(SCRIPT_LIB* library, SCRIPT* script, XVECTOR<XVARIANT*>* params, XVARIANT* returnvalue)
151{
152 if(!library) return;
153 if(!script) return;
154 if(!params) return;
155 if(!returnvalue) return;
156
157 returnvalue->Set();
158
159 switch(GEN_XSYSTEM.GetPlatform())
160 {
161 case XSYSTEM_PLATFORM_UNKNOWN :
162 default : (*returnvalue) = ""; break;
163 case XSYSTEM_PLATFORM_WINDOWS : (*returnvalue) = "Windows"; break;
164 case XSYSTEM_PLATFORM_LINUX : (*returnvalue) = "Linux"; break;
165 case XSYSTEM_PLATFORM_LINUX_EMBEDDED : (*returnvalue) = "LinuxEmbedded"; break;
166 case XSYSTEM_PLATFORM_ANDROID : (*returnvalue) = "Android"; break;
167 case XSYSTEM_PLATFORM_STM32 : (*returnvalue) = "STM32"; break;
168 };
169}

259void Call_System_GetEnviromentVar(SCRIPT_LIB* library, SCRIPT* script, XVECTOR<XVARIANT*>* params, XVARIANT* returnvalue)
260{
261 if(!library) return;
262 if(!script) return;
263 if(!params) return;
264 if(!returnvalue) return;
265
266 returnvalue->Set();
267
268 if(!params->GetSize())
269 {
270 script->HaveError(SCRIPT_ERRORCODE_INSUF_PARAMS);
271 return;
272 }
273
274 XSTRING* namevar = (XSTRING*)params->Get(0)->GetData();
275 if(!namevar) return;
276
277 XCHAR* variable = GEN_XSYSTEM.GetEnviromentVariable(namevar->Get());
278 if(variable)
279 {
280 (*returnvalue) = variable;
281 }
282}
