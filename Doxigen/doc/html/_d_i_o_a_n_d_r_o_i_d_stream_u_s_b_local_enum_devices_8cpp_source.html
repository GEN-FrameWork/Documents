246bool DIOANDROIDSTREAMUSBLOCALENUMDEVICES::SearchAndDeleteEqualDevices()
247{
248 if(devices.IsEmpty()) return false;
249
250 bool found;
251
252 do{ found = false;
253
254 for(int c=0; c<devices.GetSize(); c++)
255 {
256 DIOSTREAMDEVICEUSB* device1 = (DIOSTREAMDEVICEUSB*)devices.Get(c);
257 if(device1)
258 {
259 for(int d=0; d<devices.GetSize(); d++)
260 {
261 DIOSTREAMDEVICEUSB* device2 = (DIOSTREAMDEVICEUSB*)devices.Get(d);
262 if(device2 && (c!=d))
263 {
264 if(!device1->GetName()->Compare(device2->GetName()->Get()))
265 {
266 GEN_DELETE device1;
267 devices.Delete(device1);
268
269 found = true;
270 break;
271 }
272 }
273 }
274
275 if(found) break;
276 }
277 }
278
279 if(!found) break;
280
281 } while(1);
282
283 return true;
284}
