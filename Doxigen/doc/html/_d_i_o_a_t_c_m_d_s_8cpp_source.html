643int DIOATCMDS::CalculeBaudRateStreamUART(int timeout)
644{
645 int baudrates[] = { 115200, 9600, 57600, 38400, 19200, 4800, 2400 , 1200 };
646 int baudrate = 0;
647
648 DIOSTREAMUARTCONFIG* diostreamcfg = (DIOSTREAMUARTCONFIG*)diostream->GetConfig();
649 if(!diostreamcfg) return 0;
650
651 if(diostreamcfg->GetType() != DIOSTREAMTYPE_UART) return 0;
652
653 for(int c=0;c<sizeof(baudrates)/sizeof(int);c++)
654 {
655 diostreamcfg->SetBaudRate(baudrates[c]);
656
657 //XTRACE_PRINT(__L("Open connection: %d bauds"),baudrates[c]);
658
659 diostream->ResetXBuffers();
660
661 diostream->Close();
662 if(!diostream->Open()) return 0;

664 if(diostream->WaitToConnected(timeout))
665 {
666 if(threadconnection) threadconnection->Ini();
667
668 echomode = false;
669 if(Basic(timeout)==DIOATCMD_ERROR_NONE) baudrate = baudrates[c];
670
671 echomode = true;
672 if(Basic(timeout)==DIOATCMD_ERROR_NONE) baudrate = baudrates[c];
673
674 if(threadconnection) threadconnection->End();
675
676 if(baudrate) break;
677
678 }
679 }
680
681 diostream->Close();
682
683 return baudrate;
