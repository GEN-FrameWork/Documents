66ANDROIDJNIOBJECT::ANDROIDJNIOBJECT(XSTRING classname, XSTRING constructorsignature, ...)
67{
68 JNIEnv* JEnv = ANDROIDJNI::GetJNIEnv();
69 XBUFFER cstr;
70
71 classname.ConvertToASCII(cstr);
72 jniclass = ANDROIDJNI::FindJNIClass((char*)cstr.Get());
73
74 cstr.Empty();
75 constructorsignature.ConvertToASCII(cstr);
76 jmethodID constructor = JEnv->GetMethodID(jniclass, "<init>", (char*)cstr.Get());
77
78 va_list params;
79
80 va_start(params, constructorsignature);
81 jobject object = JEnv->NewObjectV(jniclass, constructor, params);
82 va_end(params);
