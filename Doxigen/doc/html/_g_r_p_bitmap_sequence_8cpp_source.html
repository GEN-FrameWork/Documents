593GRPBITMAPFRAME* GRPBITMAPSEQUENCE::Update()
594{
595 if(!timer) return NULL;
596 if(frames.IsEmpty()) return NULL;
597 if(status!=GRPBITMAPSEQUENCESTATUS_PLAY) return NULL;
598
599 //if(timer->GetMeasureMilliSeconds()<timebyframe) return NULL;
600
601 timer->Reset();
602
603 if(!reverse)
604 {
605 if(frames.GetSize()!=1) frameindex++;
606
607 if(frameindex == frames.GetSize())
608 {
609 frameindex = 0;
610 nloopsmade++;
611 if((nloops)&&(nloops==nloopsmade)) status=GRPBITMAPSEQUENCESTATUS_END;
612 }
613 }
614 else
615 {
616 if(frames.GetSize()!=1) frameindex--;
617
618 if(frameindex<0)
619 {
620 frameindex = (frames.GetSize()-1);
621
622 nloopsmade++;
623 if((nloops)&&(nloops==nloopsmade)) status=GRPBITMAPSEQUENCESTATUS_END;
624 }
625 }
626
627 return (GRPBITMAPFRAME*)frames.Get(frameindex);
628}
