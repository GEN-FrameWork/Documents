119{
120 RegisterEvent(GRPXEVENT_TYPE_SCREEN_CREATING);
121 RegisterEvent(GRPXEVENT_TYPE_SCREEN_CANVASCREATING);
122 RegisterEvent(GRPXEVENT_TYPE_SCREEN_CREATED);
123 RegisterEvent(GRPXEVENT_TYPE_SCREEN_DELETING);
124 RegisterEvent(GRPXEVENT_TYPE_SCREEN_DELETED);
125
126 if(!APPFLOWCONSOLE::Ini(execparams))
127 {
128 return false;
129 }
130
131 if(GetExitType() != APPFLOWBASE_EXITTYPE_UNKNOWN)
132 {
133 return false;
134 }
135
136 if(initoptions & APPFLOWGRAPHICS_INIOPTION_CREATEMAINSCREEN)

138 if(!CreateMainScreenProcess((initoptions & APPFLOWGRAPHICS_INIOPTION_SHOWMAINSCREEN)?true:false)) return false;
139 }
140
141 return true;
142}

219bool APPFLOWGRAPHICS::CreateMainScreenProcess(bool show)
220{
221 if(mainscreen) return false;
222
223 mainscreen = GEN_GRPFACTORY.CreateScreen();
224 if(!mainscreen)
225 {
226 // XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[APP Graphics] Could not initialize the main screen ..."));
227 return false;
228 }
229
230 #ifdef GRP_OPENGL_ACTIVE
231 maincontext = GRPFACTORY::GetInstance().CreateContext();
232 if(!maincontext)
233 {
234 // XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[APP Graphics] Could not initialize the Graphics Context ..."));
235 return false;
236 }
237 #endif
238
239 SubscribeEvent(GRPXEVENT_TYPE_SCREEN_CREATING , this);
240 SubscribeEvent(GRPXEVENT_TYPE_SCREEN_CANVASCREATING , this);
241 SubscribeEvent(GRPXEVENT_TYPE_SCREEN_CREATED , this);
242 SubscribeEvent(GRPXEVENT_TYPE_SCREEN_DELETING , this);
243 SubscribeEvent(GRPXEVENT_TYPE_SCREEN_DELETED , this);
244
245 #ifndef ANDROID
246
247 GRPXEVENT grpeventini(this, GRPXEVENT_TYPE_SCREEN_CREATING);
248 grpeventini.SetScreen(mainscreen);
249 PostEvent(&grpeventini);
250
251 bool status = !grpeventini.GetError();
252
253 if(status)
254 {
255 status = mainscreen->Create(show);
256 if(status)
257 {
258 GRPXEVENT grpeventini(this, GRPXEVENT_TYPE_SCREEN_CANVASCREATING);
259 grpeventini.SetScreen(mainscreen);
260 PostEvent(&grpeventini);
261
262 GRPVIEWPORT* viewport = mainscreen->GetViewport(0);
263 if(!viewport)
264 {
265 if(!mainscreen->CreateViewport(GRPVIEWPORT_ID_MAIN , 0.0f, 0.0f, (float)mainscreen->GetWidth() , (float)mainscreen->GetHeight(), 0, 0, mainscreen->GetWidth(), mainscreen->GetHeight()))
266 {
267 return false;
268 }
269 }
270 }
271 }
272
273 if(status)
274 {
275 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[App Graphics] Screen created"));
276
277 GRPXEVENT grpeventend(this, GRPXEVENT_TYPE_SCREEN_CREATED);
278 grpeventend.SetScreen(mainscreen);
279 grpeventend.SetError(status);
280 PostEvent(&grpeventend);
281 }
282 else
283 {
284 //XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[App Graphics] Error to creating screen..."));
285 }
286
287 if(!status) return false;
288
289 #ifdef INP_ACTIVE
290 if(initoptions & APPFLOWGRAPHICS_INIOPTION_INPUT)
291 {
292 keyboard = GEN_INPMANAGER.GetDevice(INPDEVICE_TYPE_KEYBOARD);
293 if(!keyboard)
294 {
295 keyboard = INPFACTORY::GetInstance().CreateDevice(INPDEVICE_TYPE_KEYBOARD, mainscreen);
296 if(keyboard) GEN_INPMANAGER.AddDevice(keyboard);
297 }
298
299 mouse = GEN_INPMANAGER.GetDevice(INPDEVICE_TYPE_MOUSE);
300 if(!mouse)
301 {
302 mouse = INPFACTORY::GetInstance().CreateDevice(INPDEVICE_TYPE_MOUSE, mainscreen);
303 if(mouse) GEN_INPMANAGER.AddDevice(mouse);
304 }
305
306 touchscreen = GEN_INPMANAGER.GetDevice(INPDEVICE_TYPE_TOUCHSCREEN);
307 if(!touchscreen)
308 {
309 touchscreen = INPFACTORY::GetInstance().CreateDevice(INPDEVICE_TYPE_TOUCHSCREEN, mainscreen);
310 if(touchscreen) GEN_INPMANAGER.AddDevice(touchscreen);
311 }
312
313 }
314 #endif
315
316 #endif
317
318 return true;
319}

331bool APPFLOWGRAPHICS::DeleteMainScreenProcess()
332{
333 if(!mainscreen)
334 {
335 return false;
336 }
337
338 GRPXEVENT grpeventini(this, GRPXEVENT_TYPE_SCREEN_DELETING);
339 grpeventini.SetScreen(mainscreen);
340 PostEvent(&grpeventini);
341
342 UnSubscribeEvent(GRPXEVENT_TYPE_SCREEN_CREATING , this);
343 UnSubscribeEvent(GRPXEVENT_TYPE_SCREEN_CANVASCREATING , this);
344 UnSubscribeEvent(GRPXEVENT_TYPE_SCREEN_CREATED , this);
345
346 if(mainscreen)
347 {
348 mainscreen->Delete();
349 GRPFACTORY::GetInstance().DeleteScreen(mainscreen);
350 mainscreen = NULL;
351 }
352
353 GRPXEVENT grpeventend(this, GRPXEVENT_TYPE_SCREEN_DELETED);
354 grpeventini.SetScreen(mainscreen);
355 PostEvent(&grpeventend);
356
357 UnSubscribeEvent(GRPXEVENT_TYPE_SCREEN_DELETED , this);
358 UnSubscribeEvent(GRPXEVENT_TYPE_SCREEN_CREATED , this);
359
360 return true;
361}
