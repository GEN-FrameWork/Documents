196bool XFILEXDB::OpenFile()
197{
198 if(!Open(this->xpath)) return false;
199
200 GetPrimaryFile()->Read((XBYTE*)&nrecords,sizeof(XDWORD));
201
202 if(!nrecords)
203 {
204 Close();
205 return false;
206 }
207
208 indexmap.SetIsMulti(true);
209 for(int c=0;c<nrecords;c++)
210 {
211 XDWORD ID = 0;
212 XDWORD filepos = 0;
213
214 GetPrimaryFile()->Read((XBYTE*)&ID , sizeof(XDWORD));
215 GetPrimaryFile()->Read((XBYTE*)&filepos , sizeof(XDWORD));
216
217 //XTRACE_PRINTCOLOR(1,__L("ID: %08X"), ID);
218
219 indexmap.Add(ID,filepos);
220 }
221
222 indexmap.SetIsMulti(false);
223
224 isopen=true;
225
226 return true;
227}

256XBUFFER* XFILEXDB::GetRecord(XDWORD ID)
257{
258 if(!isopen)
259 {
260 if(!OpenFile()) return NULL;
261 }
262
263 if(!nrecords) return NULL;
264
265 XDWORD filepos = GetPosition(ID);
266 if(!filepos) return NULL;
267
268 XDWORD size;
269
270 if(!GetPrimaryFile()->SetPosition(filepos)) return NULL;
271
272 if(!GetPrimaryFile()->Read((XBYTE*)&size ,sizeof(XDWORD))) return NULL;
273
274 XBUFFER* xbuffer = GEN_NEW XBUFFER();
275 if(xbuffer)
276 {
277 if(size)

282
