308bool XWINDOWSTRACE::GetHandleNet(XTRACE_TARGET* target)
309{
310 #if(defined(DIO_ACTIVE) && defined(DIO_STREAMUDP_ACTIVE))
311
312 SOCKADDR_IN addr;
313 SOCKET handle;
314
315 memset(&addr,0,sizeof(SOCKADDR_IN));
316
317 handle = socket(AF_INET, SOCK_DGRAM, 0);
318 if(handle == INVALID_SOCKET) return false;
319
320 if(!target->GetIP()[0]) return false;
321
322
323 addr.sin_family = AF_INET;
324 addr.sin_port = htons(target->GetPort());
325
326 if(target->GetIP()[0] == '*')
327 {
328 addr.sin_addr.s_addr = htonl(INADDR_BROADCAST);
329 }
330 else
331 {
332 #ifndef BUILDER
333 inet_pton(addr.sin_family, target->GetIP(), &addr.sin_addr.s_addr);
334 #else
335 addr.sin_addr.s_addr = inet_addr(target->GetIP());
336 #endif
337 }
338
339 connect(handle,(LPSOCKADDR)&addr,sizeof(SOCKADDR_IN));
340
341 XQWORD NEThandle = (XQWORD)handle;

343 target->SetNETHandle(NEThandle);

345 #endif
346
