671bool XLOGBASE::AddEntry(XLOGLEVEL level, XCHAR* sectionID, bool inmemory, XBYTE* data, XDWORD size, XDWORD sizeline, bool showoffset, bool showtext)
672{
673 if(!IsActive()) return false;
674 if(!mutex) return false;
675
676 mutex->Lock();
677
678 if(!PassFilters(level, sectionID))
679 {
680 mutex->UnLock();
681 return false;
682 }
683
684 XSTRING strdata;
685 XDWORD _size = 0;
686 XDWORD _sizeline = sizeline;
687 int index = 0;
688 XDWORD lines = 0;
689
690 while(_size<size)
691 {
692 XSTRING string;
693
694 if(_sizeline > (size - _size)) _sizeline = (size - _size);
695
696 if(showoffset)
697 {
698 strdata.Format(__L("%04X"),index);
699 string += strdata;
700 string += __L(" ");
701 }
702
703 for(XDWORD c=0; c<_sizeline; c++)
704 {
705 strdata.Format(__L("%02X "),data[index]);
706 string += strdata;
707 _size++;
708 index++;
709 }
710
711 if(_sizeline != sizeline)
712 {
713 for(XDWORD c=0; c<(sizeline-_sizeline); c++)
714 {
715 strdata.Format(__L(" "));
716 string += strdata;
717 }
718 }
719
720 if(showtext)
721 {
722 index -= _sizeline;
723 string += __L(" ");
724
725 for(XDWORD c=0; c<_sizeline; c++)
726 {
727 XCHAR character = (XCHAR)data[index];
728
729 if((character<0x20) || (character>0x80) || (character==__C('%')) || (character==__C('\\'))) character = __C('.');
730
731 strdata.Format(__L("%c"),character);
732 string += strdata;
733 index++;
734 }
735 }
736
737 mutex->UnLock();
738
739 AddEntry(level, lines?NULL:sectionID, inmemory, string.Get());
740
741 mutex->Lock();
742
743 lines++;
744 }
745
746 mutex->UnLock();
747
748 return true;
749}

844bool XLOGBASE::CalculateInitialStatus()
845{
846 XFILE* xfile = filelog->GetPrimaryFile();
847 if(!xfile) return false;
848
849 nlines = 0;
850 nentrys = 0;
851 size = xfile->GetSize();
852
853 entrys.DeleteContents();
854 entrys.DeleteAll();
855
856 XDWORD sizeBOM = 0;
857 XFILETXTFORMATCHAR formatchar = filelog->GetFormatCharFromFile(&sizeBOM);
858 int sizebytescharacter = filelog->SizeOfCharacter(formatchar);
859
860 xfile->SetPosition(sizeBOM);
861
862 bool endfile;
863 XDWORD br;
864 XBUFFER dataline;
865 XBYTE* readbuffer = GEN_NEW XBYTE[XFILETXT_MAXBUFFER];
866 if(!readbuffer) return false;
867
868 memset(readbuffer, 0, XFILETXT_MAXBUFFER);
869
870 do{ XDWORD bufferpos = 0;
871
872 br = XFILETXT_MAXBUFFER;
873 endfile = !xfile->Read(readbuffer, &br);
874 if(!br) break;
875
876 do{ XFILETXTTYPELF typeLF;
877 XDWORD sizeLF = 0;
878 XDWORD sizeline = 0;
879 bool endline = filelog->GetSizeOfLine(formatchar, &readbuffer[bufferpos], typeLF, sizeLF, sizeline, (br-bufferpos));
880
881 if(endline)
882 {
883 if(readbuffer[bufferpos] != 0x20)
884 {
885 nentrys++;
886
887 lastentry = GEN_NEW XLOGENTRY();
888 if(lastentry)
889 {
890 lastentry->position = bufferpos;
891 entrys.Add(lastentry);
892 }
893 }
894
895 if(lastentry) lastentry->size += (sizeline + sizeLF);
896 nlines++;
897
898 sizeline *= sizebytescharacter;
899 sizeline += (sizeLF*sizebytescharacter);
900 bufferpos += sizeline;
901 }
902 else
903 {
904 int lack = (br-bufferpos);
905 bufferpos += lack;
906
907 if(endfile && (bufferpos == br))
908 {
909 if(readbuffer[bufferpos] != 0x20)
910 {
911 nentrys++;
912
913 lastentry = GEN_NEW XLOGENTRY();
914 if(lastentry)
915 {
916 lastentry->position = bufferpos;
917 entrys.Add(lastentry);
918 }
919 }
920
921 if(lastentry)
922 {
923 lastentry->size += lack;
924 }
925
926 nlines++;
927 }
928 }
929
930 } while(bufferpos < br);
931
932 } while(!endfile);
933
934 GEN_DELETE_ARRAY readbuffer;
935
936 return true;
937}

1279 , xdatetime->GetSeconds());
