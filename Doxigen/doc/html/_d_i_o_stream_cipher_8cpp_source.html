316void DIOSTREAMCIPHER::ThreadRunFunction(void* param)
317{
318 DIOSTREAMCIPHER* diostreamcipher = (DIOSTREAMCIPHER*)param;
319 if(!diostreamcipher) return;
320
321 DIOSTREAM* diostream = diostreamcipher->diostream;
322 if(!diostream) return;
323
324 CIPHER* cipher = diostreamcipher->cipher;
325 if(!cipher) return;
326
327 int br = diostream->GetInXBuffer()->GetSize();
328 if(br>DIOSTREAMCIPHER_HEAD_SIZE)
329 {
330 if(diostreamcipher->mutexread) diostreamcipher->mutexread->UnLock();
331
332 XDWORD magicnumber;
333 XBYTE ciphertype;
334 XBYTE cipherpadding;
335 XDWORD sizebuffer;
336 XDWORD CRC32[2];
337 XBUFFER input;
338
339 input.Resize(DIOSTREAMCIPHER_HEAD_SIZE);
340
341 diostream->GetInXBuffer()->Get(input.Get(), DIOSTREAMCIPHER_HEAD_SIZE, 0);
342
343 diostreamcipher->hashcrc32->ResetResult();
344 CRC32[0] = diostreamcipher->hashcrc32->Do(input.Get(), DIOSTREAMCIPHER_HEAD_SIZE - sizeof(XDWORD));
345
346 input.Extract(magicnumber , 0);
347 input.Extract(ciphertype , 0);
348 input.Extract(cipherpadding , 0);
349 input.Extract(sizebuffer , 0);
350 input.Extract(CRC32[1] , 0);
351
352 if((magicnumber == DIOSTREAMCIPHER_HEAD_MAGICNUMBER) && (CRC32[0] == CRC32[1]))
353 {
354 if(diostream->GetInXBuffer()->GetSize() >= sizebuffer + DIOSTREAMCIPHER_HEAD_SIZE)
355 {
356 input.Empty();
357
358 input.Resize(DIOSTREAMCIPHER_HEAD_SIZE + sizebuffer);
359
360 diostream->GetInXBuffer()->Extract(input.Get(), 0, sizebuffer + DIOSTREAMCIPHER_HEAD_SIZE);
361 input.Extract(NULL, 0, DIOSTREAMCIPHER_HEAD_SIZE);
362
363 cipher->Uncipher(input);
364
365 diostreamcipher->GetInXBuffer()->Add(cipher->GetResult()->Get(), cipher->GetResult()->GetSize() - cipherpadding);
366 }
367 }
368
369 if(diostreamcipher->mutexread) diostreamcipher->mutexread->UnLock();
370 }
371
372
373 int bw = diostreamcipher->GetOutXBuffer()->GetSize();
374 if(bw)
375 {
376 if(diostreamcipher->mutexwrite) diostreamcipher->mutexwrite->Lock();
377
378 XBUFFER output;
379
380 cipher->Cipher(diostreamcipher->GetOutXBuffer()->Get(), diostreamcipher->GetOutXBuffer()->GetSize());
381
382 output.Add((XDWORD)DIOSTREAMCIPHER_HEAD_MAGICNUMBER);
383 output.Add((XBYTE)cipher->GetType());
384 output.Add((XBYTE)(cipher->GetResult()->GetSize() - bw));
385 output.Add((XDWORD)cipher->GetResult()->GetSize());
386
387 diostreamcipher->hashcrc32->ResetResult();
388 output.Add((XDWORD)diostreamcipher->hashcrc32->Do(output));
389 output.Add((XBYTE*)cipher->GetResult()->Get(), cipher->GetResult()->GetSize());
390
391 if(diostream->Write(output) == output.GetSize())
392 {
393 //if(diostream->WaitToFlushOutXBuffer())
394 diostreamcipher->GetOutXBuffer()->Extract(NULL, 0 , bw);
395 }
396
397 if(diostreamcipher->mutexwrite) diostreamcipher->mutexwrite->UnLock();
398 }
399}
