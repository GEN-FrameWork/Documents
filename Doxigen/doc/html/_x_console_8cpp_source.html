389bool XCONSOLE::PrintDataBlock(XBYTE* data, XDWORD size, XDWORD marginsize, XDWORD sizeline, bool showoffset, bool showtext)
390{
391 XSTRING margin;
392 XSTRING strdata;
393 XDWORD _size = 0;
394 XDWORD _sizeline = sizeline;
395 int index = 0;
396
397 for(XDWORD c=0;c<marginsize;c++)
398 {
399 margin += __L(" ");
400 }
401
402 while(_size<size)
403 {
404 XSTRING string;
405
406 string = margin;
407 if(_sizeline > (size - _size)) _sizeline = (size - _size);
408
409 if(showoffset)
410 {
411 strdata.Format(__L("%04X"),index);
412 string += strdata;
413 string += __L(" ");
414 }
415
416 for(XDWORD c=0; c<_sizeline; c++)
417 {
418 strdata.Format(__L("%02X "),data[index]);
419 string += strdata;
420 _size++;
421 index++;
422 }
423
424 if(_sizeline != sizeline)
425 {
426 for(XDWORD c=0; c<(sizeline-_sizeline); c++)
427 {
428 strdata.Format(__L(" "));
429 string += strdata;
430 }
431 }
432
433 if(showtext)
434 {
435 index -= _sizeline;
436 string += __L(" ");
437
438 for(XDWORD c=0; c<_sizeline; c++)
439 {
440 XCHAR character = (XCHAR)data[index];
441
442 if((character<0x20) || (character>0x80) || (character==__C('%')) || (character==__C('\\'))) character = __C('.');
443
444 strdata.Format(__L("%c"),character);
445 string += strdata;
446 index++;
447 }
448 }
449
450 Print(string.Get());
451 Print(__L("\n"));
452
453 if(_size > XCONSOLE_MAXSIZEDATABLOCK) break;
454 }

456 return true;

553bool XCONSOLE::WaitKey(XCHAR* text, XDWORD margin, bool prelude, XDWORD timeout)
554{
555 XTIMER* GEN_XFACTORY_CREATE(xtimer, CreateTimer())
556 if(!xtimer) return false;
557
558 xtimer->Reset();
559
560 XSTRING string;
561 XSTRING string2;
562 bool status = true;
563
564
565 while(KBHit())
566 {
567 GetChar();
568 }
569
570 while(!KBHit())
571 {
572 string = text;
573 string += __L(" \r");
574
575 string2.Format(string.Get(), (timeout - xtimer->GetMeasureSeconds()));
576
577 GEN_XSLEEP.MilliSeconds(100);
578
579 PrintMessage(string2.Get(), margin, prelude, false);
580

584 {
585 status = false;

587 }
588 }
589 }

593 GetChar();

595
596 string.Empty();
