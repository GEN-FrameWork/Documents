229bool CIPHERKEYSFILEPEM::DecodeCertificates(XVECTOR<XSTRING*>* lines)
230{
231 XVECTOR<CIPHERKEYSFILEPEM_ENTRYBUFFER*> entrysbuffer;
232 CIPHERKEYSFILEPEM_ENTRYBUFFER* entrybuffer = NULL;
233
234 ndecodeobj = 0;
235
236 for(XDWORD c=0; c<lines->GetSize(); c++)
237 {
238 XSTRING line;
239 int ini = XSTRING_NOTFOUND;
240 int end = XSTRING_NOTFOUND;
241
242 if(lines->Get(c))
243 {
244 line = lines->Get(c)->Get();
245 }
246
247 if(line.IsEmpty())
248 {
249 continue;
250 }
251
252 ini = line.Find(CIPHERKEYSFILEPEM_BEGINKEY, true);
253 if(ini != XSTRING_NOTFOUND)
254 {
255 ini += XSTRING::GetSize(CIPHERKEYSFILEPEM_BEGINKEY);
256
257 if(!entrybuffer)
258 {
259 entrybuffer = GEN_NEW CIPHERKEYSFILEPEM_ENTRYBUFFER();
260 if(entrybuffer)
261 {
262 entrybuffer->end_line = XSTRING_NOTFOUND;
263 entrybuffer->ini_line = c+1;
264 end = line.Find(CIPHERKEYSFILEPEM_FINISH, true, ini);
265 if(end != XSTRING_NOTFOUND)
266 {
267 entrybuffer->type.AdjustSize(_MAXSTR);
268 line.Copy(ini, end, entrybuffer->type);
269 entrybuffer->type.AdjustSize();
270 }
271 else
272 {
273 GEN_DELETE entrybuffer;
274 entrybuffer = NULL;
275 }
276 }
277 }
278 }
279
280 ini = line.Find(CIPHERKEYSFILEPEM_ENDKEY, true);
281 if(ini != XSTRING_NOTFOUND)
282 {
283 ini += XSTRING::GetSize(CIPHERKEYSFILEPEM_ENDKEY);
284
285 if(entrybuffer)
286 {
287 entrybuffer->end_line = c;
288 end = line.Find(CIPHERKEYSFILEPEM_FINISH, true, ini);
289 if(end != XSTRING_NOTFOUND)
290 {
291 XSTRING type;
292
293 type.AdjustSize(_MAXSTR);
294 line.Copy(ini, end, type);
295 type.AdjustSize();
296
297 if(!type.Compare(entrybuffer->type))
298 {
299 entrysbuffer.Add(entrybuffer);
300 entrybuffer = NULL;
301 }
302 else
303 {
304 GEN_DELETE entrybuffer;
305 entrybuffer = NULL;
306 }
307 }
308 else
309 {
310 GEN_DELETE entrybuffer;
311 entrybuffer = NULL;
312 }
313 }
314 }
315 }
316
317
318 for(XDWORD c=0; c<entrysbuffer.GetSize(); c++)
319 {
320 CIPHERKEYSFILEPEM_ENTRYBUFFER* entrybuffer = entrysbuffer.Get(c);
321 if(entrybuffer)
322 {
323 XSTRING datastr;
324 XSTRING datastr1;
325
326 for(int d=entrybuffer->ini_line; d<entrybuffer->end_line; d++)
327 {
328 datastr += lines->Get(d)->Get();
329 }
330
331 if(!entrybuffer->data.ConvertFromBase64(datastr))
332 {
333 XFEEDBACK_ADD(XFEEDBACK_CODE_ERRORINFUNCTION , 1, __L("CIPHERKEYSFILEPEM::DecodeCertificates invalid Convert From Base64"));
334 return false;
335 }
336 }
337 }
338
339
340 XASN1 asn1;
341
342 for(XDWORD c=0; c<entrysbuffer.GetSize(); c++)
343 {
344 CIPHERKEYSFILEPEM_ENTRYBUFFER* entrybuffer = entrysbuffer.Get(c);
345 if(entrybuffer)
346 {
347 decodeobj = NULL;
348
349 if(!entrybuffer->type.Compare(__L("CERTIFICATE")))
350 {
351 decodeobjtype = CIPHERKEYSFILETYPE_CERTIFICATEX509;
352
353 decodeobj = (CIPHERKEY*)GEN_NEW CIPHERCERTIFICATEX509();
354 if(decodeobj)
355 {
356 if(asn1.Decode(entrybuffer->data, this))
357 {
358 Key_Add((CIPHERKEY*)decodeobj);
359
360 CIPHERCERTIFICATEX509* certificate = (CIPHERCERTIFICATEX509*)decodeobj;
361 if(certificate)
362 {
363 certificate->XTraceCertificatedPropertys();
364 }
365 }
366 else
367 {
368 GEN_DELETE (CIPHERCERTIFICATEX509*)decodeobj;
369 }
370 }
371 }
372 }
373 }
374
375
376 entrysbuffer.DeleteContents();
377 entrysbuffer.DeleteAll();
378
