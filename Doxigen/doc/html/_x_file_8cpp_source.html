325bool XFILE::EraseBlock(XQWORD size)
326{
327 XBYTE* tbuffer;
328 XQWORD position;
329 XQWORD sizetomove;
330 XDWORD nblocks;
331 XDWORD rest;
332 bool status = true;
333
334 if(!isopen) return false;
335
336 if(!GetPosition(position)) return false;
337
338 sizetomove = GetSize() - position - size;
339
340 tbuffer= GEN_NEW XBYTE [XFILE_MAXBUFFER];
341 if(!tbuffer) return false;
342
343 nblocks = (XDWORD)(sizetomove / XFILE_MAXBUFFER);
344 rest = (XDWORD)(sizetomove % XFILE_MAXBUFFER);
345
346 for(XDWORD c=0; c<nblocks; c++)
347 {
348 SetPosition(position + size + ((XQWORD)c * XFILE_MAXBUFFER));
349 if(Read(tbuffer, XFILE_MAXBUFFER))
350 {
351 SetPosition(position + ((XQWORD)c * XFILE_MAXBUFFER));
352 Write(tbuffer, XFILE_MAXBUFFER);
353 }
354 }
355
356 SetPosition(position + size +((XQWORD)nblocks * XFILE_MAXBUFFER));
357 if(Read(tbuffer, rest))
358 {
359 SetPosition(position + ((XQWORD)nblocks * XFILE_MAXBUFFER));
360 Write(tbuffer, rest);
361 }
362
363 SetSize(GetSize() - size);
364
365 GEN_DELETE_ARRAY tbuffer;
366
367 return status;
368}

383bool XFILE::InsertBlock(XBYTE* buffer, XDWORD size)
384{
385 XBYTE* movebuffer;
386 XQWORD origin;
387 XQWORD position;
388
389 bool status = true;
390
391 if(!isopen) return false;
392
393 if(GetPosition(position)!=true) return false;
394
395 origin = position;
396
397 movebuffer= GEN_NEW XBYTE [XFILE_MAXBUFFER];
398 if(!movebuffer) return false;
399
400 ActualizeSize();
401
402 XQWORD sizefile = GetSize();
403 XQWORD sizetomove = sizefile-position;
404 XDWORD nblockstomove = (XDWORD)(sizetomove / XFILE_MAXBUFFER);
405 XDWORD resttomove = (XDWORD)(sizetomove % XFILE_MAXBUFFER);
406 XDWORD c;
407
408 position = GetSize() - XFILE_MAXBUFFER;
409
410 for(c=0; c<nblockstomove+1; c++)
411 {
412 XDWORD rb;
413
414 if(c>=nblockstomove)
415 {
416 rb = resttomove;
417 position = origin;
418 }
419 else
420 {
421 rb = XFILE_MAXBUFFER;
422 }
423
424 SetPosition(position);
425 Read(movebuffer, &rb);
426
427 SetPosition(position+size);
428 Write(movebuffer,rb);
429
430 position -= XFILE_MAXBUFFER;
431 }
432
433 SetPosition(origin);
434 Write(buffer,size);
435
436 if(status==true) ActualizeSize();
437
438 GEN_DELETE_ARRAY movebuffer;
439
440 return status;
441}

470bool XFILE::OverwriteContent(XCHAR* xpath)
471{
472 bool status=true;
473
474 if(!Open(xpath,false)) return false;
475
476 GetSize();
477
478 XBYTE* buffer;
479 const int sizeblock = 1024;
480 int block = (XDWORD)(cachesize/sizeblock);
481 int mod = (XDWORD)(cachesize%sizeblock);
482
483 buffer = GEN_NEW XBYTE [sizeblock];
484 if(!buffer) return false;
485
486 memset(buffer,0,sizeblock);
487
488 for(int c=0;c<block;c++)
489 {
490 if(Write(buffer,sizeblock)!=true)
491 {
492 status=false;
493 break;
494 }
495 }
496
497 if(status==true)
498 {
499 if(Write(buffer,mod)!=true) status=false;
500 }
501
502 GEN_DELETE_ARRAY buffer;
503
504 Close();
505
506 return status;
507}
