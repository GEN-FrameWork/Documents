418bool DIOOBEXPUSH::Ini(int timeout)
419{
420 if(!diostream) return false;
421 if(!diostream->GetConfig()) return false;
422
423 sendbuffer = GEN_NEW XBUFFER();
424 GEN_XFACTORY_CREATE(xfile, Create_File())
425 GEN_XFACTORY_CREATE(xtimer, CreateTimer());
426
427 issending = false;
428
429 if(xfile && sendbuffer && xtimer)
430 {
431 if(diostream->Open())
432 {
433 if(diostream->WaitToConnected(timeout))
434 {
435 xthreadobex = CREATEXTHREAD(XTHREADGROUPID_DIOSTREAM, __L("DIOOBEXPUSH::Ini"),ThreadSend,(void*)this);
436 if(xthreadobex)
437 {
438 xthreadobex->Ini(false);
439 return true;
440 }
441 }
442
443 diostream->Close();
444 }
445 }
446
447 GEN_DELETE sendbuffer;
448
449 GEN_XFACTORY.Delete_File(xfile);
450 GEN_XFACTORY.DeleteTimer(xtimer);
451
452 sendbuffer = NULL;
453 xfile = NULL;
454 xtimer = NULL;
455
456 return false;
457}

839bool DIOOBEXPUSH::CheckSending()
840{
841 DIOOBEXPUSHFILE* obexpushfile = (DIOOBEXPUSHFILE*)sendfilelist.Get(indexfilelist);
842 DIOSTREAMBLUETOOTH* sio = (DIOSTREAMBLUETOOTH*)diostream;
843 XWORD wsize = 0;
844 XDWORD dsize = 0;
845 bool islastblock = false;
846
847 if((!sio)||(!obexpushfile))
848 {
849 if(obexpushfile) obexpushfile->SetError(DIOOBEXPUSHERROR_NOTCONNECTION);
850 return false;
851 }
852
853 if(sio->GetStatus()==DIOSTREAMSTATUS_GETTINGCONNECTION)
854 {
855 if(xtimer->GetMeasureSeconds()>=(XDWORD)asktimeout)
856 {
857 asktimeelapsed = xtimer->GetMeasureMilliSeconds();
858 for(int c=0;c<(int)sendfilelist.GetSize();c++)
859 {
860 obexpushfile = (DIOOBEXPUSHFILE*)sendfilelist.Get(c);
861 if(obexpushfile) obexpushfile->SetError(DIOOBEXPUSHERROR_NOTCONNECTION);
862 }
863
864 return false;
865 }
866
867 return true;
868 }
869
870 if(sio->GetStatus()==DIOSTREAMSTATUS_DISCONNECTED)
871 {
872 haveerror=true;
873 SetEvent(DIOOBEXPUSH_XFSMEVENT_ENDPUSH);
874 }
875
876 if(GetEvent()==DIOOBEXPUSH_XFSMEVENT_NONE) // No hay nuevos Eventos
877 {
878 switch(GetCurrentState())
879 {
880 case DIOOBEXPUSH_XFSMSTATE_NONE : break;
881
882 case DIOOBEXPUSH_XFSMSTATE_CONNECT : { XBUFFER answer;
883
884 if(!ReadAnswer(answer)) break;
885
886 //XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_RED, answer);
887
888 if(answer.Get()[0]==0xA0)
889 {
890 sendblocksize = (XWORD)answer.Get()[5];
891 sendblocksize <<=8;
892 sendblocksize |= (XWORD)answer.Get()[6];
893
894 SetEvent(DIOOBEXPUSH_XFSMEVENT_INIPUSH);
895 }
896 else
897 {
898 haveerror = true;
899 SetEvent(DIOOBEXPUSH_XFSMEVENT_ENDPUSH);
900 }
901 }
902 break;
903
904 case DIOOBEXPUSH_XFSMSTATE_INIPUSH : SetEvent(DIOOBEXPUSH_XFSMEVENT_SENDHEADPUSH);
905 break;
906
907 case DIOOBEXPUSH_XFSMSTATE_SENDHEADPUSH : SetEvent(DIOOBEXPUSH_XFSMEVENT_RECVHEADPUSH);
908 break;
909
910 case DIOOBEXPUSH_XFSMSTATE_SENDBLOCKPUSH : sendbuffer->Delete();
911
912 islastblock = ((filesize - readfilesize) >= sendblocksize)? false:true;
913
914 sendbuffer->Add((!islastblock)?(XBYTE)0x02:(XBYTE)0x82);
915
916 wsize = 0;
917 sendbuffer->Add((XWORD)wsize);
918
919 if(islastblock)
920 readfilesizeblock = (filesize - readfilesize);
921 else readfilesizeblock = sendblocksize -16;
922
923 sendbuffer->Add((!islastblock)?(XBYTE)0x48:(XBYTE)0x49);
924
925 wsize+=3;
926
927 wsize += (int)readfilesizeblock;
928 sendbuffer->Add((XWORD)wsize);
929
930 { XBUFFER datafile((XDWORD)readfilesizeblock, true);
931
932 xfile->Read(datafile.Get(),(XDWORD*)&readfilesizeblock);
933 if(!readfilesizeblock)
934 {
935 haveerror = true;
936 SetEvent(DIOOBEXPUSH_XFSMEVENT_ENDPUSH);
937 break;
938 }
939
940 sendbuffer->Add(datafile);
941 }
942
943 if(inlittleendian)
944 wsize = SwapWORD((XWORD)sendbuffer->GetSize());
945 else wsize = (XWORD)sendbuffer->GetSize();
946 sendbuffer->Set((XBYTE*)&wsize,2,1);
947
948 //XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_GREEN, (*sendbuffer));
949
950 sio->Write(sendbuffer->Get(),sendbuffer->GetSize());
951 sio->WaitToFlushOutXBuffer(5);
952
953 if(filesize) obexpushfile->SetSendPercent((XBYTE)((readfilesize*100)/filesize));
954
955 SetEvent(DIOOBEXPUSH_XFSMEVENT_RECVBLOCKPUSH);
956 break;
957
958 case DIOOBEXPUSH_XFSMSTATE_RECVHEADPUSH :
959 case DIOOBEXPUSH_XFSMSTATE_RECVBLOCKPUSH : { XBUFFER answer;
960
961 if(!ReadAnswer(answer)) break;
962
963 //XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_RED, answer);
964
965 switch(answer.Get()[0])
966 {
967 case 0x90 : readfilesize+=readfilesizeblock;
968 SetEvent(DIOOBEXPUSH_XFSMEVENT_SENDBLOCKPUSH);
969 break;
970
971 case 0xA0 : haveerror = false;
972 SetEvent(DIOOBEXPUSH_XFSMEVENT_ENDPUSH);
973 break;
974
975 default : if(GetEvent()==DIOOBEXPUSH_XFSMEVENT_NONE)
976 {
977 haveerror = true;
978 SetEvent(DIOOBEXPUSH_XFSMEVENT_ENDPUSH);
979 }
980 break;
981 }
982
983 }
984 break;
985
986 case DIOOBEXPUSH_XFSMSTATE_ENDPUSH : return false;
987 }
988 }
989 else // Nuevo evento
990 {
991 if(GetEvent()<DIOOBEXPUSH_LASTEVENT)
992 {
993 CheckTransition();
994
995 switch(GetCurrentState())
996 {
997 case DIOOBEXPUSH_XFSMSTATE_NONE : break;
998
999 case DIOOBEXPUSH_XFSMSTATE_CONNECT : sendisstart = false;
1000
1001 asktimeelapsed = xtimer->GetMeasureMilliSeconds();
1002
1003 sendbuffer->Delete();
1004
1005 sendbuffer->Add((XBYTE)0x80);
1006
1007 wsize = 0x0007;
1008 sendbuffer->Add((XWORD)wsize);
1009
1010 sendbuffer->Add((XBYTE)0x10);
1011 sendbuffer->Add((XBYTE)0x00);
1012
1013 wsize = DIOOBEXPUSH_MAXBUFFER;
1014 sendbuffer->Add((XWORD)wsize);
1015
1016 //XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_GREEN, (*sendbuffer));
1017
1018 sio->Write(sendbuffer->Get(),sendbuffer->GetSize());
1019 sio->WaitToFlushOutXBuffer(5);
1020 break;
1021
1022 case DIOOBEXPUSH_XFSMSTATE_INIPUSH : { xtimer->Reset();
1023
1024 obexpushfile->SetIsSend(false);
1025 obexpushfile->SetSendPercent(0);
1026
1027 xpath = (*obexpushfile->GetXPath());
1028
1029 if((!xfile)||(!xfile->Open(xpath,true))||(!sendbuffer))
1030 {
1031 haveerror = true;
1032 SetEvent(DIOOBEXPUSH_XFSMEVENT_ENDPUSH);
1033 break;
1034 }
1035
1036 readfilesizeblock = 0;
1037
1038 sendisstart = true;
1039 filesize = xfile->GetSize();
1040 readfilesize = 0;
1041 }
1042 break;
1043
1044 case DIOOBEXPUSH_XFSMSTATE_SENDHEADPUSH : sendbuffer->Delete();
1045
1046 sendbuffer->Add((XBYTE)0x02);
1047
1048 wsize = 0;
1049 sendbuffer->Add((XWORD)wsize);
1050
1051 sendbuffer->Add((XBYTE)0x01);
1052
1053 // Unicode 16 + 1 byte of head;
1054 wsize++;
1055 wsize += (XWORD)((obexpushfile->GetNameSend()->GetSize()+1)*2);
1056 sendbuffer->Add((XWORD)wsize);
1057
1058 XBUFFER xbufferexchange;
1059 (*obexpushfile->GetNameSend())->ConvertToBufferExchange(xbufferexchange, true, true);
1060 sendbuffer->Add(xbufferexchange->Get(), wsize);
1061
1062 sendbuffer->Add((XBYTE)0xC3);
1063 sendbuffer->Add((XDWORD)filesize);
1064
1065 if(inlittleendian)
1066 wsize = SwapWORD((XWORD)sendbuffer->GetSize());
1067 else wsize = (XWORD)sendbuffer->GetSize();
1068 sendbuffer->Set((XBYTE*)&wsize,2,1);
1069
1070 //XTRACE_PRINTDATABLOCKCOLOR(XTRACE_COLOR_GREEN, (*sendbuffer));
1071
1072 sio->Write(sendbuffer->Get(),sendbuffer->GetSize());
1073 sio->WaitToFlushOutXBuffer(5);
1074 break;
1075
1076 case DIOOBEXPUSH_XFSMSTATE_RECVHEADPUSH : break;
1077
1078 case DIOOBEXPUSH_XFSMSTATE_SENDBLOCKPUSH : break;
1079
1080 case DIOOBEXPUSH_XFSMSTATE_RECVBLOCKPUSH : break;
1081
1082 case DIOOBEXPUSH_XFSMSTATE_ENDPUSH : if(xfile) xfile->Close();
1083
1084 if(!haveerror)
1085 {
1086 obexpushfile->SetIsSend(true);
1087 obexpushfile->SetSendPercent(100);
1088 obexpushfile->SetSendTimeElapsed(xtimer->GetMeasureMilliSeconds());
1089 obexpushfile->SetError(DIOOBEXPUSHERROR_NONE);
1090
1091 if((indexfilelist+1<(int)sendfilelist.GetSize()))
1092 {
1093 indexfilelist++;
1094 SetEvent(DIOOBEXPUSH_XFSMEVENT_INIPUSH);
1095 }
1096 }
1097 else
1098 {
1099 for(int c=0;c<(int)sendfilelist.GetSize();c++)
1100 {
1101 obexpushfile = (DIOOBEXPUSHFILE*)sendfilelist.Get(c);
1102 if(obexpushfile)
1103 {
1104 obexpushfile->SetIsSend(false);
1105
1106 if(sendisstart)
1107 {
1108 obexpushfile->SetError(DIOOBEXPUSHERROR_SENDING);
1109 }
1110 else
1111 {
1112 if(sio->IsRefusedConnection())
1113 {
1114 obexpushfile->SetError(DIOOBEXPUSHERROR_REFUSEDCONNECTION);
1115 }
1116 else
1117 {
1118 obexpushfile->SetError(DIOOBEXPUSHERROR_NOTCONNECTION);
1119 }
1120 }
1121 }
1122 }
1123
1124 return false;
1125 }
1126 break;
1127 }
1128 }
1129 }
1130
1131 return true;
1132}
