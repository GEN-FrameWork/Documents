408bool DIOMODBUS_CLIENT::MakeCommand(DIOMODBUS_CLIENT_FUNCTION function, XBUFFER& ask, XBUFFER& answer, int timeout)
409{
410 if(!diostream) return false;
411 if(diostream->GetStatus() != DIOSTREAMSTATUS_CONNECTED) return false;
412
413 XBUFFER datawrite;
414
415 answer.Delete();
416
417 bool status = false;
418
419 switch(mode)
420 {
421 case DIOMODBUS_CLIENTMODE_UNKNOWN : break;
422
423 case DIOMODBUS_CLIENTMODE_RTU : { datawrite.Add((XBYTE) unit);
424 datawrite.Add((XBYTE) function);
425 datawrite.Add(ask);
426
427 XWORD crc16 = CRC16(datawrite);
428
429 datawrite.Add((XWORD)crc16);
430
431 diostream->Write(datawrite);
432
433 if(!diostream->WaitToFlushOutXBuffer(timeout)) return false;
434 if(!diostream->WaitToFilledReadingBuffer(3, timeout)) return false;
435
436 if(diostream->GetInXBuffer()->GetByte(0) != unit) return false;
437 if(diostream->GetInXBuffer()->GetByte(1) != function) return false;
438
439 XDWORD sizeanswer = 0;
440
441 switch(function)
442 {
443 case DIOMODBUS_CLIENT_FUNCTION_READ_COILS : DIOMODBUS_CLIENTMODE_SIZEANSWERVAR; break;
444 case DIOMODBUS_CLIENT_FUNCTION_READ_DISCRETE_INPUTS : DIOMODBUS_CLIENTMODE_SIZEANSWERVAR; break;
445 case DIOMODBUS_CLIENT_FUNCTION_READ_HOLDING_REGISTERS : DIOMODBUS_CLIENTMODE_SIZEANSWERVAR; break;
446 case DIOMODBUS_CLIENT_FUNCTION_READ_INPUT_REGISTERS : DIOMODBUS_CLIENTMODE_SIZEANSWERVAR; break;
447 case DIOMODBUS_CLIENT_FUNCTION_WRITE_SINGLE_COIL : DIOMODBUS_CLIENTMODE_SIZEANSWERFIX; break;
448 case DIOMODBUS_CLIENT_FUNCTION_WRITE_SINGLE_REGISTER : DIOMODBUS_CLIENTMODE_SIZEANSWERFIX; break;
449 case DIOMODBUS_CLIENT_FUNCTION_WRITE_MULTIPLE_COILS : DIOMODBUS_CLIENTMODE_SIZEANSWERFIX; break;
450 case DIOMODBUS_CLIENT_FUNCTION_WRITE_MULTIPLE_REGISTERS : DIOMODBUS_CLIENTMODE_SIZEANSWERFIX; break;
451 case DIOMODBUS_CLIENT_FUNCTION_READ_FILE_RECORD : break;
452 case DIOMODBUS_CLIENT_FUNCTION_WRITE_FILE_RECORD : break;
453 case DIOMODBUS_CLIENT_FUNCTION_MASK_WRITE_REGISTER : break;
454 case DIOMODBUS_CLIENT_FUNCTION_READWRITE_MULTIPLE_REGISTERS : break;
455 case DIOMODBUS_CLIENT_FUNCTION_READ_FIFO_QUEUE : break;
456 case DIOMODBUS_CLIENT_FUNCTION_ENCAPSULATED_INTERFACE_TRANSPORT : break;
457
458 }
459
460 if(!sizeanswer) break;
461
462 if(!diostream->WaitToFilledReadingBuffer(sizeanswer, timeout)) return false;
463
464 answer.Resize(sizeanswer);
465
466 if(!diostream->Read(answer)) return false;
467
468 XWORD CRCread = 0;
469 XWORD CRCcheck;
470
471 answer.Extract(CRCread, sizeanswer-2);
472
473 CRCcheck = CRC16(answer);
474
475 if(CRCcheck != CRCread) return false;
476
477 status = true;
478 }
479 break;
480
481 case DIOMODBUS_CLIENTMODE_ASCII : break;
482
483 case DIOMODBUS_CLIENTMODE_TCPIP : break;
484 }
485
486 return status;
487}

501XWORD DIOMODBUS_CLIENT::CRC16(XBUFFER& data)
502{
503 XWORD poly16 = 0xA001;
504 XWORD LSB;
505 XWORD crc = 0xFFFF;
506
507 for(int d=0; d<(int)data.GetSize(); d++)
508 {
509 crc = ((crc^data.Get()[d]) | 0xFF00) & (crc | 0x00FF);
510
511 for(int c=0; c<8; c++)
512 {
513 LSB=(crc & 0x0001);
514 crc=crc/2;
515
516 if(LSB) crc = crc ^ poly16;
517 }
518 }
519
520 SWAPWORD(crc);
521
522 return crc;
523}
