327bool DIOSNMP_BASE::Open(XCHAR* IPtarget, int port, bool checkavailableNMS, XCHAR* localIP)
328{
329 if(!IPtarget) return false;
330 if(!IPtarget[0]) return false;
331
332 diostreamudpcfg = GEN_NEW DIOSTREAMUDPCONFIG();
333 if(!diostreamudpcfg) return false;
334
335 diostreamudpcfg->SetMode(DIOSTREAMMODE_CLIENT);
336
337 if(localIP) diostreamudpcfg->GetLocalIP()->Set(localIP);
338
339 diostreamudpcfg->GetRemoteURL()->Set(IPtarget);
340 diostreamudpcfg->SetRemotePort(port);
341
342 bool status = false;
343
344 diostreamudp = (DIOSTREAMUDP*)GEN_DIOFACTORY.CreateStreamIO(diostreamudpcfg);
345 if(diostreamudp)
346 {
347 if(diostreamudp->Open())
348 {
349 if(checkavailableNMS)
350 {
351 //--------------------------------------------
352 // Write something to received a ICMP result.
353 // and check if something is the other side.
354
355 status = diostreamudp->WaitToConnected(3);
356 if(status)
357 {
358 XBYTE data[] = { 0xAA,0x55 };
359 diostreamudp ->Write(data,sizeof(data));
360
361 status = diostreamudp->WaitToDisconnected(3);
362 }
363
364 } else status = true;
365 }
366 }
367
368 if(!status) Close();
369
370 return status;
371}

627bool DIOSNMP::Get(DIOSNMP_OPERATION operation, XCHAR* community, XCHAR* OIDstr, DIOSNMP_XBER& xberout)
628{
629 #define MAX_READBUFFER 2048
630
631 if(!diostreamudpcfg) return false;
632 if(!diostreamudp) return false;
633
634 XBUFFER xbuffer;
635 XBUFFER xbufferdata;
636 XDWORD number = 0;
637 DIOSNMP_XBER xber;
638 DIOSNMP_XBER value;
639 DIOSNMP_XBER bindings;
640 DIOSNMP_XBER request;
641 DIOSNMP_XBER data_next_request;
642 DIOSNMP_XBER header;
643
644 // ----------------------------------------------------------------
645 // Get-Next
646
647 xber.SetOID(OIDstr); value.Sequence_AddTo(xber); // OID
648 xber.SetNULL(); value.Sequence_AddTo(xber); // NULL
649
650 XRAND* GEN_XFACTORY_CREATE(xrand, CreateRand())
651 if(xrand)
652 {
653 number = xrand->Max(65000);
654 GEN_XFACTORY.DeleteRand(xrand);
655 }
656
657 xber.SetINTEGER(number); bindings.Sequence_AddTo(xber); // Request-ID
658 xber.SetINTEGER(0); bindings.Sequence_AddTo(xber); // Error-status
659 xber.SetINTEGER(0); bindings.Sequence_AddTo(xber); // Error-index
660
661 request.Sequence_AddTo(value);
662 bindings.Sequence_AddTo(request);
663
664 data_next_request.Sequence_AddCommandTo(DIOSNMP_OPERATION_GETNEXTREQUEST, bindings);
665
666 xber.SetINTEGER(version); header.Sequence_AddTo(xber); // Version
667 xber.SetOCTETSTRING(community); header.Sequence_AddTo(xber); // Community
668
669 header.Sequence_AddTo(data_next_request);
670
671 header.GetDump(xbuffer);
672
673 XDWORD bw = diostreamudp->Write(xbuffer.Get(), xbuffer.GetSize());
674 if(bw!= xbuffer.GetSize()) return false;
675
676 xbuffer.Empty();
677 xbuffer.Resize(MAX_READBUFFER);
678
679 XTIMER* GEN_XFACTORY_CREATE(timeout, CreateTimer())
680 if(!timeout) return false;
681
682 do{ XDWORD sizeread = diostreamudp->GetInXBuffer()->GetSize();
683 if(sizeread)
684 {
685 XDWORD br = diostreamudp->Read(xbuffer.Get(), sizeread);
686 if(br)
687 {
688 timeout->Reset();
689 xbufferdata.Add(xbuffer.Get(), br);
690 }
691 }
692
693 if(timeout->GetMeasureSeconds() > 2) break;
694
695 } while(1);
696
697 GEN_XFACTORY.DeleteTimer(timeout);
698
699 if(xbufferdata.GetSize())
700 {
701 xberout.SetTagType(xbufferdata.Get()[0]);
702 xberout.SetSize(xbufferdata.Get()[1]);
703 xberout.GetData()->Add(&xbufferdata.Get()[2], xbufferdata.GetSize());
704 }
705
706 return xbufferdata.GetSize()?true:false;
707}
