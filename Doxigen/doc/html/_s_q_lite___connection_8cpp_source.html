105bool SQLITE_CONNECTION::Connect()
106{
107 if(!database)
108 {
109 database->Error(__L("DATABASE not initialized"));
110 return false;
111 }
112
113 // First check if options provided a db name
114
115 DB_SQL_STRING databasename;
116 DB_SQL_STRING databasetimeout;
117
118 if(!FindOption(__L("DATABASE"), &databasename))
119 {
120 database->Error(__L("DATABASE argument not supplied"));
121 return false;
122 }
123
124 // Attempt connection
125 XPATH xpathdatabase;
126
127 if(!FindOption(__L("PATH"), &xpathdatabase))
128 {
129 database->Error(__L("DATABASE argument not supplied"));
130 return false;
131 }
132
133 xpathdatabase.Add(databasename);
134
135 XBUFFER dbname;
136
137 xpathdatabase.ConvertToASCII(dbname);
138 int returncode = sqlite3_open(dbname.GetPtrChar(), &static_cast<SQLITE_DATABASE*>(database)->sqlite3database);
139 if(returncode)
140 {
141 database->Error((char*)sqlite3_errmsg(static_cast<SQLITE_DATABASE*>(database)->sqlite3database));
142 sqlite3_close(static_cast<SQLITE_DATABASE*>(database)->sqlite3database);
143
144 return false;
145 }
146
147 if(FindOption(__L("TIMEOUT"), &databasetimeout))
148 {
149 int timeoutseconds = databasetimeout.ConvertToInt();
150
151 returncode = sqlite3_busy_timeout(static_cast<SQLITE_DATABASE*>(database)->sqlite3database,timeoutseconds*1000);
152
153 if(returncode)
154 {
155 database->Error((char*)sqlite3_errmsg(static_cast<SQLITE_DATABASE*>(database)->sqlite3database));
156 sqlite3_close(static_cast<SQLITE_DATABASE*>(database)->sqlite3database);
157
158 return false;
159 }
160 }
161
162 return true;
163}
