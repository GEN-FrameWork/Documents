326
327 tcgetattr( STDIN_FILENO, &oldt ); /* Obtenemos atributos del terminal */
328 newt = oldt;
329 newt.c_lflag &= ~ICANON;
330 newt.c_lflag &= ~ECHO; /* Eliminamos el Echo */
