359bool MAINPROCANDROID::OnTouchEvent(AInputEvent* event)
360{
361 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE , __L("[ANDROID Event] INI OnTouchEvent"));
362
363 int32_t action = AMotionEvent_getAction(event);
364 size_t pointercount = AMotionEvent_getPointerCount(event);
365
366 if(!mouse) mouse = (INPANDROIDDEVICEMOUSE*)GEN_INPMANAGER.GetDevice(INPDEVICE_TYPE_MOUSE);
367
368 if(mouse)
369 {
370 for(size_t c = 0; c<pointercount; ++c)
371 {
372 int32_t pointerID = c;
373 int32_t pointerIndex = AMotionEvent_getPointerId(event,c);
374
375 if(pointerIndex >= mouse->GetCursors()->GetSize())
376 {
377 INPCURSOR* cursor = GEN_NEW INPCURSOR();
378 if(cursor)
379 {
380 cursor->SetID(INPCURSOR_ID(INPCURSOR_ID_TOUCHSCREEN1+pointerIndex));
381 cursor->SetHavePreSelect(false);
382
383 mouse->GetCursors()->Add(cursor);
384 }
385 }
386
387 if(pointerIndex >= mouse->GetButtons()->GetSize())
388 {
389 INPBUTTON* button = GEN_NEW INPBUTTON();
390 if(button)
391 {
392 button->SetID(INPBUTTON_ID_TOUCHSCREEN);
393 mouse->GetButtons()->Add(button);
394 }
395 }
396
397 float x = AMotionEvent_getX(event, pointerID);
398 float y = AMotionEvent_getY(event, pointerID);
399 float pressure = AMotionEvent_getPressure(event, pointerID);
400
401 switch(action & AMOTION_EVENT_ACTION_MASK)
402 {
403 case AMOTION_EVENT_ACTION_UP :
404 case AMOTION_EVENT_ACTION_POINTER_UP : { mouse->AddPosition(pointerIndex, x, y, false);
405 if(mouse->GetButton(pointerIndex))
406 {
407 mouse->GetButton(pointerIndex)->SetPressed(false);
408 mouse->GetButton(pointerIndex)->SetPressure(pressure);
409
410 if(mouse->GetButton(pointerIndex)->GetState() != INPBUTTON_STATE_UP)
411 {
412 mouse->GetButton(pointerIndex)->SetState(INPBUTTON_STATE_RELEASED);
413 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("> ANDROID Motion %d x %f y %f p %f a %d RELEASED"), pointerID,x,y,pressure,action);
414
415 }
416 else
417 {
418 mouse->GetButton(pointerIndex)->SetState(INPBUTTON_STATE_UP);
419 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("> ANDROID Motion %d x %f y %f p %f a %d UP"), pointerID,x,y,pressure,action);
420 }
421 }
422 }
423 break;
424 case AMOTION_EVENT_ACTION_MOVE :
425 case AMOTION_EVENT_ACTION_DOWN :
426 case AMOTION_EVENT_ACTION_POINTER_DOWN : { mouse->AddPosition(pointerIndex,x,y, true);
427 if(mouse->GetButton(pointerIndex))
428 {
429 mouse->GetButton(pointerIndex)->SetPressed(true);
430 mouse->GetButton(pointerIndex)->SetPressure(pressure);
431
432 if(mouse->GetButton(pointerIndex)->GetState() == INPBUTTON_STATE_UP)
433 {
434 mouse->GetButton(pointerIndex)->SetState(INPBUTTON_STATE_PRESSED);
435 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("> ANDROID Motion %d x %f y %f p %f a %d PRESSED"), pointerID,x,y,pressure,action);
436 }
437 else
438 {
439 mouse->GetButton(pointerIndex)->SetState(INPBUTTON_STATE_HOLD);
440 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("> ANDROID Motion %d x %f y %f p %f a %d HOLD"), pointerID,x,y,pressure,action);
441 }
442 }
443 }
444 break;
445
446 case AMOTION_EVENT_ACTION_CANCEL : if (mouse->GetButton(pointerIndex))
447 mouse->GetButton(pointerIndex)->SetState(INPBUTTON_STATE_UP);
448 break;
449 }
450 }
451 }
452
453
454 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE , __L("[ANDROID Event] END OnTouchEvent"));
455
456 return true;
457}

471bool MAINPROCANDROID::OnKeyboardEvent(AInputEvent* event)
472{
473 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE , __L("[ANDROID Event] INI OnKeyboardEvent Android code [%04x]"), AKeyEvent_getKeyCode(event));
474
475 INPDEVICE* keyboard = androidmain.GetKeyboard();
476 if(!keyboard) return false;
477
478 INPBUTTON* button = keyboard->GetButtonByCode(AKeyEvent_getKeyCode(event));
479 if(!button) return false;
480
481 XDWORD keyEvent = AKeyEvent_getAction(event);
482 switch(keyEvent)
483 {
484 case AKEY_EVENT_ACTION_UP : button->SetPressed(false);
485 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("> ANDROID Keyboard %d ID %d RELEASED"), keyEvent, button->GetID());
486 break;
487
488 case AKEY_EVENT_ACTION_DOWN : button->SetPressed(true);
489 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("> ANDROID Keyboard %d ID %d PRESSED"), keyEvent, button->GetID());
490 break;
491 }
492
493 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE , __L("[ANDROID Event] END OnKeyboardEvent"));
494
495 return false;
496}

952bool MAINPROCANDROID::GetPackageResourcePath(struct android_app* app, XSTRING& path)
953{
954 JNIEnv* env;
955 JavaVM* vm = app->activity->vm;
956 bool status = false;
957
958 vm->AttachCurrentThread(&env, NULL);
959
960 jclass clazz = env->GetObjectClass(app->activity->clazz);
961 jmethodID methodj = env->GetMethodID(clazz, "getPackageResourcePath", "()Ljava/lang/String;");
962
963 jstring jstr = (jstring)env->CallObjectMethod(app->activity->clazz, methodj);
964 if(jstr != NULL)
965 {
966 const char* str = env->GetStringUTFChars(jstr, NULL);
967 path = str;
968 env->ReleaseStringUTFChars(jstr, str);
969
970 status = true;
971 }
972
973 vm->DetachCurrentThread();
974
975 return status;
976}
