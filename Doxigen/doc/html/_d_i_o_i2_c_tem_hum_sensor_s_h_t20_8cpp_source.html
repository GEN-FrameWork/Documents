362bool DIOI2CTEMHUMSENSORSHT20::CheckCRC(XWORD datasensor, XBYTE checkvalue)
363{

365
366 remainder |= checkvalue;
367
368 XDWORD divisor = (XDWORD)DIOI2CTEMHUMSENSORSHT20_SHIFTED_DIVISOR;
369
370 for(int c=0; c < 16 ; c++)
371 {
372 if(remainder & (XDWORD)1 << (23 - c))
373 {
374 remainder ^= divisor;
