563bool DIOLINUXWIFIMANAGERMODE::WPA_GetListRemoteAP(XVECTOR<XSTRING*>& nameAPs)
564{
565 XSTRING prefixcommand;
566 XSTRING command;
567 XSTRING output;
568 bool status;
569
570
571 nameAPs.DeleteContents();
572 nameAPs.DeleteAll();
573
574 prefixcommand.Format(DIOLINUXWIFIMANAGERMODE_PREFIXCOMMAND, netinterface.Get());
575
576 command.Format(__L("%s scan") , prefixcommand.Get());
577 status = WPA_CLI_Command(command, output);
578 if(!status) return false;
579
580 GEN_XSLEEP.Seconds(3);
581
582 command.Format(__L("%s scan_results") , prefixcommand.Get());
583 status = WPA_CLI_Command(command, output);
584
585 int indexfr = output.Find(__L("\n"), true);
586 if(indexfr == XSTRING_NOTFOUND) return false;
587
588 indexfr++;
589
590 int index = 0;
591 int indexstart = 0;
592
593 do{
594 index = output.Find(__L("\n"), true, indexfr);
595 if(index == XSTRING_NOTFOUND ) break;
596
597 for(int c=index; c>0 ; c--)
598 {
599 if(output.Get()[c] == __C(']'))
600 {
601 indexstart = c;
602 break;
603 }
604 }
605
606 indexfr = index+1;
607
608 if(indexstart)
609 {
610 XSTRING* newstr = GEN_NEW XSTRING();
611 if(newstr)
612 {
613 output.Copy(indexstart+1, index, (*newstr));
614 newstr->DeleteCharacter(__C('\t'), XSTRINGCONTEXT_FROM_FIRST);
615
616 nameAPs.Add(newstr);
617 }
618 }
619
620 } while(index != XSTRING_NOTFOUND);
621
622 return true;
623}
