1037XDWORD XFILERIFF::AdjustSizeOfLists(XFILERIFF_LIST_NODE* node)
1038{
1039 XFILERIFF_LIST* list = NULL;
1040 XDWORD currentsize = 0;
1041
1042 list = node->GetData();
1043 if(!list) return 0;
1044
1045 if(!list->IsTypeList()) return list->GetSize();
1046
1047 for(int c=0; c<node->GetNChildren(); c++)
1048 {
1049 XFILERIFF_LIST_NODE* subnode = node->GetChild(c);
1050 if(subnode)
1051 {
1052 XFILERIFF_LIST* sublist = subnode->GetData();
1053 if(sublist)
1054 {
1055 if(sublist->IsTypeList())
1056 {
1057 currentsize += (AdjustSizeOfLists(subnode) + (sizeof(XDWORD) * 3));
1058 }
1059 else
1060 {
1061 currentsize += (sublist->GetSize() + (sizeof(XDWORD) * 2));
1062 }
1063 }
1064 }
1065 }
1066
1067 list->SetSize(currentsize + sizeof(XDWORD));
1068
1069 if(!WriteListToFile(list)) return false;
1070
1071 return currentsize;
