800bool GRPSCREEN::CreateViewport(XCHAR* ID, float posx, float posy, float width, float height, XDWORD canvasposx, XDWORD canvasposy, XDWORD canvaswidth, XDWORD canvasheight)
801{
802 GRPVIEWPORT* viewport = GEN_NEW GRPVIEWPORT();
803 if(!viewport) return false;
804
805 viewport->GetID()->Set(ID);
806 viewport->SetPosition(posx, posy);
807 viewport->SetSize(width, height);
808
809 GRPPROPERTIES canvasproperties;
810 float cwidth = (float)canvaswidth;
811 float cheight = (float)canvasheight;
812
813 if(width > cwidth) cwidth = width;
814 if(height > cheight) cheight = height;
815
816 canvasproperties.CopyPropertysFrom(this);
817 canvasproperties.SetPosition(canvasposx, canvasposy);
818 canvasproperties.SetSize((XDWORD)cwidth, (XDWORD)cheight);
819
820 if(viewport->CreateCanvas(canvasproperties))
821 {
822 viewport->SetCanvasPosition(posx, posy);
823 viewports.Add(viewport);
824
825 } else return false;
826
827 return true;
828 }

1042bool GRPSCREEN::Rotate(XBYTE* target, GRPCANVAS* screencanvas)
1043{
1044 XDWORD pixelsize = screencanvas->GetBytesperPixel();
1045 XBYTE* source = screencanvas->Buffer_Get();
1046
1047 if(!source || !target)
1048 {
1049 return false;
1050 }
1051
1052 switch(rotation)
1053 {
1054 case GRPSCREENROTATION_NONE : break;
1055
1056 case GRPSCREENROTATION_90_CLOCKWISE : { int target_offset = 0; //((x * height) + (height - y - 1)) * pixelsize;
1057 int source_offset = 0; //(y * width + x) * pixelsize;
1058
1059 for(XDWORD y=0; y<10 /*height*/; y++)
1060 {
1061 for(XDWORD x=0; x<width; x++)
1062 {
1063 //target_offset = (y * width + x) * pixelsize;
1064 //source_offset = (y * height + (width - x - 1)) * pixelsize;
1065
1066 target_offset += pixelsize;
1067 source_offset += pixelsize;
1068
1069 memcpy(target + target_offset, source + source_offset, pixelsize);
1070 }
1071 }
1072 }
1073 break;
1074
1075 case GRPSCREENROTATION_90_ANTICLOCKWISE :
1076 break;
1077
1078 case GRPSCREENROTATION_180 :
1079 break;
1080 }
1081
1082 return true;
1083}
