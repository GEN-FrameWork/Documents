355bool XFILEDFU::ReadAll()
356{
357 if(!xfilebase) return false;
358
359 bool status = false;
360
361 images.DeleteContents();
362 images.DeleteAll();
363
364 xfilebase->SetPosition(0);
365
366 //------------------------------------------
367 // Read Prefix
368
369 status = xfilebase->Read(prefix.signature, XFILEDFU_PREFIX_SIGNATUREPREXIF_MAX);
370 if(status) status = xfilebase->Read(&prefix.version, 1);
371 if(status) status = xfilebase->Read((XBYTE*)&prefix.imagesize, sizeof(XDWORD));
372 if(status) status = xfilebase->Read(&prefix.ntargets, sizeof(XBYTE));
373
374 if(status)
375 {
376 for(int c=0; c<prefix.ntargets; c++)
377 {
378 XFILEDFU_IMAGE* image = Read_Image();
379 if(image)images.Add(image);
380 }
381 }
382
383 //------------------------------------------
384 // Read Suffix
385
386 status =xfilebase->Read((XBYTE*)&suffix.BCDdevice, sizeof(XWORD));
387 if(status) status = xfilebase->Read((XBYTE*)&suffix.IDproduct, sizeof(XWORD));
388 if(status) status = xfilebase->Read((XBYTE*)&suffix.IDvendor, sizeof(XWORD));
389 if(status) status = xfilebase->Read((XBYTE*)&suffix.BCDDFU, sizeof(XWORD));
390 if(status) status = xfilebase->Read((XBYTE*)&suffix.signature, XFILEDFU_PREFIX_SIGNATURESUFFIX_MAX);
391 if(status) status = xfilebase->Read(&suffix.length, 1);
392 if(status) status = xfilebase->Read((XBYTE*)&suffix.CRC, sizeof(XDWORD));
393
394
395 //--------------------------------------------
396 // Check CRC used in DFU.
397
398 status = CheckCRC();
399
400 return status;
401}

476XFILEDFU_IMAGE* XFILEDFU::Read_Image()
477{
478 XFILEDFU_IMAGE* image = GEN_NEW XFILEDFU_IMAGE();
479 if(!image) return NULL;
480
481 bool status = false;
482
483 status = xfilebase->Read(image->signature, XFILEDFU_PREFIX_SIGNATUREIMAGE_MAX);
484 if(status) xfilebase->Read(&image->alternatesetting, 1);
485 if(status) xfilebase->Read((XBYTE*)&image->targetnameexist, sizeof(XDWORD));
486 if(status) xfilebase->Read((XBYTE*)&image->targetname, XFILEDFU_IMAGEPREFIX_NAME_MAX);
487 if(status) xfilebase->Read((XBYTE*)&image->targetsize, sizeof(XDWORD));
488 if(status) xfilebase->Read((XBYTE*)&image->nelements, sizeof(XDWORD));
489
490 for(XDWORD c=0; c<image->nelements; c++)
491 {
492 XFILEDFU_ELEMENT* element = Read_Element();
493 if(element) image->elements.Add(element);
494 }
495
496 if(!status)
497 {
498 GEN_DELETE image;
499 return NULL;
500 }
501
502 return image;
503}

516{
517 XFILEDFU_ELEMENT* element = GEN_NEW XFILEDFU_ELEMENT();
518 if(!element) return NULL;
519
520 bool status = false;
521
522 status = xfilebase->Read((XBYTE*)&element->address, sizeof(XDWORD));
523 if(status) xfilebase->Read((XBYTE*)&element->size, sizeof(XDWORD));
524 if(status)
525 {
526 status = false;
527 if(element->size)
528 {
529 status = element->data.Resize(element->size);
530 if(status) status = xfilebase->Read(element->data.Get(), element->data.GetSize());
531 }
532 }
533
534 if(!status)
535 {
536 GEN_DELETE element;
537 return NULL;
538 }
539
540 return element;

553bool XFILEDFU::CheckCRC()
554{
555 XBUFFER buffer;
556 bool status;
557
558 xfilebase->SetPosition(0);
559 buffer.Resize((XDWORD)(xfilebase->GetSize() - sizeof(XDWORD)));
560
561 status = xfilebase->Read(buffer.Get(), buffer.GetSize());
562 if(status)
563 {
564 XDWORD CRC = 0xFFFFFFFF;
565 for (XDWORD i = 0; i < buffer.GetSize(); i++)
566 {
567 CRC = CRCtable[(CRC ^ buffer.Get()[i]) & 0xff] ^ (CRC >> 8);
568 }
569
570 if(CRC != suffix.CRC) status = false;
571 }
572
573 return status;
574}
