249DIOSTREAM* DIOWINDOWSFACTORY::CreateStreamIO(DIOSTREAMCONFIG* config)
250{
251 if(!config)
252 {
253 return NULL;
254 }
255
256 DIOSTREAM* _class = NULL;
257
258 if(!config->IsTLS())
259 {
260 switch(config->GetType())
261 {
262 case DIOSTREAMTYPE_UNKNOWN : return NULL;
263
264 #ifdef DIO_STREAMUART_ACTIVE
265 case DIOSTREAMTYPE_UART : _class = GEN_NEW DIOWINDOWSSTREAMUART();
266 break;
267 #endif

277 #endif

281 break;
282 #endif

284 #ifdef DIO_STREAMTCPIP_ACTIVE
285 case DIOSTREAMTYPE_TCPIP : if(config->GetMode() == DIOSTREAMMODE_SERVERMULTISOCKET)
286 {
287 _class = GEN_NEW DIOWINDOWSSTREAMTCPIPSERVER();
288 }
289 else
290 {
291 _class = GEN_NEW DIOWINDOWSSTREAMTCPIP();
292 }
293 break;
294 #endif
295
296 #if defined(DIO_STREAMBLUETOOTH_ACTIVE) || defined(DIO_STREAMBLUETOOTHLE_ACTIVE)

323 //_class = GEN_NEW DIOWINDOWSSTREAMTCPIPSERVER();
