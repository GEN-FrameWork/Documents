470 bool Received_Message (XBUFFER& readbuffer, XBYTE& typemsg, XWORD& legacyversion, XWORD& sizemsg)
471 {
472 XBUFFER message;
473 XDWORD sizeread;
474 bool status = false;
475
476 readbuffer.Resize(5);
477 sizeread = Read(readbuffer.Get(), readbuffer.GetSize());
478 if(sizeread == readbuffer.GetSize())
479 {
480 readbuffer.Get(typemsg);
481 readbuffer.Get(legacyversion);
482 readbuffer.Get(sizemsg);
483
484 message.Resize(sizemsg);
485
486 sizeread = Read(message.Get(), message.GetSize());
487 if(sizeread == message.GetSize())
488 {
489 readbuffer.Add(message);
490 status = true;
491 }
492 }
493
494 if(!status)
495 {
496 readbuffer.Empty();
497 }

534 }

539 GEN_XFACTORY.DeleteRand(xrand);

636 bool Update ()
637 {
638 if(!fsmachine)
639 {
640 return false;
641 }
642
643 if(fsmachine->GetEvent() == DIOSTREAMTLS_XFSMEVENT_NONE) // Not GEN_NEW event
644 {
645 switch(fsmachine->GetCurrentState())
646 {
647 case DIOSTREAMTLS_XFSMSTATE_NONE : break;
648 case DIOSTREAMTLS_XFSMSTATE_OPEN : break;
649 case DIOSTREAMTLS_XFSMSTATE_HANDSHAKE : break;
650 case DIOSTREAMTLS_XFSMSTATE_CONNECTED : break;
651 case DIOSTREAMTLS_XFSMSTATE_DISCONNECTED : break;
652
653 case DIOSTREAMTLS_XFSMSTATE_CLOSE : fsmachine->SetEvent(DIOSTREAMTLS_XFSMEVENT_NONE);
654 break;
655 }
656 }
657 else // New event
658 {
659 if(fsmachine->GetEvent() < DIOSTREAMTLS_LASTEVENT)
660 {
661 fsmachine->CheckTransition();
662
663 switch(fsmachine->GetCurrentState())
664 {
665 case DIOSTREAMTLS_XFSMSTATE_NONE : break;
666
667 case DIOSTREAMTLS_XFSMSTATE_OPEN : fsmachine->SetEvent(DIOSTREAMTLS_XFSMEVENT_HANDSHAKE);
668 break;
669
670 case DIOSTREAMTLS_XFSMSTATE_HANDSHAKE : if(HandShake_Client_ClientHello())
671 {
672 fsmachine->SetEvent(DIOSTREAMTLS_XFSMEVENT_CONNECTED);
673 }
674 else
675 {
676 fsmachine->SetEvent(DIOSTREAMTLS_XFSMEVENT_DISCONNECTED);
677 }
678 break;
679
680 case DIOSTREAMTLS_XFSMSTATE_CONNECTED : break;
681
682 case DIOSTREAMTLS_XFSMSTATE_DISCONNECTED : fsmachine->SetEvent(DIOSTREAMTLS_XFSMEVENT_CLOSE);
683 break;
684
685 case DIOSTREAMTLS_XFSMSTATE_CLOSE : break;
686 }
687 }
688 }
689
690 return true;
691 }
