216bool XWINDOWSTHREAD::Ini(bool run)
217{
218 statusfunc = XTHREADSTATUS_STOP;
219
220 thhandle = CreateThread( NULL, stacksize, Callback, this, 0, NULL);
221 if(thhandle == NULL)
222 {
223 return false;
224 }
225
226 if(run)
227 {
228 if(!Run(true))
229 {
230 CloseHandle(thhandle);
231 return false;
232 }
233 }
234
235 if((priority != XTHREADPRIORITY_UNKNOWN) && (priority != XTHREADPRIORITY_NORMAL))
236 {
237 switch(priority)
238 {
239 case XTHREADPRIORITY_LOW : SetThreadPriority(thhandle, THREAD_PRIORITY_LOWEST); break;
240 case XTHREADPRIORITY_HIGH : SetThreadPriority(thhandle, THREAD_PRIORITY_HIGHEST); break;
241 case XTHREADPRIORITY_REALTIME : SetThreadPriority(thhandle, THREAD_PRIORITY_TIME_CRITICAL); break;
242 default : break;
243 }
244 }
245
246 return true;
247}
