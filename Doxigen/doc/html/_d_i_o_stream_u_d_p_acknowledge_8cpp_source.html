232XDWORD DIOSTREAMUDPACKNOWLEDGE::Write(XBYTE* buffer, XDWORD size)
233{
234 if(!diostream) return 0;
235 if(!size) return 0;
236
237 if(addressremote.IsEmpty()) return 0;
238
239 DIOSTREAMUDP* diostreamudp = (DIOSTREAMUDP*)diostream;
240 XDWORD bw = 0;
241
242 if(diostreamudp->WriteDatagram(addressremote, portremote, buffer, size)) bw = size;
243
244 if(diostreamudp->WaitToWriteDatagramsEmpty(timeout))
245 {
246 //XTRACE_PRINTCOLOR(1, __L("Write UDP %s:%d : %s"), addressremote.Get(), portremote, bw==size??__L("[Ok.]"):__L("[Error!]"));
247
248 if(diostreamudp->WaitToGetFirstDatagram(false, timeout) != DIOSTREAMUDP_NOTFOUND)
249 {
250 XSTRING address;
251 XWORD port = 0;
252 XBUFFER xbuffer;
253 XDWORD hashresult[2] = { 0, 0 };
254
255 diostreamudp->ReadDatagram(address, port, xbuffer);
256
257 xbuffer.Extract(hashresult[0]);
258
259 HASHCRC32 hashcrc32;
260
261 hashcrc32.ResetResult();
262 hashcrc32.Do(buffer, size);
263
264 hashresult[1] = hashcrc32.GetResultCRC32();
265
266 if(hashresult[0] != hashresult[1])
267 {
268 //XTRACE_PRINTCOLOR(1, __L("Write received bad CRC!! %08X <> %08X"), hashresult, hashcrc32.GetResultCRC32());
269 bw = 0;
270 }
271 }
272 else
273 {
274 //XTRACE_PRINTCOLOR(1, __L("Write received NOT read"));
275 bw = 0;
276 }
277 }
278 else
279 {
280 //XTRACE_PRINTCOLOR(1, __L("Do not write"));
281 bw = 0;
282 }
283
284 if(!bw)
285 {
286 diostream->Close();
287 diostream->Open();
288 diostream->WaitToConnected(5);
289 }
290
291 return bw;
292}

434void DIOSTREAMUDPACKNOWLEDGE::ThreadRunFunction(void* param)
435{
436 DIOSTREAMUDPACKNOWLEDGE* diostreamudpacknowledge = (DIOSTREAMUDPACKNOWLEDGE*)param;
437 if(!diostreamudpacknowledge) return;
438
439 DIOSTREAMUDP* diostreamudp = diostreamudpacknowledge->diostream;
440 if(!diostreamudp) return;
441
442 XBUFFER data;
443 XSTRING address;
444 XWORD port;
445
446 if(diostreamudp->ReadDatagram(address, port, data))
447 {
448 HASHCRC32 hashcrc32;
449
450 hashcrc32.ResetResult();
451 if(hashcrc32.Do(data.Get(), data.GetSize()))
452 {
453 XBUFFER result;
454 XDWORD hashresult = hashcrc32.GetResultCRC32();
455
456 result.Add(hashresult);
457
458 diostreamudp->WriteDatagram(address, port, result);
459 if(diostreamudp->WaitToWriteDatagramsEmpty(diostreamudpacknowledge->timeout))
460 {
461 diostreamudpacknowledge->AddDatagram(false, address, port, data.Get(), data.GetSize());
462 }
463 }
464 }
465}
