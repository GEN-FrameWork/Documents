404GRPBITMAPFILE_TYPE GRPBITMAPFILE::GetTypeFromExtenxion(XCHAR* xpath)
405{
406 XPATH _xpath;
407 XSTRING ext;
408 GRPBITMAPFILE_TYPE type = GRPBITMAPFILE_TYPE_UNKNOWN;
409
410 _xpath = xpath;
411
412 _xpath.GetExt(ext);
413
414 if(!ext.Compare(__L(".jpg"), true)) type = GRPBITMAPFILE_TYPE_JPG;
415 if(!ext.Compare(__L(".jpeg"), true)) type = GRPBITMAPFILE_TYPE_JPG;
416 if(!ext.Compare(__L(".png"), true)) type = GRPBITMAPFILE_TYPE_PNG;
417 if(!ext.Compare(__L(".bmp"), true)) type = GRPBITMAPFILE_TYPE_BMP;
418 if(!ext.Compare(__L(".tga"), true)) type = GRPBITMAPFILE_TYPE_TGA;
419
420 return type;
421}

436GRPBITMAP* GRPBITMAPFILE::Load(XCHAR* xpath, GRPPROPERTYMODE forcemode)

636GRPBITMAPSEQUENCE* GRPBITMAPFILE::LoadSequence(XCHAR* namemask, int inisequence, int endsequence)
637{
638 if(xpath.IsEmpty()) return NULL;
639
640 GRPBITMAPSEQUENCE* bitmapsequence = GEN_NEW GRPBITMAPSEQUENCE();
641 if(!bitmapsequence) return NULL;
642
643 XSTRING namefile;
644
645 //xpath.GetNamefileExt(namefile);
646 //if(!namefile.IsEmpty()) return NULL;
647
648 // The sequence is in individual files in a directory. Sequence numbers must be continuos
649 int nfiles = inisequence;
650
651 while((nfiles < endsequence)|| (endsequence < 0))
652 {
653 XPATH xpathtemp;
654 bool existfile = false;
655
656 namefile.Format(namemask, nfiles);
657 xpathtemp.Create(2, xpath.Get(), namefile.Get());
658
659 XFILE* GEN_XFACTORY_CREATE(xfile, Create_File())
660 if(xfile)
661 {
662 existfile = xfile->Exist(xpathtemp);
663 GEN_XFACTORY.Delete_File(xfile);
664 }
665
666 if(!existfile) break;
667
668 GRPBITMAP* bitmap = Load(xpathtemp);
669 if(!bitmap) break;
670
671 bitmapsequence->AddFrame(bitmap);
672
673 nfiles++;
674 }
675
676 return bitmapsequence;
677}
