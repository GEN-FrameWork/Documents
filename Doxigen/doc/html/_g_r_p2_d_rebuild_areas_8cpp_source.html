368bool GRP2DREBUILDAREAS::CreateRebuildArea(double x, double y, double width, double height, void* extra_data)
369{
370 double _x = (x - excessedge);
371 double _y = (y - excessedge);
372 double _width = (width + (excessedge*2));
373 double _height = (height + (excessedge*2));
374
375 if(_x < 0) _x = 0;
376 if(_y < 0) _y = 0;
377
378 GRPBITMAP* bitmap = GetBitmap(_x, _y, _width, _height);
379 if(!bitmap) return false;
380
381 GRP2DREBUILDAREA* newarea = GEN_NEW GRP2DREBUILDAREA();
382 if(!newarea)
383 {
384 GEN_DELETE bitmap;
385 return false;
386 }
387
388 newarea->SetXPos(_x);
389 newarea->SetYPos(_y);
390 newarea->SetBitmap(bitmap);
391 newarea->SetExtraData(extra_data);
392
393 return areas.Add(newarea);
394}
