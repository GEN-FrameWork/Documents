220bool CIPHERRSA::SetKey(CIPHERKEY* key, bool integritycheck)
221{
222 if(!key) return false;
223
224 switch(key->GetType())
225 {
226 case CIPHERKEYTYPE_SYMMETRICAL : return false;
227
228 case CIPHERKEYTYPE_RSA_PUBLIC : { CIPHERKEYPUBLICRSA* publickey = (CIPHERKEYPUBLICRSA*)key;
229
230 publickey->Get(context.N, context.E); // Assign key public to context
231
232 if(integritycheck)
233 {
234 if(!CheckContextPublicKey()) return false;
235 }
236 }
237 break;
238
239 case CIPHERKEYTYPE_RSA_PRIVATE : { CIPHERKEYPRIVATERSA* privatekey = (CIPHERKEYPRIVATERSA*)key;
240 XMPINTEGER P1;
241 XMPINTEGER Q1;
242
243 privatekey->Get(context.P, context.Q, context.D); // Assign key public to context
244
245 if(!P1.SubtractionSigned(&context.P, 1)) return false;
246 if(!Q1.SubtractionSigned(&context.Q, 1)) return false;
247 if(!context.DP.Module(&context.DP, &context.D, &P1)) return false;
248 if(!context.DQ.Module(&context.DQ, &context.D, &Q1)) return false;
249 if(!context.QP.ModularInverse(&context.Q, &context.P)) return false;
250
251 if(integritycheck)
252 {
253 if(!CheckContextPrivateKey()) return false;
254 }
255
256 }
257 break;
258 default : break;
259
260 }
261
262 return CIPHER::SetKey(key);
263}

438{
439 if(!hash) return false;
440
441 XBER beroui;
442 XBER bernull;
443 XBER berhash;
444 XBER berseq1;
445 XBER berseq2;

448
449 if(!OUINoSignstring) return false;

451 if(!OUINoSignstring->GetSize()) return false;

541bool CIPHERRSA::GenerateKeys(XDWORD nbits, int exponent, CIPHERKEYPUBLICRSA& publickey, CIPHERKEYPRIVATERSA& privatekey, XMPINTEGER_FUNCRANDOM funcrandom, void* paramrandom)
542{
543 XMPINTEGER P1;
544 XMPINTEGER Q1;
545 XMPINTEGER H;
546 XMPINTEGER G;
547 bool status;
548
549 if(nbits<128 || exponent<3 ) return false;
550
551 P1.Ini();
552 Q1.Ini();
553 H.Ini();
554 G.Ini();
555
556 // find primes P and Q with Q < P so that: GCD( E, (P-1)*(Q-1) ) == 1
557 status = context.E.LeftSet(exponent);
558 if(status)
559 {
560 do{ if(!context.P.GeneratePrime((nbits+1)>>1, 0, funcrandom?funcrandom:(XMPINTEGER_FUNCRANDOM)GenerateRandom, NULL)) return false;
561 if(!context.Q.GeneratePrime((nbits+1)>>1, 0, funcrandom?funcrandom:(XMPINTEGER_FUNCRANDOM)GenerateRandom, NULL)) return false;
562
563 if(context.P.CompareSignedValues(context.Q) < 0) context.P.Swap(&context.Q);
564
565 if(context.P.CompareSignedValues(context.Q) == 0) continue;
566
567 if(!context.N.Multiplication(&context.P, &context.Q)) return false;
568
569 if(context.N.GetMSB()!=nbits) continue;
570
571 if(!P1.SubtractionSigned(&context.P, 1)) return false;
572 if(!Q1.SubtractionSigned(&context.Q, 1)) return false;
573
574 if(!H.Multiplication(&P1, &Q1)) return false;
575
576 if(!G.GreatestCommonDivisor(&G, &context.E, &H)) return false;
577
578 } while(G.CompareSignedValues(1) != 0);
579
580
581 // D = E^-1 mod ((P-1)*(Q-1))
582 // DP = D mod (P - 1)
583 // DQ = D mod (Q - 1)
584 // QP = Q^-1 mod P
585
586 if(!context.D.ModularInverse(&context.E, &H)) return false;
587 if(!context.DP.Module(&context.DP, &context.D, &P1)) return false;
588 if(!context.DP.Module(&context.DQ, &context.D, &Q1)) return false;
589 if(!context.QP.ModularInverse(&context.Q, &context.P)) return false;
590
591 context.len = (context.N.GetMSB() + 7) >> 3;
592 }
593
594 P1.End();
595 Q1.End();
596 H.End();
597 G.End();
598
599 if(!status)
600 {
601 context.End();
602 }
603 else
604 {
605 publickey.Set(context.N, context.E);
606 privatekey.Set(context.P, context.Q, context.D);
607 }
608
609 return status;
610}

642bool CIPHERRSA::CheckContextPrivateKey()
643{
644 XMPINTEGER PQ;
645 XMPINTEGER DE;
646 XMPINTEGER P1;
647 XMPINTEGER Q1;
648 XMPINTEGER H;
649 XMPINTEGER I;
650 XMPINTEGER G;
651 XMPINTEGER G2;
652 XMPINTEGER L1;
653 XMPINTEGER L2;
654 XMPINTEGER DP;
655 XMPINTEGER DQ;
656 XMPINTEGER QP;
657
658 if(!CheckContextPublicKey()) return false;
659
660 if((!context.P.GetLimbs()) || (!context.Q.GetLimbs()) || (!context.D.GetLimbs())) return false;
661
662 PQ.Ini();
663 DE.Ini();
664 P1.Ini();
665 Q1.Ini();
666 H.Ini();
667 I.Ini();
668 G.Ini();
669 G2.Ini();
670 L1.Ini();
671 L2.Ini();
672 DP.Ini();
673 DQ.Ini();
674 QP.Ini();
675
676 if(!PQ.Multiplication(&context.P, &context.Q)) return false;
677 if(!DE.Multiplication(&context.D, &context.E)) return false;
678 if(!P1.SubtractionSigned(&context.P, 1)) return false;
679 if(!Q1.SubtractionSigned(&context.Q, 1)) return false;
680 if(!H.Multiplication(&P1, &Q1)) return false;
681 if(!G.GreatestCommonDivisor(&G, &context.E, &H)) return false;
682 if(!G2.GreatestCommonDivisor(&G2, &P1, &Q1)) return false;
683 if(!L1.Division(&L1, &L2, &H, &G2 )) return false;
684 if(!I.Module(&I, &DE, &L1)) return false;
685 if(!DP.Module( &DP, &context.D, &P1)) return false;
686 if(!DQ.Module(&DQ, &context.D, &Q1)) return false;
687 if(!QP.ModularInverse(&context.Q, &context.P)) return false;
688
689 // Check for a valid PKCS1v2 private key
690
691 if((PQ.CompareSignedValues(context.N ) != 0) || (DP.CompareSignedValues(context.DP) != 0) ||
692 (DQ.CompareSignedValues(context.DQ) != 0) || (QP.CompareSignedValues(context.QP) != 0) ||
693 (L2.CompareSignedValues(0) != 0) || (I.CompareSignedValues(1) != 0) ||
694 (G.CompareSignedValues(1) != 0)) return false;
695
696 PQ.End();
697 DE.End();
698 P1.End();
699 Q1.End();
700 H.End();
701 I.End();
702 G.End();
703 G2.End();
704 L1.End();
705 L2.End();
706 DP.End();
707 DQ.End();
708 QP.End();
709
710 return true;
711}

731bool CIPHERRSA::Cipher_PKCS1_V15(XBYTE* buffer, XDWORD size, XBUFFER& output, CIPHERKEYTYPE keytypetouse, CIPHERRSAPKCS1VERSION pkcs1version, XMPINTEGER_FUNCRANDOM funcrandom, void* paramrandom)
732{
733 XBUFFER input;
734 int nb_pad;
735 bool status;
736
737 if(!funcrandom) return false;
738
739 int keysize = GetKeySizeInBytes(keytypetouse);
740 if(!keysize) return false;
741
742 //if(keysize < (size + 11)) return false;
743
744 nb_pad = keysize - 3 - size;
745
746 input.Add((XBYTE)0);
747
748 if(keytypetouse == CIPHERKEYTYPE_RSA_PUBLIC)
749 {
750 XBYTE data = 0;
751
752 input.Add((XBYTE)CIPHERRSA_CIPHERDATA);
753
754 while(nb_pad > 0)
755 {
756 if(funcrandom(&data, 1, paramrandom))
757 {
758 input.Add((XBYTE)data);
759 nb_pad--;
760 }
761 }
762 }
763 else
764 {
765 input.Add((XBYTE)CIPHERRSA_SINGDATA);
766
767 while(nb_pad-- > 0)
768 {
769 input.Add((XBYTE)0xFF);
770 }
771 }
772
773
774 input.Add((XBYTE)0);
775 input.Add(buffer,size);
776
777 status = false;
778
779 switch(keytypetouse)
780 {
781 case CIPHERKEYTYPE_RSA_PUBLIC : status = DoRSAPublicOperation(input, output); break;
782 case CIPHERKEYTYPE_RSA_PRIVATE : status = DoRSAPrivateOperation(input, output, funcrandom, paramrandom); break;
783 default : break;
784 }
785
786 return status;
787}

807bool CIPHERRSA::Uncipher_PKCS1_V15(XBYTE* buffer, XDWORD size, XBUFFER& output, CIPHERKEYTYPE keytypetouse, CIPHERRSAPKCS1VERSION pkcs1version, XMPINTEGER_FUNCRANDOM funcrandom, void* paramrandom)
808{
809 XBUFFER input;
810 XBUFFER _output;
811 XBYTE pad_chk;
812 int pad_count = 0;
813 XBYTE pad_done = 0;
814 bool status;
815
816 if(!funcrandom) return false;
817
818 int keysize = GetKeySizeInBytes(keytypetouse);
819 if(!keysize) return false;
820
821 if((keysize < 16) || (keysize > XMPINTEGER_MAXSIZE)) return false;
822
823 input.Add(buffer, size);
824
825 status = false;
826
827 switch(keytypetouse)
828 {
829 case CIPHERKEYTYPE_RSA_PUBLIC : status = DoRSAPublicOperation(input , _output); break;
830 case CIPHERKEYTYPE_RSA_PRIVATE : status = DoRSAPrivateOperation(input, _output, funcrandom, paramrandom); break;
831 default : break;
832 }
833
834 if(!status) return false;
835
836 int index = 0;
837
838 pad_chk = 0;
839 pad_chk |= _output.GetByte(index++);
840
841 pad_chk |= (_output.GetByte(index++)) ^ ((keytypetouse == CIPHERKEYTYPE_RSA_PRIVATE)? CIPHERRSA_CIPHERDATA : CIPHERRSA_SINGDATA);
842
843 for(XDWORD i=0; i<size-3; i++)
844 {
845 pad_done |= (_output.GetByte(index+i) == 0);
846 pad_count += (pad_done == 0);
847 }
848
849 index += pad_count;
850 pad_chk |= _output.GetByte(index++); // Must be zero
851
852
853 if(pad_chk) return false;
854
855 output.Add(&_output.Get()[index], (size-index));
856
857 return true;
858}

908bool CIPHERRSA::DoRSAPrivateOperation(XBUFFER& input, XBUFFER& output, XMPINTEGER_FUNCRANDOM funcrandom, void* paramrandom)
909{
910 XMPINTEGER T;
911 XMPINTEGER T1;
912 XMPINTEGER T2;
913
914 #if !defined(CIPHERRSA_RSANOCRT)
915 XMPINTEGER* Vi;
916 XMPINTEGER* Vf;
917
918 Vi = &context.Vi;
919 Vf = &context.Vf;
920 #endif
921
922 T.Ini();
923 T1.Ini();
924 T2.Ini();
925
926 if(!T.SetFromXBuffer(input)) return false;
927
928 if(T.CompareSignedValues(context.N)>=0) return false;
929
930 #if defined(CIPHERRSA_RSANOCRT)
931 if(!T.SlidingWindowExponentiation(&T, &context.D, &context.N, &context.RN)) return false;
932 #else
933 if(funcrandom != NULL)
934 {
935 // Blinding T = T * Vi mod N
936 if(!PrepareBlinding(Vi, Vf, funcrandom, paramrandom)) return false;
937 if(!T.Multiplication(&T, Vi )) return false;
938 if(!T.Module( &T, &T, &context.N)) return false;
939 }
940
941 // faster decryption using the CRT
942 // T1 = input ^ dP mod P
943 // T2 = input ^ dQ mod Q
944
945 if(!T1.SlidingWindowExponentiation(&T, &context.DP, &context.P, &context.RP)) return false;
946 if(!T2.SlidingWindowExponentiation(&T, &context.DQ, &context.Q, &context.RQ)) return false;
947
948 // T = (T1 - T2) * (Q^-1 mod P) mod P
949 if(!T.SubtractionSigned(&T1, &T2)) return false;
950 if(!T1.Multiplication(&T, &context.QP)) return false;
951 if(!T.Module(&T, &T1, &context.P)) return false;
952
953 // T = T2 + T * Q
954 if(!T1.Multiplication(&T, &context.Q )) return false;
955 if(!T.AdditionSigned(&T2, &T1)) return false;
956
957 if(funcrandom!= NULL)
958 {
959 // Unblind T = T * Vf mod N
960 if(!T.Multiplication(&T, Vf)) return false;
961 if(!T.Module(&T, &T, &context.N)) return false;
962 }
963 #endif
964
965 int keysize = GetKeySizeInBytes(CIPHERKEYTYPE_RSA_PRIVATE);
966 if(!keysize) return false;
967
968 if(!T.GetToXBuffer(output, keysize)) return false;
969
970 T.End();
971 T1.End();
972 T2.End();
973
974 return true;
975}

978#if !defined(CIPHERRSA_RSANOCRT)
993bool CIPHERRSA::PrepareBlinding(XMPINTEGER* Vi, XMPINTEGER* Vf, XMPINTEGER_FUNCRANDOM funcrandom, void* paramrandom)
994{
995 int count = 0;
996
997 if(context.Vf.GetLimbs()!= NULL)
998 {
999 // We already have blinding values, just update them by squaring
1000 if(!context.Vi.Multiplication(&context.Vi, &context.Vi)) return false;
1001 if(!context.Vi.Module(&context.Vi, &context.Vi, &context.N )) return false;
1002 if(!context.Vf.Multiplication(&context.Vf, &context.Vf)) return false;
1003 if(!context.Vf.Module(&context.Vf, &context.Vf, &context.N )) return false;
1004 }
1005 else
1006 {
1007 int keysize = GetKeySizeInBytes(CIPHERKEYTYPE_RSA_PRIVATE);
1008 if(!keysize) return false;
1009
1010 // Unblinding value: Vf = random number, invertible mod N
1011 do{ if(count++ > 10) return false;
1012
1013 if(!context.Vf.FillRandom(keysize-1, funcrandom, paramrandom)) return false;
1014 if(!context.Vi.GreatestCommonDivisor(&context.Vi, &context.Vf, &context.N )) return false;
1015
1016 } while(context.Vi.CompareSignedValues(1)!=0);
1017
1018 // Blinding value: Vi = Vf^(-e) mod N
1019 if(!context.Vi.ModularInverse( &context.Vf, &context.N)) return false;
1020 if(!context.Vi.SlidingWindowExponentiation(&context.Vi, &context.E, &context.N, &context.RN )) return false;
1021 }
1022
1023 if(Vi != &context.Vi)
1024 {
1025 if(!Vi->CopyFrom(&context.Vi)) return false;
1026 if(!Vf->CopyFrom(&context.Vf)) return false;
1027 }
1028
1029 return true;
1030}
