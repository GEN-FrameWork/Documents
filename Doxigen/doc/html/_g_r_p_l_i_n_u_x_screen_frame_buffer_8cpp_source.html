144bool GRPLINUXSCREENFRAMEBUFFER::Create(bool show)
145{
146 Clean();
147
148 handlefb = open(GRPLINUXSCREENFRAMEBUFFER_DEVICE, O_RDWR);
149 if(handlefb!=-1)
150 {
151 int consoletty = open("/dev/tty1", O_RDWR);
152 if(ioctl(consoletty, KDSETMODE, KD_GRAPHICS) != -1)
153 {
154 if(!ScreenResolution(width, height))
155 {
156 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[LINUX FrameBuffer] Change screen mode: error [%d]"), errno);
157 }
158
159 int zero = 0;
160 if(ioctl(handlefb, FBIO_WAITFORVSYNC, &zero) == -1)
161 {
162 XSTRING errorstr;
163
164 errorstr = strerror(errno);
165
166 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[LINUX FrameBuffer] Frame Buffer VSYNC: %s"), errorstr.Get());
167 }
168
169 return GRPSCREEN::Create(show);
170 }
171 }
172
173 XTRACE_PRINTCOLOR(XTRACE_COLOR_RED, __L("[LINUX FrameBuffer] Open device screen: error [%d]"), errno);
174
175 return false;

190bool GRPLINUXSCREENFRAMEBUFFER::Update(GRPCANVAS* canvas)
191{
192 if(handlefb == -1)
193 {
194 return false;
195 }
196
197 if(ioctl(handlefb, FBIO_WAITFORVSYNC, 0))
198 {
199 //printf("VSync failed.\n");
200 }
201
202 XBYTE* fbp = (XBYTE*)mmap(0, canvas->Buffer_GetSize(), PROT_READ | PROT_WRITE, MAP_SHARED, handlefb, 0);
203 if(fbp == (XBYTE*)-1)
204 {
205 return false;
206 }
207
208 if(rotation != GRPSCREENROTATION_NONE)
209 {
210 Rotate(fbp, canvas);
211 }
212 else
213 {
214 memcpy(fbp, (XBYTE*)canvas->Buffer_Get(), canvas->Buffer_GetSize());
215 }
216
217 munmap((XBYTE*)fbp, canvas->Buffer_GetSize());
218
219 return true;
220}

355 {
356 return false;
357 }
358
359 screen_width = variable_info.xres;
360 screen_height = variable_info.yres;
361

364

372 SetWidth(screen_width);
