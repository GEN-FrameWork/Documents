111bool GRPVIDEOFILEAVI::Open(XCHAR* path)
112{
113 if(!fileRIFF->Open(path)) return false;
114
115 if(!fileRIFF->ReadAllLists()) return false;
116
117 XDWORD sizedata = 0;
118
119 memset((XBYTE*)&mainheader , 0, sizeof(GRPVIDEOFILEAVI_MAINHEADER));
120 memset((XBYTE*)&streamheader , 0, sizeof(GRPVIDEOFILEAVI_STREAMHEADER));
121 memset((XBYTE*)&bitmapheader , 0, sizeof(GRPVIDEOFILEAVI_BITMAPHEADER));
122
123 XFILERIFF_LIST* chunk_mainheader = fileRIFF->GetChunk(__L("avih"), __L("hdrl"));
124 if(!chunk_mainheader) return false;
125
126 fileRIFF->GetChunkData(chunk_mainheader, (XBYTE*)&mainheader, sizedata);
127
128 XFILERIFF_LIST* strl_list = fileRIFF->GetList(__L("strl"), __L("hdrl"));
129 if(!strl_list) return false;
130
131 XFILERIFF_LIST* strh_chunk = fileRIFF->GetChunk(__L("strh"), __L("strl"));
132 if(!strh_chunk) return false;
133
134 fileRIFF->GetChunkData(strh_chunk, (XBYTE*)&streamheader, sizedata);
135
136 XFILERIFF_LIST* strf_chunk = fileRIFF->GetChunk(__L("strf"), __L("strl"));
137 if(!strf_chunk) return false;
138
139 fileRIFF->GetChunkData(strf_chunk, (XBYTE*)&bitmapheader, sizedata);
140
141 propertys.width = mainheader.width;
142 propertys.height = mainheader.height;
143 propertys.nframes = mainheader.totalframes;
144 propertys.framerate = streamheader.rate;
145
146 frames_node = fileRIFF->GetListNode(__L("movi"));
147 if(!frames_node) return false;
148
149 XFILERIFF_LIST* index_list = fileRIFF->GetChunk(__L("idx1"), __L("AVI "));
150 if(!index_list) return false;
151
152 fileRIFF->GetFileBase()->SetPosition(index_list->GetPositionFileData());
153
154 for(XDWORD index=0; index < propertys.nframes; index++)
155 {
156 GRPVIDEOFILEAVI_INDEXENTRY* indexentry = GEN_NEW GRPVIDEOFILEAVI_INDEXENTRY;
157 if(!indexentry) return false;
158
159 if(!fileRIFF->GetFileBase()->Read((XBYTE*)indexentry, sizeof(GRPVIDEOFILEAVI_INDEXENTRY))) return false;
160
161 indexentrys.Add(indexentry);
162 }
163
164 iscreate = false;
165
166 return true;
167}

345XBYTE* GRPVIDEOFILEAVI::GetDataFrame(XDWORD index_frame, XDWORD& sizeframe)
346{
347 if(!frames_node) return NULL;
348 if(index_frame >= propertys.nframes) return NULL;
349
350 XFILERIFF_LIST_NODE* frame_node = frames_node->GetChild(index_frame);
351 if(!frame_node) return NULL;
352
353 XFILERIFF_LIST* frame_list = frame_node->GetData();
354 if(!frame_list) return NULL;
355
356 XDWORD datasize = frame_list->GetSize();
357
358 GRPVIDEOFILEAVI_INDEXENTRY* indexentry = indexentrys.Get(index_frame);
359 if(!indexentry) return NULL;
360
361 if(datasize != indexentry->chunklength) return NULL;
362
363 XBYTE* data = GEN_NEW XBYTE[datasize];
364 if(!data) return NULL;
365
366 if(!fileRIFF->GetFileBase()->SetPosition(frame_list->GetPositionFileData()))
367 {
368 GEN_DELETE_ARRAY data;
369 return NULL;
370 }
371
372 if(!fileRIFF->GetFileBase()->Read(data, datasize))
373 {
374 GEN_DELETE_ARRAY data;
375 return NULL;
376 }
377
378 sizeframe = datasize;
379
380 return data;
381}

396bool GRPVIDEOFILEAVI::AddFrame(XBYTE* dataframe, XDWORD dataframesize)
397{
398 if(!dataframe) return false;
399 if(!dataframesize) return false;
400
401 if(!movi_node) return false;
402
403
404 XDWORD framesize = dataframesize;
405
406 //framesize += ((framesize % 2)?1:0);
407
408 //--------------------------------------------------------------------
409 // AVI->movi->00dc
410
411 XFILERIFF_LIST_NODE* frame_node = fileRIFF->CreateChunkNode(GRPVIDEOFILEAVI_TYPECHUNKFRAME, framesize);
412 if(!frame_node) return false;
413
414 fileRIFF->UpdateFilePosition(frame_node);
415 if(!fileRIFF->WriteListToFile(frame_node->GetData(), dataframe, framesize)) return false;
416
417 mainheader.totalframes++;
418 movi_node->AddChild(frame_node);
419
420
421 GRPVIDEOFILEAVI_INDEXENTRY* indexentry = GEN_NEW GRPVIDEOFILEAVI_INDEXENTRY();
422 if(indexentry)
423 {
424 indexentry->chunkID = fileRIFF->GetTypeFromString(GRPVIDEOFILEAVI_TYPECHUNKFRAME);
425 indexentry->flags = 16;
426 indexentry->chunklength = framesize;
427 indexentry->chunkoffset = offsetforentrys;
428
429 indexentrys.Add(indexentry);
430
431 //offsetforentrys += (framesize + sizeof(GRPVIDEOFILEAVI_INDEXENTRY));
432
433 offsetforentrys += (framesize + 8);
434 }
435
436 return true;
437}
