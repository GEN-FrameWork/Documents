529bool DIOPCAP_FILTERS::Entrys_Check(DIOPCAPFRAME* frame)
530{
531 bool valid = false;
532
533 if(!frame)
534 {
535 return false;
536 }
537
538 if(!entrys.GetSize())
539 {
540 return true;
541 }
542
543 for(XDWORD c=0; c<entrys.GetSize(); c++)
544 {
545 DIOPCAP_FILTER_ENTRY* entry = entrys.Get(c);
546 if(entry)
547 {
548 XDWORD protocoltype = entry->GetProtocolType();
549
550 if((protocoltype & DIOPCAPPROTOCOL_TYPE_ICMP) == frame->GetProtocolType())
551 {
552 valid = true;
553 }
554
555 if((protocoltype & DIOPCAPPROTOCOL_TYPE_UDP) == frame->GetProtocolType())
556 {
557 valid = true;
558 }
559
560 if((protocoltype & DIOPCAPPROTOCOL_TYPE_TCP) == frame->GetProtocolType())
561 {
562 valid = true;
563 }
564
565 if(valid)
566 {
567 if(!entry->Ports_GetAll()->GetSize())
568 {
569 return valid;
570 }
571
572 for(XDWORD d=0; d<entry->Ports_GetAll()->GetSize(); d++)
573 {
574 XDWORD port_target = entry->Ports_GetAll()->Get(d);
575 XDWORD port_source = (port_target & ~DIOPCAP_FILTERS_PORT_TARGET);
576
577 if(port_source == frame->GetSourcePort())
578 {
579 return valid;
580 }
581
582 if(port_target == frame->GetTargetPort())
583 {
584 return valid;
585 }
586 }
587 }
588 }
589 }
590
591 return false;
592}
