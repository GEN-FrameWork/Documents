365bool APPFLOWCHECKRESOURCESHARDWARE::CheckTotalCPUUsageStatus()
366{
367 if(!cfg)
368 {
369 return false;
370 }
371
372 int cpuusage = GEN_XSYSTEM.GetCPUUsageTotal();
373
374 if(cpuusage != XSYSTEM_CPUUSAGE_ERROR)
375 {
376 if(cpuusage > maxtotalCPUusage)
377 {
378 maxtotalCPUusage = cpuusage;
379 }
380
381 ntotalCPUusage += cpuusage;
382 ntotalCPUsamples++;
383 }
384
385 if(ntotalCPUsamples > 5)
386 {
387 if(GetCPUTotalCPUUsageAverange() > (XDWORD)cfg->CheckResourcesHardware_GetTotalCPUUsageLimitPercent())
388 {
389 APPFLOWCHECKRESOURCESHARDWARE_XEVENT xevent(this, APPFLOWCHECKRESOURCESHARDWARE_XEVENT_TYPE_TOTALCPUUSAGELIMIT);
390 xevent.SetActualTotalCPUUsage(cpuusage);
391 PostEvent(&xevent);
392 }
393 }
394
395 return true;
