337bool DIOCLIPROTOCOLBUS::ReceivedCommand(XSTRING& originID, XSTRING& command, XVECTOR<XSTRING*>& params, XSTRING& answer)
338{
339 bool status = false;
340
341 answer.Empty();
342
343 DIOCLIPROTOCOLCOMMAND* protocolcommand = GetCommand(command.Get());
344 if(protocolcommand)
345 {
346 if(protocolcommand->GetNParams() < (int)params.GetSize()) return false;
347
348 if(!command.Compare(DIOCLIPROTOCOLBUS_COMMAND_VERSION, true))
349 {
350 answer.Format(__L("%d.%d.%d"), version, subversion, subversionerror);
351 status = true;
352 }
353
354 if(!command.Compare(DIOCLIPROTOCOLBUS_COMMAND_PING, true))
355 {
356 status = true;
357 }
358
359 if(!command.Compare(DIOCLIPROTOCOLBUS_COMMAND_ENUM, true))
360 {
361 enum_mutex->Lock();
362
363 enum_sendoriginID = originID;
364
365 enum_mutex->UnLock();
366
367 threadsendenumrequest->Ini();
368 }
369
370 if(!command.Compare(DIOCLIPROTOCOLBUS_COMMAND_ENUMREQUEST, true))
371 {
372 bool found = false;
373
374 for(XDWORD c=0; c<enum_remotedevices.GetSize(); c++)
375 {
376 XSTRING* device = enum_remotedevices.Get(c);
377 if(device)
378 {
379 if(!device->Compare(originID, true)) found = true;
380 }
381 }
382
383 if(!found)
384 {
385 XSTRING* origin = GEN_NEW XSTRING();
386 if(origin)
387 {
388 (*origin) = originID;
389 enum_remotedevices.Add(origin);
390 }
391 }
392
393 status = true;
394 }
395 }
396
397 return status;
398}
