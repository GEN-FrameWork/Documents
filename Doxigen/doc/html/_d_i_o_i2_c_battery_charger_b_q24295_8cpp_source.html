148void DIOI2CBATTERYCHARGERBQ24295::Update()
149{
150 if(!charging) return;
151
152 // For an unknown reason the stat pin generates an square signal and not holds the low level when is charging
153 // so I have to read the register to know when the charge is done
154 //I read the register every 5 seconds
155
156 XBYTE read_reg_fault;
157 XBYTE read_reg_status;
158
159 if(ReadRegister(DIOI2CBATTERYCHARGERBQ24295_FAULT_REG_REG09 , read_reg_fault) != 0) return; //TODO-GFR detect FAULT errors
160 if(ReadRegister(DIOI2CBATTERYCHARGERBQ24295_SYS_STATUS_REG08 , read_reg_status) != 0) return;
161
162 if(((read_reg_status >> 4) & 0x03) == 0x03) fullycharge = true;
163}

318bool DIOI2CBATTERYCHARGERBQ24295::Initialize()
319{
320 // Charger GPIO Inputs
321 GEN_DIOGPIO.SetMode(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_STAT] , DIOGPIO_MODE_INPUT);
322
323 // Charger GPIO Outputs
324 GEN_DIOGPIO.SetMode(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_QON] , DIOGPIO_MODE_OUTPUT);
325 GEN_DIOGPIO.SetMode(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_CE] , DIOGPIO_MODE_OUTPUT);
326
327 // Charger GPIO Interrupts
328 DIOGPIO_ENTRY* interrupt = GEN_DIOGPIO.GPIOEntry_GetByID(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_INT]);
329 if(interrupt)
330 {
331 interrupt->SetIntFunctionPointer(DIOI2CBATTERYCHARGERBQ24295::Handler_Interrupt);
332 interrupt->SetIntParamPointer((void*)this);
333 GEN_DIOGPIO.SetMode(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_INT], DIOGPIO_MODE_INTERRUPT);
334 }
335
336 // Charger disabled
337 GEN_DIOGPIO.SetValue(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_CE] , true);
338
339 // Shipping Mode off
340 GEN_DIOGPIO.SetValue(GPIOentryID[DIOI2CBATTERYCHARGERBQ24295_GPIOENTRYID_QON] , false);
341
342 charging = false;
343 fault = 0;
344 fullycharge = false;
345
346 // Configure the charger ILMAX to 500mA
347 XBYTE read_reg;
348
349 if(!ReadRegister(DIOI2CBATTERYCHARGERBQ24295_INPUT_SRC_CTRL_REG00, read_reg)) return false;
350
351 // IINLIM bits (010) (500mA)
352 read_reg &= 0xF8;
353 read_reg |= 0x02;
354
355 if(!WriteRegister(DIOI2CBATTERYCHARGERBQ24295_INPUT_SRC_CTRL_REG00, read_reg)) return false;
356
357 // Disable the charger watchdog timer and the safety timer
358 if(!ReadRegister(DIOI2CBATTERYCHARGERBQ24295_CHRG_TERM_TIMER_CTRL_REG05, read_reg)) return false;
359
360 // Disable timer
361 read_reg &= 0xC7;
362 if(!WriteRegister(DIOI2CBATTERYCHARGERBQ24295_INPUT_SRC_CTRL_REG00, read_reg)) return false;
363
364 return true;
365}

400bool DIOI2CBATTERYCHARGERBQ24295::ReadRegister(XBYTE offset, XBYTE* data, XDWORD sizedata)
401{
402 if(!diostream) return false;
403
404 bool status;
405
406 status = diostream->Write(&offset, 1)?true:false;
407 if(status) status = diostream->WaitToFlushOutXBuffer(timeout);
408
409 if(!status) return false;
410
411 status = diostream->WaitToFilledReadingBuffer(sizedata, timeout);
412 if(status) status = (diostream->Read(data, sizedata) == sizedata)?true:false;
413
414 return status;
415}
