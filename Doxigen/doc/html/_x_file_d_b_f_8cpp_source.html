566bool XFILEDBF_HEADER::Load(XFILE* xfile)
567{
568 // ---------------------------
569 // Read header
570 if(!xfile->SetPosition(8)) return false;
571
572 if(!xfile->Read((XBYTE*)&headersize,2)) return false;
573 if(!headersize) return 0;
574
575 if(headerdata)
576 {
577 GEN_DELETE_ARRAY headerdata;
578 headerdata = NULL;
579 }
580
581 headerdata = GEN_NEW XBYTE[headersize];
582 if(!headerdata) return false;
583
584 if(!xfile->SetPosition(0)) return false;
585 if(!xfile->Read(headerdata,headersize)) return false;
586
587 int index=XFILEDBF_FIELDSTART;
588
589 while(headerdata[index]!=0x0D)
590 {
591 XFILEDBF_FIELDTYPE type;
592
593 switch(headerdata[index+11])
594 {
595 case 'C' : type=XFILEDBF_FIELDTYPE_CHARACTER; break;
596 case 'D' : type=XFILEDBF_FIELDTYPE_DATE; break;
597 case 'F' : type=XFILEDBF_FIELDTYPE_FLOATING; break;
598 case 'N' : type=XFILEDBF_FIELDTYPE_BINARYHEX; break;
599 case 'L' : type=XFILEDBF_FIELDTYPE_LOGICAL; break;
600 case 'M' : type=XFILEDBF_FIELDTYPE_MEMO; break;
601 case 'V' : type=XFILEDBF_FIELDTYPE_VARIABLE; break;
602 case 'P' : type=XFILEDBF_FIELDTYPE_PICTURE; break;
603 case 'B' : type=XFILEDBF_FIELDTYPE_BINARY; break;
604 case 'G' : type=XFILEDBF_FIELDTYPE_GENERAL; break;
605 case '2' : type=XFILEDBF_FIELDTYPE_SHORTINT; break;
606 case '4' : type=XFILEDBF_FIELDTYPE_LONGINT; break;
607 case '8' : type=XFILEDBF_FIELDTYPE_DOUBLE; break;
608 default : type=XFILEDBF_FIELDTYPE_UNKNOWN; break;
609 }
610
611 if(type!=XFILEDBF_FIELDTYPE_UNKNOWN)
612 {
613 XSTRING name((char*)&headerdata[index]);
614
615 if(!CreateField(type, name.Get(), headerdata[index+16])) return false;
616 }
617
618 index+=XFILEDBF_FIELDSIZE;
619 }
620
621 return true;
622}

636bool XFILEDBF_HEADER::Create(XFILE* fileb)
637{
638 headersize = (XWORD)(XFILEDBF_FIELDSTART + fields.GetSize()*XFILEDBF_FIELDSIZE+ 1);
639
640 headerdata = GEN_NEW XBYTE [headersize];
641 if(!headerdata) return false;
642
643 memset(headerdata,0,headersize);
644
645 headerdata[0] = 0x03;
646
647 XDATETIME* GEN_XFACTORY_CREATE(xdatetime, CreateDateTime())
648 if(xdatetime)
649 {
650 xdatetime->Read();
651
652 headerdata[1] = xdatetime->GetYear()-1900;
653 headerdata[2] = xdatetime->GetMonth();
654 headerdata[3] = xdatetime->GetDay();
655
656 GEN_DELETE xdatetime;
657 }
658
659 XDWORD* nf = (XDWORD*)&headerdata[4];
660 (*nf)=0;
661
662 XWORD* hs = (XWORD*)(&headerdata[8]);
663 (*hs)=headersize;
664
665 XWORD* rs = (XWORD*)(&headerdata[10]);
666 (*rs)=GetSizeRecord();
667
668
669 int index=XFILEDBF_FIELDSTART;
670
671 for(XDWORD c=0;c<fields.GetSize();c++)
672 {
673 XFILEDBF_FIELD* field = (XFILEDBF_FIELD*)fields.Get(c);
674 char type = ' ';
675
676 if(field)
677 {
678 memset(&headerdata[index],0,XFILEDBF_FIELDSIZE);
679
680
681 XBUFFER charstr;
682
683 field->name.ConvertToASCII(charstr);
684
685 XDWORD size = (int)strlen(charstr);
686 if(size>XFILEDBF_FIELDSIZE) size = XFILEDBF_FIELDSIZE;
687
688 memcpy((char*)(&headerdata[index]), charstr.Get(), size);
689
690 switch(field->type)
691 {
692 case XFILEDBF_FIELDTYPE_CHARACTER : type='C'; break;
693 case XFILEDBF_FIELDTYPE_DATE : type='D'; break;
694 case XFILEDBF_FIELDTYPE_FLOATING : type='F'; break;
695 case XFILEDBF_FIELDTYPE_BINARYHEX : type='N'; break;
696 case XFILEDBF_FIELDTYPE_LOGICAL : type='L'; break;
697 case XFILEDBF_FIELDTYPE_MEMO : type='M'; break;
698 case XFILEDBF_FIELDTYPE_VARIABLE : type='V'; break;
699 case XFILEDBF_FIELDTYPE_PICTURE : type='P'; break;
700 case XFILEDBF_FIELDTYPE_BINARY : type='B'; break;
701 case XFILEDBF_FIELDTYPE_GENERAL : type='G'; break;
702 case XFILEDBF_FIELDTYPE_SHORTINT : type='2'; break;
703 case XFILEDBF_FIELDTYPE_LONGINT : type='4'; break;
704 case XFILEDBF_FIELDTYPE_DOUBLE : type='8'; break;
705 }
706
707 headerdata[index+11]=type;
708 headerdata[index+16]=field->size;
709
710 index+=XFILEDBF_FIELDSIZE;
711 }
712 }
713
714 headerdata[headersize-1] = 0x0D;
715
716 if(!fileb->SetPosition(0)) return false;
717 if(!fileb->Write(headerdata,headersize)) return false;
718
719 return true;
720}
