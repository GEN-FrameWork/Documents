615bool XDRIVEIMAGEMANAGER::ReadToIMGFile(XDRIVEIMAGEMANAGER_DRIVE& drive, XPATH& pathfileIMG, XQWORD nsectorsperblock)
616{
617 XFILE* xfileIMG;
618 XBUFFER readbuffer;
619 XQWORD nsectorstoread = nsectorsperblock;
620 XQWORD nsectorsread = 0;
621 bool status = 0;
622
623 if(!drive.GetSize()) return status;
624
625 GEN_XFACTORY_CREATE(xfileIMG, Create_File())
626 if(xfileIMG)
627 {
628 if(xfileIMG->Create(pathfileIMG))
629 {
630 XQWORD indexsector=0;
631
632 while(indexsector<drive.GetNumberSectors())
633 {
634 if((indexsector + nsectorstoread) > drive.GetNumberSectors()) nsectorstoread = (drive.GetNumberSectors() - indexsector);
635
636 XDRIVEIMAGEMANAGER_XEVENT xevent(this, XDRIVEIMAGEMANAGER_XEVENT_TYPE_READTOIMGFILE);
637
638 xevent.SetDrive(&drive);
639 xevent.SetIndexSector(indexsector);
640 xevent.SetNumberSectorsInOperation(nsectorstoread);
641 xevent.SetError(XDRIVEIMAGEMANAGER_ERROR_NONE);
642
643 status = Read(drive, indexsector, nsectorstoread, readbuffer);
644 if(!status) xevent.SetError(XDRIVEIMAGEMANAGER_ERROR_READSECTORS);
645
646 if(status)
647 {
648 status = xfileIMG->Write(readbuffer);
649 if(!status) xevent.SetError(XDRIVEIMAGEMANAGER_ERROR_WRITEIMGFILE);
650 }
651
652 if(status) nsectorsread += nsectorstoread;
653
654 lastoperationpercent = (((float)nsectorsread*100)/(float)drive.GetNumberSectors());
655
656 xevent.SetError(lasterror);
657 xevent.SetOperationPercent(lastoperationpercent);
658
659 PostEvent(&xevent);
660
661 if(!status) break;
662
663 indexsector += nsectorstoread;
664 }
665
666 xfileIMG->Close();
667 }
668
669 GEN_XFACTORY.Delete_File(xfileIMG);
670 }
671
672 return status;
673}
