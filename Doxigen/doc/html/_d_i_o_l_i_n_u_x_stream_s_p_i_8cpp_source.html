109bool DIOLINUXSTREAMSPI::Open()
110{
111 if(!config) return false;
112
113 XBUFFER charstr;
114
115 (*config->GetLocalDeviceName()).ConvertToASCII(charstr);
116 handle = open(charstr.GetPtrChar(), O_RDWR);
117 //XTRACE_PRINTCOLOR(((handle<0)?4:1), __L("SPI Open %s : %s"), config->GetLocalDeviceName()->Get(), ((handle<0)?__L("Error"):__L("Ok")));
118 if(handle<0) return false;
119
120 // Set SPI parameters.
121 // Why are we reading it afterwriting it? I've no idea, but for now I'm blindly
122 // copying example code I've seen online...
123
124 XBYTE _mode = config->GetSPIMode();
125 XBYTE _bits = config->GetNBitsWord();
126 XDWORD _speed = config->GetSpeed();
127
128 if (ioctl(handle, SPI_IOC_WR_MODE , &_mode) < 0) return false;
129 if (ioctl(handle, SPI_IOC_WR_BITS_PER_WORD , &_bits) < 0) return false;
130 if (ioctl(handle, SPI_IOC_WR_MAX_SPEED_HZ , &_speed) < 0) return false;
131
132 if (ioctl(handle, SPI_IOC_RD_MODE , &_mode) < 0) return false;
133 if (ioctl(handle, SPI_IOC_RD_BITS_PER_WORD , &_bits) < 0) return false;
134 if (ioctl(handle, SPI_IOC_RD_MAX_SPEED_HZ , &_speed) < 0) return false;
135
136 return DIOSTREAMSPI::Open();
137}

190 transf.rx_buf = (unsigned long)bufferread;

194 transf.bits_per_word = config->GetNBitsWord();
