333{

383void XBUFFERBITS::WriteU1(XDWORD v)
384{
385 bits_left--;
386
387 if(! IsEOF())
388 {
389 // FIXME this is slow, but we must clear bit first
390 // is it better to memset(0) the whole buffer during bs_init() instead?
391 // if we don't do either, we introduce pretty nasty bugs
392 (*(p)) &= ~(0x01 << bits_left);
393 (*(p)) |= ((v & 0x01) << bits_left);
394 }
395
396 if(bits_left == 0)
397 {
398 p++;
399 bits_left = 8;

459 XDWORD len;

464 }

467 v++;
468
469 if (v >= 0x01000000)
470 {
