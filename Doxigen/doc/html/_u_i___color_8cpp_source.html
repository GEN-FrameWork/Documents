300bool UI_COLOR::SetFromString(XCHAR* string)
301{
302 SetInvalid();
303
304 if(!string)
305 {
306 return false;
307 }
308
309 XSTRING colorstr;
310 int ncommas = 0;
311 int _alpha = 0;
312 bool status = false;
313
314 colorstr = string;
315 if(colorstr.IsEmpty())
316 {
317 return false;
318 }
319
320 for(XDWORD c=0; c<colorstr.GetSize(); c++)
321 {
322 XCHAR character = colorstr.Get()[c];
323 if(character == __C(','))
324 {
325 ncommas++;
326 }
327 }
328
329 switch(ncommas)
330 {
331 case 0 : // only name
332 { XSTRING* resolved_colorstr = GEN_UI_COLORS.Get(colorstr);
333 if(resolved_colorstr)
334 {
335 status = SetFromString(resolved_colorstr->Get());
336 }
337 }
338 break;
339
340 case 1 : // name + int alpha
341 { XSTRING _colorstr;
342 XSTRING* resolved_colorstr = NULL;
343
344 _colorstr.AdjustSize(_MAXSTR);
345 colorstr.UnFormat(__L("%s,%d"), _colorstr.Get(), &_alpha);
346 _colorstr.AdjustSize();
347
348 resolved_colorstr = GEN_UI_COLORS.Get(_colorstr);
349 if(resolved_colorstr)
350 {
351 status = SetFromString(resolved_colorstr->Get());
352 }
353
354 alpha = GetAlphaForPercent(_alpha);
355 }
356 break;
357
358 case 2 : // int colors
359 status = colorstr.UnFormat(__L("%d,%d,%d"), &red, &green, &blue);
360 alpha = 255;
361 break;
362
363 case 3 : // int colors + int alpha
364 status = colorstr.UnFormat(__L("%d,%d,%d,%d"), &red, &green, &blue, &_alpha);
365 alpha = GetAlphaForPercent(_alpha);
366 break;
367
368 default : break;
369 }
370
371 if(!status)
372 {
373 SetInvalid();
374 }
375
376 valid = true;
377
378 return status;
379}
