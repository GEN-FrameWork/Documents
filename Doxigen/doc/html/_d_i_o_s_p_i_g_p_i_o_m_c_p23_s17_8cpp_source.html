101bool DIOSPIGPIOMCP23S17::IniDevice()
102{

149bool DIOSPIGPIOMCP23S17::Configure()
150{
151 XBYTE ioconfig = DIOSPIGPIOMCP23S17_BANK_OFF |
152 DIOSPIGPIOMCP23S17_INT_MIRROR_OFF |
153 DIOSPIGPIOMCP23S17_SEQOP_OFF |
154 DIOSPIGPIOMCP23S17_DISSLW_OFF |
155 DIOSPIGPIOMCP23S17_HAEN_ON |
156 DIOSPIGPIOMCP23S17_ODR_OFF |
157 DIOSPIGPIOMCP23S17_INTPOL_LOW;
158
159 XBYTE addr = 0;
160
161 if(!Write_Register(DIOSPIGPIOMCP23S17_IOCON, addr, ioconfig)) return false;
162
163 // I/O direction
164 if(!Write_Register(DIOSPIGPIOMCP23S17_IODIRA , addr, 0x00)) return false;
165 if(!Write_Register(DIOSPIGPIOMCP23S17_IODIRB , addr, 0xFF)) return false;
166
167 // GPIOB pull ups
168 if(!Write_Register(DIOSPIGPIOMCP23S17_GPPUB , addr, 0xFF)) return false;
169
170 return true;
171}

187bool DIOSPIGPIOMCP23S17::Read_Register(XBYTE reg, XBYTE addr, XBYTE& data)
188{
189 XBYTE control_byte = GetControlByte(DIOSPIGPIOMCP23S17_READ_CMD, addr);
190 XBYTE tx_buffer[3] = { control_byte, reg, 0 };

192
193 data = 0;
194

265bool DIOSPIGPIOMCP23S17::Write_Bit(XBYTE reg, XBYTE addr, XBYTE bitnum, XBYTE data)
266{
267 XBYTE reg_data = 0;
268
269 if(!Read_Register(reg, addr, reg_data)) return false;
270
271 if(data)
272 {
273 reg_data |= (1 << bitnum); // set
274 }
275 else
276 {
277 reg_data &= 0xff ^ (1 << bitnum); // clear
278 }
279
280 return Write_Register(reg, addr, reg_data);
