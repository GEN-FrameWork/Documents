341void Call_TerminateApplicationWithWindow(SCRIPT_LIB* library, SCRIPT* script, XVECTOR<XVARIANT*>* params, XVARIANT* returnvalue)
342{
343 if((!library) || (!script) || (!params) || (!returnvalue))
344 {
345 return;
346 }
347
348 returnvalue->Set();
349
350 if(params->GetSize()<2)
351 {
352 script->HaveError(SCRIPT_ERRORCODE_INSUF_PARAMS);
353 return;
354 }
355
356 XVECTOR<XPROCESS*> applist;
357 XSTRING appname = (*params->Get(0));
358 XSTRING windowstitle = (*params->Get(1));
359 int windowsposx = 0;
360 bool status = false;
361
362 if(GEN_XPROCESSMANAGER.Application_GetRunningList(applist, true))
363 {
364 for(XDWORD c=0; c<applist.GetSize(); c++)
365 {
366 if(applist.Get(c)->GetName()->Find(appname, true)!= XSTRING_NOTFOUND)
367 {
368 if(applist.Get(c)->GetWindowTitle()->Find(windowstitle, false) != XSTRING_NOTFOUND)
369 {
370 status = GEN_XPROCESSMANAGER.Application_Terminate((*applist.Get(c)));
371 break;
372 }
373 }
374 }
375 }
376
377 applist.DeleteContents();
378 applist.DeleteAll();
379
380 (*returnvalue) = status;
381}
