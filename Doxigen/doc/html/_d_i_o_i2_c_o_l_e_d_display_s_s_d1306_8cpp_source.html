130bool DIOI2COLEDDISPLAYSSD1306::IniDevice()
131{
132 if(!DIODEVICEI2C::IniDevice()) return false;
133
134 if(diostream->Open())
135 {
136 if(diostream->WaitToConnected(timeout))
137 {
138 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_DISPLAY_OFF); // 0xAE
139 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_DISPLAY_CLOCK_DIV_RATIO_SET); // 0xD5
140 SendCommand(0xF0); // Suggest ratio
141 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_MULTIPLEX_RADIO_SET); // 0xA8
142 SendCommand((64 - 1)); // Display_Height - 1
143
144 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_DISPLAY_OFFSET_SET); // 0xD3 + No offset
145 SendCommand(00); // No offset
146 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_CHARGE_PUMP_SET); // 0x8D
147 SendCommand(0x14); // Enable charge pump
148 SendCommand((DIOI2COLEDDISPLAYSSD1306_COMMAND_DISPLAY_START_LINE_SET | 0x00)); // 0x40 | Start line
149 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_MEMORY_ADDRESSING_SET); // 0x20
150 SendCommand(0x00); // Horizontal Addressing - Operate like KS0108
151 SendCommand((DIOI2COLEDDISPLAYSSD1306_COMMAND_SEGMENT_REMAP_SET | 0x01)); // 0xA0 - Left towards Right
152 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_COM_OUTPUT_SCAN_DEC); // 0xC8 - Up towards Down
153 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_COM_PINS_SET); // 0xDA
154 //SendCommand(0x12); // Sequential COM pin configuration
155
156 if((width == 128) && (height == 64)) SendCommand(0x12); // Sequential COM pin configuration
157 if((width == 128) && (height == 32)) SendCommand(0x02); // Sequential COM pin configuration
158 if((width == 96) && (height == 16)) SendCommand(0x02); // Sequential COM pin configuration
159
160 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_CONSTRAST_SET); // 0x81
161 SendCommand(0xFF); // 0xFF
162 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_PRECHARGE_PERIOD_SET); // 0xD9
163 SendCommand(0xF1); // 0xF1
164
165 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_VCOMH_DESELECT_LEVEL_SET); // 0xDB
166 SendCommand(0x20); // 0x20
167
168 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_DISPLAY_ALL_ON_RESUME); // 0xA4
169 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_DISPLAY_NORMAL); // 0xA6

199bool DIOI2COLEDDISPLAYSSD1306::Clear(XBYTE color)
200{
201 XBYTE nblocks;
202
203 nblocks = (width >> 4) * (height >> 3);
204
205 // Set columns
206 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_COLUMN_ADDRESS_SET); // 0x21
207 SendCommand(0x00); // Start
208 SendCommand(width - 1); // End
209
210 //Set rows
211 SendCommand(DIOI2COLEDDISPLAYSSD1306_COMMAND_PAGE_ADDRESS_SET); // 0x22
212 SendCommand(0x00); // Start
213 SendCommand(height - 1); // End
214
215 XBYTE buffer[16];
216
217 memset(buffer, color, 16);
218
219 //Send buffer
220 for(XBYTE c=0; c<nblocks; c++)
221 {
222 SendData(buffer, 16);
223 }
224
225 return true;
226}

280bool DIOI2COLEDDISPLAYSSD1306::PutPixel(XWORD x, XWORD y, bool color)
281{
282 if(!buffer) return false;
283
284 // Goto to point
285 SetXY((XBYTE)x, (XBYTE)y);
286
287 XBYTE data = 0;
288
289 // Read data
290 data = buffer[DIOI2COLEDDISPLAYSSD1306_POINTER(x, y)];
291
292 //Set pixel
293 if(color)
294 DIOI2COLEDDISPLAYSSD1306_BITSET(data, y % 8);
295 else DIOI2COLEDDISPLAYSSD1306_BITCLEAR(data, y % 8);
296
297 //Sent data
298 buffer[DIOI2COLEDDISPLAYSSD1306_POINTER(x, y)] = data;
299
300 return true;
