112bool DIOBUSPIRATE::Ini(int timeout)
113{
114 if(!diostream) return false;
115

141DIOBUSPIRATETYPEPROCOTOL DIOBUSPIRATE::GetActualProtocol(int timeout)
142{
143 if(!diostream) return DIOBUSPIRATETYPEPROCOTOL_UNKNOWN;
144 if(diostream->GetStatus() != DIOSTREAMSTATUS_CONNECTED) return DIOBUSPIRATETYPEPROCOTOL_UNKNOWN;
145
146 XSTRING promptstatus;
147
148 WriteCommand(__L(""), timeout);
149 ReadPromptStatus(promptstatus, timeout);
150
151 if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_HIZ, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_HIZ;
152 else if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_1WIRE, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_1WIRE;
153 else if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_UART, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_UART;
154 else if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_I2C, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_I2C;
155 else if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_SPI, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_SPI;
156 else if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_2WIRE, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_2WIRE;
157 else if(promptstatus.Find(DIOBUSPIRATE_COMMANDMENU_PROTOCOL_3WIRE, false) != XSTRING_NOTFOUND) return DIOBUSPIRATETYPEPROCOTOL_3WIRE;
158
159 return DIOBUSPIRATETYPEPROCOTOL_UNKNOWN;
160}

178bool DIOBUSPIRATE::SetMode(DIOBUSPIRATETYPEPROCOTOL protocol, bool issoftware, int speed, bool issniffer, int timeout)
179{
180 if(!xtimerout) return false;
181 if(!diostream) return false;
182 if(diostream->GetStatus() != DIOSTREAMSTATUS_CONNECTED) return false;
183
184 XSTRING option;
185 bool status = false;
186
187 switch(protocol)
188 {
189 case DIOBUSPIRATETYPEPROCOTOL_I2C : if(WriteCommand(__L("m"), timeout))
190 {
191 option.Format(__L("%d"), protocol);
192 if(WriteCommand(option, timeout))
193 {
194 option.Format(__L("%d"), issoftware?1:2);
195 if(WriteCommand(option, timeout))
196 {
197 option.Format(__L("%d"), speed);
198 if(WriteCommand(option, timeout))
199 {
200 if(issniffer)
201 {
202 option.Format(__L("(2)"), speed);
203 status = WriteCommand(option, timeout);
204
205 } else status = true;
206 }
207 }
208 }
209 }
210
211 GEN_XSLEEP.Seconds(1);
212 diostream->GetInXBuffer()->Empty();
213 break;
214
215 default : break;
216
217 }
218
219 return status;
220}

304bool DIOBUSPIRATE::ReadPromptStatus(XSTRING& promptstatus, int timeout)
305{
306 if(!diostream) return false;
307 if(diostream->GetStatus() != DIOSTREAMSTATUS_CONNECTED) return false;
308
309 promptstatus.Empty();
310
311 if(!xtimerout) return false;
312
313 xtimerout->Reset();
314
315 while(1)
316 {
317 if(diostream->WaitToFilledReadingBuffer(1, 1))
318 {
319 XBYTE data;
320 if(diostream->Read(&data, 1))
321 {
322 if(data != '>')
323 {
324 promptstatus.Add(data);
325 }
326 else return true;
327
328 xtimerout->Reset();
329 }
330 }
331
332 if((int)xtimerout->GetMeasureSeconds() >= timeout) break;
333 }
334
335 return false;
336}

351bool DIOBUSPIRATE::WriteCommand(XCHAR* command, int timeout)
352{
353 if(!diostream) return false;
354 if(diostream->GetStatus() != DIOSTREAMSTATUS_CONNECTED) return false;
355
356 XSTRING _command;
357 bool status;
358
359 _command = command;
360 _command += __L("\r");
361
362 XBUFFER charstr;
363
364 _command.ConvertToASCII(charstr);
365 status = diostream->Write((XBYTE*)charstr.Get(), _command.GetSize())?true:false;
366 if(status) status = diostream->WaitToFlushOutXBuffer(timeout);
367
368 return status;
369}
