162bool DIOI2CLIGHTSENSORTSL2561::Get(XWORD& fullspectrum, XWORD& infrared, XDWORD& lux, bool autogain)
163{
164 fullspectrum = 0;
165 infrared = 0;
166 lux = 0;
167
168 if(!diostream) return false;
169
170 XWORD fs, ir, hi, lo;
171 bool status = true;
172
173 if(!autogain)
174 {
175 // autogain not requested. Executing a single sensor read
176
177 status = GetData(fullspectrum, infrared);
178 lux = CalculateLux(fullspectrum, infrared);
179
180 return status;
181 }
182
183 // autogain requested
184 switch(integrationtime)
185 {
186 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_13MS : hi = DIOI2CLIGHTSENSORTSL2561_AGC_THI_13MS;
187 lo = DIOI2CLIGHTSENSORTSL2561_AGC_TLO_13MS;
188 break;
189
190 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_101MS : hi = DIOI2CLIGHTSENSORTSL2561_AGC_THI_101MS;
191 lo = DIOI2CLIGHTSENSORTSL2561_AGC_TLO_101MS;
192 break;
193
194 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_402MS : hi = DIOI2CLIGHTSENSORTSL2561_AGC_THI_402MS;
195 lo = DIOI2CLIGHTSENSORTSL2561_AGC_TLO_402MS;
196 break;
197
198 default : hi = DIOI2CLIGHTSENSORTSL2561_AGC_THI_402MS;
199 lo = DIOI2CLIGHTSENSORTSL2561_AGC_TLO_402MS;
200 break;
201 }
202
203 // try to adjust the gain
204 status = GetData(fs, ir);
205 if(!status) return false;
206
207 if((fs < lo) && (gain == DIOI2CLIGHTSENSORTSL2561GAIN_1X))
208 {
209 // light too low with this gain
210 // raise the gain and redo the reading
211 SetGain(DIOI2CLIGHTSENSORTSL2561GAIN_16X);
212
213 status = GetData(fs, ir);
214 if(!status) return false;
215
216 // now consider the reading valid after being adjusted
217 fullspectrum = fs;
218 infrared = ir;
219
220 lux = CalculateLux(fullspectrum, infrared);
221
222 return true;
223 }
224
225
226 if((fs > hi) && (gain == DIOI2CLIGHTSENSORTSL2561GAIN_16X))
227 {
228 // light too high with this gain
229 // lower the gain and redo the reading
230 SetGain(DIOI2CLIGHTSENSORTSL2561GAIN_1X);
231
232 status = GetData(fs, ir);
233 if(!status) return false;
234
235 // now consider the reading valid after being adjusted
236 fullspectrum = fs;
237 infrared = ir;
238 lux = CalculateLux(fullspectrum, infrared);
239
240 return true;
241 }
242
243
244 // the reading was valid without gain adjustment (or chip limits encountered!)
245 fullspectrum = fs;
246 infrared = ir;
247
248 lux = CalculateLux(fullspectrum, infrared);
249
250 return true;
251}

346bool DIOI2CLIGHTSENSORTSL2561::ReadData16(XBYTE reg, XWORD& value, int timeout)
347{
348 XBUFFER xbuffer;
349
350 value = 0;
351
352 xbuffer.Add((XBYTE)reg);
353 if(!diostream->Write(xbuffer)) return false;
354 if(!diostream->WaitToFlushOutXBuffer(timeout)) return false;
355
356 xbuffer.Resize(2);
357 if(!diostream->WaitToFilledReadingBuffer(2, timeout)) return false;
358 if(!diostream->Read(xbuffer)) return false;
359 if(xbuffer.GetSize() < 2) return false;
360
361 value = xbuffer.GetByte(1);
362 value <<= 8;
363 value |= xbuffer.GetByte(0);
364
365 return true;
366}

399 if(status) status = ReadData16((DIOI2CLIGHTSENSORTSL2561_COMMAND_BIT | DIOI2CLIGHTSENSORTSL2561_WORD_BIT | DIOI2CLIGHTSENSORTSL2561REGISTER_CHAN1_LOW), infrared , timeout);

449XDWORD DIOI2CLIGHTSENSORTSL2561::CalculateLux(XWORD broadband, XWORD inflared)
450{
451 XDWORD chscale;
452 XDWORD channel1;
453 XDWORD channel0;
454 XDWORD clipthreshold;
455 XDWORD ratio1 = 0;
456 int ratio;
457 XDWORD b, m;
458 int temp;
459 XDWORD lux;
460
461 // Make sure the sensor isn't saturated!
462 switch(integrationtime)
463 {
464 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_13MS : clipthreshold = DIOI2CLIGHTSENSORTSL2561_CLIPPING_13MS; break;
465 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_101MS : clipthreshold = DIOI2CLIGHTSENSORTSL2561_CLIPPING_101MS; break;
466 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_402MS : clipthreshold = DIOI2CLIGHTSENSORTSL2561_CLIPPING_402MS; break;
467 default : clipthreshold = DIOI2CLIGHTSENSORTSL2561_CLIPPING_402MS; break;
468 }
469
470 // return 0 lux if the sensor is saturated
471 if((broadband > clipthreshold) || (inflared > clipthreshold)) return 0;
472
473
474 // get the correct scale depending on the intergration time
475 switch(integrationtime)
476 {
477 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_13MS : chscale = DIOI2CLIGHTSENSORTSL2561_LUX_CHSCALE_TINT0; break;
478 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_101MS : chscale = DIOI2CLIGHTSENSORTSL2561_LUX_CHSCALE_TINT1; break;
479 case DIOI2CLIGHTSENSORTSL2561INTEGRATIONTIME_402MS : chscale = (1 << DIOI2CLIGHTSENSORTSL2561_LUX_CHSCALE); break;
480 default : chscale = (1 << DIOI2CLIGHTSENSORTSL2561_LUX_CHSCALE); break; // No scaling ... integration time = 402ms
481 }
482
483 // scale for gain (1x or 16x)
484 if (!gain) chscale = chscale << 4;
485
486 // scale the channel values
487 channel0 = (broadband * chscale) >> DIOI2CLIGHTSENSORTSL2561_LUX_CHSCALE;
488 channel1 = (inflared * chscale) >> DIOI2CLIGHTSENSORTSL2561_LUX_CHSCALE;
489
490 // find the ratio of the channel values (Channel1/Channel0)
491 if (channel0 != 0) ratio1 = (channel1 << (DIOI2CLIGHTSENSORTSL2561_LUX_RATIOSCALE+1)) / channel0;
492
493 // round the ratio value
494 ratio = (ratio1 + 1) >> 1;
495
496 #ifdef TSL2561_PACKAGE_CS
497 if ((ratio >= 0) && (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K1C))
498 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B1C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M1C; }
499 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K2C)
500 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B2C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M2C; }
501 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K3C)
502 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B3C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M3C; }
503 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K4C)
504 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B4C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M4C; }
505 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K5C)
506 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B5C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M5C; }
507 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K6C)
508 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B6C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M6C; }
509 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K7C)
510 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B7C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M7C; }
511 else if (ratio > DIOI2CLIGHTSENSORTSL2561_LUX_K8C)
512 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B8C; m = DIOI2CLIGHTSENSORTSL2561_LUX_M8C; }
513 #else
514 if ((ratio >= 0) && (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K1T))
515 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B1T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M1T; }
516 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K2T)
517 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B2T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M2T; }
518 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K3T)
519 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B3T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M3T; }
520 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K4T)
521 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B4T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M4T; }
522 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K5T)
523 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B5T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M5T; }
524 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K6T)
525 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B6T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M6T; }
526 else if (ratio <= DIOI2CLIGHTSENSORTSL2561_LUX_K7T)
527 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B7T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M7T; }
528 else if (ratio > DIOI2CLIGHTSENSORTSL2561_LUX_K8T)
529 { b = DIOI2CLIGHTSENSORTSL2561_LUX_B8T; m = DIOI2CLIGHTSENSORTSL2561_LUX_M8T; }
530 #endif
531
532 temp = ((channel0 * b) - (channel1 * m));
533
534 // do not allow negative lux value
535 if (temp < 0) temp = 0;
536
537 // round lsb (2^(LUX_SCALE-1))
538 temp += (1 << (DIOI2CLIGHTSENSORTSL2561_LUX_LUXSCALE-1));
539
540 // strip off fractional portion
541 lux = (temp >> DIOI2CLIGHTSENSORTSL2561_LUX_LUXSCALE);
542
543 // Signal I2C had no errors
544 return lux;
545}
