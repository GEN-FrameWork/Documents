181 for(aa = adapter_addresses; aa != NULL; aa = aa->Next)

183 DIOSTREAMDEVICEIP* device = GEN_NEW DIOSTREAMDEVICEIP();

186 device->SetIsActive((aa->OperStatus == IfOperStatusUp?true:false));

188 char buffer[BUFSIZ];
