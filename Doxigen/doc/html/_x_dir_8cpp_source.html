479bool XDIR::Copy(XPATH& xpathsource,XPATH& xpathtarget)
480{
481 #define COPYFILE_MAXBUFFER 2048
482
483 bool status = true;
484
485 XFILE* GEN_XFACTORY_CREATE(fsource, Create_File())
486 XFILE* GEN_XFACTORY_CREATE(ftarget, Create_File());
487
488 if(fsource && ftarget)
489 {
490 if(fsource->Open(xpathsource))
491 {
492 if(ftarget->Create(xpathtarget))
493 {
494 XBYTE* buffer = GEN_NEW XBYTE[COPYFILE_MAXBUFFER];
495 if(buffer)
496 {
497 XDWORD rsize = 0;
498
499 do { rsize = COPYFILE_MAXBUFFER;
500
501 memset(buffer,0,rsize);
502
503 fsource->Read(buffer,&rsize);
504
505 if(rsize) ftarget->Write(buffer,rsize);
506
507 } while(rsize == COPYFILE_MAXBUFFER);
508
509 GEN_DELETE_ARRAY buffer;
510 }
511
512 ftarget->Close();
513
514 } else status = false;
515
516 fsource->Close();
517
518 } else status = false;
519
520 } else status = false;
521
522 GEN_XFACTORY.Delete_File(fsource);
523 GEN_XFACTORY.Delete_File(ftarget);
524
525 return status;
526}
