102void ANDROIDNATIVEACTIVITY_EVENTLOOP::Run(ANDROIDNATIVEACTIVITY_HANDLER* activityhandler, ANDROIDNATIVEINPUT_HANDLER* inputhandler)
103{
104 int32_t result;
105 int32_t events;
106 android_poll_source* source;
107
108 //app_dummy();
109
110 this->activityhandler = activityhandler;
111 this->inputhandler = inputhandler;
112
113 while (true)
114 {
115 // Event processing loop.
116 while((result = ALooper_pollOnce(enabled?0:-1, NULL, &events, (void**)&source)) >= 0)
117 {
118 // An event has to be processed.
119 if(source != NULL) source->process(application, source);
120
121 // Application is getting destroyed.
122 if(application->destroyRequested) return;
123 }
124
125 if((enabled) && (!quit))
126 {
127 if(activityhandler->OnStep() != STATUS_OK)
128 {
129 quit = true;
130 ANativeActivity_finish(application->activity);
131 }
132 }
133 }
134}

193int32_t ANDROIDNATIVEACTIVITY_EVENTLOOP::ProcessInputEvent(AInputEvent* event, android_app* application)
194{
195 int32_t eventtype = AInputEvent_getType(event);
196
197 this->application = application;
198
199 switch(eventtype)
200 {
201 case AINPUT_EVENT_TYPE_MOTION : switch (AInputEvent_getSource(event))
202 {
203 case AINPUT_SOURCE_TOUCHSCREEN : return inputhandler->OnTouchEvent(event); break;
204 case AINPUT_SOURCE_TRACKBALL : return inputhandler->OnTrackballEvent(event); break;
205 }
206 break;
207
208 case AINPUT_EVENT_TYPE_KEY : return inputhandler->OnKeyboardEvent(event); break;
209 }
210
211 return 0;
212}

225void ANDROIDNATIVEACTIVITY_EVENTLOOP::ProcessApplicationEvent(int32_t command, android_app* application)
226{
227 this->application = application;
228
229 switch(command)
230 {
231 case APP_CMD_CONFIG_CHANGED : activityhandler->OnConfigurationChanged();
232 break;
233
234 case APP_CMD_INIT_WINDOW : activityhandler->OnCreateWindow();
235 break;
236
237 case APP_CMD_DESTROY : activityhandler->OnDestroy();
238 break;
239
240 case APP_CMD_GAINED_FOCUS : Activate();
241 activityhandler->OnGainFocus();
242 break;
243
244 case APP_CMD_LOST_FOCUS : activityhandler->OnLostFocus();
245 Deactivate();
246 break;
247
248 case APP_CMD_LOW_MEMORY : activityhandler->OnLowMemory();
249 break;
250
251 case APP_CMD_PAUSE : activityhandler->OnPause();
252 Deactivate();
253 break;
254
255 case APP_CMD_RESUME : Activate();
256 activityhandler->OnResume();
257 break;
258
259 case APP_CMD_SAVE_STATE : activityhandler->OnSaveState(&application->savedState, &application->savedStateSize);
260 break;
261
262 case APP_CMD_START : activityhandler->OnStart();
263 break;
264
265 case APP_CMD_STOP : activityhandler->OnStop();
266 break;
267
268 case APP_CMD_TERM_WINDOW : activityhandler->OnDestroyWindow();
269 Deactivate();
270 break;
271
272 default : break;
273 }
274}
