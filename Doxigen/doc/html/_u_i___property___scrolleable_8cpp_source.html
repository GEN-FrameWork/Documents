381void UI_PROPERTY_SCROLLEABLE::Scroll_SetStep(UI_PROPERTY_SCROLLEABLE_TYPE type, double step)
382{
383 switch(type)
384 {
385 case UI_PROPERTY_SCROLLEABLE_TYPE_UNKWOWN :
386 case UI_PROPERTY_SCROLLEABLE_TYPE_ALL :
387 default : break;
388
389 case UI_PROPERTY_SCROLLEABLE_TYPE_HORIZONTAL : if(!horizontal.active) break;
390
391 if((horizontal.displacement + step) > 0)
392 {
393 horizontal.displacement = 0;
394 break;
395 }
396
397 if(horizontal.limit != -1)
398 {
399 if((horizontal.displacement + step) < horizontal.limit)
400 {
401 horizontal.displacement = horizontal.limit;
402 break;
403 }
404 }
405
406 horizontal.step = step;
407 horizontal.displacement += step;
408 break;
409
410 case UI_PROPERTY_SCROLLEABLE_TYPE_VERTICAL : if(!vertical.active) break;
411
412 if((vertical.displacement + step) > 0)
413 {
414 vertical.displacement = 0;
415 break;
416 }
417
418 if(vertical.limit != -1)
419 {
420 if((vertical.displacement + step) < vertical.limit)
421 {
422 vertical.displacement = vertical.limit;
423 break;
424 }
425 }
426
427 vertical.step = step;
