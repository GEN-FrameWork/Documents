272{

333bool GRPVECTORFILE::DetectFileFormatText(XPATH& pathfile)
334{
335 XFILETXTFORMATCHAR formatchar = XFILETXTFORMATCHAR_UNKNOWN;
336 bool isText = false;
337
338 XFILETXT* fileTXT = GEN_NEW XFILETXT();
339 if(fileTXT)
340 {
341 if(fileTXT->Open(pathfile))
342 {
343 formatchar = fileTXT->GetFormatCharFromFile();
344
345 fileTXT->Close();
346 }
347
348 GEN_DELETE fileTXT;
349 }
350
351 if(formatchar != XFILETXTFORMATCHAR_ASCII) return true;
352
353 XFILE* file = GEN_XFACTORY.Create_File();
354 if(file)
355 {
356 if(file->Open(pathfile))
357 {
358 #define MAX_BUFFER_TEST 1024*5
359
360 XBUFFER buffer;
361 XDWORD size = MAX_BUFFER_TEST;
362 XDWORD controlchars = 0;
363
364 buffer.Resize(MAX_BUFFER_TEST);
365 file->Read(buffer.Get(), &size);
366
367 if(size)
368 {
369 for(XDWORD c=0; c<size; c++)
370 {
371 if(((buffer.Get()[c] >= 0x00) && (buffer.Get()[c] < 0x0A)) ||
372 ((buffer.Get()[c] > 0x0D) && (buffer.Get()[c] < 0x20)))
373 {
374 controlchars++;
375 }
376 }
377
378 int percent = (controlchars*100/size);
379 if(percent < 10) isText = true;
380 }
381
382 file->Close();
383 }
384
385 GEN_DELETE file;
386 }
387
388 return isText;
389}
