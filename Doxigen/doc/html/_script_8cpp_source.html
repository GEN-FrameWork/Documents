296{

298

300
301 XDWORD ID = GEN_SCRIPT_CACHE.GenerateID(xpath);
302
303 XSTRING* _script = GEN_SCRIPT_CACHE.Cache_Get(ID);
304 if(_script)
305 {
306 script.Empty();
307 script += _script->Get();
308
309 GetNameScript()->Format(__L("ID%08X"), ID);

435bool SCRIPT::LoadScriptAndRun(XVECTOR<XSTRING*>* listscripts, SCRFUNCADJUSTLIBRARYS adjustlibrarys)
436{
437 if(!listscripts)
438 {
439 return false;
440 }
441
442 bool status = false;
443
444 for(XDWORD c=0; c<listscripts->GetSize(); c++)
445 {
446 XSTRING* linescripts = listscripts->Get(c);
447 if(linescripts)
448 {
449 if(!linescripts->IsEmpty())
450 {
451 XVECTOR<XSTRING*> namescripts;
452
453 linescripts->Split(__C(','), namescripts);
454
455 XSTRING* namescript = namescripts.Get(0);
456 if(namescript)
457 {
458 SCRIPT::EliminateExtraChars(namescript);
459
460 SCRIPT* script = SCRIPT::Create(namescript->Get());
461 if(script)
462 {
463 XPATH allpath;
464 XDWORD ID = 0;
465 bool incache = false;
466
467 if(adjustlibrarys)
468 {
469 adjustlibrarys(script);
470 }
471
472 GEN_XPATHSMANAGER.GetPathOfSection(XPATHSMANAGERSECTIONTYPE_SCRIPTS, allpath);
473
474 for(XDWORD d=0; d<namescripts.GetSize(); d++)
475 {
476 allpath += __C(',');
477 allpath += namescript->Get();
478 }
479
480 #ifdef SCRIPT_CACHE_ACTIVE
481
482 allpath.Slash_Normalize(false);
483 ID = GEN_SCRIPT_CACHE.GenerateID(allpath);
484 XSTRING* _script = GEN_SCRIPT_CACHE.Cache_Get(ID);
485 if(_script)
486 {
487 (*script->GetScript()) += _script->Get();
488
489 incache = true;
490 status = !_script->IsEmpty();
491
492 script->GetNameScript()->Format(__L("ID%08X"), ID);
493 }
494
495 #endif
496
497 if(!incache)
498 {
499 for(XDWORD d=0; d<namescripts.GetSize(); d++)
500 {
501 namescript = namescripts.Get(d);
502 if(namescript)
503 {
504 XPATH xpath;
505
506 SCRIPT::EliminateExtraChars(namescript);
507
508 GEN_XPATHSMANAGER.GetPathOfSection(XPATHSMANAGERSECTIONTYPE_SCRIPTS, xpath);
509 xpath.Slash_Add();
510 xpath += namescript->Get();
511
512 status = script->LoadAdd(xpath);
513 if(!status)
514 {
515 break;
516 }
517 }
518 }
519 }
520
521 #ifdef SCRIPT_CACHE_ACTIVE
522 if(status && !incache)
523 {
524 GEN_SCRIPT_CACHE.Cache_Add(ID, script->GetScript());
525 }
526 #endif
527
528 if(status)
529 {
530 script->Run();
531 }
532
533 GEN_DELETE script;
534 script = NULL;
535
536 }
537 }
538
539 namescripts.DeleteContents();
540 namescripts.DeleteAll();
541
542 }
543 }
544 }
545
546 return status;
547}
