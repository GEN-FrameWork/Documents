275bool HASHSHA2::Do(XBYTE* input, XQWORD size)
276{
277 XBYTE digest[128];
278
279 memset(digest , 0, 128);
280
281 switch(typesha2)
282 {
283 case HASHSHA2TYPE_UNKNOWN : return false;
284
285 case HASHSHA2TYPE_224 : Sha2_224(input, (XDWORD)size, digest);
286 result->Add(digest, HASHSHA2_224_DIGEST_SIZE);
287 break;
288
289 case HASHSHA2TYPE_256 : Sha2_256(input, (XDWORD)size, digest);
290 result->Add(digest, HASHSHA2_256_DIGEST_SIZE);
291 break;
292
293 case HASHSHA2TYPE_384 : Sha2_384(input, (XDWORD)size, digest);
294 result->Add(digest, HASHSHA2_384_DIGEST_SIZE);
295 break;
296
297 case HASHSHA2TYPE_512 : Sha2_512(input, (XDWORD)size, digest);
298 result->Add(digest, HASHSHA2_512_DIGEST_SIZE);
299 break;
300
301 default : return false;
302 }
303
304 return true;
305}

342void HASHSHA2::Sha2_256_Transf(HASHSHA2_256_CTX* ctx, XBYTE* input, XDWORD blocknb)
343{
344 XDWORD w[64];
345 XDWORD wv[8];
346 XDWORD t1;
347 XDWORD t2;
348 XBYTE* subblock;
349 int i;
350 int j;
351
352 for(i=0; i<(int)blocknb; i++)
353 {
354 subblock = input + (i << 6);
355
356 for(j=0; j<16; j++)
357 {
358 PACK32(&subblock[j << 2], &w[j]);
359 }
360
361 for(j=16; j<64; j++)
362 {
363 SHA256_SCR(j);
364 }
365
366 for(j=0; j<8; j++)
367 {
368 wv[j] = ctx->h[j];
369 }
370
371 for(j=0; j<64; j++)
372 {
373 t1 = wv[7] + SHA256_F2(wv[4]) + CH(wv[4], wv[5], wv[6]) + Sha256_k[j] + w[j];
374 t2 = SHA256_F1(wv[0]) + MAJ(wv[0], wv[1], wv[2]);
375
376 wv[7] = wv[6];
377 wv[6] = wv[5];
378 wv[5] = wv[4];
379 wv[4] = wv[3] + t1;
380 wv[3] = wv[2];
381 wv[2] = wv[1];
382 wv[1] = wv[0];
383 wv[0] = t1 + t2;
384 }
385
386 for(j=0; j<8; j++)
387 {
388 ctx->h[j] += wv[j];
389 }
390 }
391}

449void HASHSHA2::Sha2_256_Update(HASHSHA2_256_CTX* ctx, XBYTE* input, XDWORD size)
450{
451 XDWORD blocknb;
452 XDWORD new_len, rem_len, tmp_len;
453 XBYTE* shifted_input;
454
455 tmp_len = HASHSHA2_256_BLOCK_SIZE - ctx->size;
456 rem_len = size < tmp_len ? size : tmp_len;
457
458 memcpy(&ctx->block[ctx->size], input, rem_len);
459
460 if(ctx->size + size < HASHSHA2_256_BLOCK_SIZE)
461 {
462 ctx->size += size;
