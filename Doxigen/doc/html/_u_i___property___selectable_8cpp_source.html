186UI_PROPERTY_SELECTABLE_STATE UI_PROPERTY_SELECTABLE::SetSelectableStateFromString(XCHAR* selectablestr)
187{
188 if(!selectablestr) return UI_PROPERTY_SELECTABLE_STATE_DEACTIVE;
189
190 XSTRING _selectablestr;
191
192 selectable_state = UI_PROPERTY_SELECTABLE_STATE_DEACTIVE;
193
194 _selectablestr = selectablestr;
195
196 if(!_selectablestr.Compare(__L("active"), true)) selectable_state = UI_PROPERTY_SELECTABLE_STATE_ACTIVE;
197 else if(!_selectablestr.Compare(__L("deactive"), true)) selectable_state = UI_PROPERTY_SELECTABLE_STATE_DEACTIVE;
198 else if(!_selectablestr.Compare(__L("preselect"), true)) selectable_state = UI_PROPERTY_SELECTABLE_STATE_PRESELECT;
199 else if(!_selectablestr.Compare(__L("selected"), true)) selectable_state = UI_PROPERTY_SELECTABLE_STATE_SELECTED;
200
201 return selectable_state;
202}
