186 void QueryPerformanceData (PERF_DATA_BLOCK** perfdata, DWORD objectindex, DWORD counterindex)
187 {
188 //
189 // Since i want to use the same allocated area for each query,
190 // i declare CBUFFER as static.
191 // The allocated is changed only when RegQueryValueEx return ERROR_MORE_DATA
192 //
193 static CBUFFER buffer(XWINDOWSSYSTEM_TOTALBYTES);
194 DWORD buffersize = buffer.GetSize();
195 LONG result;
196
197 XSTRING keyname;
198 keyname.Format(__L("%d"), objectindex);
199
200 buffer.Reset();
201 while((result = RegQueryValueEx( HKEY_PERFORMANCE_DATA , keyname.Get()
202 , NULL
203 , NULL
204 , buffer
205 , &buffersize )) == ERROR_MORE_DATA )
206 {
207 // Get a buffer that is big enough.
208
209 buffersize += XWINDOWSSYSTEM_BYTEINCREMENT;
210 buffer.Realloc(buffersize);
211 }
212
213 *perfdata = (PPERF_DATA_BLOCK) buffer.buffer;
214 }

217 T GetCounterValue (PPERF_OBJECT_TYPE perfobj, DWORD counterindex, LPCTSTR instancename)
218 {
219 PPERF_COUNTER_DEFINITION perfcntr = NULL;
220 PPERF_INSTANCE_DEFINITION perfinst = NULL;
221 PPERF_COUNTER_BLOCK counterblock = NULL;
222
223 // Get the first counter.
224
225 perfcntr = FirstCounter( perfobj );
226
227 for(DWORD j=0; j<perfobj->NumCounters; j++ )
228 {
229 if (perfcntr->CounterNameTitleIndex == counterindex) break;
230
231 // Get the next counter.
232 perfcntr = NextCounter( perfcntr );
233 }
234
235 if(perfobj->NumInstances == PERF_NO_INSTANCES)
236 {
237 counterblock = (PPERF_COUNTER_BLOCK) ((LPBYTE) perfobj + perfobj->DefinitionLength);
238 }
239 else
240 {
241 perfinst = FirstInstance( perfobj );
242
243 // Look for instance instancename
244 //_bstr_t bstrInstance;
245 //_bstr_t bstrInputInstance = instancename;
246
247 XSTRING instancenamestr;
248 XSTRING instancenamestr2;
249
250 instancenamestr = instancename;
251
252 for(int k=0; k<perfobj->NumInstances; k++)
253 {
254 instancenamestr2 = (XCHAR*)((PBYTE)perfinst + perfinst->NameOffset);
255
256 if(!instancenamestr.Compare(instancenamestr2))
257 {
258 counterblock = (PPERF_COUNTER_BLOCK) ((LPBYTE) perfinst + perfinst->ByteLength);
259 break;
260 }
261
262 // Get the next instance.
263
264 perfinst = NextInstance( perfinst );
265 }
266 }
267
268 if(counterblock)
269 {
270 T* value = NULL;
271
272 value = (T*)((LPBYTE) counterblock + perfcntr->CounterOffset);
273 return *value;
274 }
275
276 return -1;
277 }

280 T GetCounterValueForProcessID (PPERF_OBJECT_TYPE perfobj, DWORD counterindex, DWORD processID)
281 {
282 int PROC_ID_COUNTER = 784;
283 BOOL processIDexist = FALSE;
284 PPERF_COUNTER_DEFINITION perfcntr = NULL;
285 PPERF_COUNTER_DEFINITION therequestedperfcntr = NULL;
286 PPERF_COUNTER_DEFINITION procIDperfcntr = NULL;
287 PPERF_INSTANCE_DEFINITION perfinst = NULL;
288 PPERF_COUNTER_BLOCK counterblock = NULL;
289
290 // Get the first counter.
291
292 perfcntr = FirstCounter(perfobj);
293
294 for(DWORD j=0; j<perfobj->NumCounters; j++)
295 {
296 if(perfcntr->CounterNameTitleIndex == PROC_ID_COUNTER)
297 {
298 procIDperfcntr = perfcntr;
299 if(therequestedperfcntr) break;
300 }
301
302 if(perfcntr->CounterNameTitleIndex == counterindex)
303 {
304 therequestedperfcntr = perfcntr;
305 if(procIDperfcntr) break;
306 }
307
308 // Get the next counter.
309
310 perfcntr = NextCounter( perfcntr );
311 }
312
313 if(perfobj->NumInstances == PERF_NO_INSTANCES)
314 {
315 counterblock = (PPERF_COUNTER_BLOCK) ((LPBYTE) perfobj + perfobj->DefinitionLength);
316 }
317 else
318 {
319 perfinst = FirstInstance( perfobj );
320
321 for(int k=0; k<perfobj->NumInstances; k++)
322 {
323 counterblock = (PPERF_COUNTER_BLOCK) ((LPBYTE) perfinst + perfinst->ByteLength);
324 if(counterblock)
325 {
326 T _processID = 0;
327
328 _processID = *(T*)((LPBYTE) counterblock + procIDperfcntr->CounterOffset);
329 if((XDWORD)(_processID) == processID)
330 {
331 processIDexist = TRUE;
332 break;
333 }
334 }
335
336 // Get the next instance.
337
338 perfinst = NextInstance( perfinst );
339 }
340 }
341
342 if(processIDexist && counterblock)
343 {
344 T *value = NULL;
345 value = (T*)((LPBYTE) counterblock + therequestedperfcntr->CounterOffset);
346 return *value;
347 }
348
349 return -1;
350 }
