127bool INPLINUXDEVICEMOUSEX11::Update()
128{
129 if(!created || !enabled)
130 {
131 return false;
132 }
133
134 if(!grpscreenx11)
135 {
136 return false;
137 }
138
139 if(!grpscreenx11->HasFocus())
140 {
141 return false;
142 }
143
144 Display* display = grpscreenx11->GetDisplay();
145 Window root;
146 Window child;
147 int rootx;
148 int rooty;
149 int mousex = -1;
150 int mousey = -1;
151 unsigned int mask;
152
153 XQueryPointer(display, (*grpscreenx11->GetWindow()), &root, &child, &rootx, &rooty, &mousex, &mousey, &mask);
154
155 INPCURSOR* cursor = GetCursor(INPCURSOR_ID_MOUSE);
156 if(cursor)
157 {
158 if((mousex>=0)&&(mousey>=0)&&(mousex < grpscreenx11->GetWidth()) && (mousey < grpscreenx11->GetHeight()))
159 {
160 cursor->SetIsChanged(false);
161
162 if((cursor->GetX() != mousex) || (cursor->GetY() != mousey))
163 {
164 mousey = (grpscreenx11->GetHeight()-mousey);
165 cursor->Set(mousex,mousey);
166 cursor->SetIsChanged(true);
167 }
168 }
169 else
170 {
171 mousex = -1;
172 mousey = -1;
173
174 cursor->Set(mousex,mousey);
175 cursor->SetIsChanged(true);
176 }
177
178 INPBUTTON* button = GetButton(INPBUTTON_ID_MOUSE_LEFT);
179 if(button) cursor->AddPointToMotion(button->IsPressed());
180 }
181
182 //-------------------------------------------------------------
183 // Read Buttons
184
185 for(XDWORD c=0; c<buttons.GetSize(); c++)
186 {
187 INPBUTTON* button=buttons.Get(c);
188 if(button)
189 {
190 if(button->GetState() == INPBUTTON_STATE_RELEASED)
191 {
192 button->SetState(INPBUTTON_STATE_UP);
193 }
194 else
195 {
196 if(button->GetState() == INPBUTTON_STATE_PRESSED)
197 {
198 button->SetState(INPBUTTON_STATE_HOLD);
199 }
200 }
201 }
202 }
203
204 XEvent event;
205
206 if(XPending(display))
207 {
208 XNextEvent(display, &event);
209
210 switch(event.type)
211 {
212 case ButtonPress : { INPBUTTON* button = GetButtonByCode(event.xbutton.button);
213 if(button)
214 {
215 switch (button->GetState())
216 {
217 case INPBUTTON_STATE_UP : button->SetState(INPBUTTON_STATE_PRESSED); break;
218 case INPBUTTON_STATE_HOLD : button->SetState(INPBUTTON_STATE_HOLD); break;
219 case INPBUTTON_STATE_PRESSED : button->SetState(INPBUTTON_STATE_HOLD); break;
220 case INPBUTTON_STATE_RELEASED : button->SetState(INPBUTTON_STATE_PRESSED); break;
221 default : break;
222 }
223
224 button->SetPressed(true);
225 }
226 }
227 break;
228
229 case ButtonRelease : { INPBUTTON* button = GetButtonByCode(event.xbutton.button);
230 if(button)
231 {
232 switch (button->GetState())
233 {
234 case INPBUTTON_STATE_UP : button->SetState(INPBUTTON_STATE_UP); break;
235 case INPBUTTON_STATE_HOLD : button->SetState(INPBUTTON_STATE_RELEASED); break;
236 case INPBUTTON_STATE_PRESSED : button->SetState(INPBUTTON_STATE_RELEASED); break;
237 case INPBUTTON_STATE_RELEASED : button->SetState(INPBUTTON_STATE_RELEASED); break;
238 default : break;
239 }
240
241 button->SetPressed(false);
242 }
243 }
244 break;
245
246 default : break;
247 }
248 }
249
250
251 return true;
252}
