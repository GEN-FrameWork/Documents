167bool DIOANDROIDPCAP::CreateListNetInterfaces()
168{
169 pcap_if_t* allnetinterfaces;
170 pcap_if_t* netinterface;
171 char errbuf[PCAP_ERRBUF_SIZE];
172
173 // Retrieve the device list
174 if(pcap_findalldevs(&allnetinterfaces, errbuf) == -1) return false;
175
176 // Create list
177 for(netinterface=allnetinterfaces; netinterface; netinterface=netinterface->next)
178 {
179 DIOPCAPNETINTERFACE* _netinterface = GEN_NEW DIOPCAPNETINTERFACE();
180 if(_netinterface)
181 {
182 XSTRING string;
183
184 string = netinterface->name; _netinterface->SetName(string);
185 string = netinterface->description; _netinterface->SetDescription(string);
186
187 netinterfaces.Add(_netinterface);
188 }
189 }
190
191 pcap_freealldevs(allnetinterfaces);
192
193 return true;
194}
