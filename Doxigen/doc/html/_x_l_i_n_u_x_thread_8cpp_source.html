306void* XLINUXTHREAD::Callback(void* thread)
307{
308 XLINUXTHREAD* th = NULL;
309 XTHREADFUNCTION func = NULL;
310 void* param;
311
312 th = (XLINUXTHREAD*)thread;
313 if(!th) return NULL;
314
315 func = (XTHREADFUNCTION)th->GetFunction();
316 if(!func) return NULL;
317
318 param = th->GetParam();
319
320 th->gotofunction = true;
321
322 if((th->priority != XTHREADPRIORITY_UNKNOWN) &&
323 (th->priority != XTHREADPRIORITY_NORMAL))
324 {
325 struct sched_param param; // scheduling priority
326
327 int policy = SCHED_FIFO; // scheduling policy
328
329 switch(th->priority)
330 {
331 #ifdef SCHED_IDLE
332 case XTHREADPRIORITY_LOW : policy = SCHED_IDLE;
333 param.sched_priority = sched_get_priority_min(policy);
334 break;
335 #endif
336
337 case XTHREADPRIORITY_HIGH : policy = SCHED_FIFO;
338 param.sched_priority = sched_get_priority_max(policy)/2;
339 break;
340
341 case XTHREADPRIORITY_REALTIME : policy = SCHED_FIFO;
342 param.sched_priority = sched_get_priority_max(policy);
343 break;
344
345 default : break;
346 }
347
348 pthread_setschedparam(th->thhandle, policy, &param);
349 }
350
351 while(1)
352 {
353 if(!th) break;
354
355 if(th->statusfunc == XTHREADSTATUS_EXIT) break;
356 if(th->statusfunc == XTHREADSTATUS_END) break;
357
358 if(th->statusfunc == XTHREADSTATUS_RUN)
359 {
360 if(func)
361 {
362 th->isinfunction = true;
363 func(param);
364 th->isinfunction = false;
365
366 } else break;
367 }
368
369 if(!th) break;
370
371 if(th->waityield) usleep(th->waityield*1000);
372 }
373
374 if(th) th->statusfunc = XTHREADSTATUS_END;
375
376 pthread_exit(0);
377
378 return NULL;
379}
