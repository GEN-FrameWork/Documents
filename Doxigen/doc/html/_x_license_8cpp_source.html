184{
185 XSTRING origin;
186 XBUFFER originbuffer;

188 XUUID ID;
189 XBUFFER* result = NULL;
190
191 ID.Empty();
192
193 //GEN_XSYSTEM.GetOperativeSystemID(origin);
194
195 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetBIOSSerialNumber()->Get());
196 origin.AddFormat(__L(" %s"), GEN_XSYSTEM.GetCPUSerialNumber()->Get());
197
198 origin.ConvertToUTF8(originbuffer);
199

202 {
203 return false;
204 }
205

215 {
216 return false;
217 }

258{

261 XBUFFER xbufferkey;

348bool XLICENSE::CipherExpirationDate(bool cipher, XSTRING& applicationID, XSTRING& expirationdate)
349{
350 XBUFFER xbufferkey;
351 CIPHERAES cipherAES;
352 CIPHERKEYSYMMETRICAL key;
353 bool status = false;
354
355 if(!GetBufferKeyFromMachineID(applicationID, xbufferkey))
356 {
357 return false;
358 }
359
360 key.Set(xbufferkey);
361
362 cipherAES.SetKey(&key);
363 cipherAES.SetChainingMode(CIPHERCHAININGMODE_CBC);
364 cipherAES.SetPaddingType(XBUFFER_PADDINGTYPE_SPACES);
365
366 if(cipher)
367 {
368 XBUFFER charstr;
369
370 expirationdate.ConvertToASCII(charstr);
371
372 status = cipherAES.Cipher((XBYTE*)charstr.Get(), expirationdate.GetSize());
373 if(status)
374 {
375 expirationdate.Empty();
376 cipherAES.GetResult()->ConvertToBase64(expirationdate);
377 return true;
378 }
379 }
380 else
381 {
382 XBUFFER xbuffer;
383
384 xbuffer.ConvertFromBase64(expirationdate);
385
386 status = cipherAES.Uncipher(xbuffer);
387 if(status)
388 {
389 expirationdate.Empty();
390
391 for(XDWORD c=0; c<cipherAES.GetResult()->GetSize(); c++)
392 {
393 expirationdate.Add((XCHAR)cipherAES.GetResult()->GetByte(c));
394 }
395
396 expirationdate.DeleteCharacter(0x20, XSTRINGCONTEXT_TO_END);
397
398 return true;
399 }
400 }
401
402 return status;
403}

649bool XLICENSE::CheckMasterCreation(XPATH& xpath, XLICENSEID& licenseID, XSTRING& applicationID, int expirationdays)
650{
651 XFILEINI fileini;
652 XSTRING licensestring;
653
654 if(!fileini.Open(xpath))
655 {
656 return false;
657 }
658
659 fileini.ReadValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_LICENSE, licensestring);
660 fileini.Close();
661
662 if(!applicationID.Compare(licensestring))
663 {
664 XFILEINI fileini;
665 XSTRING licenseIDstring;
666 XSTRING licensestring;
667 XSTRING licenseexpiration;
668 XBUFFER licencia;
669 XDATETIME* expirationdatetime;
670
671 licenseID.GetID()->GetToString(licenseIDstring);
672 if(!Generate(licenseID, &applicationID, &licencia))
673 {
674 return false;
675 }
676
677 licensestring.ConvertHexStringFromBuffer(licencia);
678
679 GEN_XFACTORY_CREATE(expirationdatetime, CreateDateTime())
680 if(!expirationdatetime)
681 {
682 return false;
683 }
684
685 expirationdatetime->SetToZero();
686 if(expirationdays)
687 {
688 expirationdatetime->Read();
689 expirationdatetime->AddDays((int)expirationdays);
690 }
691
692 expirationdatetime->GetDateTimeToString(XDATETIME_FORMAT_STANDARD, licenseexpiration);
693 GEN_XFACTORY.DeleteDateTime(expirationdatetime);
694
695 if(!CipherExpirationDate(true, applicationID, licenseexpiration))
696 {
697 return false;
698 }
699
700 if(!fileini.Create(xpath))
701 {
702 return false;
703 }
704
705 fileini.CreateSection(XLICENSE_FILESECTION);
706 fileini.WriteValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_ID , licenseIDstring);
707 fileini.WriteValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_LICENSE , licensestring);
708 fileini.WriteValue(XLICENSE_FILESECTION, XLICENSE_FILEVALUE_EXPIRATION, licenseexpiration);
709
710 fileini.Close();
711
712 return true;
713 }
714
715 return false;
716}
