326bool POSTGRESQL_DATABASE::GetTables()
327{
328 DB_SQL_STRING querystring;
329 querystring.Format(__L("SELECT relname FROM pg_class WHERE relkind='r';"));
330
331 DB_SQL_QUERY* query = CreateQuery();
332 DB_SQL_VARIANT variant;
333
334 if (query!=NULL)
335 {
336 query->Set(querystring.Get());
337 if (this->Execute(query))
338 {
339 DB_SQL_RESULT* result=query->GetResult();
340 if(result!=NULL) //hay un result
341 {
342 result->GetNumRows();
343 while(result->HasNext())
344 {
345 result->ProcessRow();
346
347 DB_SQL_ROW* row = result->GetRow();
348 for(XDWORD e=0; e<result->GetNumColumns(); e++)
349 {
350 variant=row->Get(e);
351 if(variant.GetType()==(XVARIANT_TYPE)DB_SQL_VARIANT_TYPE_STRING)
352 {
353 DB_SQL_STRING* tablename = GEN_NEW DB_SQL_STRING();
354 if(tablename)
355 {
356 tablename->Set((XCHAR*)variant);
357 tables.Add(tablename);
358 }
359 }
360 }
361 result->Next();
362 }
363 }
364 else return false;
365 }
366 delete(query);
367 }
368 else return false;
369
370 return true;
371}
