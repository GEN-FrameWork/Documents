685bool DIOPCAPFRAME::GetHeader(DIOPCAPUDPHEADER& header)
686{
687 DIOPCAPIPHEADER ipheader;
688 if(!GetHeader(ipheader))
689 {
690 return false;
691 }
692
693 int iplen = (ipheader.ver_ihl & 0xf) * 4;
694
695 UDP_header = (DIOPCAPUDPHEADER*)((XBYTE*)data->Get() + sizeof(DIOPCAPETHERNETHEADER) + iplen);
696 if(!UDP_header)
697 {
698 return false;
699 }
700
701 memcpy((XBYTE*)&header,(XBYTE*)(UDP_header),sizeof(DIOPCAPUDPHEADER));
702
703 if(hardwareuselittleendian)
704 {
705 SWAPWORD(header.sourceport); // Source port
706 SWAPWORD(header.targetport); // Target port
707 SWAPWORD(header.datagramlen); // Datagram length
708 SWAPWORD(header.checksum); // Checksum
