178XDWORD SNDOPENALBUFFER::GenerateNote(XDWORD frequency, XDWORD duration, XDWORD samplerate)
179{
180 if(!frequency || !duration)
181 {
182 return false;
183 }
184
185 XDWORD size = (XDWORD)((duration/1000) * samplerate);
186
187 xbuffer.SetLocalHardwareUseLittleEndian(false);
188
189 for(XDWORD c=0; c<size; c++)
190 {
191 XWORD data = (short)(32760 * sin(2 * PI * c * frequency/samplerate));
192 xbuffer.Add((XWORD)data);
193 }
194
195 return size;
196}
