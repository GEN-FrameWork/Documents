290bool DIOSTREAMUARTCONFIG::GetToString(XSTRING* string)
291{
292 if(!string) return false;
293
294 XSTRING xstringtmp;
295
296 string->Empty();
297
298 if(!GetLocalDeviceName()->IsEmpty())
299 {
300 string->Add(GetLocalDeviceName()->Get());
301 string->Add(__L(","));
302 }
303 else
304 {
305 if(!GetPort()) return false;
306 string->Format(__L("%d,"), GetPort());
307 }
308
309 if(!GetBaudRate()) return false;
310 xstringtmp.Format(__L("%d,"), GetBaudRate());
311 string->Add(xstringtmp);
312
313 switch(GetDataBits())
314 {
315 case DIOSTREAMUARTDATABIT_5 : string->Add(__L("5,")); break;
316 case DIOSTREAMUARTDATABIT_6 : string->Add(__L("6,")); break;
317 case DIOSTREAMUARTDATABIT_7 : string->Add(__L("7,")); break;
318 case DIOSTREAMUARTDATABIT_8 : string->Add(__L("8,")); break;
319 default : return false;
320 }
321
322 switch(GetParity())
323 {
324 case DIOSTREAMUARTPARITY_NONE : string->Add(__L("N,")); break;
325 case DIOSTREAMUARTPARITY_ODD : string->Add(__L("O,")); break;
326 case DIOSTREAMUARTPARITY_EVEN : string->Add(__L("E,")); break;
327 case DIOSTREAMUARTPARITY_MARK : string->Add(__L("M,")); break;
328 case DIOSTREAMUARTPARITY_SPACE : string->Add(__L("S,")); break;
329 default : return false;
330 }
331
332 switch(GetFlowControl())
333 {
334 case DIOSTREAMUARTFLOWCONTROL_NONE : string->Add(DIOSTREAMUART_FLOWCONTROL_STR_NONE); break;
335 case DIOSTREAMUARTFLOWCONTROL_HARD : string->Add(DIOSTREAMUART_FLOWCONTROL_STR_SOFT); break;
336 case DIOSTREAMUARTFLOWCONTROL_SOFT : string->Add(DIOSTREAMUART_FLOWCONTROL_STR_HARD); break;
337 default : return false;
338 }

340 return true;

372bool DIOSTREAMUARTCONFIG::SetFromString(XCHAR* string)
373{
374 if(!string) return false;
375
376 XSTRING _string = string;
377 XSTRING devicename;
378 int baudrate;
379 int databits;
380 XCHAR parity;
381 XSTRING stopbits;
382 XSTRING flowcontrol;
383
384 devicename.AdjustSize(_MAXSTR);
385 stopbits.AdjustSize(_MAXSTR);
386 flowcontrol.AdjustSize(_MAXSTR);
387
388 _string.UnFormat(__L("%s,%d,%d,%c,%s,%s"), devicename.Get(), &baudrate, &databits, &parity, stopbits.Get(), flowcontrol.Get());
389
390 devicename.AdjustSize();
391 stopbits.AdjustSize();
392 flowcontrol.AdjustSize();
393
394
395 if(devicename.HaveOnlyNumbers())
396 {
397 SetPort(devicename.ConvertToInt());
398 if(!GetPort()) return false;
399 }
400 else GetLocalDeviceName()->Set(devicename);
401
402 if(!baudrate) return false;
403 SetBaudRate(baudrate);
404
405 switch(databits)
406 {
407 case 5 : SetDataBits(DIOSTREAMUARTDATABIT_5); break;
408 case 6 : SetDataBits(DIOSTREAMUARTDATABIT_6); break;
409 case 7 : SetDataBits(DIOSTREAMUARTDATABIT_7); break;
410 case 8 : SetDataBits(DIOSTREAMUARTDATABIT_8); break;
411 default : return false;
412 }
413
414 switch(parity)
415 {
416 case 'N' : SetParity(DIOSTREAMUARTPARITY_NONE); break;
417 case 'O' : SetParity(DIOSTREAMUARTPARITY_ODD); break;
418 case 'E' : SetParity(DIOSTREAMUARTPARITY_EVEN); break;
419 case 'M' : SetParity(DIOSTREAMUARTPARITY_MARK); break;
420 case 'S' : SetParity(DIOSTREAMUARTPARITY_SPACE); break;
421 default : return false;
422 }
423
424 if(!stopbits.Compare(__L("1")))
425 {
426 SetStopBits(DIOSTREAMUARTSTOPBITS_ONE);
427 }
428 else
429 {
430 if(!stopbits.Compare(__L("1.5")))
431 {
432 SetStopBits(DIOSTREAMUARTSTOPBITS_ONEANDAHALF);
433 }
434 else
435 {
436 if(!stopbits.Compare(__L("2")))
437 {
438 SetStopBits(DIOSTREAMUARTSTOPBITS_TWO);
439
440 } else return false;
441 }
442 }
443
444 if(!flowcontrol.Compare(DIOSTREAMUART_FLOWCONTROL_STR_NONE, true))
445 {
446 SetFlowControl(DIOSTREAMUARTFLOWCONTROL_NONE);
447 }
448 else
449 {
450 if(!stopbits.Compare(DIOSTREAMUART_FLOWCONTROL_STR_SOFT) || !stopbits.Compare(__L("SOFT")))
451 {
452 SetFlowControl(DIOSTREAMUARTFLOWCONTROL_SOFT);
453 }
454 else
455 {
456 if(!stopbits.Compare(DIOSTREAMUART_FLOWCONTROL_STR_HARD) || !stopbits.Compare(__L("HARD")))
457 {
458 SetFlowControl(DIOSTREAMUARTFLOWCONTROL_HARD);
459
460 } else return false;
461 }
462 }
463
464 return true;
465}
