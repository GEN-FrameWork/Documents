407bool DIOI2CAIRQUALITYCCS811::SetEnvironmentalData(float relativehumidity, float temperature)
408{
409 if((temperature < -25) || (temperature > 50)) return false;
410 if((relativehumidity < 0) || (relativehumidity > 100)) return false;
411
412 XDWORD rhum = (XDWORD)(relativehumidity * 1000); //42.348 becomes 42348
413 XDWORD temp = (XDWORD)(temperature * 1000); //23.2 becomes 23200
414
415 XBYTE envdata[4];
416
417 //Split value into 7-bit integer and 9-bit fractional
418
419 //Incorrect way from datasheet.
420 //envData[0] = ((rH % 1000) / 100) > 7 ? (rH / 1000 + 1) << 1 : (rH / 1000) << 1;
421 //envData[1] = 0; //CCS811 only supports increments of 0.5 so bits 7-0 will always be zero
422 //if (((rH % 1000) / 100) > 2 && (((rH % 1000) / 100) < 8))
423 //{
424 // envData[0] |= 1; //Set 9th bit of fractional to indicate 0.5%
425 //}
426
427 //Correct rounding. See issue 8: https://github.com/sparkfun/Qwiic_BME280_CCS811_Combo/issues/8
428 envdata[0] = (rhum + 250) / 500;
429 envdata[1] = 0; //CCS811 only supports increments of 0.5 so bits 7-0 will always be zero
430
431 temp += 25000; //Add the 25C offset
432
433 //Split value into 7-bit integer and 9-bit fractional
434 //envData[2] = ((temp % 1000) / 100) > 7 ? (temp / 1000 + 1) << 1 : (temp / 1000) << 1;
435 //envData[3] = 0;
436 //if (((temp % 1000) / 100) > 2 && (((temp % 1000) / 100) < 8))
437 //{
438 // envData[2] |= 1; //Set 9th bit of fractional to indicate 0.5C
439 //}
440
441 //Correct rounding
442 envdata[2] = (temp + 250) / 500;
443 envdata[3] = 0;
444
445 return WriteRegister(DIOI2CAIRQUALITYCCS811_OFFSET_ENV_DATA, envdata, sizeof(envdata));
446}

491bool DIOI2CAIRQUALITYCCS811::ReadNTC(float& temperature, float& resistance)
492{
493 XBYTE data[4];
494 XWORD vrefcounts = 0;
495 XWORD ntccounts = 0;
496
497 memset(data, 0, sizeof(data));
498
499 bool status = ReadRegister(DIOI2CAIRQUALITYCCS811_OFFSET_NTC, data, sizeof(data));
500
501 vrefcounts = ((XWORD)data[0] << 8) | data[1];
502 ntccounts = ((XWORD)data[2] << 8) | data[3];
503 resistance = ((float)ntccounts * referenceresistance / (float)vrefcounts);
504
505 temperature = (float)log((long)resistance);
506 temperature = (float)(1 / (0.001129148 + (0.000234125 * temperature) + (0.0000000876741 * temperature * temperature * temperature)));
507 temperature = (float)(temperature - 273.15); // Convert Kelvin to Celsius
508
509 return status;
510}
