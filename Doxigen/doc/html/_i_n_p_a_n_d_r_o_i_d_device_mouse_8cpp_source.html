159bool INPANDROIDDEVICEMOUSE::AddPosition(int index, float x, float y, bool ispressed)
160{
161 if(!grpscreen) return false;
162
163 INPCURSOR* cursor = GetCursor(index);
164 if(!cursor) return false;
165
166 cursor->SetIsChanged(false);
167
168 if((cursor->GetX() != x) || (cursor->GetY() != y))
169 {
170 GRPANDROIDSCREEN* _grpscreen = (GRPANDROIDSCREEN*)grpscreen;
171
172 // to place the origin coordinates top left instead of bottom left
173 y = (_grpscreen->GetHeight()) - y;
174
175 cursor->Set(x,y);
176 cursor->SetIsChanged(true);
177 }
178
179 cursor->AddPointToMotion(ispressed);
180
181 return true;
182}
