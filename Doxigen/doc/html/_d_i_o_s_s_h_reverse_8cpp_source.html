221bool DIOSSHREVERSE::DownloadCFG(XCHAR* URL, XSTRING& publicIP, XSTRING& localIP)
222{
223 if(!URL) return false;

225 XBUFFER xbuffer;

230 DIOWEBCLIENT* webclient = GEN_NEW DIOWEBCLIENT();

233 _URL.Set(URL);
234 _URL.Slash_Add();
235 _URL.Add(DIOSSHREVERSE_CFGREMOTENAMEFILE);

237 status = webclient->Get(_URL, xbuffer, NULL, 3);
238 if(status)
239 {

247 if(status)
248 {
249 int c = 1;

252 XSTRING section;

254

256
257 if(!fileini->SelectSection(section)) break;
258
259 status = fileini->ReadValue(__L("isactive"), value);
260 if(status)
261 {
262 if(value.ConvertToBoolean()
263 {
264 DIOURL* urlorigin = GEN_NEW DIOURL();
265 if(urlorigin)
266 {
267 XSTRING _publicIP;
268 XSTRING _localremoteIP;

270 status = fileini->ReadValue(__L("urlorigin"), (*urlorigin));

275 urlorigin->ResolveURL(_publicIP);
276
277 if(!publicIP.Compare(_publicIP))

279 validaccess = true;

281 status = fileini->ReadValue(__L("localremoteip"), _localremoteIP);
282 if(status)
283 {
284 if(_localremoteIP.GetSize())
285 {
286 if(_localremoteIP.Compare(localIP)) validaccess = false;
287 }
288 }
289 }
290
291 if(validaccess)
292 {
293 localIP = __L("localhost");
294
295 status = fileini->ReadValue(__L("urltarget"), value);
296 if(status) GetURLTarget()->Set(value);
297
298 status = fileini->ReadValue(__L("localip"), value);

300

303

306

316 }

318

320 }
321 }
322 }

326 }

328 GEN_DELETE webclient;

330 return result;
331}
