328int DIOLINUXSTREAMTCPIP::Accept(int socket, void* addr, void* addrlen, XDWORD usec)
329{
330 fd_set fds;
331 struct timeval time_out;
332 int status;
333
334 if(socket==-1) return -1;
335
336 FD_ZERO(&fds);
337 FD_SET((unsigned int)socket,&fds);
338
339 time_out.tv_sec = 0;
340 time_out.tv_usec = usec;
341
342 status = (int)select((int)(socket)+1, &fds, NULL, NULL, &time_out);
343 if(status>0)
344 status = accept(socket, (struct sockaddr*)addr, (socklen_t*)addrlen);
345 else status = -1;
346
347 return status;
348}

362int DIOLINUXSTREAMTCPIP::IsReadyConnect(int socket)
363{
364 if(socket==-1) return -1;
365
366 struct timeval tv;
367 int rc;
368 fd_set fdr;
369 fd_set fdw;
370 fd_set fds;
371
372 FD_ZERO(&fdr);
373 FD_ZERO(&fdw);
374 FD_ZERO(&fds);
375
376 FD_SET((unsigned int)socket, &fdr);
377 FD_SET((unsigned int)socket, &fdw);
378 FD_SET((unsigned int)socket, &fds);
379
380 tv.tv_sec = 0;
381 tv.tv_usec = 100;
382
383 rc = select((int)(socket)+1, &fdr, &fdw, &fds, &tv);
384 if(rc == -1) return -1;
385
386 int status1 = FD_ISSET(socket, &fdr) ? 1 : 0;
387 int status2 = FD_ISSET(socket, &fdw) ? 1 : 0;
388 int status3 = FD_ISSET(socket, &fds) ? 1 : 0;
389
390 if(config->IsServer())
391 {
392 if(status1 || status2 ) return 1;
393 }
394 else
395 {
396 if(status1 || status2)
397 {
398 int optval;
399 socklen_t optlen = sizeof(optval);
400
401 if(getsockopt(socket,SOL_SOCKET, SO_ERROR, &optval, &optlen) < 0 ) return -1;
402 if(optval) return -1;
403
404 return 1;
405 }
406 }
407
408 if(status3) return -1;
409
410 return 0;
411}

423bool DIOLINUXSTREAMTCPIP::GetHandleServer()
424{
425 int handleserver = (int)config->GetHandleMultiServer();
426 struct sockaddr_in loc_addr = { 0 };
427 struct sockaddr_in rem_addr = { 0 };
428 XSTRING IPstring;
429
430 if(!config->GetXMutexMultiServer())
431 {
432 return false;
433 }
434
435 config->GetXMutexMultiServer()->Lock();
436
437 if(handleserver < 0)
438 {
439 handleserver = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
440 if(handleserver < 0)
441 {
442 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
443 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
444 config->GetXMutexMultiServer()->UnLock();
445
446 return false;
447 }
448
449 int opt = 1;
450 if(setsockopt(handleserver, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt)) < 0)
451 {
452 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
453 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
454 config->GetXMutexMultiServer()->UnLock();
455
456 return false;
457 }
458
459 #ifdef SO_REUSEPORT
460 opt = 1;
461 if(setsockopt(handleserver, SOL_SOCKET, SO_REUSEPORT, &opt, sizeof(opt)) < 0)
462 {
463 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
464 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
465 config->GetXMutexMultiServer()->UnLock();
466
467 return false;
468 }
469 #endif
470
471 config->GetLocalIP()->GetXString(IPstring);
472
473 loc_addr.sin_family = AF_INET;
474 if(!config->GetLocalIP()->IsEmpty())
475 {
476 XBUFFER charstr;
477
478 IPstring.ConvertToASCII(charstr);
479 loc_addr.sin_addr.s_addr = inet_addr(charstr.GetPtrChar());
480
481 } else loc_addr.sin_addr.s_addr = htonl(INADDR_ANY);
482
483 loc_addr.sin_port = htons(config->GetRemotePort());
484
485 if(bind(handleserver, (struct sockaddr *)&loc_addr, sizeof(loc_addr)) < 0)
486 {
487 if(errno == EADDRINUSE)
488 {
489 SetLastDIOError(DIOSTREAMERROR_ADDRINUSE);
490
491 } else SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
492
493 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
494
495 config->GetXMutexMultiServer()->UnLock();
496
497 return false;
498 }
499
500 if(listen(handleserver, DIOSTREAMTCPIP_MAXLISTEN) < 0)
501 {
502 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
503 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
504 config->GetXMutexMultiServer()->UnLock();
505
506 return false;
507 }
508
509 fcntl(handleserver, F_SETFL, fcntl(handleserver, F_GETFL,0) | O_NONBLOCK);
510
511 config->SetHandleMultiServer(handleserver);
512 }
513
514 config->GetXMutexMultiServer()->UnLock();
515
516 if(enumservers)
517 {
518 if(enumservers->GetXTimer())
519 {
520 if(enumservers->GetXTimer()->GetMeasureSeconds() >= DIOSTREAMTCPIP_TIMEOUTENUMSERVERSIGNAL)
521 {
522 enumservers->SendSignal();
523 enumservers->GetXTimer()->Reset();
524 }
525 }
526 }
527
528 socklen_t sizeaddr = sizeof(struct sockaddr_in);
529 handlesocket = Accept(handleserver, (struct sockaddr*)&rem_addr, &sizeaddr, 100/*50*1000*/);
530 if(handlesocket >= 0)
531 {
532 int opt = 1;
533 if(setsockopt(handlesocket, SOL_SOCKET, SO_KEEPALIVE , &opt, sizeof(opt)) < 0)
534 {
535 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
536 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
537
538 return false;
539 }
540
541 int flags = fcntl(handlesocket, F_GETFL, 0);
542 fcntl(handlesocket, F_SETFL, flags | O_NONBLOCK);
543
544 clientIP.Set(inet_ntoa(rem_addr.sin_addr));
545 }
546 else
547 {
548 if(errno == EADDRINUSE)
549 {
550 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
551 SetLastDIOError(DIOSTREAMERROR_ADDRINUSE);
552
553 return false;
554 }
555 }
556
557 return true;
558}

570bool DIOLINUXSTREAMTCPIP::GetHandleClient()
571{
572 if(config->GetRemoteURL()->IsEmpty())
573 {
574 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
575 return false;
576 }
577
578 handlesocket = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
579 if(handlesocket == -1)
580 {
581 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
582 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
583 return false;
584 }
585
586 if(!config->GetLocalIP()->IsEmpty())
587 {
588 struct sockaddr_in loc_addr;
589 XSTRING IPstring;
590
591 config->GetLocalIP()->GetXString(IPstring);
592
593 memset(&loc_addr, 0, sizeof(struct sockaddr_in));
594
595 loc_addr.sin_family = AF_INET;
596
597 XBUFFER charstr;
598
599 IPstring.ConvertToASCII(charstr);
600 loc_addr.sin_addr.s_addr = inet_addr(charstr.GetPtrChar());
601 loc_addr.sin_port = 0; //htons(diostream->config->GetRemotePort());
602
603 if(bind(handlesocket, (struct sockaddr *)&loc_addr, sizeof(loc_addr)) < 0 )
604 {
605 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
606 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
607 return false;
608 }
609 }
610
611 struct sockaddr_in rem_addr;
612 DIOIP remoteIP;
613 XSTRING IPstring;
614
615 memset(&rem_addr, 0, sizeof(struct sockaddr_in));
616
617 if(!config->GetRemoteURL()->ResolveURL(remoteIP))
618 {
619 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
620 SetLastDIOError(DIOSTREAMERROR_URLNOTVALID);
621 return false;
622 }
623
624 remoteIP.GetXString(IPstring);
625
626 rem_addr.sin_family = AF_INET;
627
628 XBUFFER charstr;
629
630 IPstring.ConvertToASCII(charstr);
631 rem_addr.sin_addr.s_addr = inet_addr(charstr.GetPtrChar());
632 clientIP.Set(charstr.GetPtrChar());
633 rem_addr.sin_port = htons(config->GetRemotePort());
634
635 int opt = 1;
636 if(setsockopt(handlesocket, SOL_SOCKET, SO_KEEPALIVE , &opt, sizeof(opt)) < 0)
637 {
638 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
639 SetLastDIOError(DIOSTREAMERROR_UNKNOWN);
640 return false;
641 }
642
643 int flags = fcntl(handlesocket, F_GETFL, 0);
644 fcntl(handlesocket, F_SETFL, flags | O_NONBLOCK);
645
646 if(connect(handlesocket, (struct sockaddr *)&rem_addr, sizeof(rem_addr)) < 0)
647 {
648 if(errno == EADDRINUSE)
649 {
650 SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
651 SetLastDIOError(DIOSTREAMERROR_ADDRINUSE);
652 return false;
653 }
654 }
655
656 DIOSTREAM_XEVENT xevent(this, DIOSTREAM_XEVENT_TYPE_GETTINGCONNECTION);
657 xevent.SetDIOStream(this);
658 PostEvent(&xevent);
659
660 return true;
661}

673void DIOLINUXSTREAMTCPIP::ThreadConnection(void* data)
674{
675 DIOLINUXSTREAMTCPIP* diostream = (DIOLINUXSTREAMTCPIP*)data;
676 if(!diostream) return;
677
678 if(diostream->GetEvent()==DIOLINUXTCPIPFSMEVENT_NONE) // No hay nuevos Eventos
679 {
680 switch(diostream->GetCurrentState())
681 {
682 case DIOLINUXTCPIPFSMSTATE_NONE : break;
683
684 case DIOLINUXTCPIPFSMSTATE_GETTINGCONNECTION : if(diostream->handlesocket == -1)
685 {
686 if(diostream->config->IsServer())
687 {
688 diostream->GetHandleServer();
689 }
690 else
691 {
692 diostream->GetHandleClient();
693 }
694 }
695 else
696 {
697 switch(diostream->IsReadyConnect(diostream->handlesocket))
698 {
699 case -1: diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
700 break;
701
702 case 1: diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_CONNECTED);
703 diostream->status = DIOSTREAMSTATUS_CONNECTED;
704 break;
705
706 default: diostream->status = DIOSTREAMSTATUS_GETTINGCONNECTION;
707 break;
708 }
709 }
710 break;
711
712 case DIOLINUXTCPIPFSMSTATE_CONNECTED : break;
713
714 case DIOLINUXTCPIPFSMSTATE_WAITINGTOREAD : { struct timeval waitd;
715 fd_set read_flags;
716 fd_set write_flags;
717
718 if(diostream->handlesocket==-1)
719 {
720 diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
721 break;
722 }
723
724 waitd.tv_sec = 0;
725 waitd.tv_usec = 100;
726
727 FD_ZERO(&read_flags);
728 FD_ZERO(&write_flags);
729
730 if(!diostream->IsBlockRead()) FD_SET(diostream->handlesocket, &read_flags);
731 if(!diostream->IsBlockWrite())
732 {
733 if(diostream->outbuffer->GetSize()) FD_SET(diostream->handlesocket, &write_flags);
734 }
735
736 int error = select((int)(diostream->handlesocket)+1, &read_flags, &write_flags, (fd_set*)NULL, &waitd);
737 if(error == -1) //Socket ready for reading
738 {
739 diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
740 break;
741 }
742
743 if(FD_ISSET(diostream->handlesocket, &read_flags)) //Socket ready for reading
744 {
745 XBYTE buffer[DIOSTREAM_MAXBUFFER];
746
747 FD_CLR(diostream->handlesocket, &read_flags);
748
749 XDWORD size = recv(diostream->handlesocket,(char*)buffer, DIOSTREAM_MAXBUFFER, 0);
750 if(size==-1)
751 {
752 diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
753 break;
754 }
755
756 if(size)
757 {
758 diostream->inbuffer->Add(buffer,size);
759 }
760 else
761 {
762 diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
763 break;
764 }
765 }
766
767 if(FD_ISSET(diostream->handlesocket, &write_flags)) //Socket ready for writing
768 {
769 FD_CLR(diostream->handlesocket, &write_flags);
770
771 int esize = diostream->outbuffer->GetSize();
772 if(esize>DIOSTREAM_MAXBUFFER) esize = DIOSTREAM_MAXBUFFER;
773
774 if(esize)
775 {
776 diostream->outbuffer->SetBlocked(true);
777 XDWORD size = send(diostream->handlesocket,(char*)diostream->outbuffer->Get(), esize,0);
778 diostream->outbuffer->SetBlocked(false);
779
780 if(size == -1)
781 {
782 diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_DISCONNECTING);
783 break;
784 }
785
786 if(size) diostream->outbuffer->Extract(NULL, 0, size);
787 }
788 }
789 }
790 break;
791
792 case DIOLINUXTCPIPFSMSTATE_SENDINGDATA : break;
793
794 case DIOLINUXTCPIPFSMSTATE_DISCONNECTING : break;
795
796 }
797 }
798 else
799 {
800 if(diostream->GetEvent()<DIOLINUXTCPIP_LASTEVENT)
801 {
802 diostream->CheckTransition();
803
804 switch(diostream->GetCurrentState())
805 {
806 case DIOLINUXTCPIPFSMSTATE_NONE : break;
807
808 case DIOLINUXTCPIPFSMSTATE_GETTINGCONNECTION : if(diostream->enumservers)
809 {
810 if(diostream->enumservers->GetXTimer()) diostream->enumservers->GetXTimer()->AddSeconds(DIOSTREAMTCPIP_TIMEOUTENUMSERVERSIGNAL);
811 }
812
813 if(diostream->config->IsServer())
814 {
815 diostream->config->SetCounterMultiServer(diostream->config->GetCounterMultiServer()+1);
816 }
817 break;
818
819 case DIOLINUXTCPIPFSMSTATE_CONNECTED : { DIOSTREAM_XEVENT xevent(diostream, DIOSTREAM_XEVENT_TYPE_CONNECTED);
820 xevent.SetDIOStream(diostream);
821 diostream->PostEvent(&xevent);
822
823 diostream->SetEvent(DIOLINUXTCPIPFSMEVENT_WAITINGTOREAD);
824 }
825 break;
826
827 case DIOLINUXTCPIPFSMSTATE_WAITINGTOREAD : break;
828
829 case DIOLINUXTCPIPFSMSTATE_SENDINGDATA : break;
830
831 case DIOLINUXTCPIPFSMSTATE_DISCONNECTING : { DIOSTREAM_XEVENT xevent(diostream, DIOSTREAM_XEVENT_TYPE_DISCONNECTED);
832 xevent.SetDIOStream(diostream);
833
834 diostream->PostEvent(&xevent);
835
836 diostream->threadconnection->Run(false);
837 diostream->status = DIOSTREAMSTATUS_DISCONNECTED;
838 }
839 break;
840
841 }
842 }
843 }
