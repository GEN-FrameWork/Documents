406bool DIOCLIPROTOCOL::SendCommandArg(XCHAR* command, XSTRING* target, XSTRING* answer, int timeoutanswer, va_list* arg)
407{
408 DIOCLIPROTOCOLCOMMAND* protocolcommand = GetCommand(command);
409 if(!protocolcommand) return false;
410
411 if(!arg) return false;
412
413 XSTRING tosend;
414
415 if(answer)
416 {
417 answer->Empty();
418 }
419 else
420 {
421 tosend.Format(__L("%c"), DIOCLIPROTOCOL_MARK_DONOTANSWER);
422 }
423
424 if(!ID.IsEmpty())
425 {
426 tosend.AddFormat(__L("%s%s"), ID.Get(), DIOCLIPROTOCOL_MARK_ORIGIN);
427 }
428
429 if(target)
430 {
431 if(!target->IsEmpty())
432 {
433 tosend.AddFormat(__L("%s%s"), target->Get(), DIOCLIPROTOCOL_MARK_TARGET);
434 }
435 }
436 else
437 {
438 tosend.AddFormat(__L("%s%s"), DIOCLIPROTOCOL_MARK_BROADCAST, DIOCLIPROTOCOL_MARK_TARGET);
439 }
440
441 tosend.AddFormat(__L("%s"), protocolcommand->GetCommand());
442
443 int nparams = protocolcommand->GetNParams();
444 if(nparams)
445 {
446 tosend.AddFormat(__L(" "));
447
448 for(int c=0;c<nparams;c++)
449 {
450 XCHAR* param = (XCHAR*)va_arg((*arg), XCHAR*);
451 if(param)
452 {
453 tosend += param;
454 if(c != nparams-1) tosend += __L(",");
455 }
456 }
457 }
458
459 if(activeCRC)
460 {
461 HASHCRC32 CRC32;
462
463 XBUFFER charstr;
464
465 tosend.ConvertToASCII(charstr);
466 CRC32.Do((XBYTE*)charstr.Get(), tosend.GetSize());
467
468 XDWORD CRC32result = CRC32.GetResultCRC32();
469 tosend.AddFormat(__L("%s%08X"), DIOCLIPROTOCOL_MARK_CRC32, CRC32result);
470 }
471
472 tosend += __L("\n\r");
473
474
475 XBUFFER charstr;
476
477 tosend.ConvertToASCII(charstr);
478 diostream->Write((XBYTE*)charstr.Get(), tosend.GetSize());
479
480 tosend.DeleteCharacter(__C('\n'));
481 tosend.DeleteCharacter(__C('\r'));
482
483 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[Protocol CLI] %s Send Ask: %s -> %s %s"), tosend.Get(), ID.Get(), (target?target->Get():DIOCLIPROTOCOL_MARK_BROADCAST), command);
484
485 bool status = true;
486
487 if(answer)
488 {
489 if(xtimerout) xtimerout->Reset();
490
491 while(1)
492 {
493 if(!xtimerout)
494 {
495 status = false;
496 break;
497 }
498
499 if(exitproccess)
500 {
501 status = false;
502 break;
503 }
504
505 if(xtimerout->GetMeasureSeconds() >= (XDWORD)timeoutanswer)
506 {
507 status = false;
508 break;
509 }
510
511 DIOCLIPROTOCOLANSWER* protocolanswer = GetFirstAnswer(command);
512 if(protocolanswer)
513 {
514 answer->Add(protocolanswer->GetAnswer()->Get());
515 DeleteFirstAnswer(command);
516 break;
517 }
518 }
519 }
520
521 return status;
522}

736 }

739 int indexcommand = laststringreceived.Find(mark_answer, true);
740 if(indexcommand != XSTRING_NOTFOUND)
741 {
742 laststringreceived.Copy(0, indexcommand, command);
743 laststringreceived.DeleteCharacters(0, indexcommand + mark_answer.GetSize());

748 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("[Protocol CLI] Received answer: %s -> %s %s : (%s)"), originID.Get(), targetID.Get(), command.Get(), answer.Get());

752 if(AddAnswer(originID, command, answer))
753 {
754
755 }
756 }
757 }
758 }
759 }
760
761 laststringreceived.Empty();
762 }
763 }
764 else
765 {
766 if((character >= 0x20) && (character <= 0x7F)) laststringreceived.Add(character);
767 }
768 }
769 }
