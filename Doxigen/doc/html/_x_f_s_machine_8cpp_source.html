204bool XFSMACHINESTATE::DeleteTransition(int outputID)

207

412bool XFSMACHINE::AddState(int state, int event, int tostate,...)
413{
414 XFSMACHINESTATE* _state;
415 int narg = 0;
416 va_list arg;
417
418 va_start(arg, tostate);
419 while(va_arg(arg, int) != XFSMACHINESTATE_EVENTDEFEND)
420 {
421 narg++;
422 }
423 va_end(arg);
424
425 if((narg)&&(narg%2)) return false;
426
427 narg+=2;
428
429 _state = GEN_NEW XFSMACHINESTATE(state,(narg/2));

431
432 _state->AddTransition(event,tostate);
433
434 va_start(arg,tostate);
435
436 int _event;
437 int _tostate;
438
439 do { _event = va_arg(arg, int);

443 _state->AddTransition(_event,_tostate);

448 va_end(arg);
449
450 if(!AddState(_state))
