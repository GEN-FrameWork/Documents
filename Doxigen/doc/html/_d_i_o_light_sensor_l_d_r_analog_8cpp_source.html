195bool DIOLIGHTSENSORLDRANALOG::ReadDirect(XDWORD& level)
196{
197 if(!diogpio) return false;
198 if(!xtimer) return false;
199 if(!IsInitialized()) return false;
200
201 level = 0;
202
203 xtimer->Reset();
204
205 // Discharge capacitor
206 diogpio->SetMode(pindata, false);
207 diogpio->Set(pindata , false);
208 GEN_XSLEEP.MilliSeconds(100);
209
210 diogpio->SetMode(pindata, true);
211 while(!diogpio->Get(pindata))
212 {
213 level++;
214 if(!(level%100))
