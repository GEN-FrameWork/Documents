172bool GRPBITMAPFILEBMPHEADER::Load(XFILE* file)
173{
174 if(!file->Read((XBYTE*)&headerinfo.type , sizeof(XWORD))) return false;
175 if(!file->Read((XBYTE*)&headerinfo.size , sizeof(XDWORD))) return false;
176 if(!file->Read((XBYTE*)&headerinfo.reserved1 , sizeof(XWORD))) return false;
177 if(!file->Read((XBYTE*)&headerinfo.reserved2 , sizeof(XWORD))) return false;
178 if(!file->Read((XBYTE*)&headerinfo.offsetdata , sizeof(XDWORD))) return false;
179
180 if(!file->Read((XBYTE*)&headerinfo.headersize , sizeof(XDWORD))) return false;
181 if(!file->Read((XBYTE*)&headerinfo.width , sizeof(XDWORD))) return false;
182 if(!file->Read((XBYTE*)&headerinfo.height , sizeof(XDWORD))) return false;
183 if(!file->Read((XBYTE*)&headerinfo.planes , sizeof(XWORD))) return false;
184 if(!file->Read((XBYTE*)&headerinfo.bitsperpixel , sizeof(XWORD))) return false;
185 if(!file->Read((XBYTE*)&headerinfo.compression , sizeof(XDWORD))) return false;
186 if(!file->Read((XBYTE*)&headerinfo.sizebmp , sizeof(XDWORD))) return false;
187 if(!file->Read((XBYTE*)&headerinfo.xpixelspermeter , sizeof(XDWORD))) return false;
188 if(!file->Read((XBYTE*)&headerinfo.ypixelspermeter , sizeof(XDWORD))) return false;
189 if(!file->Read((XBYTE*)&headerinfo.colorused , sizeof(XDWORD))) return false;
190 if(!file->Read((XBYTE*)&headerinfo.colorimportant , sizeof(XDWORD))) return false;
191
192
193 // Check: "BM" signature in the file for WINDOWS NO OS/2 valid format.
194 if(headerinfo.type!=0x4D42) return false;
195
196 switch(headerinfo.bitsperpixel)
197 {
198 case 8 : mode = GRPPROPERTYMODE_08_INDEX;
199 break;
200
201 case 16 : { XDWORD maskcolor[4];
202 if(!file->Read((XBYTE*)maskcolor, 16)) return false;
203
204 if((maskcolor[0] == 0x00007c00) && (maskcolor[1] == 0x000003E0) && (maskcolor[2] == 0x0000001F) && (maskcolor[3] == 0x00008000)) mode = GRPPROPERTYMODE_16_RGBA_5551;
205 if((maskcolor[0] == 0x0000F800) && (maskcolor[1] == 0x000007E0) && (maskcolor[2] == 0x0000001F) && (maskcolor[3] == 0x00000000)) mode = GRPPROPERTYMODE_16_RGB_565;
206 }
207 break;
208
209
210 case 24 : mode = GRPPROPERTYMODE_24_RGB_888;
211 break;
212
213 case 32 : mode = GRPPROPERTYMODE_32_RGBA_8888;
214 break;
215 }
216
217
218 DeleteAllPalette();
219
220 int ncpalette = headerinfo.colorused;
221
222 if(ncpalette)
223 {
224 for(int c=0;c<ncpalette;c++)
225 {
226 XBYTE red;
227 XBYTE green;
228 XBYTE blue;
229 XBYTE alpha;
230
231 if(!file->Read(&blue ,1)) return false;
232 if(!file->Read(&green ,1)) return false;
233 if(!file->Read(&red ,1)) return false;
234 if(!file->Read(&alpha ,1)) return false;
235
236 GRPBITMAPFILEBMPCOLORPALETE* colorpalette = GEN_NEW GRPBITMAPFILEBMPCOLORPALETE(red, green, blue, alpha);
237 if(colorpalette) palette.Add((XBYTE)c, colorpalette);
238 }
239 }
240
241 return true;
242}

525GRPBITMAP* GRPBITMAPFILEBMP::CreateBitmapFromFile(XPATH& xpath, GRPPROPERTYMODE forcemode)
526{
527 if(!Open(xpath)) return NULL;
528
529 GRPBITMAPFILEBMPHEADER header;
530 GRPBITMAP* bitmap = NULL;
531//XBYTE* buffer = NULL;
532 bool status = true;
533
534 if(!ReadHeaderFile(&header))
535 {
536 Close();
537 return NULL;
538 }
539
540 int width = header.GetWidth();
541 int height = header.GetHeight();
542 int bytesppixel = (header.GetBitsPerPixel()/8);
543
544 bitmap = GRPFACTORY::GetInstance().CreateBitmap(width, height, header.GetMode());
545 if(bitmap)
546 {
547 int sizebmp = (((header.GetWidth()*bytesppixel)+3)&(~3))*header.GetHeight();
548 XBYTE* databuffer = NULL;
549 int index = 0;
550
551 databuffer = GEN_NEW XBYTE[sizebmp];
552 if(databuffer)
553 {
554 file->SetPosition(header.GetOffsetData());
555
556 if(!file->Read(databuffer, sizebmp)) return NULL;
557
558 for(int y=height-1; y>-1; y--)
559 {
560 for(int x=0; x<width; x++)
561 {
562 GRP2DCOLOR_RGBA8 color;
563
564 switch(header.GetMode())
565 {
566 case GRPPROPERTYMODE_08_INDEX : header.GetColorPalette(databuffer[index], color);
567 color.a = 255;
568 bitmap->PutPixel(x, y, (GRP2DCOLOR*)&color);
569 break;
570
571 case GRPPROPERTYMODE_16_RGBA_5551 : { XWORD rgb = databuffer[index+1];
572 rgb <<= 8;
573 rgb |= databuffer[index+0];
574
575 color.r = (rgb >> 7) & 0xF8;
576 color.g = (rgb >> 2) & 0xF8;
577 color.b = (rgb << 3) & 0xF8;
578 color.a = (rgb & 0x8000)?255:0;
579
580 bitmap->PutBlendPixel(x, y, (GRP2DCOLOR*)&color, (int)color.a);
581 }
582 break;
583
584 case GRPPROPERTYMODE_16_RGB_565 : { XWORD rgb = databuffer[index+1];
585 rgb <<= 8;
586 rgb |= databuffer[index+0];
587
588 color.r = (rgb >> 8) & 0xF8;
589 color.g = (rgb >> 3) & 0xFC;
590 color.b = (rgb << 3) & 0xF8;
591 color.a = 255;
592
593 bitmap->PutPixel(x, y, (GRP2DCOLOR*)&color);
594 }
595 break;
596
597 case GRPPROPERTYMODE_24_RGB_888 : color.r = databuffer[index+2];
598 color.g = databuffer[index+1];
599 color.b = databuffer[index+0];
600 color.a = 255;
601
602 bitmap->PutPixel(x, y, (GRP2DCOLOR*)&color);
603 break;
604
605 case GRPPROPERTYMODE_32_RGBA_8888 : color.r = databuffer[index+2];
606 color.g = databuffer[index+1];
607 color.b = databuffer[index+0];
608 color.a = databuffer[index+3];
609
610 bitmap->PutBlendPixel(x, y, (GRP2DCOLOR*)&color, (int)color.a);
611 break;
612 default: break;
613 }
614
615 index += bytesppixel;
616 }
617 }
618
619 } else status = false;
620
621 GEN_DELETE_ARRAY databuffer;
622
623 } else status = false;
624
625
626 if(!status)
627 {
628 if(bitmap)
629 {
630 GRPFACTORY::GetInstance().DeleteBitmap(bitmap);
631 bitmap = NULL;
632 }
633 }
634
635 Close();
636
637 return ForceModeOfBitmap(forcemode, bitmap);
638}
