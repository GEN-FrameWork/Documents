104bool DIOWEBHEADER::Read(DIOSTREAMTCPIP* diostream, int timeout)
105{

107
108 XSTRING line;
109 bool status;
110
111 DeleteAllLines();

113 XTIMER* GEN_XFACTORY_CREATE(xtimerout, CreateTimer())
114 if(!xtimerout) return false;
115
116 do{ status = diostream->ReadStr(line);
117 if(status)
118 {

357bool DIOWEBHEADER::GetFormat(bool isserver, int type, void* data)
358{
359 if(lines.IsEmpty()) return false;
360
361 if(!data) return false;
362
363 XSTRING line;
364 int version = 0;
365 int subversion = 0;
366
367 line =((XSTRING*)lines.Get(0))->Get();
368
369 if(isserver)
370 {
371 int code;
372 XSTRING codestring(DIOWEBHEADER_MAXLINE);
373
374 line.UnFormat(__L("HTTP/%d.%d %d %s"),&version,&subversion,&code,codestring.Get());
375 codestring.AdjustSize();
376
377 switch(type)
378 {
379 case 0: (*(int*)data) = version; break;
380 case 1: (*(int*)data) = subversion; break;
381 case 2: (*(int*)data) = code; break;
382 case 3: (*(XSTRING*)data) = codestring; break;
383 }
384 }
385 else
386 {
387 XSTRING method(DIOWEBHEADER_MAXLINE);
388 XSTRING resource(DIOWEBHEADER_MAXLINE);
389
390 if(line.GetSize())
391 {
392 int foundindex = line.Find(__L("HTTP/"), true);
393
394 if(foundindex != XSTRING_NOTFOUND) line.UnFormat(__L("%s %s HTTP/%d.%d"), method.Get(), resource.Get(), &version, &subversion);
395 }
396
397 method.AdjustSize();
398 resource.AdjustSize();
399
400 switch(type)
401 {
402 case 0: if(data) (*(XSTRING*)data) = method; break;
403 case 1: if(data) (*(XSTRING*)data) = resource; break;
404 case 2: if(data) (*(int*)data) = version; break;
405 case 3: if(data) (*(int*)data) = subversion; break;
406 }
407 }
408
409 return true;
410}
