182bool UI_MANAGER::Load(XPATH& pathfile, GRPSCREEN* screen, int viewportindex)
183{
184 bool status = false;
185
186 if(pathfile.Find(__L(".zip"), true) != XSTRING_NOTFOUND)
187 {
188 iszippedfile = true;
189 }
190
191 if(!iszippedfile)
192 {
193 status = LoadLayout(pathfile, screen, viewportindex);
194 }
195 else
196 {
197 unzipfile = GEN_NEW XFILEUNZIP();
198 if(!unzipfile)
199 {
200 return false;
201 }
202
203 if(unzipfile->Open(pathfile))
204 {
205 XSTRING origin_drive;
206 XPATH origin_path;
207 XSTRING origin_namefile;
208 XSTRING origin_ext;
209
210 pathfile.GetDrive(origin_drive);
211 pathfile.GetPath(origin_path);
212 pathfile.GetNamefile(origin_namefile);
213 pathfile.GetExt(origin_ext);
214
215 unzippathfile = origin_drive;
216 unzippathfile += origin_path;
217
218 XSTRING namefile;
219
220 namefile = origin_namefile;
221 namefile += __L(".xml");
222
223 status = unzipfile->DecompressFile(namefile, unzippathfile, namefile.Get());
224 if(status)
225 {
226 XPATH unzippathfile_tmp;
227
228 unzippathfile_tmp = unzippathfile;
229 unzippathfile_tmp += namefile;
230
231 status = LoadLayout(unzippathfile_tmp, screen, viewportindex);
232
233 DeleteTemporalUnZipFile(unzippathfile_tmp);
234 }
235 }

238 CloseUnZipFile();
239 }
240 }
