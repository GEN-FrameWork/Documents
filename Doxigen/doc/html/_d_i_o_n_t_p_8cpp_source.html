128bool DIONTP::GetTimeResponse(XCHAR* urlntpserver, int timeout, bool hardwareuselittleendian, DIONTPRESPONSE& response)
129{
130 if(!xtimer) return false;
131 if((!diostreamudpcfg) || (!diostreamudp)) return false;
132
133 bool status = false;
134
135 memset(&response, 0, sizeof(DIONTPRESPONSE));
136
137 diostreamudpcfg->SetMode(DIOSTREAMMODE_CLIENT);
138 diostreamudpcfg->SetIsUsedDatagrams(false);
139 diostreamudpcfg->GetRemoteURL()->Set(urlntpserver);
140 diostreamudpcfg->SetRemotePort(DIONTP_DEFAULTPORT);
141
142 if(!diostreamudp->Open()) return false;
143
144 DIONTPBASICINFO basicinfo;
145
146 memset((XBYTE*)&basicinfo, 0, sizeof(DIONTPBASICINFO));
147
148 basicinfo.livnmode = 27; //Encoded representation which represents NTP Client Request & NTP version 3.0
149
150 if(diostreamudp->Write((XBYTE*)&basicinfo,sizeof(DIONTPBASICINFO)))
151 {
152 xtimer->Reset();
153
154 DIONTPFULLPACKET fullpacket;
155 XDWORD br = 0;
156 while(!br)
157 {
158 memset((XBYTE*)&fullpacket,0 ,sizeof(DIONTPFULLPACKET));
159
160 XDWORD size = diostreamudp->GetInXBuffer()->GetSize();
161 if(size >= sizeof(DIONTPBASICINFO))
162 {
163 br = diostreamudp->Read((XBYTE*)&fullpacket, size);
164 if(br)
165 {
166 response.stratum = fullpacket.basic.stratum;
167 response.leapindicator = (fullpacket.basic.livnmode & 0xC0) >> 6;
168 response.originatetime = ConvertTimePacket(fullpacket.basic.originatetimestamp , hardwareuselittleendian);
169 response.receivetime = ConvertTimePacket(fullpacket.basic.receivetimestamp , hardwareuselittleendian);
170 response.transmittime = ConvertTimePacket(fullpacket.basic.transmittimestamp , hardwareuselittleendian);
171
172 long long roundtripdelay = (long long)(response.destinationtime - response.originatetime) - (long long)(response.receivetime - response.transmittime);
173 long long localclockoffset = (long long)(response.receivetime - response.originatetime) + (long long)(response.transmittime - response.destinationtime);
174 response.roundtripdelay = (double)(roundtripdelay);
175 response.localclockoffset = (double)(localclockoffset/2);
176
177 status = true;
178 break;
179 }
180 }
181 else
182 {
183 //XTRACE_PRINTCOLOR(XTRACE_COLOR_PURPLE, __L("ntp [%d] %d"), sizeof(DIONTPBASICINFO), size);
184 }
185
186 if((int)xtimer->GetMeasureSeconds()>= timeout)
187 {
188 status = false;
189 break;
190 }
191 }
192 }
193
194 diostreamudp->Close();
195
196 return status;
197}

341XQWORD DIONTP::ConvertTimePacket(DIOTIMEPACKET& timepacket, bool hardwareuselittleendian)
342{
343 XDWORD dwhigh = (hardwareuselittleendian)?SwapDWORD(timepacket.integer) : timepacket.integer;
344 XDWORD dwlow = (hardwareuselittleendian)?SwapDWORD(timepacket.fractional) : timepacket.fractional;
345 XQWORD timedata;
346
347 timedata = ((XQWORD) dwhigh) << 32;
348 timedata |= dwlow;
349
350 return timedata;
351}
