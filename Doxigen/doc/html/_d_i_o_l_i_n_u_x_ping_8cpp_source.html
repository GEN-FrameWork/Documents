148 urltarget->ResolveURL(targetIP);

170 XDWORD crc32result;
171 XSTRING address;
172 XBUFFER xbuffer;

174 XRAND* GEN_XFACTORY_CREATE(xrand, CreateRand())
175 if(!xrand) break;
176
177 memset((XBYTE*)&echorequest, 0, sizeof(DIOPING_ECHOREQUEST));

181 echorequest.icmphdr.code = 0;
182 echorequest.icmphdr.ID = ID;
183 echorequest.icmphdr.seq = nseq++;
184
185 //Fill in some data to send
186 for(int nret=0; nret<DIOPING_REQ_TICKETSIZE; nret++)

191 GEN_XFACTORY.DeleteRand(xrand);

197 echorequest.applicationID = applicationID;
198 echorequest.time = (XDWORD)xtimer->GetMicroSecondsTickCounter(); // Save tick count when sent
199 echorequest.icmphdr.checksum = CalculeCheckSum((XWORD *)&echorequest, sizeof(DIOPING_ECHOREQUEST)); // Put data in packet and compute checksum
200
201 XDWORD size = sendto(handle,(char*)&echorequest, sizeof(DIOPING_ECHOREQUEST), 0, (sockaddr*)&targetaddr, sizeof(struct sockaddr_in));
202 if(size != sizeof(DIOPING_ECHOREQUEST))
