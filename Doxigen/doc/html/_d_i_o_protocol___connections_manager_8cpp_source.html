1625void DIOPROTOCOL_CONNECTIONSMANAGER::ThreadProtocolConnections(void* param)
1626{
1627 DIOPROTOCOL_CONNECTIONSMANAGER* protocolconnections = (DIOPROTOCOL_CONNECTIONSMANAGER*)param;
1628 if(!protocolconnections) return;
1629
1630 if(protocolconnections->isserver)
1631 {
1632 protocolconnections->ManageProtocolConnectionsServer();
1633 }
1634 else
1635 {
1636 protocolconnections->ManageProtocolConnectionsClient();
1637 }
1638
1639 int c=0;
1640 while(c < protocolconnections->ProtocolConnections_GetNAvailable())
1641 {
1642 DIOPROTOCOL_CONNECTION* protocolconnection = (DIOPROTOCOL_CONNECTION*)protocolconnections->protocolconnections.Get(c);
1643 if(protocolconnection)
1644 {
1645 DIOPROTOCOL* protocol = protocolconnection->GetDIOProtocol();
1646 if(protocol)
1647 {
1648 if(protocol->IsConnected())
1649 {
1650 if(!protocol->IsInitialized())
1651 {
1652 if(protocolconnections->isserver)
1653 {
1654 protocolconnection->SetInUse(true, DIOPROTOCOL_CONNECTIONSINUSEID_INIPROTOCOL);
1655
1656 protocol->GetDIOStream()->GetXTimerNotActivity()->Reset();
1657 protocolconnections->SendEvent(DIOPROTOCOL_CONNECTIONSMANAGER_XEVENT_TYPE_INITPROTOCOL, protocolconnection);
1658 if(protocol->IsInitialized()) protocol->CMD_SendIsInitServerProtocol(true);
1659
1660 protocolconnection->SetInUse(false, DIOPROTOCOL_CONNECTIONSINUSEID_INIPROTOCOL);
1661 }
1662 else
1663 {
1664 XTIMER* GEN_XFACTORY_CREATE(timerout, CreateTimer())
1665
1666 while(1)
1667 {
1668 if(protocolconnection->GetDIOProtocol()->IsDisconnected()) break;
1669 if(protocol->IsServerInitialized()) break;
1670
1671 if(timerout)
1672 {
1673 if(timerout->GetMeasureSeconds() > 5) break;
1674 }
1675
1676 GEN_XSLEEP.MilliSeconds(10);
1677 }
1678
1679 GEN_XFACTORY.DeleteTimer(timerout);
1680
1681
1682 if(protocol->IsServerInitialized())
1683 {
1684 protocolconnection->SetInUse(true, DIOPROTOCOL_CONNECTIONSINUSEID_INIPROTOCOL);
1685
1686 protocol->GetDIOStream()->GetXTimerNotActivity()->Reset();
1687 protocolconnections->SendEvent(DIOPROTOCOL_CONNECTIONSMANAGER_XEVENT_TYPE_INITPROTOCOL, protocolconnection);
1688
1689 protocolconnection->SetInUse(false, DIOPROTOCOL_CONNECTIONSINUSEID_INIPROTOCOL);
1690 }
1691 }
1692
1693 XTRACE_PRINTCOLOR((protocol->IsInitialized()?2:4), __L("Protocol Connection Manager: %s Init Protocol %s") , protocolconnections->isserver?__L("Server"):__L("Client"), protocol->IsInitialized()?__L("Ok!"):__L("ERROR!"));
1694
1695 if(!protocol->IsInitialized())
1696 {
1697 protocolconnections->ProtocolConnections_Disconnect(c);
1698 }
1699 }
1700 else
1701 {
1702 if(protocol->GetDIOStream()->GetXTimerNotActivity()->GetMeasureSeconds() > DIOPROTOCOL_CONNECTIONS_DEFAULTIMECHECKCONNECTIONS)
1703 {
1704 XDWORD milliseconds;
1705
1706 bool status = protocol->CMD_Ping(milliseconds);
1707 if(!status)
1708 {
1709 protocolconnections->ProtocolConnections_Disconnect(c);
1710
1711 XTRACE_PRINTCOLOR(status?1:4, __L("Protocol Connection Manager: Check connection protocol %s"), status?__L("Ok!"):__L("ERROR!"));
1712 }
1713 }
1714 }
1715 }
1716 }
1717 }
1718
1719 c++;
1720 }
