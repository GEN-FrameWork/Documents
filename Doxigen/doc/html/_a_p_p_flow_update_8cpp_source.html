368{
369 if(!cfg) return false;
370
371 this->cfg = cfg;
372
373 this->update_cfg.CopyFrom(updatecfg);
374
375 //--------------------------------------------------------------------------------------

420bool APPFLOWUPDATE::EventManager(APPFLOWUPDATE_EVENTOPERATION eventoperation, XSUBJECT* xsubject, XOBSERVER* observer)
421{
422 XDWORD events[] = { DIOAPPLICATIONUPDATE_XEVENT_TYPE_GETVERSIONAVAILABLE ,
423 DIOAPPLICATIONUPDATE_XEVENT_TYPE_CHECKVERSIONAVAILABLE ,
424 DIOAPPLICATIONUPDATE_XEVENT_TYPE_DOWNLOADFILE ,
425 DIOAPPLICATIONUPDATE_XEVENT_TYPE_UNZIPFILE ,
426 DIOAPPLICATIONUPDATE_XEVENT_TYPE_DOWNLOADFILES ,
427 DIOAPPLICATIONUPDATE_XEVENT_TYPE_BACKUPORIGINALFILE ,
428 DIOAPPLICATIONUPDATE_XEVENT_TYPE_BACKUPORIGINALFILES ,
429 DIOAPPLICATIONUPDATE_XEVENT_TYPE_APP_RELEASERESOURCES ,
430 DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEFILE ,
431 DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEEXECFILE ,
432 DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEFILES ,
433 DIOAPPLICATIONUPDATE_XEVENT_TYPE_APP_END ,
434 DIOAPPLICATIONUPDATE_XEVENT_TYPE_RESTOREUPDATEFILE ,
435 DIOAPPLICATIONUPDATE_XEVENT_TYPE_RESTOREUPDATEFILES ,
436 };
437
438
439 for(XDWORD c=0; c< sizeof(events)/sizeof(XDWORD); c++)
440 {
441 switch(eventoperation)
442 {
443 case APPFLOWUPDATE_EVENTOPERATION_REGISTER : RegisterEvent(events[c]); break;
444 case APPFLOWUPDATE_EVENTOPERATION_DEREGISTER : DeRegisterEvent(events[c]); break;
445 case APPFLOWUPDATE_EVENTOPERATION_SUBSCRIBE : SubscribeEvent(events[c] , xsubject, observer); break;
446 case APPFLOWUPDATE_EVENTOPERATION_UNSUBSCRIBE : UnSubscribeEvent(events[c] , xsubject, observer); break;
447 default : return false;
448 }
449 }
450
451 return true;
452}

464bool APPFLOWUPDATE::Do()
465{
466 DIOAPPLICATIONUPDATE_VERSIONDATA applicationversiondata;
467 XPATH xpathupdate;
468 XPATH xpathbackup;
469 XSTRING string;
470 XSTRING stringresult;
471 bool status = false;
472 XSTRING dir_mask;
473 DIOAPPLICATIONUPDATE_ERROR error;
474
475 dir_mask = DIOAPPLICATIONUPDATE_RESTOREDIR_ID;
476 dir_mask += DIOAPPLICATIONUPDATE_RESTOREDIR_MASK;
477
478 if(!dioappupdate) return false;
479
480 if(update_cfg.GetDolog()) EventManager(APPFLOWUPDATE_EVENTOPERATION_SUBSCRIBE, dioappupdate, this);
481
482 // --------------------------------------------------------------------------------------------
483 // Get last update available
484 if(dioappupdate->GetVersionAvailable((*update_cfg.GetURL()), update_cfg.Get_Port(), applicationversiondata, 20, error))
485 {
486 // Check last update available
487 if(dioappupdate->IsNewVersion(applicationversiondata))
488 {
489 // Have GEN_NEW version available
490
491 // Download GEN_NEW version to update available to subdir root/update[x.x.x] Update Version
492 xpathupdate = update_cfg.GetPathRootApp()->Get();
493 xpathupdate.Slash_Add();
494 xpathupdate.AddFormat(dir_mask.Get() , applicationversiondata.GetVersion()
495 , applicationversiondata.GetSubVersion()
496 , applicationversiondata.GetSubVersionError());
497
498 status = dioappupdate->Download((*update_cfg.GetURL()), update_cfg.Get_Port(), xpathupdate, applicationversiondata, 60, error);
499
500 //-----------------------------------------------------------------------
501 DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_DOWNLOADFILES);
502
503 xevent.SetVersionData(&applicationversiondata);
504 xevent.SetError(error);
505
506 PostEvent(&xevent);
507 //-----------------------------------------------------------------------
508
509 if(status)
510 {
511 // New version update downloaded
512
513
514 //-----------------------------------------------------------------------
515 // Eliminate resources APPLICATION
516 { DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_APP_RELEASERESOURCES);
517 xevent.SetVersionData(&applicationversiondata);
518 xevent.SetError(error);
519
520 PostEvent(&xevent);
521 }
522 //-----------------------------------------------------------------------
523
524
525 // Actual files to be updated available to subdir root/update[x.x.x] Actual version
526 xpathbackup = update_cfg.GetPathRootApp()->Get();;
527 xpathbackup.Slash_Add();
528
529 xpathbackup.AddFormat(dir_mask.Get() , update_cfg.GetAppVersion()->GetVersion()
530 , update_cfg.GetAppVersion()->GetSubVersion()
531 , update_cfg.GetAppVersion()->GetSubVersionError());
532
533 // Change actual files for the GEN_NEW version
534 status = dioappupdate->ChangeFilesFromDownload(applicationversiondata, xpathupdate, xpathbackup, error);
535
536 //-----------------------------------------------------------------------
537 { DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEFILES);
538
539 xevent.SetVersionData(&applicationversiondata);
540 xevent.SetError(error);
541
542 PostEvent(&xevent);
543 }
544 //-----------------------------------------------------------------------
545
546 if(!status)
547 {
548 status = dioappupdate->Restore(xpathbackup, error);
549
550 //-----------------------------------------------------------------------
551 DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_RESTOREUPDATEFILES);
552
553 xevent.SetVersionData(&applicationversiondata);
554 xevent.SetError(error);
555
556 PostEvent(&xevent);
557 //-----------------------------------------------------------------------
558 }
559
560 if(status)
561 {
562 if(applicationversiondata.SystemMustBeInit())
563 {
564 GEN_XSYSTEM.ShutDown(XSYSTEM_CHANGESTATUSTYPE_REBOOT);
565 }
566 else
567 {
568 // Exit & Restart of Application
569
570 DIOAPPLICATIONUPDATE_XEVENT xevent(this, DIOAPPLICATIONUPDATE_XEVENT_TYPE_APP_END);
571
572 xevent.SetVersionData(&applicationversiondata);
573 xevent.SetError(error);
574
575 PostEvent(&xevent);
576
577 dioappupdate->RestartApplication(error);
578 }
579 }
580 }
581 }
582 }
583
584 if(update_cfg.GetDolog()) EventManager(APPFLOWUPDATE_EVENTOPERATION_UNSUBSCRIBE, dioappupdate, this);
585
586 return status;
587}

751bool APPFLOWUPDATE::LogOfEvent(DIOAPPLICATIONUPDATE_XEVENT* event)
752{
753 DIOAPPLICATIONUPDATE_FILE* updatefile = event->GetActualFileToUpdate();
754 XSTRING string;
755
756 switch(event->GetEventType())
757 {
758 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_GETVERSIONAVAILABLE : if(event->GetError() == DIOAPPLICATIONUPDATE_NOT_ERROR)
759 {
760 string.Format( __L("Update Version available %d.%d.%d")
761 , event->GetVersionData()->GetVersion()
762 , event->GetVersionData()->GetSubVersion()
763 , event->GetVersionData()->GetSubVersionError());
764 }
765 else
766 {
767 string.Format( __L("No version available to update"));
768 }
769 break;
770
771 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_CHECKVERSIONAVAILABLE : string.Format( __L("Version %d.%d.%d is ")
772 , event->GetVersionData()->GetVersion()
773 , event->GetVersionData()->GetSubVersion()
774 , event->GetVersionData()->GetSubVersionError());
775
776 switch(event->GetVersionComparation())
777 {
778 case -1 : string.Add(__L("old")); break;
779 case 0 : string.Add(__L("equal")); break;
780 case 1 : string.Add(__L("GEN_NEW")); break;
781 }
782
783 if(event->GetVersionComparation() <=0) string.AddFormat(__L(". It will not be updated"));
784
785 break;
786
787
788 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_DOWNLOADFILE : if(updatefile) string.Format( __L("Download file %s (%dk).")
789 , updatefile->GetXPathFile()->Get()
790 , updatefile->GetSize()/1024);
791
792 break;
793
794 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_UNZIPFILE : if(updatefile) string.Format( __L("Unzip file %s (%dk) -> %s.")
795 , updatefile->GetXPathFile()->Get()
796 , updatefile->GetSize()/1024
797 , event->GetActualUnzipFileName()->Get());
798 break;
799
800 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_DOWNLOADFILES : string.Format(__L("Downloads file %s."), (event->GetError() == DIOAPPLICATIONUPDATE_NOT_ERROR?__L("Ok."):__L("ERROR!")));
801 break;
802
803 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_BACKUPORIGINALFILE : if(updatefile) string.Format( __L("Backup original file %s (%dk).")
804 , updatefile->GetXPathFile()->Get()
805 , updatefile->GetSize()/1024);
806 break;
807
808 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_BACKUPORIGINALFILES : string.Format(__L("Backup Original file %s."), (event->GetError() == DIOAPPLICATIONUPDATE_NOT_ERROR?__L("Ok."):__L("ERROR!")));
809 break;
810
811 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_APP_RELEASERESOURCES : string.Format(__L("Application release blocked resources"));
812 break;
813
814
815 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEFILE : if(updatefile) string.Format( __L("Copy update file %s (%dk).")
816 , updatefile->GetXPathFile()->Get()
817 , updatefile->GetSize()/1024);
818 break;
819
820 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEEXECFILE : if(updatefile) string.Format( __L("Notify update EXEC file %s (%dk).")
821 , updatefile->GetXPathFile()->Get()
822 , updatefile->GetSize()/1024);
823 break;
824
825 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_UPDATEFILES : string.Format(__L("Update files %s."), (event->GetError() == DIOAPPLICATIONUPDATE_NOT_ERROR?__L("Ok."):__L("ERROR!")));
826 break;
827
828
829 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_APP_END : string.Format(__L("Terminate application"));
830 break;
831
832 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_RESTOREUPDATEFILE : if(updatefile) string.Format( __L("Restore file %s from backup (%dk).")
833 , updatefile->GetXPathFile()->Get()
834 , updatefile->GetSize()/1024);
835 break;
836
837 case DIOAPPLICATIONUPDATE_XEVENT_TYPE_RESTOREUPDATEFILES : string.Format(__L("Restore update files %s."), (event->GetError() == DIOAPPLICATIONUPDATE_NOT_ERROR?__L("Ok."):__L("ERROR!")));
838 break;
839 }
840
841 if(string.GetSize())
842 {
843 XLOG::GetInstance().AddEntry((event->GetError() == DIOAPPLICATIONUPDATE_NOT_ERROR)?XLOGLEVEL_INFO:XLOGLEVEL_ERROR, DIOAPPLICATIONUPDATE_LOG_SECTIONID_APPFLOWUPDATE, false, string.Get());
844 return true;
845 }
846
847 return false;
848}

891bool APPFLOWUPDATE::CreateTaskUpdate(bool doajust)
892{
893 if(!cfg) return false;
894
895 XSCHEDULERTASK* xtask = GEN_NEW XSCHEDULERTASK(xscheduler);
896 if(!xtask) return false;
897
898 bool toactive = false;
899
900 XDATETIME* GEN_XFACTORY_CREATE(start, CreateDateTime())
901 XDATETIME* GEN_XFACTORY_CREATE(end, CreateDateTime())
902 if(start && end)
903 {
904 if(!cfg->ApplicationUpdate_GetCheckTime()->IsEmpty())
905 {
906 start->Read();
907
908 start->GetDateTimeFromString((*cfg->ApplicationUpdate_GetCheckTime()), XDATETIME_FORMAT_ADDTIME);
909 start->SetSeconds(0);
910 start->SetMilliSeconds(0);
911
912 start->AddDays(1);
913
914 end->Set(start);
915 end->AddSeconds(5);
916
917 xtask->SetTimeLimits(start, end);
918
919 toactive = true;
920 }
921 else
922 {
923 int minutescadence = cfg->ApplicationUpdate_GetCheckCadence();
924 if(minutescadence)
925 {
926 start->Read();
927
928 if(doajust)
929 {
930 int minutes = ((start->GetMinutes() / minutescadence) + ((start->GetMinutes() % minutescadence)?1:0)) * minutescadence;
931 if(minutes == start->GetMinutes()) minutes += minutescadence;
932 if(minutes >= 60)
933 {
934 start->AddHours(1);
935 minutes = 0;
936 }
937
938 start->SetMinutes(minutes);
939
940 } else start->AddMinutes(minutescadence);
941
942 start->SetSeconds(0);
943 start->SetMilliSeconds(0);
944
945 xtask->SetTimeLimits(start);
946
947 toactive = true;
948 }
949 }
950 }
951
952 if(toactive)
953 {
954 xtask->SetID(APPFLOWUPDATE_TASKID_CHECKUPDATEAPPLICATION);
955 xtask->SetNCycles(1);
956 xtask->SetIsActive(true);
957
958 xscheduler->Task_Add(xtask);
959 }
960 else GEN_DELETE xtask;
961
962 GEN_XFACTORY.DeleteDateTime(start);
963 GEN_XFACTORY.DeleteDateTime(end);
964
965 return true;
966}
