601bool XFILECFG::Load(XPATH& xpath)
602{
603 bool status = false;
604
605 if(values.IsEmpty()) return status;
606
607 if(fileini->IsOpen())
608 {
609 EndFile();
610 }
611
612 if(IniFile(xpath))
613 {
614 AjustRemarks();
615
616 for(int c=0;c<(int)values.GetSize();c++)
617 {
618 XFILECFGVALUE* cfgvalue = (XFILECFGVALUE*)values.Get(c);
619 if(cfgvalue)
620 {
621 XSTRING strvalue;
622
623 if(fileini->ReadValue(cfgvalue->GetGroup()->Get(), cfgvalue->GetID()->Get() , strvalue))
624 {
625 switch(cfgvalue->GetType())
626 {
627 case XFILECFG_VALUETYPE_UNKNOWN : break;
628
629 case XFILECFG_VALUETYPE_INT : { int* value = (int*)cfgvalue->GetValue();
630 if(!value) return false;
631
632 (*value) = strvalue.ConvertToInt();
633 }
634 break;
635
636 case XFILECFG_VALUETYPE_MASK : { XWORD* value = (XWORD*)cfgvalue->GetValue();
637 if(!value) return false;
638
639 strvalue.UnFormat(__L("%04X"), value);
640 }
641 break;
642
643 case XFILECFG_VALUETYPE_FLOAT : { float* value = (float*)cfgvalue->GetValue();
644 if(!value) return false;
645
646 (*value) = strvalue.ConvertToFloat();
647 }
648 break;
649
650 case XFILECFG_VALUETYPE_STRING : { XSTRING* value = (XSTRING*)cfgvalue->GetValue();
651 if(!value) return false;
652
653 (*value) = strvalue.Get();
654
655 value->DeleteCharacter(__C('\t'));
656 value->DeleteCharacter(__C('\n'));
657 value->DeleteCharacter(__C('\r'));
658
659 value->DeleteCharacter(__C(' '), XSTRINGCONTEXT_TO_END);
660 }
661 break;
662
663 case XFILECFG_VALUETYPE_BOOLEAN : { bool* value = (bool*)cfgvalue->GetValue();
664 if(!value) return false;
665
666 (*value) = strvalue.ConvertToBoolean();
667 }
668 break;
669 }
670 }
671 }
672 }
673
674 status = EndFile();
675 }
676
677 return status;
678}
