301bool XFEEDBACK_CONTROL::AddFeedbak(char const* namefile, XFEEDBACK_CODE code,int line, ...)
302{
303 if(!isactive)
304 {
305 return false;
306 }
307
308 XFEEDBACK* feedback = GEN_NEW XFEEDBACK();
309 if(feedback)
310 {
311 feedback->SetCode(code);
312 feedback->GetModuleName()->Set(namefile);
313 feedback->SetModuleLine(line);
314
315 XSTRING text;
316
317 if(GetFeedbakText(code, text))
318 {
319 if(!text.IsEmpty())
320 {
321 va_list arg;
322
323 va_start(arg, line);
324
325 switch(code)
326 {
327 case XFEEDBACK_CODE_UNKNOWN : feedback->GetText()->Set(text);
328 break;
329
330 case XFEEDBACK_CODE_INVALIDPARAMFUNCTION : { int indexparam = va_arg(arg, int);
331 XCHAR* nameparam = va_arg(arg, XCHAR*);
332 XCHAR* namefunction = va_arg(arg, XCHAR*);
333
334 if(nameparam && namefunction)
335 {
336 feedback->GetText()->Format(text.Get(), indexparam, nameparam, namefunction);
337 }
338 }
339 break;
340
341 case XFEEDBACK_CODE_INVALIDMEMORYALLOCATION : { XCHAR* namevariable = va_arg(arg, XCHAR*);
342 XCHAR* namefunction = va_arg(arg, XCHAR*);
343
344 if(namevariable && namefunction)
345 {
346 feedback->GetText()->Format(text.Get(), namevariable, namefunction);
347 }
348 }
349 break;
350
351 case XFEEDBACK_CODE_ERRORINFUNCTION : { XCHAR* namefunction = va_arg(arg, XCHAR*);
352 XCHAR* errortext = va_arg(arg, XCHAR*);
353
354 if(namefunction && errortext)
355 {
356 feedback->GetText()->Format(text.Get(), namefunction, errortext);
357 }
358 }
359 break;
360
361 default : feedback->GetText()->Set(text);
362 break;
363 }
364
365 va_end(arg);
366
367
368
369 GEN_XLOG.AddEntry(XLOGLEVEL_WARNING, XFEEDBACK_CONTROL_LOG_SECTIONID, false , __L("%05d line %d, module %s -> %s"), feedbacks.GetSize() + 1
370 , feedback->GetModuleLine()
371 , feedback->GetModuleName()->Get()
372 , feedback->GetText()->Get());
373
374
375
376 return feedbacks.Add(feedback);
377 }
378 }
379 }
380
381 GEN_DELETE feedback;
382
383 return false;
384}
