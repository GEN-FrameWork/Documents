895bool DIOCOREPROTOCOL_HEADER::Deserialize()
896{
897 XSTRING string;
898
899 string.Empty();
900 Primitive_Extract<XSTRING>(string, DIOCOREPROTOCOL_HEADER_VAR_ID_MESSAGE);
901 ID_message.SetFromString(string);
902
903 #ifdef DIOCOREPROTOCOL_HUMANFORMAT_ACTIVE

906 #else
907 Primitive_Extract<int>((int&)message_type, DIOCOREPROTOCOL_HEADER_VAR_MESSAGE_TYPE);
908 #endif
909
910 #ifdef DIOCOREPROTOCOL_HUMANFORMAT_ACTIVE
911 Primitive_Extract<XSTRING>(string, DIOCOREPROTOCOL_HEADER_VAR_OPERATION);
912 GetOperationFromString(&string, operation);
913 #else
914 Primitive_Extract<int>((int&)operation, DIOCOREPROTOCOL_HEADER_VAR_OPERATION);
915 #endif
916
917 Primitive_Extract<XSTRING>(operation_param, DIOCOREPROTOCOL_HEADER_VAR_OPERATION_PARAM);
918
919 string.Empty();
920 Primitive_Extract<XSTRING>(string, DIOCOREPROTOCOL_HEADER_VAR_DATETIME_SEND);
921 datetime_send->GetDateTimeFromStringISO8601(string, XDATETIME_FORMAT_ISO8601_STANDARD | XDATETIME_FORMAT_TIMEWITHMILLISECONDS);
922
923 #ifdef DIOCOREPROTOCOL_HUMANFORMAT_ACTIVE
924 Primitive_Extract<XSTRING>(string, DIOCOREPROTOCOL_HEADER_VAR_CONTENT_TYPE);
925 GetContentTypeFromString(&string, content_type);
926 #else
927 Primitive_Extract<int>((int&)content_type, DIOCOREPROTOCOL_HEADER_VAR_CONTENT_TYPE);
928 #endif
929
930 Primitive_Extract<XDWORD>(block_index, DIOCOREPROTOCOL_HEADER_VAR_BLOCK_INDEX);
931
932 Primitive_Extract<XDWORD>(block_amount, DIOCOREPROTOCOL_HEADER_VAR_BLOCK_AMOUNT);
933
934 Primitive_Extract<XDWORD>(content_size, DIOCOREPROTOCOL_HEADER_VAR_CONTENT_SIZE);
935
936 Primitive_Extract<XDWORD>(content_compresssize, DIOCOREPROTOCOL_HEADER_VAR_CONTENT_COMPRESSSIZE);
937
938 string.Empty();
939 Primitive_Extract<XSTRING>(string, DIOCOREPROTOCOL_HEADER_VAR_CONTENT_CRC32);
940 string.UnFormat(__L("%08X"), &content_CRC32);
941
942 return true;
