106GRPVECTORFILERESULT GRPVECTORFILEDXF::DetectType()
107{
108 GRPVECTORFILERESULT result;
109
110 if(pathfile.IsEmpty())
111 {
112 type = GRPVECTORFILETYPE_UNKNOWN;
113
114 return GRPVECTORFILERESULT_ERRORNOTFILE;
115 }
116
117 XSTRING extension;
118 pathfile.GetExt(extension);
119
120 if(extension.IsEmpty())
121 {
122 type = GRPVECTORFILETYPE_UNKNOWN;
123
124 return GRPVECTORFILERESULT_ERRORNOTFILE;
125 }
126
127 if(!extension.Compare(GRPVECTORFILEDXF_EXTENSION, true))
128 {
129 type = GRPVECTORFILETYPE_DXF;
130 }
131
132 if(DetectFileFormatText(pathfile))
133 {
134 XFILETXT* file = GEN_NEW XFILETXT();
135 if(file)
136 {
137 if(file->Open(pathfile))
138 {
139 int nlines = 100;
140
141 file->ReadNLines(nlines);
142
143 nlines = file->GetNLines();
144
145 if(ParserHaveAnySection(file, nlines))
146 {
147 result = GRPVECTORFILERESULT_OK;
148 }
149 else
150 {
151 result = GRPVECTORFILERESULT_ERRORINVALIDFORMAT;
152 }
153
154 file->Close();
155 }
156 else
157 {
158 result = GRPVECTORFILERESULT_ERRORNOTFILE;
159 }
160
161 GEN_DELETE file;
162 }
163 else
164 {
165 result = GRPVECTORFILERESULT_ERRORNOTMEMORY;
166 }
167 }
168 else
169 {
170 result = GRPVECTORFILERESULT_ERRORINVALIDFORMAT;
171 }
172
173 return result;
174}

186GRPVECTORFILERESULT GRPVECTORFILEDXF::Load()
187{
188 GRPVECTORFILEDXFRESULT result = GRPVECTORFILERESULT_ERRORNOTMEMORY;
189
190 XFILETXT* file = GEN_NEW XFILETXT();
191 if(!file)
192 {
193 return GRPVECTORFILERESULT_ERRORNOTMEMORY;
194 }
195
196 if(file->Open(pathfile))
197 {
198 XTIMER* xtimer = GEN_XFACTORY.CreateTimer();
199 if(xtimer)
200 {
201 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("Reading TXT file ..."));
202
203 xtimer->Reset();
204
205 file->ReadAllFile();
206
207 //XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, __L("Read TXT file: %ld Milliseconds."), xtimer->GetMeasureMilliSeconds());
208
209 GEN_XFACTORY.DeleteTimer(xtimer);
210 }
211
212 result = ParserTextFile (file);
213
214 file->Close();
215 }
216 else
217 {
218 result = GRPVECTORFILERESULT_ERRORNOTFILE;
219 }
220
221 GEN_DELETE file;
222
223 return result.TakeAsBase();
224}

325GRPVECTORFILEDXFRESULT GRPVECTORFILEDXF::ParserTextSections(XFILETXT* fileTXT, XVECTOR<GRPVECTORFILEDXFTEXTSECTION*>& sections)
326{
327 GRPVECTORFILEDXFTEXTSECTION* section = NULL;
328 GRPVECTORFILEDXFRESULT result = GRPVECTORFILERESULT_ERRORUNKNOWN;
329 int c = 0;
330
331 do{ if(!section)
332 {
333 section = GEN_NEW GRPVECTORFILEDXFTEXTSECTION ();
334 }
335
336 XSTRING* line = fileTXT->GetLine(c);
337 ParserTextFilePrepareLine(line);
338
339 if(line && !line->Compare(__L("0"),true))
340 {
341 line = fileTXT->GetLine(c+1);
342 ParserTextFilePrepareLine(line);
343
344 if(line && !line->Compare(__L("SECTION"), false))
345 {
346 line = fileTXT->GetLine(c+2);
347 ParserTextFilePrepareLine(line);
348 if(line && !line->Compare(__L("2"),true))
349 {
350 c += 3;
351
352 line = fileTXT->GetLine(c);
353 ParserTextFilePrepareLine(line);
354
355 section->iniline = c+1;
356 section->name = line->Get();
357
358 c++;
359 }
360 }
361
362 if(line && !line->Compare(__L("ENDSEC"), false))
363 {
364 section->endline = c-1;
365 section->type = GRPVECTORFILEDXFTEXTSECTION::GetTypeSection(section->name);
366
367 #ifdef XTRACE_ACTIVE
368 if(section->type == GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_UNKNOWN)
369 {
370 XSTRING message;
371
372 message.Format(__L("section %s Unknown"), section->name.Get());
373
374 GRPVECTORFILE_XEVENT vfEvent(this, GRPVECTORFILE_XEVENTTYPE_PARTUNKNOWN);
375
376 vfEvent.SetType(GRPVECTORFILETYPE_DXF);
377 vfEvent.GetPath()->Set(fileTXT->GetPrimaryFile()->GetPathNameFile());
378 vfEvent.GetMsg()->Set(message);
379
380 PostEvent(&vfEvent);
381 }
382 #endif
383
384 GRPVECTORFILEDXFTEXTSECTION* sectionType = GRPVECTORFILEDXFTEXTSECTION::CreateInstance(section->type);
385 if(sectionType)
386 {
387 section->CopyTo(sectionType);
388 sectionType->SetGrpVectorFile(this);
389 sections.Add(sectionType);
390
391 GEN_DELETE section;
392 }
393 else
394 {
395 sections.Add(section);
396 }
397
398 section = NULL;
399 }
400
401 if(line && !line->Compare(__L("EOF"), false))
402 {
403 c++;
404 break;
405 }
406 }
407
408 c++;
409
410 } while(c < fileTXT->GetNLines());
411
412 if(section && (section->type == GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_UNKNOWN))
413 {
414 GEN_DELETE section;
415 }
416
417 if(!sections.IsEmpty())
418 {
419 result = GRPVECTORFILERESULT_OK;
420 }
421 else
422 {
423 result = GRPVECTORFILEDXFRESULT_NOTSECTIONS;
424 }
425
426 return result;
427}

494GRPVECTORFILEDXFRESULT GRPVECTORFILEDXF::ParserTextFile(XFILETXT* fileTXT)
495{
496 GRPVECTORFILEDXFRESULT result = GRPVECTORFILERESULT_ERRORUNKNOWN;
497
498 sections.DeleteContents();
499 sections.DeleteAll();
500
501 result = ParserTextSections (fileTXT, sections);
502 if(result == (GRPVECTORFILEDXFRESULT)GRPVECTORFILERESULT_OK)
503 {
504 for(XDWORD c=0; c<sections.GetSize(); c++)
505 {
506 GRPVECTORFILEDXFTEXTSECTION* section = sections.Get(c);
507 if(section)
508 {
509 result = section->ParserTextSection(fileTXT);
510 if(result != (GRPVECTORFILEDXFRESULT)GRPVECTORFILERESULT_OK)
511 {
512 break;
513 }
514 }
515 }
516
517 if(!sections.GetSize())
518 {
519 result = GRPVECTORFILEDXFRESULT_NOTSECTIONS;
520 }
521 }
522
523 GRPVECTORFILEDXFTEXTSECTIONHEADER* sectionheader = (GRPVECTORFILEDXFTEXTSECTIONHEADER*)GetSection(GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_HEADER);
524 if(sectionheader)
525 {
526 config.SetHeader(sectionheader);
527 }
528
529 #ifdef XTRACE_ACTIVE
530 ShowTraceAllSections();
531 #endif
532
533 return result;
534}

585bool GRPVECTORFILEDXF::ShowTraceAllSections()
586{
587 for(XDWORD c=0; c<sections.GetSize(); c++)
588 {
589 GRPVECTORFILEDXFTEXTSECTION* section = sections.Get(c);
590 if(section)
591 {
592 XSTRING line;
593
594 line.Format(__L("[Vector file DXF load] Section [%d, %d]: %s"), section->iniline, section->endline, section->name.Get());
595
596 switch(section->type)
597 {
598 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_UNKNOWN :
599 default : break;
600
601 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_HEADER : { GRPVECTORFILEDXFTEXTSECTIONHEADER* sectionheader = (GRPVECTORFILEDXFTEXTSECTIONHEADER*)GetSection (section->type);
602 if(sectionheader)
603 {
604 line.AddFormat(__L(" -> NVariables (%d)"), sectionheader->GetVariables()->GetSize());
605 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, line.Get());
606 }
607 }
608 break;
609
610 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_CLASSES :
611 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_TABLES : break;
612
613 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_BLOCKS : { GRPVECTORFILEDXFTEXTSECTIONBLOCKS* sectionblocks = (GRPVECTORFILEDXFTEXTSECTIONBLOCKS*)GetSection(section->type);
614 if(sectionblocks)
615 {
616 line.AddFormat(__L(" -> (%d)"), sectionblocks->GetBlocks()->GetSize());
617 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, line.Get());
618
619 for(XDWORD d=0; d<sectionblocks->GetBlocks()->GetSize(); d++)
620 {
621 GRPVECTORFILEDXFTEXTBLOCK* block = sectionblocks->GetBlocks()->Get(d);
622 if(block)
623 {
624 if(!block->IsEndBlock())
625 {
626 GRPVECTORFILEDXFTEXTSECTIONENTITIES* sectionentities = block->GetSectionEntities();
627 if(sectionentities)
628 {
629 line.Format(__L("[Vector file DXF load] Block (%2d) [%s] ENTITIES "), d+1, block->GetName()->Get());
630 GenerateSectionsEntities(sectionentities,line);
631 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, line.Get());
632 }
633 }
634 }
635 }
636 }
637 }
638 break;
639
640 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_ENTITIES : { GRPVECTORFILEDXFTEXTSECTIONENTITIES* sectionentities = (GRPVECTORFILEDXFTEXTSECTIONENTITIES*)GetSection (section->type);
641 if(sectionentities)
642 {
643 GenerateSectionsEntities(sectionentities,line);
644 XTRACE_PRINTCOLOR(XTRACE_COLOR_BLUE, line.Get());
645 }
646 }
647 break;
648 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_OBJECTS :
649 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_ACDSDATA :
650 case GRPVECTORFILEDXFTEXTSECTION_TYPESECTION_THUMBNAILIMAGE : break;
651 }
652 }
653 }
654
655 return true;
656}
