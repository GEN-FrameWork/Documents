379bool INPCURSORMOTION::AddFromLine(INPCURSORMOTION_REDUCEDMODE reducedmode, int reducedfactor, int x1, int y1, int x2, int y2)
380{
381 int x, xinc1, xinc2;
382 int y, yinc1, yinc2;
383 int i, deltax, deltay, npixels;
384 int d, dinc1, dinc2;
385
386 // Calculate deltax and deltay for initialisation
387 deltax = abs(x2 - x1);
388 deltay = abs(y2 - y1);
389
390 // Initialize all vars based on which is the independent variable
391 if(deltax>=deltay)
392 {
393 // x is independent variable }
394 npixels = deltax + 1;
395 d = (2 * deltay) - deltax;
396 dinc1 = deltay >> 1;
397 dinc2 = (deltay - deltax) >> 1;
398 xinc1 = 1;
399 xinc2 = 1;
400 yinc1 = 0;
401 yinc2 = 1;
402 }
403 else
404 {
405 // y is independent variable }
406 npixels = deltay + 1;
407 d = (2 * deltax) - deltay;
408 dinc1 = deltax >> 1;
409 dinc2 = (deltax - deltay) >> 1;
410 xinc1 = 0;
411 xinc2 = 1;
412 yinc1 = 1;
413 yinc2 = 1;
414 }
415
416 // Make sure x and y move in the right directions
417 if(x1>x2)
418 {
419 xinc1 = -xinc1;
420 xinc2 = -xinc2;
421 }
422
423 if(y1>y2)
424 {
425 yinc1 = -yinc1;
426 yinc2 = -yinc2;
427 }
428
429 // Start drawing at
430 x = x1;
431 y = y1;
432
433
434 int steps = CalculateReducedModeSteps(reducedmode,reducedfactor,npixels);
435 int incstep = steps;
436 bool addpoint = false;
437
438 // Draw the pixels }
439 for(i=0;i<npixels;i++)
440 {
441 if(incstep == steps)
442 {
443 addpoint = true;
444 incstep = 0;
445 }
446
447 if(addpoint)
448 AddPoint((float)x, (float)y, 0);
449 else incstep++;
450
451 addpoint = false;
452
453 if(d<0)
454 {
455 d = d + dinc1;
456 x = x + xinc1;
457 y = y + yinc1;
458 }
459 else
460 {
461 d = d + dinc2;
462 x = x + xinc2;
463 y = y + yinc2;
464 }
465 }
466
467 return false;
468}

486bool INPCURSORMOTION::AddFromCircle(INPCURSORMOTION_REDUCEDMODE reducedmode,int reducedfactor,int xc, int yc, int radio)
487{
488 float deg = 0;
489 int npixels = 0;

491 do{ deg+=0.005f;

493 npixels++;
494
495 } while (deg <= 6.4);
496
497 deg = 0;
498
499 int steps = CalculateReducedModeSteps(reducedmode,reducedfactor,npixels);

519
520 } while (deg <= 6.4);
521
522 return true;

851INPCURSORMOTION_DIR INPCURSORMOTION::GetDirectionByAngle(bool indetail)
852{
853 if(points.GetSize() < 2) return INPCURSORMOTION_DIR_UNKNOWN;
854
855 float angle = GetAngle();
856
857 if(indetail)
858 {
859 if((angle > 337.5f) || (angle <= 22.5f)) return INPCURSORMOTION_DIR_RIGHT;
860 if((angle > 22.5f) && (angle <= 67.5f)) return INPCURSORMOTION_DIR_RIGHTUP;
861 if((angle > 67.5f) && (angle <= 112.5f)) return INPCURSORMOTION_DIR_UP;
862 if((angle > 112.5f) && (angle <= 157.5f)) return INPCURSORMOTION_DIR_LEFTUP;
863 if((angle > 157.5f) && (angle <= 202.5f)) return INPCURSORMOTION_DIR_LEFT;
864 if((angle > 202.5f) && (angle <= 247.5f)) return INPCURSORMOTION_DIR_LEFTDOWN;
865 if((angle > 247.5f) && (angle <= 292.5f)) return INPCURSORMOTION_DIR_DOWN;
866 if((angle > 292.5f) && (angle <= 337.5f)) return INPCURSORMOTION_DIR_RIGHTDOWN;
867 }
868 else
869 {
870 if((angle > 270) || (angle <= 45)) return INPCURSORMOTION_DIR_RIGHT;
871 if((angle > 45) && (angle <= 135)) return INPCURSORMOTION_DIR_UP;
872 if((angle > 135) && (angle <= 180)) return INPCURSORMOTION_DIR_LEFT;
873 if((angle > 180) && (angle <= 270)) return INPCURSORMOTION_DIR_DOWN;
874 }
875
876 return INPCURSORMOTION_DIR_UNKNOWN;
877}

891INPCURSORMOTION_DIR INPCURSORMOTION::GetDirectionByDifferential(XDWORD* differential)
892{
893 INPCURSORMOTION_DIR dir = INPCURSORMOTION_DIR_UNKNOWN;
894
895 if(differential) (*differential) = 0;
896
897 if(points.GetSize() < 2) return dir;
898
899 int max_up = 0;
900 int max_down = 0;
901 int max_right = 0;
902 int max_left = 0;
903
904 int x_origin = (int)points.Get(0)->GetX();
905 int y_origin = (int)points.Get(0)->GetY();
906
907 for(XDWORD c=1; c<points.GetSize(); c++)
908 {
909 max_right = __MAX(((int)points.Get(c)->GetX() - x_origin), max_right);
910 max_left = __MAX((x_origin - (int)points.Get(c)->GetX()), max_left);
911
912 max_up = __MAX(((int)points.Get(c)->GetY() - y_origin), max_up);
913 max_down = __MAX((y_origin - (int)points.Get(c)->GetY()), max_down);
914 }
915
916 int x = __MAX(max_right, max_left);
917 int y = __MAX(max_up , max_down);
918
919 if(x > y)
920 {
921 if(differential) (*differential) = x;
922
923 dir = INPCURSORMOTION_DIR_LEFT;
924 if(max_right > max_left) dir = INPCURSORMOTION_DIR_RIGHT;
925 }
926 else
927 {
928 if(differential) (*differential) = y;
929
930 dir = INPCURSORMOTION_DIR_DOWN;
931 if(max_up > max_down) dir = INPCURSORMOTION_DIR_UP;
932 }
933
934 return dir;
935}

1005int INPCURSORMOTION::CalculateReducedModeSteps(INPCURSORMOTION_REDUCEDMODE mode,int factor,int npixels)
1006{
1007 int steps = 0;
1008
1009 switch(mode)
1010 {
1011 case INPCURSORMOTION_REDUCEDMODE_NONE : break;
1012 case INPCURSORMOTION_REDUCEDMODE_BYSTEPS : steps = factor; break;
1013 case INPCURSORMOTION_REDUCEDMODE_NMAX : steps = (npixels/factor); break;

1016
1017 return steps;
